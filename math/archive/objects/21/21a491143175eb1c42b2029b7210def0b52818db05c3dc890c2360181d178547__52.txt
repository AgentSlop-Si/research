14: def make_matrix(paths,n=256):
15:     edges=scan_path_edges(paths)
16:     a=np.array([v[0] for v in edges]);b=np.array([v[1] for v in edges])
17:     rr=np.r_[a,b,a,b];cc=np.r_[a,b,b,a]
18:     vv=np.r_[np.ones(2*len(a),dtype=np.int64),-np.ones(2*len(a),dtype=np.int64)]
19:     return coo_matrix((vv,(rr,cc)),shape=(n,n)).tocsr()
20: 
21: def lam(p):
22:     L=make_matrix(p)
23:     return float(sorted(eigsh(L,k=2,which='SM',tol=1e-9,return_eigenvectors=False))[1])

25: def gf2_rank_cols(cols,k):
26:     rank=0;basis={}
27:     for x in cols:
28:         v=int(x)
29:         while v:
30:             b=v.bit_length()-1
31:             if b in basis:v^=basis[b]
32:             else:
33:                 basis[b]=v;rank+=1;break
34:     return rank
35: 
36: def random_cols(rng,k):
37:     while True:
38:         v=[int(i) for i in rng.integers(0,1<<k,size=k)]
39:         if gf2_rank_cols(v,k)==k:return v

42:     n=1<<k;p=np.arange(n,dtype=np.int64)
43:     cm=sum(1<<c for c in controls)
44:     t=1<<target
45:     for i in range(n):
46:         if (i&cm)==cm:p[i]=i^t
