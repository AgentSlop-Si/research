   1 #!/usr/bin/env python3
   2 """Independent inert ZIP/source audit. Never imports or executes archive members."""
   3 from pathlib import Path,PurePosixPath
   4 import hashlib,io,json,re,stat,zipfile
   5 
   6 BASE=Path(__file__).resolve().parent
   7 known=set(json.loads((BASE/'PRIOR_HASH_REFERENCE.json').read_text())['hashes'])
   8 meta=json.loads((BASE/'SOURCE_IDENTITIES.json').read_text())
   9 rows=[]; checks=[]; archives=[]
  10 def sha(b): return hashlib.sha256(b).hexdigest()
  11 def save_json(name,data): (BASE/name).write_text(json.dumps(data,indent=2)+'\n')
  12 def walk(blob,chain,out,depth=0):
  13     assert depth<32
  14     with zipfile.ZipFile(io.BytesIO(blob)) as z:
  15         members={}
  16         for item in z.infolist():
  17             if item.is_dir(): continue
  18             rel=PurePosixPath(item.filename)
  19             assert not rel.is_absolute() and '..' not in rel.parts
  20             assert not stat.S_ISLNK(item.external_attr>>16)
  21             assert item.filename not in members,'Duplicate archive path'
  22             data=z.read(item); members[item.filename]=data
  23             dst=out/rel; dst.parent.mkdir(parents=True,exist_ok=True)
  24             if dst.exists(): assert dst.read_bytes()==data,'Extracted bytes differ'
  25             else: dst.write_bytes(data)
  26             rows.append({'archive':chain,'member':item.filename,'path':str(dst.relative_to(BASE)),
  27                          'sha256':sha(data),'bytes':len(data),'new_to_combined_prior':sha(data) not in known})
  28             if zipfile.is_zipfile(io.BytesIO(data)):
  29                 walk(data,chain+'!'+item.filename,out/(item.filename+'.unpacked'),depth+1)
  30         for name,data in members.items():
  31             if PurePosixPath(name).name not in ('SHA256SUMS.txt','SHA256SUMS','MANIFEST.sha256'): continue
  32             for line in data.decode().splitlines():
  33                 if not line.strip(): continue
  34                 m=re.fullmatch(r'([0-9a-fA-F]{64})\s+\*?(.+)',line)
  35                 assert m,(chain,name,line)
  36                 target=m[2]; rel=PurePosixPath(target)
  37                 assert not rel.is_absolute() and '..' not in rel.parts
  38                 candidates={str(PurePosixPath(name).parent/rel),target}
  39                 present=sorted(c for c in candidates if c in members)
  40                 assert present,(chain,name,target,'Missing checksum target')
  41                 hashes={sha(members[c]) for c in present}
  42                 assert len(hashes)==1,(chain,name,target,'Ambiguous checksum target')
  43                 actual=next(iter(hashes)); assert actual==m[1].lower(),(chain,name,target,'Failed checksum')
  44                 checks.append({'archive':chain,'manifest':name,'target':target,'resolved_target':present[0],
  45                                'sha256':actual,'passed':True})
  46 for s in meta:
  47     f=BASE/'sources'/s['filename']; data=f.read_bytes()
  48     assert len(data)==s['bytes']
  49     archives.append({**s,'sha256':sha(data),'identical_to_prior_source_content':sha(data) in known})
  50     walk(data,f.name,BASE/'extracted'/f.stem)
  51 new={}
  52 for r in rows:
  53     if r['new_to_combined_prior']: new.setdefault(r['sha256'],r)
  54 summary={'top_level_archives':len(meta),'recursive_member_occurrences':len(rows),
  55          'unique_member_contents':len({r['sha256'] for r in rows}),'new_unique_member_contents':len(new),
  56          'embedded_manifest_checks':len(checks),'failed_manifest_checks':0,
  57          'new_text_data_contents':sum(not r['member'].endswith(('.zip','.pyc')) for r in new.values()),
  58          'new_nested_archives':sum(r['member'].endswith('.zip') for r in new.values()),
  59          'new_bytecode_contents':sum(r['member'].endswith('.pyc') for r in new.values()),
  60          'all_extracted_bytes_match':True,'uploaded_programs_executed':False}
  61 for n,v in [('SOURCE_ARCHIVES.json',archives),('SOURCE_MEMBER_INVENTORY.json',rows),
  62             ('NEW_UNIQUE_CONTENT.json',list(new.values())),('SOURCE_MANIFEST_VERIFICATION.json',checks),
  63             ('PROVENANCE_SUMMARY.json',summary)]: save_json(n,v)
  64 print(json.dumps(summary,indent=2))
