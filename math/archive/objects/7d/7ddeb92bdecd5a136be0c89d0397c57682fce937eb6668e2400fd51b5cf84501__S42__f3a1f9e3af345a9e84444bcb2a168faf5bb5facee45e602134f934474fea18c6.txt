  16 def qdata(N,p,w):

  46 def state_step(state,bit,W,p):

  48  for j,v in enumerate(vals):

  52 def slot_budget(N):
  53  n=N.bit_length()-1;sizes=[]
  54  for r in range(N):
  55   if r==0:sizes.append(2*n);continue
  56   t=(r&-r).bit_length()-1;v=r>>t;h=0

  62 def two_run(x,w,p):

  64  out=[[0]*s for s in slot_budget(N)];series=[[0]*min(s,n+1)for s in slot_budget(N)];v=(1,0,x[:]);visits=0
  65  def put(rows,j,k,value):assert k<len(rows[j]);rows[j][k]=(rows[j][k]+value)%p
  66  for a in range(n+1):
  67   mass=sum(v[2])%p;y=v
  68   for b in range(n):
  69    stride,c,vals=y;visits+=len(vals);assert len(vals)==max(N//2**(a+b),1)
  70    for j,value in enumerate(vals):

  72     if a<n:put(out,node,a+b+1,-lam*value)

  74    put(out,0,a+b,-mass)
  75    if a<n:put(out,0,a+b+1,lam*mass)

  81  aa=[x[:]]

  85   aa.append(v)
  86  expected=[[0]*(2*n)for _ in range(N)];truncated=[[0]*(n+1)for _ in range(N)]
  87  for a in range(n+1):

  89   vals=numerator;orig=aa[a];mass=sum(vals)%p;omass=sum(orig)%p
  90   for b in range(n):

  92    if a+b<=n:
  93     for i in range(N):truncated[i][a+b]=(truncated[i][a+b]+orig[i]-omass*int(i==0))%p
  94    vals=[sum(vals[j]for j in range(N)if 2*j%N==i)%p for i in range(N)]
  95    orig=[sum(orig[j]for j in range(N)if 2*j%N==i)%p for i in range(N)]
  96  for i in range(N):
  97   assert out[i]+[0]*(2*n-len(out[i]))==expected[i]
  98   assert series[i]+[0]*(n+1-len(series[i]))==truncated[i]
  99   for d in range(n+1):assert expected[i][d]==(truncated[i][d]-(lam*truncated[i][d-1]if d else 0))%p
 100  h=[rng.randrange(p)for _ in x];contract=[sum(h[i]*expected[i][d]for i in range(N))%p for d in range(2*n)]
 101  assert contract==[sum(h[i]*(out[i][d]if d<len(out[i])else 0)for i in range(N))%p for d in range(2*n)]
 102  return out,series,{'N':N,'p':p,'state_visits':visits,'full_numerator_slots':sum(map(len,out)),'series_slots':sum(map(len,series)),'degree_bound':2*n-1}
 103 def haar(v,w,N,z,p):
 104  if len(v)==1:return [(2*z-1)*pow(1-z,-1,p)*v[0]%p]
 105  h=len(v)//2;child=haar([(v[j]+v[j+h])%p for j in range(h)],w*w%p,N//2,z,p);out=[0]*len(v)
 106  for j in range(h):out[2*j]=z*child[j]%p;out[2*j+1]=pow(w,2*j+1,p)*(v[j]-v[j+h])%p
 107  return out
