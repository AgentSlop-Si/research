  22 def eye(n: int) -> Matrix:
  23     return [[int(i == j) for j in range(n)] for i in range(n)]
  24 
  25 
  26 def mm(a: Matrix, b: Matrix, p: int) -> Matrix:
  27     columns = list(zip(*b))
  28     return [[sum(u * v for u, v in zip(row, col)) % p for col in columns] for row in a]
  29 
  30 
  31 def mv(a: Matrix, x: Sequence[int], p: int) -> list[int]:
  32     return [sum(u * v for u, v in zip(row, x)) % p for row in a]
  33 
  34 
  35 def add(a: Matrix, b: Matrix, p: int) -> Matrix:
  36     return [[(u + v) % p for u, v in zip(row, other)] for row, other in zip(a, b)]
  37 
  38 
  39 def scale(a: Matrix, c: int, p: int) -> Matrix:
  40     return [[c * v % p for v in row] for row in a]
  41 
  42 
  43 def inverse(a: Matrix, p: int) -> Matrix:
  44     n = len(a)
  45     aug = [row[:] + ident for row, ident in zip(a, eye(n))]
  46     for j in range(n):
  47         pivot = next((r for r in range(j, n) if aug[r][j] % p), None)
  48         if pivot is None:
  49             raise ValueError("Singular test matrix")
  50         aug[j], aug[pivot] = aug[pivot], aug[j]
  51         inv = pow(aug[j][j], -1, p)
  52         aug[j] = [v * inv % p for v in aug[j]]
  53         for r in range(n):
  54             if r == j:
  55                 continue
  56             alpha = aug[r][j]
  57             if alpha:
  58                 aug[r] = [(u - alpha * v) % p for u, v in zip(aug[r], aug[j])]
  59     return [row[n:] for row in aug]
  60

  62 def determinant(a: Matrix, p: int) -> int:
  63     a = [row[:] for row in a]
  64     result = 1
  65     for j in range(len(a)):
  66         pivot = next((r for r in range(j, len(a)) if a[r][j] % p), None)
  67         if pivot is None:
  68             return 0
  69         if pivot != j:
  70             a[j], a[pivot] = a[pivot], a[j]
  71             result = -result % p
  72         v = a[j][j]
  73         result = result * v % p
  74         inv = pow(v, -1, p)
  75         for r in range(j + 1, len(a)):
  76             q = a[r][j] * inv % p
  77             for c in range(j + 1, len(a)):
  78                 a[r][c] = (a[r][c] - q * a[j][c]) % p
  79     return result
  80

  82 def q_matrix(N: int, w: int, p: int) -> Matrix:
  83     # Construct from graph edges, independently of the butterfly apply() method.
  84     a = [[0] * N for _ in range(N)]
  85     for s in range(N):
  86         a[2 * s % N][s] = 1
  87         a[(2 * s + 1) % N][s] = pow(w, s, p)
  88     return a
  89 
  90 
  91 def direct_dft(x: Sequence[int], w: int, p: int) -> list[int]:
  92     return [sum(value * pow(w, k * j, p) for j, value in enumerate(x)) % p
  93             for k in range(len(x))]
  94 
  95 
  96 def radix2_dft(x: Sequence[int], w: int, p: int) -> list[int]:
  97     # A separate ordinary even/odd recursive algorithm for larger exact controls.
  98     N = len(x)
  99     if N == 1:
 100         return [x[0] % p]
 101     even = radix2_dft(x[::2], w * w % p, p)
 102     odd = radix2_dft(x[1::2], w * w % p, p)
 103     out = [0] * N
 104     phase = 1
 105     for k in range(N // 2):
 106         t = phase * odd[k] % p
 107         out[k] = (even[k] + t) % p
 108         out[k + N // 2] = (even[k] - t) % p
 109         phase = phase * w % p
 110     return out

 113 def matrix_from_map(length: int, func: Callable[[list[int]], list[int]]) -> Matrix:
 114     cols = [func([int(i == j) for i in range(length)]) for j in range(length)]
 115     return [list(row) for row in zip(*cols)]

 118 def independent_phi(r: int) -> int:
 119     bits = bin(r)[2:]
 120     return sum(int(bits[t]) * (int(bits[:t], 2) if t else 0) for t in range(len(bits)))

 123 def bitreverse(r: int, bits: int) -> int:
 124     return int(f'{r:0{bits}b}'[::-1], 2) if bits else 0

 127 def test_fourier() -> dict:
 128     rng = random.Random(2026101001)
 129     basis_vectors = fresh_vectors = entries = 0
 130     records = []
 131     for field in FIELDS:
 132         for N in (2, 4, 8, 16, 32, 64, 128):
 133             plan = FixedGraph(N, field)
 134             count = Counts()
 135             for j in range(N):
 136                 x = [int(i == j) for i in range(N)]
 137                 expected = [pow(plan.w, k * j, field.p) for k in range(N)]
 138                 assert plan.fourier(x) == expected
 139                 basis_vectors += 1
 140                 entries += N
 141             for x in ([1] * N, [rng.randrange(field.p) for _ in range(N)],
 142                       [1 if i % 2 else field.p - 1 for i in range(N)]):
 143                 assert plan.fourier(x, count) == direct_dft(x, plan.w, field.p)
 144                 fresh_vectors += 1
 145                 entries += N
 146             n = plan.n
 147             expected_mul = n * (N // 2 - 1) + N - n - 1
 148             assert count.additions == 3 * N * n
 149             assert count.multiplications == 3 * expected_mul
 150             assert all(plan.phi[j] == independent_phi(j) for j in range(N))
 151             assert all(0 <= v < N for v in plan.phi)
 152             assert sum(v == 0 for v in plan.phi) == n + 1
 153             records.append({'p': field.p, 'N': N, 'online_additions': N * n,
 154                             'online_multiplications': expected_mul,
 155                             'root_table_multiplications': plan.root_table_multiplications})
 156         for N in (256, 1024, 4096):
 157             plan = FixedGraph(N, field)
 158             for _ in range(2):
 159                 x = [rng.randrange(field.p) for _ in range(N)]
 160                 assert plan.fourier(x) == radix2_dft(x, plan.w, field.p)
 161                 fresh_vectors += 1
 162                 entries += N
 163     return {'basis_vectors': basis_vectors, 'fresh_vectors': fresh_vectors,
 164             'output_entries': entries, 'operation_counts': records,
 165             'warning': 'The actual Fourier executor remains Theta(N log N).'}

 168 def test_partial_powers() -> dict:
 169     cases = nonzeros = 0
 170     for field in FIELDS:
 171         p = field.p
 172         for N in (2, 4, 8, 16, 32, 64):
 173             plan = FixedGraph(N, field)
 174             Q = q_matrix(N, plan.w, p)
 175             power = eye(N)
 176             for t in range(plan.n + 1):
 177                 L, m = 1 << t, N >> t
 178                 formula = [[0] * N for _ in range(N)]
 179                 for r in range(N):
 180                     prefix, v = divmod(r, L)
 181                     br = bitreverse(v, t)
 182                     for a in range(L):
 183                         s = a * m + prefix
 184                         formula[r][s] = pow(plan.w, s * br + independent_phi(v), p)
 185                 assert power == formula, (p, N, t)
 186                 assert all(sum(value != 0 for value in row) == L for row in power)
 187                 assert sum(power[r][0] != 0 for r in range(N)) == L
 188                 assert [r for r in range(N) if power[r][0]] == list(range(L))
 189                 nonzeros += N * L
 190                 cases += 1
 191                 power = mm(Q, power, p)
 192     return {'matrix_identities': cases, 'nonzero_entries': nonzeros,
 193             'consequence': 'Q^t has exactly N*2^t nonzeros for 0<=t<=log2(N).'}

 196 def test_branches_and_haar() -> dict:
 197     rng = random.Random(2026101002)
 198     branch_cases = haar_cases = schur_cases = 0
 199     for field in FIELDS:
 200         p = field.p
 201         for N in (2, 4, 8, 16, 32, 64, 128, 256):
 202             plan = FixedGraph(N, field)
 203             z = 7
 204             for _ in range(3):
 205                 x = [rng.randrange(p) for _ in range(N)]
 206                 assert plan.apply(x) == [(a + b) % p for a, b in zip(
 207                     plan.branch_apply(x, 0), plan.branch_apply(x, 1))]
 208                 for bit in (0, 1):
 209                     y = plan.branch_solve(x, bit, z)
 210                     a = plan.branch_apply(y, bit)
 211                     assert [(u - z * v) % p for u, v in zip(y, a)] == x
 212                     branch_cases += 1
 213                 v = [rng.randrange(p) for _ in range(N // 2)]
 214                 injected = [0] * N
 215                 injected[1::2] = v
 216                 solved = plan.branch_solve(injected, 0, z)
 217                 expected = [plan.roots[r] * (solved[r] - solved[r + N // 2]) % p
 218                             for r in range(N // 2)]
 219                 actual = plan.haar_apply(v, z)
 220                 assert actual == expected
 221                 assert plan.haar_inverse(actual, z) == v
 222                 haar_cases += 1
 223             if N <= 32:
 224                 Q = q_matrix(N, plan.w, p)
 225                 G0 = matrix_from_map(N, lambda v: plan.branch_solve(v, 0, z))
 226                 m = N // 2
 227                 H = matrix_from_map(m, lambda v: plan.haar_apply(v, z))
 228                 J = [[int(r == 2 * s + 1) for s in range(m)] for r in range(N)]
 229                 C = [[(plan.roots[r] * (int(j == r) - int(j == r + m))) % p
 230                       for j in range(N)] for r in range(m)]
 231                 inv_small = inverse(add(eye(m), scale(H, -z, p), p), p)
 232                 correction = mm(mm(mm(mm(G0, J, p), inv_small, p), C, p), G0, p)
 233                 rhs = add(G0, scale(correction, z, p), p)
 234                 lhs = inverse(add(eye(N), scale(Q, -z, p), p), p)
 235                 assert lhs == rhs
 236                 assert sum(v != 0 for row in H for v in row) == m * plan.n
 237                 schur_cases += 1
 238     return {'branch_vector_solves': branch_cases, 'haar_apply_inverse_pairs': haar_cases,
 239             'complete_schur_matrix_checks': schur_cases,
 240             'warning': 'H and H^-1 are cheap. No cheap inverse of I-zH is supplied.'}

 243 def test_traces_and_determinants() -> dict:
 244     trace_cases = det_jets = periodic_states = 0
 245     records = []
 246     for field in FIELDS:
 247         p = field.p
 248         for N in (2, 4, 8, 16, 32, 64):
 249             plan = FixedGraph(N, field)
 250             traces, states = plan.trace_prefix(extra=2)
 251             assert states == 8 * N - 2
 252             Q = q_matrix(N, plan.w, p)
 253             power = eye(N)
 254             for k, value in enumerate(traces, 1):
 255                 power = mm(Q, power, p)
 256                 assert sum(power[i][i] for i in range(N)) % p == value
 257                 trace_cases += 1
 258             periodic_states += states
 259             # Independent determinant-polynomial interpolation, not Newton identities.
 260             if N <= 16:
 261                 zs = list(range(N + 1))
 262                 values = [determinant(add(eye(N), scale(Q, -z, p), p), p) for z in zs]
 263                 V = [[pow(z, k, p) for k in range(N + 1)] for z in zs]
 264                 coefficients = mv(inverse(V, p), values, p)
 265                 coefficients += [0] * 4
 266                 assert coefficients[:plan.n + 3] == plan.determinant_jet(extra=2)
 267                 det_jets += 1
 268             records.append({'p': p, 'N': N, 'traces': traces,
 269                             'periodic_states': states, 'determinant_jet': plan.determinant_jet(extra=2)})
 270     return {'exact_trace_checks': trace_cases, 'independent_determinant_jets': det_jets,
 271             'enumerated_periodic_states': periodic_states, 'records': records}

 274 def dual_trace_power(W0: Matrix, W1: Matrix, exponent: int, p: int) -> tuple[int, int]:
 275     # Compute in F[epsilon]/epsilon^2 by repeated dense products.
 276     n = len(W0)
 277     base, tangent = eye(n), [[0] * n for _ in range(n)]
 278     for _ in range(exponent):
 279         tangent = add(mm(tangent, W0, p), mm(base, W1, p), p)
 280         base = mm(base, W0, p)
 281     return sum(base[i][i] for i in range(n)) % p, sum(tangent[i][i] for i in range(n)) % p

 284 def test_marked_cycle_gradient() -> dict:
 285     rng = random.Random(2026101003)
 286     input_cases = coordinates = 0
 287     scalar_independence = 0
 288     for field in FIELDS:
 289         p = field.p
 290         for N in (2, 4, 8, 16):
 291             plan = FixedGraph(N, field)
 292             Q = q_matrix(N, plan.w, p)
 293             L = plan.n + 2
 294             old_scalar = None
 295             for _ in range(2):
 296                 x = [rng.randrange(p) for _ in range(N)]
 297                 W0 = [row[:] + [x[i]] for i, row in enumerate(Q)] + [[0] * (N + 1)]
 298                 expected_fourier = direct_dft(x, plan.w, p)
 299                 expected_graph = [0] * N
 300                 for k, r in enumerate(plan.reversal):
 301                     expected_graph[r] = plan.roots[plan.phi[r]] * expected_fourier[k] % p
 302                 gradient = []
 303                 for j in range(N):
 304                     W1 = [[0] * (N + 1) for _ in range(N + 1)]
 305                     W1[N][j] = 1
 306                     scalar, derivative = dual_trace_power(W0, W1, L, p)
 307                     gradient.append(derivative * pow(L, -1, p) % p)
 308                 assert gradient == expected_graph
 309                 if old_scalar is not None:
 310                     assert scalar == old_scalar  # value at t=0 forgets x, gradient does not.
 311                     scalar_independence += 1
 312                 old_scalar = scalar
 313                 input_cases += 1
 314                 coordinates += N
 315     return {'arbitrary_input_cases': input_cases, 'dual_number_derivative_coordinates': coordinates,
 316             'input_independent_scalar_controls': scalar_independence,
 317             'warning': 'Dense dual arithmetic is an identity oracle, NOT the conjectured fast trace circuit.'}

 321 def test_bordered_determinant() -> dict:
 322     rng = random.Random(2026101005)
 323     cases = affine_checks = 0
 324     for field in FIELDS:
 325         p = field.p
 326         for N in (2, 4, 8, 16):
 327             plan = FixedGraph(N, field)
 328             Q = q_matrix(N, plan.w, p)
 329             # Full polynomial interpolation is an independent small-size oracle.
 330             V = [[pow(z, k, p) for k in range(N + 2)] for z in range(N + 2)]
 331             Vi = inverse(V, p)
 332             d = plan.determinant_jet(extra=2)
 333             for _ in range(3):
 334                 x = [rng.randrange(p) for _ in range(N)]
 335                 t = [rng.randrange(p) for _ in range(N)]
 336                 W = [row[:] + [x[i]] for i, row in enumerate(Q)] + [t[:] + [0]]
 337                 vals = [determinant(add(eye(N + 1), scale(W, -z, p), p), p)
 338                         for z in range(N + 2)]
 339                 e = mv(Vi, vals, p) + [0] * 4
 340                 # Truncated formal quotient E/D has constant coefficient 1.
 341                 quotient = []
 342                 for j in range(plan.n + 3):
 343                     quotient.append((e[j] - sum(d[k] * quotient[j-k]
 344                                               for k in range(1, j+1))) % p)
 345                 fx = direct_dft(x, plan.w, p)
 346                 qx = [0] * N
 347                 for k, r in enumerate(plan.reversal):
 348                     qx[r] = plan.roots[plan.phi[r]] * fx[k] % p
 349                 expected = sum(a*b for a,b in zip(t,qx)) % p
 350                 assert -quotient[plan.n + 2] % p == expected
 351                 z = 7
 352                 A = add(eye(N), scale(Q, -z, p), p)
 353                 D = determinant(A, p)
 354                 E = determinant(add(eye(N + 1), scale(W, -z, p), p), p)
 355                 twot = [2*v % p for v in t]
 356                 W2 = [row[:] + [x[i]] for i, row in enumerate(Q)] + [twot + [0]]
 357                 E2 = determinant(add(eye(N + 1), scale(W2, -z, p), p), p)
 358                 assert (E2-D) % p == 2*(E-D) % p
 359                 cases += 1
 360                 affine_checks += 1
 361     return {'independent_polynomial_quotient_checks': cases,
 362             'exact_affinity_in_markers_checks': affine_checks,
 363             'warning': 'Dense determinant interpolation is only an independent test oracle.'}

 366 def test_simple_paths() -> dict:
 367     records = []
 368     for n in range(1, 8):
 369         N = 1 << n
 370         simple = 0
 371         for s in range(N):
 372             for emitted in range(N):
 373                 current = s
 374                 visited = {s}
 375                 ok = True
 376                 for t in range(n - 1, -1, -1):
 377                     current = (2 * current + ((emitted >> t) & 1)) % N
 378                     if current in visited:
 379                         ok = False
 380                     visited.add(current)
 381                 simple += ok
 382         lower = max(0, N * N - math.comb(n + 1, 2) * N)
 383         assert simple >= lower
 384         records.append({'N': N, 'all_input_output_paths': N * N,
 385                         'internally_simple_paths': simple, 'proved_lower_bound': lower})
 386     return {'records': records, 'consequence': 'Even simple marked paths remain quadratically numerous.'}

 389 def test_negative_controls() -> dict:
 390     rng = random.Random(2026101004)
 391     missing_phase = alias_rejections = singular_rejections = 0
 392     for field in FIELDS:
 393         p = field.p
 394         for N in (4, 8, 16, 32):
 395             plan = FixedGraph(N, field)
 396             x = [rng.randrange(1, p) for _ in range(N)]
 397             y = plan.power(x, plan.n)
 398             wrong = [y[r] for r in plan.reversal]
 399             assert wrong != direct_dft(x, plan.w, p)
 400             missing_phase += 1
 401         for N in (4, 8):
 402             plan = FixedGraph(N, field)
 403             Q = q_matrix(N, plan.w, p)
 404             b = 1 << (plan.n.bit_length())  # b > log2(N), b divides p-1 in our fields.
 405             xi, a = field.root(b), 7
 406             total = [[0] * N for _ in range(N)]
 407             for j in range(b):
 408                 z = a * pow(xi, j, p) % p
 409                 green = inverse(add(eye(N), scale(Q, -z, p), p), p)
 410                 total = add(total, scale(green, pow(xi, -j * plan.n, p), p), p)
 411             total = scale(total, pow(b, -1, p), p)
 412             Qn, Qb = eye(N), eye(N)
 413             for _ in range(plan.n):
 414                 Qn = mm(Q, Qn, p)
 415             for _ in range(b):
 416                 Qb = mm(Q, Qb, p)
 417             rhs = scale(mm(Qn, inverse(add(eye(N), scale(Qb, -pow(a, b, p), p), p), p), p),
 418                         pow(a, plan.n, p), p)
 419             assert total == rhs
 420             assert total != scale(Qn, pow(a, plan.n, p), p)
 421             alias_rejections += 1
 422         plan = FixedGraph(8, field)
 423         for f in (lambda: plan.branch_solve([1] * 8, 0, 1),
 424                   lambda: plan.haar_inverse([1] * 4, field.inverse(2))):
 425             try:
 426                 f()
 427                 raise AssertionError('A singular control was not rejected.')
 428             except ZeroDivisionError:
 429                 singular_rejections += 1
 430     return {'missing_phase_rejections': missing_phase, 'resolvent_alias_rejections': alias_rejections,
 431             'singular_parameter_rejections': singular_rejections}

 434 def main() -> None:
 435     parser = argparse.ArgumentParser(description=__doc__)
 436     parser.add_argument('--output', type=Path, default=Path('RESULTS.json'))
 437     args = parser.parse_args()
 438     tests = [test_fourier, test_partial_powers, test_branches_and_haar,
 439              test_traces_and_determinants, test_marked_cycle_gradient,
 440              test_bordered_determinant, test_simple_paths, test_negative_controls]
 441     results = {'status': 'Exact algebra verified; NO sub-N-log-N Fourier algorithm constructed.',
 442                'fields': [{'p': f.p, 'primitive_generator': f.generator} for f in FIELDS],
 443                'tests': {}}
 444     for test in tests:
 445         print(test.__name__, flush=True)
 446         results['tests'][test.__name__] = test()
 447     results['passed_groups'] = len(tests)
 448     args.output.write_text(json.dumps(results, indent=2) + '\n', encoding='utf-8')
 449     print(f'PASS: {len(tests)}/{len(tests)} groups. Results: {args.output}')
