36:     def test_formula_matches_bruteforce_covariance(self) -> None:
37:         # Tiny independent enumeration over every odd a and every beta.
38:         n, m, output = 8, 2, 3
39:         brute = np.zeros((n, n), dtype=np.complex128)
40:         exact_basis_output = np.exp(-2j * np.pi * output * np.arange(n) / n) / math.sqrt(n)
41:         count = 0
42:         for a in range(1, n, 2):
43:             for beta in range(n):
44:                 state = ChirpState(a, beta)
45:                 coeff = np.empty(n, dtype=np.complex128)
46:                 for j in range(n):
47:                     e = np.zeros(n, dtype=np.complex128)
48:                     e[j] = 1.0
49:                     coeff[j] = direct_row_estimate(e, m, state)[output] - exact_basis_output[j]
50:                 brute += np.outer(np.conj(coeff), coeff)
51:                 count += 1
52:         brute /= count
53:         formula = covariance_matrix_formula(n, m, output)
54:         self.assertLess(float(np.max(np.abs(brute - formula))), 2e-12)

78:         rng = np.random.default_rng(2)

80:         x = rng.normal(size=n) + 1j * rng.normal(size=n)
81:         exact = exact_unitary_dft(x)
