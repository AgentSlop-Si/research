   7 @lru_cache(None)
   8 def mul(a,b,d):
   9     if d == 1: return a & b
  10     h=d//2; mask=(1<<h)-1
  11     a0,a1=a&mask,a>>h; b0,b1=b&mask,b>>h
  12     t=mul(a1,b1,h)
  13     lo=mul(a0,b0,h)^mul(t,1<<(h-1),h)
  14     hi=mul(a0,b1,h)^mul(a1,b0,h)^t
  15     return lo^(hi<<h)
  16 
  17 def power(a,n,d):
  18     z=1
  19     while n:
  20         if n&1:z=mul(z,a,d)
  21         a=mul(a,a,d);n//=2
  22     return z

  24 def s_eval(i,a,d):
  25     for _ in range(i):a=mul(a,a,d)^a
  26     return a

  28 @lru_cache(None)
  29 def s_poly(i):
  30     if i==0:return 2
  31     f=s_poly(i-1); z=f
  32     for j in range(f.bit_length()):
  33         if f>>j&1:z^=1<<(2*j)
  34     return z

  36 def poly_mul(a,b):
  37     z=0
  38     while b:
  39         if b&1:z^=a
  40         a<<=1;b>>=1
  41     return z

  43 @lru_cache(None)
  44 def x_poly(j):
  45     z=1;i=0
  46     while j:
  47         if j&1:z=poly_mul(z,s_poly(i))

  51 def slow_basis(bits,m):
  52     out=0
  53     for j in range((1<<m)-1,-1,-1):
  54         if bits>>j&1:
  55             bits^=x_poly(j);out|=1<<j
  56     assert bits==0
  57     return out

  59 def to_y(f,m,k,h):
  60     # Coefficients are h-bit vectors. y=x^(2^k)+x.
  61     if m==k:return f
  62     H=1<<(m-1);R=1<<(m-1-k)
  63     mask=(1<<(H*h))-1
  64     high=f>>(H*h); low=f&mask
  65     q=high^(high>>((H-R)*h))
  66     rem=low^((q<<(R*h))&mask)
  67     return to_y(rem,m-1,k,h)^(to_y(q,m-1,k,h)<<(H*h))
  68 
  69 def packed_basis(f,m,h=1):
  70     if m<=1:return f
  71     k=1<<((m-1).bit_length()-1)
  72     D=1<<k
  73     fy=to_y(f,m,k,h)
  74     upper=packed_basis(fy,m-k,h*D)
  75     out=0; width=h*D; mask=(1<<width)-1
  76     for i in range(1<<(m-k)):
  77         out^=packed_basis((upper>>(i*width))&mask,k,h)<<(i*width)
  78     return out

  80 def direct_eval(f,a,d):
  81     y=0
  82     for j in range(f.bit_length()-1,-1,-1):
  83         y=mul(y,a,d)^(f>>j&1)
  84     return y

  86 def fafft(coeff,k,l,alpha,d,stats):
  87     if k==0:return [(alpha,coeff[0])]
  88     half=1<<(k-1); c=s_eval(k-1,alpha,d)
  89     stats['nodes']+=1;stats['coefficient_units']+=half;stats['twiddle_steps']+=k-1
  90     q0=[coeff[i]^mul(c,coeff[i+half],d) for i in range(half)]
  91     if l and l&(l-1)==0:
  92         return fafft(q0,k-1,l+1,alpha,d,stats)
  93     q1=[q0[i]^coeff[i+half] for i in range(half)]
  94     return fafft(q0,k-1,l if l==0 else l+1,alpha,d,stats)+fafft(q1,k-1,l+1,alpha^(1<<(k-1)),d,stats)

  96 def full_evaluation(f,d):
  97     bits=packed_basis(f,d)
  98     coeff=[bits>>j&1 for j in range(1<<d)]
  99     stats=dict(nodes=0,coefficient_units=0,twiddle_steps=0)
 100     reps=fafft(coeff,d,0,0,d,stats);out={}
 101     for a,y in reps:
 102         start=a
 103         while True:
 104             assert a not in out
 105             out[a]=y
 106             a=mul(a,a,d);y=mul(y,y,d)
 107             if a==start:break
 108     assert len(out)==1<<d
 109     stats['representatives']=len(reps)
 110     return out,stats

 112 def primitive(d):
 113     n=(1<<d)-1
 114     for a in range(1,1<<d):
 115         x=1;seen=set()
 116         for _ in range(n):seen.add(x);x=mul(x,a,d)
 117         if len(seen)==n:return a

 120 def test_chunk_tables(d,chunks):
 121     h=d//chunks; mask=(1<<h)-1
 122     tables={}
 123     for i in range(chunks):
 124         for j in range(chunks):
 125             tables[i,j]=[mul(a<<(i*h),b<<(j*h),d) for a in range(1<<h) for b in range(1<<h)]
 126     for a in range(1<<d):
 127         for b in range(1<<d):
 128             z=0
 129             for i in range(chunks):
 130                 for j in range(chunks):z^=tables[i,j][(((a>>(i*h))&mask)<<h)|((b>>(j*h))&mask)]
 131             assert z==mul(a,b,d)
 132     return (1<<d)**2

 134 @lru_cache(None)
 135 def recurrence(k,l):
 136     if not k:return (0,0,1)
 137     children=[recurrence(k-1,l if l==0 else l+1)]
 138     if l==0 or l&(l-1):children.append(recurrence(k-1,l+1))
 139     return ((1<<(k-1))+sum(z[0] for z in children),k-1+sum(z[1] for z in children),sum(z[2] for z in children))
