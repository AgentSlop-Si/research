  26 def invert(c, a):
  27     """Counted, pivoted Gauss-Jordan; even zero-factor eliminations are charged."""
  28     d = len(a)
  29     work = [row[:] + [int(i == j) for j in range(d)]
  30             for i, row in enumerate(a)]
  31     swaps = 0
  32     for k in range(d):
  33         pivot = next((i for i in range(k, d) if work[i][k]), None)
  34         if pivot is None:
  35             raise ValueError('singular port closure')

  37             work[pivot], work[k] = work[k], work[pivot]

  17 def rank_one_add(p,functional,target,vector):
  18     """y += vector*(sum_j functional[j]*register[j]); all sources restored.
  19 
  20     Source registers and target registers must be disjoint. No clean scratch.
  21     A source pivot temporarily stores the linear functional, then is uncomputed.
  22     """
  23     functional={i:a%p for i,a in functional.items() if a%p}
  24     vector=[a%p for a in vector]
  25     if not functional or not any(vector):
  26         return [],{'functional_nonzeros':len(functional),'output_nonzeros':sum(bool(a) for a in vector),
  27                    'scalar_instructions':0,'source_pivot_inverse':0}
  28     if any(target<=i<target+len(vector) for i in functional):
  29         raise ValueError('source and target registers must be disjoint')
  30     pivot=min(functional)
  31     before=[('scale',pivot,functional[pivot])]
  32     # Order is immaterial: these source-to-pivot shears commute. Preserve
  33     # insertion order so preparing a rank-one ADD is linear, without sorting.
  34     before.extend(('shear',pivot,i,a) for i,a in functional.items() if i!=pivot)
  35     middle=[('shear',target+i,pivot,a) for i,a in enumerate(vector) if a]
  36     word=before+middle+inverse(before,p)
  37     m,k=len(functional),sum(bool(a) for a in vector)

  39                  'scalar_instructions':6*m+3*k-4,'source_pivot_inverse':1}
