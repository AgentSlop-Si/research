   1 """Exactly certify degree<=d Boolean spectral completion failures.
   2 
   3 Fourier-character grading via cyclic-shift averaging ensures it suffices to
   4 search monomials whose known frequency indices sum to the target modulo N.
   5 A full (M+1)x(M+1) minor of these M columns + target, nonzero mod a split
   6 prime, implies the same over characteristic-zero complex cyclotomic fields.
   7 """
   8 from __future__ import annotations
   9 from itertools import combinations, combinations_with_replacement
  10 from spectral_boolean import primitive_root
  11 
  12 
  13 def graded_monomials(n,known,target,d):
  14     out=[]
  15     for j in range(1,d+1):
  16         for idx in combinations_with_replacement(known,j):
  17             if sum(idx)%n==target:out.append(idx)
  18     return out
  19 
  20 
  21 def degree_d_inputs(n,d):
  22     for size in range(d+1):
  23         yield from combinations(range(n),size)
  24 
  25 
  26 def evaluations_for_support(n,p,powers,known,mons,target,support):
  27     sources={s:sum(powers[s][j] for j in support)%p for s in known}
  28     vals=[]
  29     for mon in mons:
  30         value=1
  31         for s in mon:value=value*sources[s]%p
  32         vals.append(value)
  33     vals.append(sum(powers[target][j] for j in support)%p)
  34     return vals
  35 
  36 
  37 def rank_certificate(n:int,p:int,target:int,degree:int=3,known=None,max_rows=None):
  38     """Find M+1 binary samples giving a nonzero augmented determinant.
  39 
  40     If found, polynomial recovery in degree <=d is impossible over C as well.
  41     If not found, absence of a certificate is NOT existence of completion.
  42     """
  43     if known is None:known=tuple(s for s in range(n) if s!=target)
  44     known=tuple(known)
  45     w=primitive_root(n,p)
  46     powers=[[pow(w,(s*j)%n,p) for j in range(n)] for s in range(n)]
  47     mons=graded_monomials(n,known,target,degree)
  48     m=len(mons)
  49     pivots={}
  50     selected=[]
  51     rows_seen=0
  52     for subset in degree_d_inputs(n,degree):
  53         values=evaluations_for_support(n,p,powers,known,mons,target,subset)
  54         v=values[:]
  55         for pivot,basis in pivots.items():
  56             c=v[pivot]
  57             if c:
  58                 v=[(a-c*b)%p for a,b in zip(v,basis)]
  59         lead=next((j for j,t in enumerate(v) if t),None)
  60         rows_seen+=1
  61         if lead is not None:
  62             inv=pow(v[lead],-1,p)
  63             pivots[lead]=[(t*inv)%p for t in v]
  64             selected.append(subset)
  65             if len(selected)==m+1:
  66                 return {"status":"no_completion_degree_at_most_d_over_C",
  67                         "n":n,"prime":p,"root":w,"degree":degree,
  68                         "target":target,"known":known,"monomials":mons,
  69                         "certificate_supports":selected,"rows_examined":rows_seen,
  70                         "rank_aug":m+1,"num_columns":m,
  71                         "certificate_check":"determinant of augmented square evaluation matrix is nonzero mod p"}
  72         if max_rows is not None and rows_seen>=max_rows:break
  73     return {"status":"inconclusive","n":n,"prime":p,"degree":degree,
  74             "target":target,"known":known,"monomials":mons,
  75             "rows_examined":rows_seen,"rank_aug":len(selected),
  76             "num_columns":m,"certificate_supports":selected}
  77 
  78 
  79 def determinant_mod(matrix,p):
  80     n=len(matrix)
  81     assert all(len(r)==n for r in matrix)
  82     a=[r[:] for r in matrix]
  83     det=1
  84     for c in range(n):
  85         row=next((i for i in range(c,n) if a[i][c]),None)
  86         if row is None:return 0
  87         if row!=c:
  88             a[c],a[row]=a[row],a[c]
  89             det=-det
  90         pivot=a[c][c]%p
  91         det=det*pivot%p
  92         inv=pow(pivot,-1,p)
  93         for i in range(c+1,n):
  94             mult=a[i][c]*inv%p
  95             if mult:
  96                 for j in range(c,n):
  97                     a[i][j]=(a[i][j]-mult*a[c][j])%p
  98     return det%p
  99

 101 def verify_certificate(result:dict):
 102     if result['status'] != 'no_completion_degree_at_most_d_over_C':return False
 103     n,p,target=result['n'],result['prime'],result['target']
 104     known=tuple(result['known']); mons=[tuple(m) for m in result['monomials']]
 105     supports=[tuple(v) for v in result['certificate_supports']]
 106     if len(supports)!=len(mons)+1:return False
 107     if any(len(s)>result['degree'] or sorted(set(s))!=list(s) for s in supports):return False
 108     if mons!=graded_monomials(n,known,target,result['degree']):return False
 109     w=primitive_root(n,p)
 110     if result.get('root')!=w:return False
 111     powers=[[pow(w,(s*j)%n,p) for j in range(n)] for s in range(n)]
 112     a=[evaluations_for_support(n,p,powers,known,mons,target,s) for s in supports]
 113     return determinant_mod(a,p)!=0
