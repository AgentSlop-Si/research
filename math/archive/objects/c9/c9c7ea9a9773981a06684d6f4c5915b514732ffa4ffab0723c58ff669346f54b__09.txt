  17 def trim(a):
  18  a=list(a)

  22 def dense(p):

  29 def mulpoly(a,b):
  30  c=[0]*(len(a)+len(b)-1)
  31  for i,x in enumerate(a):
  32   if x:
  33    for j,y in enumerate(b):
  34     if y:c[i+j]+=x*y
  35  return trim(c)

  37 def divexact(a,b):
  38  a=trim(a);b=trim(b);q=[0]*max(0,(len(a)-len(b)+1))
  39  while a and len(a)>=len(b):
  40   assert a[-1]%b[-1]==0
  41   z=a[-1]//b[-1];k=len(a)-len(b);q[k]=z
  42   for j,v in enumerate(b):a[k+j]-=z*v
  43   a=trim(a)
  44  assert not a
  45  return trim(q)

  47 def modrem(a,b,p):
  48  a=trim([v%p for v in a]);b=trim([v%p for v in b]);ib=pow(b[-1],-1,p)
  49  while len(a)>=len(b):
  50   z=a[-1]*ib%p;k=len(a)-len(b)
  51   for j,v in enumerate(b):a[k+j]=(a[k+j]-z*v)%p
  52   a=trim(a)
  53  return a

  55 def gcdmod(a,b,p):
  56  a=trim([v%p for v in a]);b=trim([v%p for v in b])
  57  while b:a,b=b,modrem(a,b,p)
  58  return [x*pow(a[-1],-1,p)%p for x in a]

  60 def detinteger(a):
  61  b=[r[:] for r in a];sgn=1;prev=1
  62  for k in range(len(b)-1):
  63   t=next((i for i in range(k,len(b)) if b[i][k]),None)
  64   if t is None:return 0
  65   if t!=k:b[t],b[k]=b[k],b[t];sgn=-sgn
  66   z=b[k][k]
  67   for i in range(k+1,len(b)):
  68    for j in range(k+1,len(b)):
  69     w=b[i][j]*z-b[i][k]*b[k][j];assert w%prev==0;b[i][j]=w//prev
  70    b[i][k]=0
  71   prev=z
  72  return sgn*b[-1][-1]

  75 def monomialdet(a):
  76  # Entry is (integer sign, nonnegative exponent); enumerate every term.
  77  out={};n=len(a)
  78  for s,sg in PERMS[n]:
  79   deg=0;v=sg
  80   for i,j in enumerate(s):v*=a[i][j][0];deg+=a[i][j][1]
  81   out[deg]=out.get(deg,0)+v
  82  return dense([[k,v] for k,v in out.items() if v])

  92 def monomialframe(pos,i,sig):
  93  return [[(sig**(((po[i]-po[k])%64)//32),(po[i]-po[k])%32) for k in range(32)] for po in pos]

  95 def evalframe(pos,i,sig,taus,f):
  96  return [[sg*pow(t,q,f.p)*f.inv(1-sig*pow(t,32,f.p))%f.p for sg,q in row] for row,t in zip(monomialframe(pos,i,sig),taus)]

  98 def makegreens(ps,t,f):
  99  out=[]
 100  for p in ps:
 101   a=f.eye(64)
 102   for j,i in enumerate(p):a[i][j]=(a[i][j]-t)%f.p
 103   g=f.inverse(a);assert f.mul(a,g)==f.eye(64);out.append(g)
 104  return out
