0080 def mersenne(exponent):
0081     require(exponent >= 3 and all(exponent % d for d in range(2, isqrt(exponent)+1)),
0082             'Composite Lucas-Lehmer exponent')
0083     p = 2**exponent - 1
0084     s = 4
0085     for _ in range(exponent - 2):
0086         s = (s*s-2) % p
0087     require(s == 0, 'Lucas-Lehmer failed')
0088     return p

0091 def exactness(h):
0092     require(h in (23, 25), 'Uncertified dimension')
0093     primes = [mersenne(e) for e in (61, 31, 19)]
0094     Z = 3*h-7
0095     B0 = (4*h-2)*Z + 27*(h+1)
0096     D = 3*(h+1)*(h-1)**2
0097     B = 2*(h-1)*B0
0098     bounds = {r: sum(comb(h,j)*factorial(j)*B**j*D**(r-j)
0099                      for j in range(r+1)) for r in (2,3,4)}
0100     require(bounds[2] < primes[0], 'Rank-two minor modulus insufficient')
0101     require(max(bounds[3],bounds[4]) < prod(primes), 'CRT minor modulus insufficient')
0102     require(3*(h+1)*(h-1) < min(primes), 'Denominator may vanish')
0103     return dict(h=h, primes=primes, prime_product=prod(primes),
0104                 single_numerator_bound=B0, common_denominator_bound=D,
0105                 correction_numerator_bound=B, minor_bounds=bounds,
0106                 modulus_gaps={2: primes[0]-bounds[2],
0107                               3: prod(primes)-bounds[3],4: prod(primes)-bounds[4]})

0110 def tree(edges, n):
0111     require(len(edges) == n-1 and len(set(edges)) == n-1, 'Wrong tree edge count')
0112     seen = {0}
0113     while True:
0114         new = seen | {v for u,v in edges if u in seen} | {u for u,v in edges if v in seen}
0115         if new == seen:
0116             break
0117         seen = new
0118     require(len(seen) == n, 'Disconnected restriction graph')

0179 @lru_cache(None)
0180 def logs(x):
0181     x = Q(x)
0182     require(x >= 1, 'Log domain')
0183     k = 0
0184     while x > 2:
0185         x /= 2
0186         k += 1
0187     def small(y):
0188         z = (y-1)/(y+1)
0189         lo = 2*sum((z**(2*j+1)/Q(2*j+1) for j in range(32)), Q())
0190         return lo, lo+2*z**65/(65*(1-z*z))
0191     lo,hi = small(x)
0192     lo2,hi2 = small(Q(2))
0193     lo,hi = lo+k*lo2,hi+k*hi2
0194     scale = 10**30
0195     return Q((lo*scale).numerator//(lo*scale).denominator,scale), \
0196         Q(-(-(hi*scale).numerator//(hi*scale).denominator),scale)

0199 def moment(m,W,rows,saving):
0200     require(type(saving) is Q and 0 < saving < 1, 'Exact positive saving required')
0201     lower,upper = Q(),Q()
0202     enclosure = {}
0203     for t,n in sorted(rows.items()):
0204         require(type(t) is int and type(n) is int and 0<t<m and n>0, 'Invalid child')
0205         lo,hi = logs(Q(m,t))
0206         u,v = saving*lo,saving*hi
0207         require(0 <= u <= v < 1, 'Exponential enclosure domain')
0208         weight = Q(t*n,m*W)
0209         lower += weight*(1+u+u*u/2+u*u*u/6)
0210         upper += weight*(1+v+v*v/(2*(1-v/3)))
0211         enclosure[t] = dict(lower=lo,upper=hi)
0212     return dict(saving=saving,exponent=1-saving,lower=lower,upper=upper,
0213                 strict_gap=1-upper,logarithms=enclosure)

0234                 'Original frame rank mass')

0262                 child_multiplicities=dict(sorted(hist.items())),parts=parts,axes=axes)
