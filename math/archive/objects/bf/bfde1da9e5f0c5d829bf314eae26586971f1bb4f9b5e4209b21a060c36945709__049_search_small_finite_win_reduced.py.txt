import torch, math, random, json, time

torch.set_default_dtype(torch.float64); cdtype=torch.complex128
n=16
a=(1+1j)/2; b=(1-1j)/2
C=torch.tensor([[a,b],[b,a]],dtype=cdtype)
T=C
for _ in range(3): T=torch.kron(T,C)
Tnorm=torch.linalg.norm(T)

def base_schedule():
    hubs=[0,1,2,3]; assign={v:hubs[(v-4)%4] for v in range(4,16)}
    return [(v,assign[v]) for v in range(4,16)]+[(0,1),(2,3),(0,2),(1,3)]+[(v,assign[v]) for v in range(4,16)]
def make_schedule(g,seed):
    rng=random.Random(seed); s=base_schedule()
    while len(s)<g:
      p,q=rng.sample(range(n),2); s.insert(rng.randrange(len(s)+1),(p,q))
    return s

def optimize(sched,seed,steps=3000):
    torch.manual_seed(seed); g=len(sched)
    # each gate params log p,log q,log r (complex); s=-q*r/p. plus input/output diagonal logs.
    z=torch.zeros((g,3,2),requires_grad=True)
    din=torch.zeros((n,2),requires_grad=True); dout=torch.zeros((n,2),requires_grad=True)
    with torch.no_grad():
      z.normal_(0,0.02); din.normal_(0,0.01); dout.normal_(0,0.01)
      # initialize local gate close to C: p=a,q=b,r=b, s=a. Logs.
      vals=[a,b,b]
      for j,v in enumerate(vals):
        z[:,j,0].fill_(math.log(abs(v))); z[:,j,1].fill_(math.atan2(v.imag,v.real))
      z.add_(0.01*torch.randn_like(z))
    pars=[z,din,dout]
    opt=torch.optim.Adam(pars,lr=0.02)
    best=1e9; state=None
    def cexp(u): return torch.exp(u[...,0]+1j*u[...,1])
    def build():
      iv=cexp(din); ov=cexp(dout)
      M=torch.diag(iv)
      pv=cexp(z)
      for j,(x,y) in enumerate(sched):
        p,q,r=pv[j,0],pv[j,1],pv[j,2]; ss=-q*r/p
        G=torch.eye(n,dtype=cdtype)
        G[x,x]=p;G[x,y]=q;G[y,x]=r;G[y,y]=ss
        M=G@M
      return torch.diag(ov)@M
    for it in range(steps):
      opt.zero_grad(); M=build(); rel=torch.linalg.norm(M-T)/Tnorm
      loss=rel**2
      loss.backward(); torch.nn.utils.clip_grad_norm_(pars,20.0); opt.step()
      v=rel.item()
      if v<best:
        best=v; state=[p.detach().clone() for p in pars]
      if best<1e-10: break
      if it in (1200,2200):
        for pg in opt.param_groups: pg['lr']*=0.25
    with torch.no_grad():
      for p,s in zip(pars,state): p.copy_(s)
    opt2=torch.optim.LBFGS(pars,lr=.3,max_iter=250,line_search_fn='strong_wolfe',tolerance_grad=1e-12,tolerance_change=1e-14)
    def closure():
      opt2.zero_grad(); rel=torch.linalg.norm(build()-T)/Tnorm; loss=rel**2; loss.backward(); return loss
    try: opt2.step(closure)
    except Exception: pass
    with torch.no_grad():
      rel=(torch.linalg.norm(build()-T)/Tnorm).item()
      # approximate intermediate condition
      iv=cexp(din); P=torch.diag(iv); pv=cexp(z); mc=1.0
      for j,(x,y) in enumerate(sched):
        p,q,r=pv[j,0],pv[j,1],pv[j,2]; ss=-q*r/p
        G=torch.eye(n,dtype=cdtype); G[x,x]=p;G[x,y]=q;G[y,x]=r;G[y,y]=ss
        P=G@P; sv=torch.linalg.svdvals(P); mc=max(mc,(sv[0]/sv[-1]).item())
      P=torch.diag(cexp(dout))@P; sv=torch.linalg.svdvals(P); mc=max(mc,(sv[0]/sv[-1]).item())
    return rel,mc

all=[]
for g in [28,29,30,31]:
  for ss in range(2):
    sched=make_schedule(g,1000+g*10+ss)
    rel,cond=optimize(sched,7000+g*10+ss,steps=1400)
    rec={'g':g,'schedule_seed':ss,'rel_error':rel,'max_intermediate_condition':cond,'schedule':sched}; all.append(rec)
    print(json.dumps({k:v for k,v in rec.items() if k!='schedule'}),flush=True)
json.dump({'model':'each local call is full diagonal-double-coset of C; B11*B22+B12*B21=0','results':all},open('adaptive_fft_cpp/small_finite_win_reduced.json','w'),indent=2)
