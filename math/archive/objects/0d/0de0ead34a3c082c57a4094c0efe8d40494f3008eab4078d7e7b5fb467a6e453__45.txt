   1	"""Fresh arbitrary bounded-input *approximate* normalized DFT via sampling.
   2	
   3	Different goal from exact FFT: given |x[j]|<=1, estimate all bins of
   4	FFT(x)/N within additive epsilon with probability >=1-failure under
   5	random sampling, using O(N (1+log log N)) Fourier arithmetic for
   6	fixed epsilon/failure and an O(N)-size explicit output.
   7	
   8	Ingredients are classical Monte Carlo concentration + pruned radix-2 FFT,
   9	not claimed to be a novel algorithm or a practical SciPy replacement.
  10	"""
  11	from __future__ import annotations
  12	from dataclasses import dataclass
  13	from math import ceil,log,log2
  14	import numpy as np
  15	
  16	@dataclass
  17	class SparseStats:
  18	    nonzeros:int=0
  19	    active_nodes:int=0
  20	    butterfly_slots:int=0
  21	    twiddle_multiplications:int=0
  22	    output_length:int=0
  23	
  24	def _pruned_fft(n:int, ids, val, tw_cache:dict, stats:SparseStats):
  25	    if len(ids)==0:
  26	        return None
  27	    if n==1:
  28	        stats.active_nodes+=1
  29	        return np.array([complex(val[0])],dtype=np.complex128)
  30	    stats.active_nodes+=1
  31	    half=n//2
  32	    mask=(ids%2)==0
  33	    e=_pruned_fft(half,ids[mask]//2,val[mask],tw_cache,stats) if np.any(mask) else None
  34	    o=_pruned_fft(half,ids[~mask]//2,val[~mask],tw_cache,stats) if np.any(~mask) else None
  35	    stats.butterfly_slots+=half
  36	    if o is None:
  37	        return np.concatenate((e,e))
  38	    if n not in tw_cache:
  39	        tw_cache[n]=np.exp((-2j*np.pi/n)*np.arange(half))
  40	    t=tw_cache[n]*o
  41	    stats.twiddle_multiplications+=half
  42	    if e is None:
  43	        return np.concatenate((t,-t))
  44	    return np.concatenate((e+t,e-t))
  45	
  46	def sparse_fft(N:int,indices,values):
  47	    """Exact DFT (to floating arithmetic roundoff) of k-sparse length N vector.
  48	    Duplicate indices are accumulated. Explicit output always length N.
  49	    """
  50	    if N<1 or (N&(N-1)):
  51	        raise ValueError('N must be a positive power of two')
  52	    ids=np.asarray(indices,dtype=np.int64)
  53	    vals=np.asarray(values,dtype=np.complex128)
  54	    if ids.shape!=vals.shape or ids.ndim!=1 or np.any(ids<0) or np.any(ids>=N):
  55	        raise ValueError('bad sparse input')
  56	    fused={}
  57	    for j,v in zip(ids,vals):fused[int(j)]=fused.get(int(j),0j)+complex(v)
  58	    active=sorted(j for j in fused if fused[j]!=0)
  59	    ids=np.asarray(active,dtype=np.int64)
  60	    vals=np.asarray([fused[j] for j in active],dtype=np.complex128)
  61	    stats=SparseStats(nonzeros=len(ids),output_length=N)
  62	    out=_pruned_fft(N,ids,vals,{},stats)
  63	    if out is None:out=np.zeros(N,dtype=np.complex128)
  64	    return out,stats
  65	
  66	def sample_count(N:int,epsilon:float,failure_probability:float=0.01):
  67	    """Union-bound Hoeffding: P(any normalized-DFT bin > epsilon)
  68	    <= 4N exp(-m epsilon^2/4) for bounded complex input.
  69	    """
  70	    if N<1 or not (0<epsilon<2) or not (0<failure_probability<1):
  71	        raise ValueError('invalid N, epsilon or failure probability')
  72	    return ceil(4/(epsilon**2)*log(4*N/failure_probability))
  73	
  74	def approximate_normalized_fft(x,epsilon:float=0.3,failure_probability:float=0.01,seed:int=0):
  75	    """Randomized all-bin estimator. NO exact DFT promise.
  76	    For every fixed bounded input, union Hoeffding gives confidence >=1-eta.
  77	    """
  78	    x=np.asarray(x,dtype=np.complex128)
  79	    N=len(x)
  80	    if N<1 or (N&(N-1)):
  81	        raise ValueError('N must be power of 2')
  82	    if np.any(abs(x)>1+1e-12):
  83	        raise ValueError('input magnitude must be <=1')
  84	    m=sample_count(N,epsilon,failure_probability)
  85	    rng=np.random.default_rng(seed)
  86	    selected=rng.integers(0,N,size=m)
  87	    unique,counts=np.unique(selected,return_counts=True)
  88	    sparse_vals=(counts/m)*x[unique]
  89	    out,stats=sparse_fft(N,unique,sparse_vals)
  90	    return out,{'samples':m,'unique_sample_positions':len(unique),
  91	                'worst_case_failure_bound':failure_probability,
  92	                'butterfly_slots':stats.butterfly_slots,
  93	                'active_nodes':stats.active_nodes,'twiddle_mults':stats.twiddle_multiplications,
  94	                'standard_butterfly_slots':N//2*(N.bit_length()-1)}
  95	
  96	if __name__=='__main__':
  97	    n=4096
  98	    inp=np.ones(n,dtype=complex)
  99	    out,stats=approximate_normalized_fft(inp,epsilon=.4,seed=5)
 100	    print('stats',stats)
 101	    print('worst observed abs error',np.max(abs(out-np.fft.fft(inp)/n)))
