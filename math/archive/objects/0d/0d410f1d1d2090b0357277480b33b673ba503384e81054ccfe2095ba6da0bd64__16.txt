   8	HERE=Path(__file__).resolve().parent;ROOT=HERE.parent

  10	from check_n16_outer_gate import LABELS

  13	def cm(a,b):
  14	 ar,ai=a;br,bi=b
  15	 return ar@br-ai@bi,ar@bi+ai@br
  16	def bounds(a,den):return [F(int(sum(abs(r)+abs(i) for r,i in zip(rr,ii))),den) for rr,ii in zip(*a)]
  17	def frac(d):return F(int(d['numerator']),int(d['denominator']))
  18	def show(x):return {'numerator':str(x.numerator),'denominator':str(x.denominator),'decimal':float(x)}
  19	def input_path(name):

  22	def propose():

  28	def certify():

  32	 seed=json.loads(seed_path.read_text());old=json.loads(input_path('EXACT_CONTRACTION_CERTIFICATE.json').read_text())

  40	 directions=tuple(np.asarray(seed['directions_'+part],dtype=object) for part in ('real','imag'))

  46	  for col in range(64):

  48	   for power in range(64):greens[j,row,col]=d**(64-power)*(L//poles[j]);row=pl[row]
  49	 hs=[];ls=[]

  51	  h=tuple(np.empty((64,64),dtype=object) for _ in range(2));largest=0

  55	   # Stream the 256 chart-direction columns to avoid storing a large tensor.
