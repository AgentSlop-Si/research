   1 #!/usr/bin/env python3
   2 from pathlib import Path
   3 import hashlib,json
   4 root=Path(__file__).resolve().parent
   5 manifest=root/'SEALED_MANIFEST.json'
   6 expected=(root/'SEALED_MANIFEST.sha256').read_text().split()[0]
   7 assert hashlib.sha256(manifest.read_bytes()).hexdigest()==expected
   8 entries=json.loads(manifest.read_text())['files']
   9 for entry in entries:
  10     p=root/entry['path']
  11     assert p.is_file(),entry['path']
  12     assert p.stat().st_size==entry['bytes'],entry['path']
  13     assert hashlib.sha256(p.read_bytes()).hexdigest()==entry['sha256'],entry['path']
  14 print('PASS',len(entries),'sealed files')
