"""Exact modular Cooley--Tukey positive control, 2 quasiseparable stages.

The block-F4 stages have QS order 1, but repeated block-diagonal F4 stages are NOT
in the global separated-generator rank-one chart (nondegenerate connectivity).
All coefficients and endpoint permutation information are exact modulo 12289.
"""
import json
import numpy as np
P=12289

def root(n):
    assert (P-1)%n==0
    for g in range(2,P):
        z=pow(g,(P-1)//n,P)
        if pow(z,n,P)==1 and all(pow(z,n//q,P)!=1 for q in set(factor(n))):return z
    raise AssertionError('root')
def factor(n):
    a=[];x=n;d=2
    while d*d<=x:
        if x%d==0:
            a.append(d)
            while x%d==0:x//=d
        d+=1
    if x>1:a.append(x)
    return a

def rank_mod(A):
    A=np.array(A,dtype=np.int64)%P;h,w=A.shape;r=0
    for c in range(w):
        if r>=h:break
        ix=np.flatnonzero(A[r:,c]);
        if not ix.size:continue
        k=r+int(ix[0]);A[[r,k]]=A[[k,r]]
        A[r,c:]=A[r,c:]*pow(int(A[r,c]),-1,P)%P
        for i in range(r+1,h):
            if A[i,c]:A[i,c:]=(A[i,c:]-int(A[i,c])*A[r,c:])%P
        r+=1
    return r

def qs_order(S):
    n=len(S)
    return max([rank_mod(S[k:,:k]) for k in range(1,n)]+[rank_mod(S[:k,k:]) for k in range(1,n)])

def fft_matrix(n):
    z=root(n)
    return np.array([[pow(z,i*j,P) for j in range(n)] for i in range(n)],dtype=np.int64)

def ct_factor(a,b,ordera=None,orderb=None):
    n=a*b;z=root(n)
    ordera=list(range(a)) if ordera is None else list(ordera)
    orderb=list(range(b)) if orderb is None else list(orderb)
    assert sorted(ordera)==list(range(a)) and sorted(orderb)==list(range(b))
    A=np.zeros((n,n),dtype=np.int64);B=np.zeros_like(A)
    for i in range(a):
        for v in range(b):
            for u in range(b):
                A[i*b+v,i*b+u] = pow(z,a*orderb[v]*orderb[u],P) # F_b
    for k2 in range(b):
        for v in range(a):
            for u in range(a):
                i=ordera[u];k1=ordera[v]
                B[k2*a+v,k2*a+u]=pow(z, b*k1*i+k2*i,P)
    mid=[ordera[u]*b+orderb.index(k2) for k2 in range(b) for u in range(a)]
    input_order=[i+a*orderb[u] for i in range(a) for u in range(b)]
    output_order=[k2+b*ordera[v] for k2 in range(b) for v in range(a)]
    M=B@A[mid,:]%P
    target=fft_matrix(n)[np.ix_(output_order,input_order)]
    assert np.array_equal(M,target)
    return A,B,mid,input_order,output_order

def audit():
    rows=[]
    for a,b,oa,ob in [(2,4,[0,1],[0,2,1,3]),(4,4,[0,2,1,3],[0,2,1,3]),(4,8,[0,2,1,3],list(range(8)))]:
        A,B,mid,in_order,out_order=ct_factor(a,b,oa,ob)
        result={'n':a*b,'radix_pair':[a,b],'rank_stage_1':qs_order(A),'rank_stage_2':qs_order(B),'number_stage_matrices':2,'exact_all_entries':True, 'middle_shuffle':mid,'input_order':in_order,'output_order':out_order}
        rows.append(result)
    # Witness that repeated block diagonal rank-1 QS fails to lie in global rank-1 chart:
    A,*_=ct_factor(4,4,[0,2,1,3],[0,2,1,3])
    pairs=[]
    for base in (0,4):
        for j in range(base,base+3):
            for i in range(j+1,base+4):
                if A[i,j]: pairs.append((i,j,int(A[i,j])))
    # Pick lower nonzero in 1st block and 2nd block.
    i,j,v=pairs[0]; x,y,w=next((x,y,w) for x,y,w in pairs if x>=4)
    assert A[x,j]==0 and v and w
    witness={'nonzero_lower_first':[i,j,v],'nonzero_lower_second':[x,y,w],'forced_nonzero_but_actual_zero':[x,j,int(A[x,j])]}
    return rows,witness
if __name__=='__main__':
    rows,witness=audit()
    print(json.dumps({'prime':P,'results':rows,'separated_chart_counterexample':witness},indent=2))
