0027 def prime(n):
0028     return n>=2 and all(n%d for d in range(2,math.isqrt(n)+1))
0029 def divisors(n):
0030     return [d for d in range(2,n+1) if n%d==0 and prime(d)]
0031 def field(q):
0032     p=2*q+1
0033     while not prime(p):p+=2*q
0034     v=next(g for g in range(2,p) if pow(g,2*q,p)==1 and all(pow(g,2*q//d,p)!=1 for d in divisors(2*q)))
0035     return p,v

0037 def direct(x,root,p):
0038     return [sum(v*pow(root,k*j,p) for j,v in enumerate(x))%p for k in range(len(x))]

0078 def floating():
0079     rng=np.random.default_rng(380421);rows=[];maxerr=0.;transfers=0;moments=0

0083         Fq=np.exp(-2j*np.pi*np.outer(np.arange(q),np.arange(q))/q)/np.sqrt(q)

0085         E=np.zeros((q,q),complex);P=[]

0091                 Pr[b*m:(b+1)*m,b*m:(b+1)*m]=np.outer(e,e.conj())
0092             P.append(Pr)

0094         J=Fq@E;errs.append(np.linalg.norm(J.conj().T@J-np.eye(q)))
0095         for b in range(k):
0096             for ell in range(k):
