   9 def z(n, m=None):
  10     return [[0] * (n if m is None else m) for _ in range(n)]
  11

  13 def eye(n):
  14     a = z(n)
  15     for i in range(n):
  16         a[i][i] = 1
  17     return a

  20 def add(a, b, sign=1):
  21     return [[(v + sign * w) % P for v, w in zip(r, s)] for r, s in zip(a, b)]

  24 def scale(a, c):
  25     return [[v * c % P for v in row] for row in a]

  28 def mm(a, b):
  29     c = z(len(a), len(b[0]))
  30     sb = [[(j, v) for j, v in enumerate(row) if v] for row in b]
  31     for i, row in enumerate(a):
  32         for k, v in enumerate(row):
  33             if v:
  34                 for j, w in sb[k]:
  35                     c[i][j] += v * w
  36         c[i] = [v % P for v in c[i]]
  37     return c

  40 def kron(a, b):
  41     out = z(len(a) * len(b), len(a[0]) * len(b[0]))
  42     for i, row in enumerate(a):
  43         for j, v in enumerate(row):
  44             if v:
  45                 for r, br in enumerate(b):
  46                     for s, w in enumerate(br):
  47                         out[i * len(b) + r][j * len(b[0]) + s] = v * w % P
  48     return out

  51 def root(n):
  52     return pow(3, (P - 1) // n, P)
  53

  55 def fourier(n, inverse=False):
  56     w = pow(root(n), -1, P) if inverse else root(n)
  57     norm = pow(n, -1, P) if inverse else 1
  58     return [[norm * pow(w, i * j, P) % P for j in range(n)] for i in range(n)]

  61 def shift(n, step=1):
  62     a = z(n)
  63     for j in range(n):
  64         a[(j + step) % n][j] = 1
  65     return a

  68 def diag(values):
  69     a = z(len(values))
  70     for i, v in enumerate(values):
  71         a[i][i] = v % P
  72     return a

  75 def transpose_grid(m):
  76     a = z(m * m)
  77     for r in range(m):
  78         for s in range(m):
  79             a[s * m + r][r * m + s] = 1
  80     return a

  83 def degree(a):
  84     return {'nnz': sum(bool(v) for row in a for v in row),
  85             'row': max(sum(bool(v) for v in row) for row in a),
  86             'column': max(sum(bool(v) for v in row) for row in zip(*a))}

  89 def basic(m):
  90     n = m * m
  91     k = transpose_grid(m)
  92     b, bi = kron(eye(m), fourier(m)), kron(eye(m), fourier(m, True))
  93     ell = kron(fourier(m), eye(m))
  94     delta = diag([pow(root(n), r * s, P) for r in range(m) for s in range(m)])
  95     qmat = mm(delta, ell)
  96     return n, k, b, bi, ell, delta, qmat

  99 def relocated_controls():
 100     rows = []
 101     for m in (2, 4, 8, 16):
 102         n, k, b, bi, ell, delta, qmat = basic(m)
 103         f, s, d, si = fourier(n), shift(n), diag([pow(root(n), j, P) for j in range(n)]), shift(n, -1)
 104         assert mm(k, mm(b, qmat)) == f
 105         actions = []
 106         for x, y in ((s, d), (d, si)):
 107             a = mm(k, mm(y, k))
 108             t = mm(bi, mm(a, b))
 109             assert mm(a, b) == mm(b, t)
 110             assert mm(t, qmat) == mm(qmat, x)
 111             assert mm(k, a) == mm(y, k)
 112             assert degree(a)['row'] == degree(a)['column'] == 1
 113             assert degree(t)['row'] == degree(t)['column'] == 1
 114             actions.append((a, t))
 115         expected_t1 = z(n)
 116         expected_t2 = z(n)
 117         for bank_index in range(m):
 118             for j in range(m):
 119                 expected_t1[bank_index * m + (j + 1) % m][bank_index * m + j] = pow(root(n), bank_index, P)
 120                 out_bank = (bank_index - 1) % m
 121                 expected_t2[out_bank * m + j][bank_index * m + j] = pow(root(m), j, P) if bank_index == 0 else 1
 122         assert actions[0][1] == expected_t1
 123         assert actions[1][1] == expected_t2
 124         rows.append({'M': m, 'N': n, 'P_degree': degree(k), 'Q_degree': degree(qmat), 'all_four_action_degrees': 1,
 125                      'Q_possible_as_two_zero_mask_layers_by_support': m <= 4})

 129 def swap_controls():
 130     rows = []
 131     for m in (2, 4, 8, 16):
 132         n, k, b, bi, ell, delta, unused = basic(m)
 133         baseline_p = mm(ell, delta)
 134         a, bb = 0, m
 135         g = eye(n)
 136         g[a][a] = g[bb][bb] = 0
 137         g[a][bb] = g[bb][a] = 1
 138         c = mm(b, mm(g, bi))
 139         u = [(b[i][a] - b[i][bb]) % P for i in range(n)]
 140         w = [(bi[a][j] - bi[bb][j]) % P for j in range(n)]
 141         rank_one = [[(int(i == j) - u[i] * w[j]) % P for j in range(n)] for i in range(n)]
 142         assert c == rank_one and mm(c, c) == eye(n)
 143         assert sum(bool(v) for v in u) == sum(bool(v) for v in w) == 2 * m
 144         new_p, new_q = mm(baseline_p, c), mm(g, k)
 145         assert mm(new_p, mm(b, new_q)) == fourier(n)
 146         action_stats = []
 147         for x in (shift(n), diag([pow(root(n), j, P) for j in range(n)])):
 148             t0 = mm(k, mm(x, k))
 149             t = mm(g, mm(t0, g))
 150             amat = mm(b, mm(t, bi))
 151             assert mm(amat, b) == mm(b, t)
 152             action_stats.append(degree(amat))
 153         assert action_stats[1]['row'] > 1
 154         rows.append({'M': m, 'N': n, 'gauge_nontrivial_vector_supports': [2 * m, 2 * m],
 155                      'gauge_matrix_degree': degree(c), 'changed_A_degrees': action_stats,
 156                      'involution_and_exact_factorization': True})
 157     return rows

 160 def clock_coset_controls():
 161     rows = []
 162     for m in (2, 4, 8):
 163         n, _, b, bi, _, _, _ = basic(m)
 164         exponents = []
 165         for bank_index in range(m):
 166             residue = m - 1 - bank_index
 167             slope = 2 * bank_index + 1
 168             for j in range(m):
 169                 exponents.append((residue + m * slope * j) % n)
 170         assert len(set(exponents)) == n
 171         t = diag([pow(root(n), e, P) for e in exponents])
 172         a = mm(b, mm(t, bi))
 173         assert degree(a)['row'] == degree(a)['column'] == 1
 174         rows.append({'M': m, 'all_active_blocks_monomial': True, 'raw_coset_permutation_verified': True})
