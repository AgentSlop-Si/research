1: """Shared output-pruned DFT plans for arbitrary requested Fourier bins.
2: 
3: For N = K*M, write a time index as n = m + M*r.  A Cooley--Tukey step gives
4: 
5:     X[j] = sum_m G[m, j mod K] * exp(-2*pi*i*m*j/N),
6: 
7: where G[m,:] is the K-point DFT of x[m + M*r] over r=0..K-1.
8: 
9: Thus one set of M length-K FFTs is shared by *all* requested outputs.  If K is
10: chosen on the order of q=|Q| (and K divides N), q arbitrary outputs cost
11: O(N log K + q*N/K) = O(N log q) arithmetic.  This is classical generic FFT
12: pruning; this module packages it for projected relation-aware Fourier views.
13: 
14: The plan stores only the final q-by-M phase matrix.  For K>=q this is at most
15: N complex numbers, independent of how scattered the requested bins are.
16: """

32: def _divisors(n: int) -> list[int]:
33:     small=[]; large=[]
34:     d=1
35:     while d*d<=n:
36:         if n%d==0:
37:             small.append(d)
38:             if d*d!=n: large.append(n//d)
39:         d+=1
40:     return small+large[::-1]

43: def candidate_radices(n:int,q:int) -> list[int]:
44:     """A small useful set of Cooley--Tukey inner sizes.
45: 
46:     The arithmetic proxy for one plan execution is log2(K)+q/K, ignoring
47:     backend constants.  Return divisors around the minimizer plus endpoints.
48:     """
49:     n=int(n); q=int(q)
50:     if n<1 or q<1 or q>n: raise ValueError('need 1<=q<=n')
51:     ds=_divisors(n)
52:     scored=[]
53:     for k in ds:
54:         score=(0.0 if k==1 else math.log2(k))+q/k
55:         scored.append((score,k))
56:     scored.sort()
57:     picks={1,n}
58:     for _,k in scored[:6]: picks.add(k)
59:     # Also include bracketing divisors around q; practical FFT constants often
60:     # move the optimum away from the arithmetic proxy.
61:     below=[d for d in ds if d<=q]
62:     above=[d for d in ds if d>=q]
63:     if below: picks.add(below[-1])
64:     if len(below)>1: picks.add(below[-2])
65:     if above: picks.add(above[0])
66:     if len(above)>1: picks.add(above[1])
67:     return sorted(picks)

70: def choose_radix(n:int,q:int) -> int:
71:     ds=candidate_radices(n,q)
72:     return min(ds,key=lambda k: ((0.0 if k==1 else math.log2(k))+q/k, k))

75: class FullFFTBinPlan:
76:     """Optimized full FFT fallback, gathering only requested bins."""
77:     strategy='full_fft'
78:     def __init__(self,n:int,bins:Sequence[int],*,backend='scipy'):
79:         self.n=int(n); self.backend=backend
80:         b=np.asarray(bins,dtype=np.int64)
81:         if b.ndim!=1 or b.size<1 or np.any(b<0) or np.any(b>=self.n):
82:             raise ValueError('bad bins')
83:         self.bins=b.astype(np.intp,copy=True); self.q=len(b)
84:     @property
85:     def plan_bytes(self): return int(self.bins.nbytes)
86:     def execute(self,x):
87:         a=np.asarray(x,dtype=np.complex128)
88:         if a.shape!=(self.n,): raise ValueError('input has wrong length')
89:         if self.backend=='scipy' and sp_fft is not None: y=sp_fft.fft(a)
90:         elif self.backend in ('scipy','numpy'): y=np.fft.fft(a)
91:         else: raise ValueError('backend must be scipy or numpy')
92:         return np.ascontiguousarray(y[self.bins])
93:     def project_stream(self,basis_vectors:Iterable[np.ndarray]):
94:         rows=[self.execute(row) for row in basis_vectors]
95:         return np.ascontiguousarray(np.stack(rows)) if rows else np.empty((0,self.q),np.complex128)

98: class ArbitraryBinsDFTPlan:
99:     """Reusable exact (up to floating-point roundoff) arbitrary-output plan."""
100:     strategy='arbitrary_pruned'
101:     def __init__(self,n:int,bins:Sequence[int],*,radix:int|str='auto',backend='scipy'):
102:         self.n=int(n); self.backend=backend
103:         if self.n<1: raise ValueError('n must be positive')
104:         b=np.asarray(bins,dtype=np.int64)
105:         if b.ndim!=1 or b.size<1 or np.any(b<0) or np.any(b>=self.n):
106:             raise ValueError('bad bins')
107:         self.bins=b.astype(np.intp,copy=True); self.q=len(self.bins)
108:         if radix=='auto': k=choose_radix(self.n,self.q)
109:         else: k=int(radix)
110:         if k<1 or self.n%k: raise ValueError('radix must be a positive divisor of n')
111:         self.k=k; self.m=self.n//self.k
112:         self.residue=np.asarray(self.bins % self.k,dtype=np.intp)
113:         mm=np.arange(self.m,dtype=np.int64)
114:         # Reduce exact integer products before converting to float angles.
115:         self.phase=np.stack([phase_row(self.n,int(b),mm) for b in self.bins])
116:     @property
117:     def plan_bytes(self):
118:         return int(self.phase.nbytes+self.bins.nbytes+self.residue.nbytes)
119:     @property
120:     def arithmetic_proxy(self):
121:         return (0.0 if self.k==1 else math.log2(self.k))+self.q/self.k
122:     def execute(self,x):
123:         a=np.asarray(x,dtype=np.complex128)
124:         if a.shape!=(self.n,): raise ValueError('input has wrong length')
125:         if self.k==1:
126:             # q is normally tiny here.  groups[:,0] is just x in the required
127:             # Cooley--Tukey ordering.
128:             col=a
129:             return np.einsum('qm,m->q',self.phase,col,optimize=False)
130:         groups=np.ascontiguousarray(a.reshape(self.k,self.m).T)
131:         if self.backend=='scipy' and sp_fft is not None:
132:             inner=sp_fft.fft(groups,axis=1)
133:         elif self.backend in ('scipy','numpy'):
134:             inner=np.fft.fft(groups,axis=1)
135:         else: raise ValueError('backend must be scipy or numpy')
136:         selected=inner[:,self.residue]  # M x q, one inner output per requested bin
137:         return np.einsum('mq,qm->q',selected,self.phase,optimize=False)
138:     def project_stream(self,basis_vectors:Iterable[np.ndarray]):
139:         rows=[self.execute(row) for row in basis_vectors]
140:         return np.ascontiguousarray(np.stack(rows)) if rows else np.empty((0,self.q),np.complex128)

151: class AutoOutputPlan:
152:     """Measured query-plan chooser over pruned radices and a full FFT.
153: 
154:     With ``sample`` supplied, nearby Cooley--Tukey radices are timed on the
155:     actual machine subject to ``max_plan_bytes``.  The fastest execution plan
156:     is retained.  ``calibration_ms`` is exposed and must be charged by callers;
157:     the tuner is intended for many repeated registrations/queries, not one-offs.
158: 
159:     Without a sample, a conservative arithmetic/memory heuristic is used.
160:     """
161:     def __init__(self,n:int,bins:Sequence[int],*,backend='scipy',sample=None,
162:                  reps=2,max_plan_bytes=128*1024*1024):
163:         self.n=int(n); self.bins=np.asarray(bins,dtype=np.intp); self.q=len(self.bins)
164:         self.backend=backend; self.max_plan_bytes=int(max_plan_bytes)
165:         if self.q<1: raise ValueError('bins must be nonempty')
166:         self.calibration_ms=0.0; self.candidates=[]; self.sample_result=None
167:         self._sample_fingerprint=None
168:         full=FullFFTBinPlan(n,bins,backend=backend)
169:         if sample is None:
170:             # Prefer a radix >= q when possible so the phase table is O(N),
171:             # then fall back to the arithmetic minimizer.
172:             ds=[k for k in candidate_radices(self.n,self.q) if self.n%k==0]
173:             feasible=[k for k in ds if self.q*(self.n//k)*16 <= self.max_plan_bytes]
174:             ge=[k for k in feasible if k>=self.q]
175:             k=min(ge) if ge else (min(feasible,key=lambda z: ((0 if z==1 else math.log2(z))+self.q/z)) if feasible else self.n)
176:             pruned=ArbitraryBinsDFTPlan(n,bins,radix=k,backend=backend)
177:             use_pruned=(self.q <= max(1,self.n//100))
178:             self.impl=pruned if use_pruned else full
179:             self.choice=PlanChoice(self.impl.strategy,k if use_pruned else None,None,None)
180:             return
181: 
182:         a=np.asarray(sample,dtype=np.complex128)
183:         if a.shape!=(self.n,): raise ValueError('sample has wrong length')
184:         reps=max(1,int(reps))
185:         t_cal=time.perf_counter_ns()
186:         self._sample_fingerprint=hashlib.blake2b(np.ascontiguousarray(a).tobytes(),digest_size=32).digest()
187:         # Full-FFT timing and result.
188:         full.execute(a)
189:         vals=[]; full_result=None
190:         for _ in range(reps):
191:             t=time.perf_counter_ns(); full_result=full.execute(a); vals.append((time.perf_counter_ns()-t)/1e6)
192:         full_ms=float(np.median(vals))
193:         best_impl=full; best_ms=full_ms; best_result=full_result; best_radix=None
194:         self.candidates.append({'strategy':'full_fft','radix':None,'execute_ms':full_ms,'plan_bytes':full.plan_bytes})
195: 
196:         for k in candidate_radices(self.n,self.q):
197:             if k>=self.n: continue
198:             phase_bytes=self.q*(self.n//k)*16
199:             if phase_bytes>self.max_plan_bytes: continue
200:             p=ArbitraryBinsDFTPlan(self.n,self.bins,radix=k,backend=backend)
201:             p.execute(a)  # warm
202:             vals=[]; result=None
203:             for _ in range(reps):
204:                 t=time.perf_counter_ns(); result=p.execute(a); vals.append((time.perf_counter_ns()-t)/1e6)
205:             tm=float(np.median(vals))
206:             self.candidates.append({'strategy':'arbitrary_pruned','radix':k,'execute_ms':tm,'plan_bytes':p.plan_bytes})
207:             if tm<best_ms:
208:                 best_impl=p; best_ms=tm; best_result=result; best_radix=k
209:             # Non-winning plans are dropped immediately; only their timing is retained.
210:         self.impl=best_impl; self.sample_result=np.ascontiguousarray(best_result)
211:         self.calibration_ms=(time.perf_counter_ns()-t_cal)/1e6
212:         self.choice=PlanChoice(self.impl.strategy,best_radix,best_ms,full_ms)
213: 
214:     @property
215:     def strategy(self): return self.impl.strategy
216:     @property
217:     def plan_bytes(self): return self.impl.plan_bytes
218:     def execute(self,x): return self.impl.execute(x)
219:     def project_stream(self,basis_vectors): return self.impl.project_stream(basis_vectors)

221:     def project_stream_with_sample(self,sample,basis_vectors):
222:         """Use the already-computed calibration output for the first basis row."""
223:         a=np.asarray(sample,dtype=np.complex128)
224:         if a.shape!=(self.n,): raise ValueError('sample has wrong length')
225:         fingerprint=hashlib.blake2b(np.ascontiguousarray(a).tobytes(),digest_size=32).digest()
226:         if self.sample_result is not None and fingerprint==self._sample_fingerprint:
227:             first=self.sample_result.copy()
228:         else:
229:             first=self.execute(a)
230:         rows=[first]
231:         rows.extend(self.execute(row) for row in basis_vectors)
232:         return np.ascontiguousarray(np.stack(rows))

236: def productive_radices(n:int,q:int,max_plan_bytes=128*1024*1024,limit=4) -> list[int]:
237:     """Radices worth trying when every trial will process a real basis row."""
238:     if int(limit)<=0: return []
239:     ds=[d for d in _divisors(int(n)) if d<int(n)]
240:     if not ds: return []
241:     targets=[]
242:     # Practical measurements often prefer a larger K than the arithmetic
243:     # minimum because larger inner FFTs improve locality and shrink the final
244:     # phase-combination table.
245:     for mul in (1,2,4,8): targets.append(max(1,int(q)*mul))
246:     targets.append(choose_radix(n,q))
247:     picks=[]
248:     for target in targets:
249:         feasible=[d for d in ds if q*(n//d)*16<=max_plan_bytes]
250:         if not feasible: continue
251:         k=min(feasible,key=lambda d:(abs(math.log(max(d,1)/max(target,1))),d))
252:         if k not in picks:picks.append(k)
253:     # Prefer the practical 4q neighborhood first, then nearby alternatives.
254:     picks.sort(key=lambda k:(abs(math.log(max(k,1)/max(4*q,1))),k))
255:     return picks[:max(1,int(limit))]

267: class ProductiveProjectionPlanner:
268:     """Autotune while doing useful basis registrations.
269: 
270:     Each candidate strategy is evaluated on a *different basis vector whose
271:     projection is required anyway*.  Hence no transform execution is thrown
272:     away.  Only construction of losing pruned plans is pure tuning overhead.
273:     After the productive trials, all remaining basis rows use the fastest
274:     observed strategy.
275:     """
276:     def __init__(self,n:int,bins:Sequence[int],*,backend='scipy',
277:                  max_plan_bytes=128*1024*1024,max_pruned_trials=3):
278:         self.n=int(n); self.bins=np.asarray(bins,dtype=np.intp); self.q=len(self.bins)
279:         self.backend=backend; self.max_plan_bytes=int(max_plan_bytes)
280:         self.max_pruned_trials=int(max_pruned_trials)
281:         self.stats=None
282: 
283:     def project(self,basis_vectors:Iterable[np.ndarray], *, expected_count:int|None=None):
284:         it=iter(basis_vectors); rows=[]; trials=[]; planning_ms=0.0
285:         full=FullFFTBinPlan(self.n,self.bins,backend=self.backend)
286:         # Strategy table stores (observed execution ms, plan object).
287:         strategies=[]
288:         try:first=next(it)
289:         except StopIteration:
290:             self.stats=ProductivePlannerStats('none',None,0.0,[],0)
291:             return np.empty((0,self.q),dtype=np.complex128)
292:         t=time.perf_counter_ns(); y=full.execute(first); ms=(time.perf_counter_ns()-t)/1e6
293:         rows.append(y); trials.append({'strategy':'full_fft','radix':None,'execute_ms':ms,'plan_bytes':full.plan_bytes})
294:         strategies.append((ms,full,None))
295:         count=1
296: 
297:         # Do not tune aggressively if the caller says there are too few rows to
298:         # amortize even plan construction.  Unknown count assumes a stream.
299:         max_trials=self.max_pruned_trials
300:         if expected_count is not None:
301:             max_trials=min(max_trials,max(0,int(expected_count)-2))
302:         for k in productive_radices(self.n,self.q,self.max_plan_bytes,max_trials):
303:             try: row=next(it)
304:             except StopIteration: break
305:             tp=time.perf_counter_ns(); p=ArbitraryBinsDFTPlan(self.n,self.bins,radix=k,backend=self.backend); planning_ms+=(time.perf_counter_ns()-tp)/1e6
306:             t=time.perf_counter_ns(); y=p.execute(row); ms=(time.perf_counter_ns()-t)/1e6
307:             rows.append(y); count+=1
308:             trials.append({'strategy':'arbitrary_pruned','radix':k,'execute_ms':ms,'plan_bytes':p.plan_bytes})
309:             strategies.append((ms,p,k))
310: 
311:         best_ms,best,best_k=min(strategies,key=lambda z:z[0])
312:         for row in it:
313:             rows.append(best.execute(row)); count+=1
314:         self.stats=ProductivePlannerStats(best.strategy,best_k,planning_ms,trials,count)
315:         return np.ascontiguousarray(np.stack(rows))
