  91 class Phase7ExtraCertificates(unittest.TestCase):
  92     def test_plaquette_holonomy_is_diagonal_gauge_invariant(self):
  93         rng=random.Random(17007)
  94         for n in (4,6,8,9,12,16,25):
  95             p,w=test_field(n)
  96             phase=[pow(w,(-a)%n,p) for a in range(n)]
  97             a=[rng.randrange(1,p) for _ in range(n)]
  98             b=[rng.randrange(1,p) for _ in range(n)]
  99             F=lambda k,j: phase[(k*j)%n]
 100             G=lambda k,j: a[k]*b[j]*F(k,j)%p
 101             for _ in range(200):
 102                 k,kk,j,jj=(rng.randrange(n) for _ in range(4))
 103                 # Cross-ratio factor cancels all row/column gauge scalars.
 104                 cross=phase[((k-kk)*(j-jj))%n]
 105                 lhs=G(k,j)*G(kk,jj)%p
 106                 rhs=G(k,jj)*G(kk,j)%p
 107                 self.assertEqual(lhs,rhs*cross%p)

 109     def test_complex_submatrix_stable_rank(self):
 110         import numpy as np
 111         rng=random.Random(1776)
 112         for n in (4,6,8,9,12,16,25):
 113             for _ in range(24):
 114                 rows=rng.sample(range(n),rng.randint(1,n))
 115                 cols=rng.sample(range(n),rng.randint(1,n))
 116                 A=np.array([[np.exp(-2j*np.pi*k*t/n) for t in cols] for k in rows])
 117                 r=np.linalg.matrix_rank(A,tol=1e-7)
 118                 self.assertGreaterEqual(r+1e-8,len(rows)*len(cols)/n)
 119                 self.assertLessEqual(np.linalg.norm(A,2),np.sqrt(n)+1e-7)

 121     def test_noncyclic_finite_abelian_example(self):
 122         # Group G=Z2 x Z4. Character values in F_17 with i=4.
 123         p=17; imaginary=4
 124         G=[(a,b) for a in range(2) for b in range(4)]
 125         def character(k,j):
 126             return pow(-1,k[0]*j[0])*pow(imaginary,k[1]*j[1],p)%p
 127         # H={0}xZ4, H^perp=Z2x{0}, each block size 4x2.
 128         row_cosets=[[v for v in G if v[0]==a] for a in range(2)]
 129         col_cosets=[[v for v in G if v[1]==b] for b in range(4)]
 130         # These column cosets have exactly 2 elements each.
 131         used=set(); cost=0
 132         for I in row_cosets:
 133             for J in col_cosets:
 134                 cost+=len(I)+len(J)
 135                 for k in I:
 136                     for j in J:
 137                         used.add((k,j))
 138                         for kk in I:
 139                             for jj in J:
 140                                 self.assertEqual(
 141                                     (character(k,j)*character(kk,jj)-
 142                                      character(k,jj)*character(kk,j))%p,0)
 143         self.assertEqual(len(used),64)
 144         self.assertEqual(cost,8*(4+2))
