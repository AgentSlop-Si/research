   1	"""Independent-style exact finite experiments for the algebraic arguments."""
   2	import json
   3	from random import Random
   4	from orbit_recovery import (orbits,representatives,binary_collision_family,
   5	    orbit_support_of_difference,dft_selected_exact,dft_bin_exact,lower_bounds,
   6	    subset_sum_cyclotomic_coeffs,target_conjugate_coeffs)
   7	from cyclotomic_interpolation import interpolation_audit,rational_interpolation_rank,rational_interpolant
   8	
   9	
  10	def run():
  11	    rng=Random(20261009)
  12	    result={'minimal_representatives':[], 'missing_orbit_ambiguity':[],
  13	            'degree_bounds':[], 'exact_interpolation':[], 'exhaustive':{}}
  14	    for n in (2,4,8,16,32,64):
  15	        O=orbits(n)
  16	        q=representatives(n)
  17	        assert len(q)==n.bit_length()
  18	        assert set(O)=={1<<j for j in range(n.bit_length())}
  19	        assert {k for ks in O.values() for k in ks}==set(range(n))
  20	        assert {n//(k & -k) if k else 1 for k in q}==set(O)
  21	        result['minimal_representatives'].append({'N':n,'sample_bins':q,
  22	             'minimal_count':len(q),'orbits':{str(m):len(ks) for m,ks in O.items()}})
  23	        for m,fs in O.items():
  24	            choose_masks=[0,1] if m==1 else [0,1,(1<<(m//2))-1]
  25	            variants=[binary_collision_family(n,m,mask) for mask in choose_masks]
  26	            x0=variants[0]
  27	            known=tuple(k for k in range(n) if k not in fs)
  28	            signature=dft_selected_exact(x0,known)
  29	            for x in variants[1:]:
  30	                assert x!=x0
  31	                assert dft_selected_exact(x,known)==signature
  32	                assert orbit_support_of_difference(x,x0)==fs
  33	            result['missing_orbit_ambiguity'].append({'N':n,'missing_conductor':m,
  34	                'orbit_size':len(fs),'ambiguity_count_lower_bound':2**(m//2) if m>1 else 2,
  35	                'tested_variants':len(variants),'support':fs})
  36	        for _ in range(10):
  37	            xs=[rng.randint(-2,2) for _ in range(n)]
  38	            ys=[rng.randint(-2,2) for _ in range(n)]
  39	            if dft_selected_exact(xs,q)==dft_selected_exact(ys,q):assert xs==ys
  40	    # Exhaustive uniqueness for N=8: 256 binary strings -> 256 different signatures.
  41	    n=8;q=representatives(n)
  42	    sigs={dft_selected_exact([(mask>>j)&1 for j in range(n)],q)
  43	          for mask in range(1<<n)}
  44	    assert len(sigs)==1<<n
  45	    result['exhaustive']={'N':n,'all_binary_inputs':1<<n,'distinct_minimal_signatures':len(sigs)}
  46	    for n in (4,8,16,32,64):
  47	        d=lower_bounds(n)
  48	        if n<=16:
  49	            # Verifies distinct subset sums in exact Q(zeta_n) coordinates.
  50	            d['distinct_primitive_samples']=len({subset_sum_cyclotomic_coeffs(n,i)
  51	                                            for i in range(1<<(n//2))})
  52	        result['degree_bounds'].append(d)
  53	    for n in (4,8):
  54	        result['exact_interpolation'].append(interpolation_audit(n))
  55	    result['degree_eight_rational_example']={
  56	        'excluded_rational_degrees':[rational_interpolation_rank(8,3,d)['max_degree']
  57	                                     for d in range(3,8) if rational_interpolation_rank(8,3,d)['no_rational_solution_proved']],
  58	        'exact_solution_degree':rational_interpolant(8,3,8)['degree'],
  59	        'all_boolean_cases_verified':rational_interpolant(8,3,8)['all_values_verified'],
  60	        'denominator_nonzero_on_all_samples':True}
  61	    return result
  62	
  63	if __name__=='__main__':
  64	    data=run()
  65	    with open('audit_results.json','w') as f:json.dump(data,f,indent=2)
  66	    print('orbit sizes, minimal measurements:', [(x['N'],x['minimal_count']) for x in data['minimal_representatives']])
  67	    print('exact interpolation:', data['exact_interpolation'])
  68	    print('checks PASS')
