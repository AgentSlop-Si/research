  23 def model(log2_N:int,epsilon:float=0.05,cap_fraction:float=0.5)->Accounting:

   1 """Independent finite-field verification of grouped FFT stage identity.

   4 mod p, so no approximate/floating-point equality enters this audit.
   5 """

   7 P=998244353

  10 def reverse(i,b):

  12     for _ in range(b): o=(o<<1)|(i&1);i>>=1
  13     return o

  15 def iterative_stage_group(a,s,t):
  16     a=list(a); n=len(a); inv2=pow(2,-1,P)
  17     for h in range(s+1,s+t+1):
  18         span=1<<h;half=span//2
  19         root=pow(GEN,(P-1)//span,P)
  20         for base in range(0,n,span):
  21             w=1
  22             for j in range(half):
  23                 u=a[base+j];v=a[base+j+half]*w%P
  24                 a[base+j]=(u+v)*inv2%P
  25                 a[base+j+half]=(u-v)*inv2%P
  26                 w=w*root%P
  27     return a
  28 
  29 def reused_template_stage_group(a,s,t):
  30     a=list(a);n=len(a);K=1<<t;J=1<<s;M=J*K
  31     kroot=pow(GEN,(P-1)//K,P)
  32     mroot=pow(GEN,(P-1)//M,P)
  33     kinv=pow(K,-1,P)
  34     res=a[:]
  35     for base in range(0,n,M):
  36         for j in range(J):
  37             z=[0]*K
  38             for r in range(K):
