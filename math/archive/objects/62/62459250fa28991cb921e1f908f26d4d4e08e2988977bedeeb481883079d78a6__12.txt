   1	"""Many-consumer relation-aware Fourier store.
   2	
   3	All consumers share one producer-side relation state
   4	
   5	    x = sum_i c_i b_i + e,
   6	
   7	while each consumer owns only a subset of a fixed union of Fourier bins.  A
   8	coefficient or point update is recorded once, O(1).  Consumer v with q_v bins
   9	and lag H_v chooses replay O(q_v H_v) or rebuild O(q_v(r+k)).
  10	
  11	A bounded global event log prevents dormant consumers from retaining unbounded
  12	history: once their history is gone they rebuild exactly from current state.
  13	"""
  14	from __future__ import annotations
  15	from dataclasses import dataclass
  16	from collections import deque
  17	from typing import Mapping, Sequence, Iterable
  18	import numpy as np
  19	
  20	
  21	@dataclass
  22	class ViewState:
  23	    name:str
  24	    indices:np.ndarray
  25	    bins:np.ndarray
  26	    cached:np.ndarray
  27	    version:int
  28	    strategy:str='initial'
  29	    backlog:int=0
  30	    query_count:int=0
  31	    basis_cache:np.ndarray|None=None
  32	    phase_cache:np.ndarray|None=None
  33	
  34	
  35	def union_frequency_sets(view_bins:Mapping[str,Sequence[int]],n:int):
  36	    if not view_bins: raise ValueError('need at least one view')
  37	    normalized={}
  38	    all_bins=[]
  39	    for name,bins in view_bins.items():
  40	        b=np.asarray(bins,dtype=np.int64)
  41	        if b.ndim!=1 or b.size<1 or np.any(b<0) or np.any(b>=n): raise ValueError(f'bad bins for {name}')
  42	        normalized[str(name)]=b.astype(np.intp,copy=True); all_bins.extend(map(int,b))
  43	    union=np.asarray(sorted(set(all_bins)),dtype=np.intp)
  44	    loc={int(v):i for i,v in enumerate(union)}
  45	    indices={name:np.asarray([loc[int(v)] for v in b],dtype=np.intp) for name,b in normalized.items()}
  46	    return union,normalized,indices

  49	class SharedProjectedFourierStore:
  50	    def __init__(self,n:int,union_bins:Sequence[int],basis_projection:np.ndarray,
  51	                 view_bins:Mapping[str,Sequence[int]],coefficients=None,
  52	                 sparse_residual:Iterable[tuple[int,complex]]=(),*,
  53	                 prepared_positions:Sequence[int]|None=None,event_capacity:int|None=None,
  54	                 promote_after:int|None=8):
  55	        self.n=int(n)
  56	        if self.n<1:raise ValueError('n must be positive')
  57	        ub=np.asarray(union_bins,dtype=np.int64)
  58	        if ub.ndim!=1 or ub.size<1 or np.any(ub<0) or np.any(ub>=self.n):raise ValueError('bad union bins')
  59	        if len(set(map(int,ub)))!=len(ub):raise ValueError('union bins must be unique')
  60	        self.union_bins=ub.astype(np.intp,copy=True); self.u=len(ub)
  61	        self._union_loc={int(v):i for i,v in enumerate(self.union_bins)}
  62	        proj=np.asarray(basis_projection,dtype=np.complex128)
  63	        if proj.ndim!=2 or proj.shape[1]!=self.u:raise ValueError('basis_projection must be (r,u)')
  64	        self.projection=np.ascontiguousarray(proj); self.r=proj.shape[0]
  65	        if coefficients is None:self.coefficients=np.zeros(self.r,dtype=np.complex128)
  66	        else:
  67	            c=np.asarray(coefficients,dtype=np.complex128)
  68	            if c.shape!=(self.r,):raise ValueError('bad coefficients')
  69	            self.coefficients=c.copy()
  70	        self.sparse={}
  71	        for j,d in sparse_residual:self._point_update(int(j),complex(d),log=False)
  72	        self._prepared_row={}; self._prepared_phase=None
  73	        if prepared_positions is not None:
  74	            pp=list(dict.fromkeys(int(x) for x in prepared_positions))
  75	            if any(j<0 or j>=self.n for j in pp):raise ValueError('bad prepared position')
  76	            self._prepared_row={j:i for i,j in enumerate(pp)}
  77	            self._prepared_phase=np.exp((-2j*np.pi/self.n)*np.outer(np.asarray(pp,float),self.union_bins.astype(float)))
  78	        self.version=0; self.events=deque(); self.event_capacity=None if event_capacity is None else max(0,int(event_capacity))
  79	        self.promote_after=None if promote_after is None else max(1,int(promote_after))
  80	        self.views={}
  81	        for name,b in view_bins.items():self.add_view(str(name),b)
  82	
  83	    def _indices_for(self,bins):
  84	        try:return np.asarray([self._union_loc[int(v)] for v in bins],dtype=np.intp)
  85	        except KeyError as e:raise ValueError(f'view bin {e.args[0]} not in registered union')
  86	
  87	    def _phase_view(self,j:int,idx:np.ndarray):
  88	        row=self._prepared_row.get(j)
  89	        if row is not None and self._prepared_phase is not None:return self._prepared_phase[row,idx]
  90	        return np.exp((-2j*np.pi*j/self.n)*self.union_bins[idx].astype(float))
  91	
  92	    def _rebuild(self,idx):
  93	        y=np.asarray(self.coefficients@self.projection[:,idx],dtype=np.complex128)
  94	        for j,d in self.sparse.items():y += d*self._phase_view(j,idx)
  95	        return y
  96	
  97	    def promote_view(self,name:str):
  98	        v=self.views[name]
  99	        if v.basis_cache is None:
 100	            v.basis_cache=np.ascontiguousarray(self.projection[:,v.indices])
 101	            # Keep prepared point phases shared at union scope; duplicating them
 102	            # per hot view can erase the memory advantage of the shared store.
 103	
 104	    def _maybe_promote(self,v:ViewState):
 105	        if self.promote_after is not None and v.basis_cache is None and v.query_count>=self.promote_after:
 106	            self.promote_view(v.name)
 107	
 108	    def _phase_for_view(self,j:int,v:ViewState):
 109	        row=self._prepared_row.get(j)
 110	        if row is not None and v.phase_cache is not None:return v.phase_cache[row]
 111	        if row is not None and self._prepared_phase is not None:return self._prepared_phase[row,v.indices]
 112	        return np.exp((-2j*np.pi*j/self.n)*v.bins.astype(float))
 113	
 114	    def _rebuild_view(self,v:ViewState):
 115	        bp=v.basis_cache if v.basis_cache is not None else self.projection[:,v.indices]
 116	        y=np.asarray(self.coefficients@bp,dtype=np.complex128)
 117	        for j,d in self.sparse.items():y += d*self._phase_for_view(j,v)
 118	        return y
 119	
 120	    def query_many(self,names):
 121	        """Answer several views, fusing them when a union rebuild is cheaper.
 122	
 123	        The cost proxy compares independent optimal replay/rebuild work against
 124	        one exact rebuild on the union of all requested bins.  This captures
 125	        cross-consumer overlap without forcing a permanently materialized union
 126	        spectrum.
 127	        """
 128	        names=list(dict.fromkeys(str(x) for x in names))
 129	        if not names:return {}
 130	        if len(names)==1:return {names[0]:self.query(names[0])}
 131	        d=self.r+len(self.sparse)
 132	        indiv=0
 133	        for name in names:
 134	            v=self.views[name];H=self.version-v.version
 135	            if H==0:continue
 136	            indiv += len(v.indices)*min(H,max(1,d))
 137	        batch_idx=np.asarray(sorted(set(int(i) for name in names for i in self.views[name].indices)),dtype=np.intp)
 138	        batch_cost=len(batch_idx)*max(1,d)
 139	        # When everyone is already current, simply return caches.
 140	        if indiv==0:return {name:self.views[name].cached.copy() for name in names}
 141	        if batch_cost<=indiv:
 142	            y=self._rebuild(batch_idx)
 143	            pos={int(idx):j for j,idx in enumerate(batch_idx)}
 144	            out={}
 145	            for name in names:
 146	                v=self.views[name]
 147	                take=np.asarray([pos[int(i)] for i in v.indices],dtype=np.intp)
 148	                v.cached=np.ascontiguousarray(y[take]);v.backlog=self.version-v.version;v.version=self.version;v.strategy='batch_rebuild'
 149	                v.query_count+=1; self._maybe_promote(v)
 150	                out[name]=v.cached.copy()
 151	            return out
 152	        return {name:self.query(name) for name in names}
 153	
 154	    def add_view(self,name:str,bins:Sequence[int]):
 155	        if name in self.views:raise ValueError('view already exists')
 156	        b=np.asarray(bins,dtype=np.intp); idx=self._indices_for(b)
 157	        self.views[name]=ViewState(name,idx,b.copy(),self._rebuild(idx),self.version)
 158	
 159	    def drop_view(self,name:str):
 160	        self.views.pop(name)
 161	        self.compact()
 162	
 163	    def _append(self,kind,key,delta):
 164	        self.version+=1; self.events.append((self.version,kind,int(key),complex(delta)))
 165	        if self.event_capacity is not None:
 166	            while len(self.events)>self.event_capacity:self.events.popleft()
 167	
 168	    def add_coefficient(self,i:int,delta:complex):
 169	        z=complex(delta)
 170	        if z==0j:return
 171	        i=int(i); self.coefficients[i]+=z; self._append('c',i,z)
 172	
 173	    def _point_update(self,j:int,d:complex,*,log=True):
 174	        if not 0<=j<self.n:raise ValueError('point outside signal')
 175	        if d==0j:return
 176	        v=self.sparse.get(j,0j)+d
 177	        if v==0j:self.sparse.pop(j,None)
 178	        else:self.sparse[j]=v
 179	        if log:self._append('p',j,d)
 180	
 181	    def add_point(self,j:int,delta:complex):self._point_update(int(j),complex(delta),log=True)
 182	
 183	    def _history_available(self,v:ViewState):
 184	        if v.version==self.version:return True
 185	        return bool(self.events) and self.events[0][0]<=v.version+1
 186	
 187	    def query(self,name:str):
 188	        v=self.views[name]; v.query_count+=1; self._maybe_promote(v); H=self.version-v.version
 189	        if H==0:
 190	            v.strategy='cached';v.backlog=0;return v.cached.copy()
 191	        d=self.r+len(self.sparse)
 192	        if H>max(1,d) or not self._history_available(v):
 193	            v.cached=self._rebuild_view(v);v.strategy='rebuild';v.backlog=H;v.version=self.version
 194	            return v.cached.copy()
 195	        for ver,kind,key,delta in self.events:
 196	            if ver<=v.version:continue
 197	            if kind=='c':
 198	                row=v.basis_cache[key] if v.basis_cache is not None else self.projection[key,v.indices]
 199	                v.cached += delta*row
 200	            else:v.cached += delta*self._phase_for_view(key,v)
 201	        v.strategy='replay';v.backlog=H;v.version=self.version
 202	        return v.cached.copy()
 203	
 204	    def compact(self):
 205	        if not self.views:self.events.clear();return
 206	        floor=min(v.version for v in self.views.values())
 207	        while self.events and self.events[0][0]<=floor:self.events.popleft()
 208	
 209	    @property
 210	    def storage_bytes(self):
 211	        phase=0 if self._prepared_phase is None else self._prepared_phase.nbytes
 212	        views=sum(v.indices.nbytes+v.bins.nbytes+v.cached.nbytes +
 213	                  (0 if v.basis_cache is None else v.basis_cache.nbytes) +
 214	                  (0 if v.phase_cache is None else v.phase_cache.nbytes) for v in self.views.values())
 215	        return int(self.projection.nbytes+self.coefficients.nbytes+phase+views+len(self.sparse)*24)
