0001 import numpy as np,json,time
0002 results=[]
0003 for m in range(2,10):
0004  n=1<<m; p=np.array([int(f'{j:0{m}b}'[::-1],2) for j in range(n)])
0005  best=(-10,None)
0006  for t in range(1,n):
0007   c=np.r_[0,np.cumsum(p<t)]
0008   for w in range(1,n):
0009    z=c[w:]-c[:-w]
0010    costs=np.full(n-w+1,2);costs[0]=costs[-1]=1
0011    lo=int(np.argmin(z+costs));hi=int(np.argmax(z-costs))
0012    # rank deficit to boundary-aware rank cap = maxdiff - (bI+bJ+1)
0013    deficit=int(z[hi]-z[lo]-costs[hi]-costs[lo]-1)
0014    if deficit>best[0]: best=(deficit,dict(n=n,t=t,w=w,I=[hi,hi+w],J=[lo,lo+w],delta=int(z[hi]-z[lo]),budgets=[int(costs[hi]),int(costs[lo])]))
0015  results.append(best);print(m,best,flush=True)
