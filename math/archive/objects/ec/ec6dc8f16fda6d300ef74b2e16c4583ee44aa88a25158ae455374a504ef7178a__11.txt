   9 def specified_cycles(n,q,seed):
  10     # Reconstruct the documented seeded input instance, not uploaded code.
  11     rng=Random(seed+17*n);orders=[];successors=[]
  12     while len(orders)<q:
  13         order=list(range(n));rng.shuffle(order)
  14         successor=[0]*n
  15         for i,v in enumerate(order):successor[v]=order[(i+1)%n]
  16         slope=(successor[1]-successor[0])%n
  17         affine=all(successor[x]==(slope*x+successor[0])%n for x in range(n))
  18         if affine or successor in successors:continue
  19         if any(all(successor[t[x]]==t[successor[x]] for x in range(n)) for t in successors):continue
  20         orders.append(order);successors.append(successor)
  21     return orders

  24 def entries(order,alpha,columns,p):
  25     n=len(order);pos=[0]*n
  26     for i,v in enumerate(order):pos[v]=i
  27     assert alpha%p and (1-pow(alpha,n,p))%p
  28     den=pow((1-pow(alpha,n,p))%p,-1,p)
  29     powers=[pow(alpha,i,p)*den%p for i in range(n)]
  30     return [[powers[(pos[i]-pos[j])%n] for j in columns] for i in range(n)]

  57 def five_term(n):

  59     cycles=specified_cycles(n,5,320031)
  60     e=[entries(c,7+2*s,list(range(6)),p) for s,c in enumerate(cycles)]
  61     left=[];jacobian=[];residual=[];last=[]
  62     for i in range(n):
  63         b=[[e[s][i][j] for s in range(5)] for j in range(5)]
  64         bi=field.inverse(b)
  65         rhs=[pow(w,-i*j,p) for j in range(5)]
  66         l=[sum(bi[s][j]*rhs[j] for j in range(5))%p for s in range(5)]
  67         assert all(l)
  68         assert all(sum(b[j][s]*l[s] for s in range(5))%p==rhs[j] for j in range(5))
  69         h=[e[s][i][5] for s in range(5)]
  70         z=[sum(h[s]*bi[s][ell] for s in range(5))%p for ell in range(5)]
  71         row=[-z[ell]*b[ell][s]*l[s]%p for ell in range(1,5) for s in range(5)]
  72         row += [h[s]*l[s]%p for s in range(5)]
  73         jacobian.append(row);left.append(l);last.append([h[s]*l[s]%p for s in range(5)])
  74         residual.append((pow(w,-5*i,p)-sum(h[s]*l[s] for s in range(5)))%p)

  76     assert all(right)

  88 def separated_groups(orders):
  89     n=len(orders[0]);m=n//2
  90     left=set(orders[0][:m]);right=set(orders[0][-m:])
  91     for order in orders[1:]:
  92         union=[v for v in order if v in left or v in right]
  93         lower=set(union[:m]);upper=set(union[m:])
  94         candidates=[(left&lower,right&upper),(left&upper,right&lower)]
  95         a,b=max(candidates,key=lambda pair:min(len(pair[0]),len(pair[1])))
  96         m//=2;left=set(sorted(a)[:m]);right=set(sorted(b)[:m])
  97     return sorted(left),sorted(right)
