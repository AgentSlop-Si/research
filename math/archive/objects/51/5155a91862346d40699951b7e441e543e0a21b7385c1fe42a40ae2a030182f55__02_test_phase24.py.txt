import sys
import unittest
import itertools
import random
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from phase24_ordering import (stable_binary_order, successor_of_order,
    cycle_rank_difference,transitions_noncyclic,predicted_binary_rank,
    binary_rank_probability,permuted_geometric_via_green_woodbury,
    direct_permuted_geometric_reference)
from phase23_resolvent import RealGreenFourier

class Phase24Tests(unittest.TestCase):
    def test_exhaustive_rank_formula_through_n16(self):
        for n in range(2,17):
            for t in range(1<<n):
                bits=[(t>>j)&1 for j in range(n)]
                order=stable_binary_order(bits)
                succ=successor_of_order(order)
                self.assertEqual(cycle_rank_difference(succ),predicted_binary_rank(bits),
                                 (n,t,bits))

    def test_exact_distribution_up_to_n16(self):
        for n in range(2,17):
            counts={r:0 for r in range(0,n,2)}
            for t in range(1<<n):
                bits=[(t>>j)&1 for j in range(n)]
                counts[predicted_binary_rank(bits)]+=1
            for r,c in counts.items():
                num,den=binary_rank_probability(n,r)
                self.assertEqual(c*den,num*(1<<n))

    def test_exact_mean_variance_up_to_n16(self):
        for n in range(3,17):
            vals=[]
            for t in range(1<<n):
                bits=[(t>>j)&1 for j in range(n)]
                vals.append(predicted_binary_rank(bits))
            size=len(vals)
            self.assertEqual(sum(vals),size*(n/2-1))
            self.assertEqual(sum((v-(n/2-1))**2 for v in vals),size*n/4)

    def test_random_all_stride_ranks(self):
        rng=random.Random(2409)
        for n in (32,64,128,256):
            for _ in range(4):
                bits=[rng.randrange(2) for _ in range(n)]
                succ=successor_of_order(stable_binary_order(bits))
                vals=[cycle_rank_difference(succ,a) for a in range(1,n)]
                self.assertGreaterEqual(min(vals),n//8,
                                        (n,min(vals),vals.index(min(vals))+1))

    def test_woodbury_one_swap_and_two_swaps(self):
        rng=random.Random(77)
        for n in (8,16,32,64):
            eng=RealGreenFourier.create(n)
            for swaps in (1,2):
                for _ in range(3):
                    order=list(range(n))
                    for _ in range(swaps):
                        a,b=rng.sample(range(n),2)
                        order[a],order[b]=order[b],order[a]
                    for z in (2,3,5):
                        if pow(z,n,998244353)==1:continue
                        actual=permuted_geometric_via_green_woodbury(order,z,eng)
                        expected=direct_permuted_geometric_reference(order,z,eng)
                        self.assertEqual(actual,expected,(n,swaps,z,order))

    def test_woodbury_identity(self):
        for n in (8,16,32):
            eng=RealGreenFourier.create(n)
            got=permuted_geometric_via_green_woodbury(list(range(n)),7,eng)
            ref=direct_permuted_geometric_reference(list(range(n)),7,eng)
            self.assertEqual(got,ref)

    def test_woodbury_stable_single_run(self):
        for n in (8,16,32):
            eng=RealGreenFourier.create(n)
            for k in (0,1,n//2,n-1,n):
                bits=[0]*k+[1]*(n-k)
                order=stable_binary_order(bits)
                for z in (2,11):
                    got=permuted_geometric_via_green_woodbury(order,z,eng)
                    ref=direct_permuted_geometric_reference(order,z,eng)
                    self.assertEqual(got,ref)

if __name__=='__main__':unittest.main()
