   6 def hword(n, offset=0):
   7     out = []
   8     bit = 1
   9     while bit < n:
  10         out += [(offset+i, offset+i+bit) for i in range(n) if not i & bit]
  11         bit *= 2
  12     return out
  13 
  14 
  15 def haarword(n, offset=0):
  16     if n == 1:
  17         return []
  18     m = n // 2
  19     return [(offset+i, offset+i+m) for i in range(m)] + haarword(m, offset)
  20 
  21 
  22 def detailword(n):
  23     out = []
  24     size = 2
  25     while size < n:
  26         out += hword(size, size)
  27         size *= 2
  28     return out

  31 def replay(n, word):
  32     rows = [[int(i == j) for j in range(n)] for i in range(n)]
  33     for a, b in word:
  34         u, v = rows[a], rows[b]
  35         rows[a] = [x+y for x, y in zip(u, v)]
  36         rows[b] = [x-y for x, y in zip(u, v)]
  37     return rows

  40 def expected_h(n):
  41     return [[(-1)**((i & j).bit_count()) for j in range(n)] for i in range(n)]
  42 
  43 
  44 def expected_d(n):
  45     rows = [[0]*n for _ in range(n)]
  46     rows[0][0] = rows[1][1] = 1
  47     size = 2
  48     while size < n:
  49         h = expected_h(size)
  50         for i in range(size):
  51             for j in range(size):
  52                 rows[size+i][size+j] = h[i][j]
  53         size *= 2
  54     return rows
