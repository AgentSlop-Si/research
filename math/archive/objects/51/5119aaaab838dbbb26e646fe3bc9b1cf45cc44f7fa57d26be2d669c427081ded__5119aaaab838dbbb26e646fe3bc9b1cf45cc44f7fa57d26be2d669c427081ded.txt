import importlib.util, json
from pathlib import Path
p=Path(__file__).with_name('check_joint_orbit.py')
s=importlib.util.spec_from_file_location('independent_local_checker',p);a=importlib.util.module_from_spec(s);s.loader.exec_module(a)
def rank(A,p):
 A=[row[:] for row in A];r=0
 for c in range(len(A[0])):
  pivot=next((i for i in range(r,len(A)) if A[i][c]),None)
  if pivot is None:continue
  A[r],A[pivot]=A[pivot],A[r];z=pow(A[r][c],-1,p);A[r]=[v*z%p for v in A[r]]
  for i in range(len(A)):
   if i!=r and A[i][c]:
    z=A[i][c];A[i]=[(x-z*y)%p for x,y in zip(A[i],A[r])]
  r+=1
 return r
out=[]
for m in range(2,45):
 p,o,L,w=a.field_for(m);eta=pow(o,m,p);mat=[]
 for h in range(m):
  q=o*pow(eta,-h,p)%p;F=1;row=[]
  for r in range(m):
   if r:F=F*(1-pow(q,r,p))%p
   row.append(pow(q,r*(r-1)//2,p)*pow(F,-1,p)%p)
  mat.append(row)
 out.append({'M':m,'p':p,'rank':rank(mat,p)})
print(json.dumps(out,indent=2))
Path(__file__).with_name('KERNEL_RANK_CONTROLS.json').write_text(json.dumps(out,indent=2)+'\n')
