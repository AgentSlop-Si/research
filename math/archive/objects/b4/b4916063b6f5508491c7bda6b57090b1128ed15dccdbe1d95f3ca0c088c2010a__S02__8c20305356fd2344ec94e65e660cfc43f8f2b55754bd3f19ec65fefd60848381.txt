  16 def divisors(n: int) -> list[int]:
  17     if n <= 0:
  18         raise ValueError('n must be positive')
  19     dset = set()
  20     for k in range(1, isqrt(n) + 1):
  21         if n % k == 0:
  22             dset.update((k, n // k))
  23     return sorted(dset)

  26 def optimal_divisor(n: int) -> int:
  27     """Subgroup order yielding minimum endpoint incidence."""
  28     return min(divisors(n), key=lambda d: (d + n // d, d))

  31 def optimal_tile_cost(n: int) -> int:
  32     d = optimal_divisor(n)
  33     return n * (d + n // d)

  36 def row_subgroup_order(rows: Sequence[int], n: int) -> int:
  37     """Size of subgroup generated by differences of listed row indices."""
  38     if not rows:
  39         raise ValueError('empty row set')
  40     base = rows[0] % n
  41     step = n
  42     for r in rows[1:]:
  43         step = gcd(step, (r - base) % n)
  44     return n // step

  47 def rank_one_rectangle(rows: Sequence[int], cols: Sequence[int], n: int) -> bool:
  48     if not rows or not cols:
  49         return False
  50     if len(set(rows)) != len(rows) or len(set(cols)) != len(cols):
  51         raise ValueError('indices must be distinct')
  52     if any(not 0 <= r < n for r in rows) or any(not 0 <= c < n for c in cols):
  53         raise ValueError('indices outside 0..n-1')
  54     k0, j0 = rows[0], cols[0]
  55     return all(((k-k0)*(j-j0)) % n == 0 for k in rows for j in cols)
  56

  58 def rectangle_envelope(rows: Sequence[int], cols: Sequence[int], n: int) -> dict:
  59     """Return the maximal coset rectangle forced by the row-difference group.
  60 
  61     For any rank-one block R x C, R subset (k0+H), C subset(j0+H^perp),
  62     with |H|=d, |H^perp|=n/d, and |R||C|<=n.
  63     """
  64     if not rank_one_rectangle(rows, cols, n):
  65         raise ValueError('not a rank-one Fourier rectangle')
  66     d = row_subgroup_order(rows, n)
  67     q = n // d
  68     return {'row_coset': [(rows[0] + q * j) % n for j in range(d)],
  69             'col_coset': [(cols[0] + d * j) % n for j in range(q)],
  70             'row_capacity': d, 'col_capacity': q,
  71             'area': len(rows)*len(cols), 'max_area': n}

  74 def subgroup_tiles(n: int, d: int | None = None) -> list[tuple[tuple[int, ...], tuple[int, ...]]]:
  75     if d is None:
  76         d = optimal_divisor(n)
  77     if d not in divisors(n):
  78         raise ValueError('d must divide n')
  79     q = n // d
  80     return [(tuple(r + q*v for v in range(d)),
  81              tuple(c + d*w for w in range(q)))
  82             for r in range(q) for c in range(d)]

  85 def verify_partition(n: int, tiles: Sequence[tuple[Sequence[int], Sequence[int]]]) -> None:
  86     hit = [0]*(n*n)
  87     for rows, cols in tiles:
  88         assert rank_one_rectangle(rows, cols, n)
  89         for k in rows:
  90             for j in cols:
  91                 hit[k*n+j] += 1
  92     assert all(x == 1 for x in hit), 'not an exact disjoint rectangle tiling'

  99 def is_prime(k: int) -> bool:
 100     if k < 2:
 101         return False
 102     for d in range(2, isqrt(k)+1):
 103         if k%d == 0:
 104             return False
 105     return True

 108 def prime_factors(k: int) -> set[int]:
 109     ans = set()
 110     d = 2
 111     while d*d <= k:
 112         if k%d == 0:
 113             ans.add(d)
 114             while k%d == 0:
 115                 k //= d
 116         d += 1
 117     if k>1: ans.add(k)
 118     return ans

 121 def test_field(n: int) -> tuple[int, int]:
 122     """Find prime p and supplied primitive n-th root w for test fixtures.
 123 
 124     Field discovery is an offline fixture step, NOT part of fast executor.
 125     """
 126     pf = prime_factors(n)
 127     for m in range(1,5000):
 128         p = m*n+1
 129         if not is_prime(p):
 130             continue
 131         for g in range(2,p):
 132             w = pow(g,(p-1)//n,p)
 133             if all(pow(w,n//q,p)!=1 for q in pf):
 134                 return p,w
 135     raise RuntimeError('no test fixture field')

 151 def roots_table(n: int, w: int, p: int) -> tuple[list[int], int]:
 152     roots=[1]*n
 153     for i in range(1,n):
 154         roots[i]=(roots[i-1]*w)%p
 155     return roots,n-1

 158 def direct_dft(x: Sequence[int], p: int, w: int) -> list[int]:
 159     n=len(x)
 160     roots,_=roots_table(n,w,p)
 161     return [sum(x[j]*roots[(-k*j)%n] for j in range(n))%p for k in range(n)]

 164 def one_pass_tiled_dft(x: Sequence[int], p: int, w: int, d: int | None = None) -> tuple[list[int], Ledger]:
 165     """Exact direct contraction of N independent rank-one subgroup tiles.
 166 
 167     Per tile, q input products + (q-1) adds build a scalar, followed by
 168     d output products+adds. No power calls inside loops; all root lookup
 169     preparation is included in ledger.
 170     """
 171     n=len(x)
 172     if d is None: d=optimal_divisor(n)
 173     if n%d: raise ValueError('d must divide n')
 174     q=n//d
 175     powers, prep=roots_table(n,w,p)
 176     ledger=Ledger(root_muls=prep)
 177     output=[0]*n
 178     for r in range(q):
 179         for c in range(d):
 180             acc=0
 181             for j in range(q):
 182                 in_index=c+d*j
 183                 v=(x[in_index]*powers[(-r*d*j)%n])%p
 184                 ledger.muls+=1;ledger.input_read_occurrences+=1
 185                 if j: ledger.adds+=1
 186                 acc=(acc+v)%p
 187             for v in range(d):
 188                 out_index=r+q*v
 189                 e=(-r*c-q*v*c)%n
 190                 contribution=acc*powers[e]%p
 191                 output[out_index]=(output[out_index]+contribution)%p
 192                 ledger.muls+=1;ledger.adds+=1
 193     ledger.output_writes=n
 194     return output,ledger

 197 def radix2_fft(a: Sequence[int], powers: Sequence[int], p:int) -> tuple[list[int], int]:
 198     """Standard radix-2 scalar circuit, counted butterflies, n | len(powers)."""
 199     n=len(a)
 200     N=len(powers)
 201     if n<=0 or (n&(n-1)) or N%n:
 202         raise ValueError('power-of-2 length divisor required')
 203     if n==1:
 204         return [a[0]%p],0
 205     ev,be=radix2_fft(a[0::2],powers,p)
 206     od,bo=radix2_fft(a[1::2],powers,p)
 207     out=[0]*n
 208     for k in range(n//2):
 209         tw=od[k]*powers[(-k*N//n)%N]%p
 210         out[k]=(ev[k]+tw)%p
 211         out[k+n//2]=(ev[k]-tw)%p
 212     return out,be+bo+n//2

 215 def shared_two_axis_dft(x: Sequence[int],p:int,w:int,d:int|None=None) -> tuple[list[int], Ledger]:
 216     """Charge both short Fourier families: classical CT, not new faster FFT."""
 217     n=len(x)
 218     if d is None:d=optimal_divisor(n)
 219     q=n//d
 220     if d<=0 or n%d or (d&(d-1)) or (q&(q-1)):
 221         raise ValueError('d and n/d must be powers of two')
 222     powers,prep=roots_table(n,w,p)
 223     ledger=Ledger(root_muls=prep)
 224     S=[]
 225     for c in range(d):
 226         a,b=radix2_fft([x[c+d*j] for j in range(q)],powers,p)
 227         S.append(a)
 228         ledger.butterflies+=b
 229         ledger.input_read_occurrences+=q
 230     out=[0]*n
 231     for r in range(q):
 232         vals=[]
 233         for c in range(d):
 234             vals.append(S[c][r]*powers[(-r*c)%n]%p)
 235             ledger.muls+=1
 236         row,b=radix2_fft(vals,powers,p)
 237         ledger.butterflies+=b
 238         for v in range(d):
 239             out[r+q*v]=row[v]
 240     ledger.muls+=ledger.butterflies
 241     ledger.adds+=2*ledger.butterflies
 242     ledger.output_writes=n
 243     return out,ledger

 246 def enumerate_rankone_tiles(n:int)->list[tuple[tuple[int,...],tuple[int,...],int,int]]:
 247     """Full subset enumeration for small n; independent small-instance MILP."""
 248     if n>8:raise ValueError('small-instance exhaustive checker only')
 249     subsets=[tuple(j for j in range(n) if mask & (1<<j))
 250              for mask in range(1,1<<n)]
 251     return [(I,J,len(I)+len(J),sum(1<<(r*n+c) for r in I for c in J))
 252             for I in subsets for J in subsets if rank_one_rectangle(I,J,n)]
