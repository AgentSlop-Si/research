0001 /* Optional local memory/undefined-behavior check.
0002  * gcc -O1 -g -std=c11 -fsanitize=address,undefined -fno-omit-frame-pointer \
0003  *     test_sanitizers.c -lm -o test_sanitizers && ./test_sanitizers
0004  */
0005 #include "leaf_fft.c"
0006 #include <stdio.h>
0007 int main(void) {
0008     cx *tables[3]={fft_table_create(4),fft_table_create(8),fft_table_create(16)};
0009     for(int i=0;i<3;++i) if(!tables[i]) return 2;
0010     double worst=0;
0011     for(size_t n=16;n<=1048576;n*=2) {
0012         plan *p=fft_plan_create(n);
0013         uint8_t *x=malloc(n);cx *reference=malloc(n*sizeof(cx)),*y=malloc(n*sizeof(cx));
0014         if(!p||!x||!reference||!y) return 2;
0015         uint32_t state=123456789;
0016         for(size_t j=0;j<n;++j) {state^=state<<13;state^=state>>17;state^=state<<5;x[j]=(uint8_t)(state&1);}
0017         fft_radix2(p,x,reference);
0018         for(int method=0;method<7;++method) {
0019             fft_execute(method,p,x,y,method>=4?tables[method-4]:NULL);
0020             for(size_t j=0;j<n;++j) {
0021                 double err=hypot(y[j].r-reference[j].r,y[j].i-reference[j].i);
0022                 if(err>worst) worst=err;
0023                 if(err>1e-9) {fprintf(stderr,"Mismatch N=%zu method=%d bin=%zu error=%g\n",n,method,j,err);return 1;}
0024             }
0025         }
0026         free(x);free(reference);free(y);fft_plan_destroy(p);
0027     }
0028     for(int i=0;i<3;++i)fft_table_destroy(tables[i]);
0029     printf("Address/undefined-behavior sanitizer check passed; max method difference %.12g\n",worst);
0030     return 0;
0031 }
