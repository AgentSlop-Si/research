"""Exact finite-field counterexample to naive three-stage rank propagation."""
from minor_certificate import rank_mod


def witness(n):
    assert n%2==0
    m=n//2
    A=[[0]*n for _ in range(n)]
    for i in range(0,n,2):
        A[i][i]=1; A[i][i+1]=1
        A[i+1][i]=1; A[i+1][i+1]=2
    upper=[]; lower=[]
    for k in range(1,n):
        upper.append(rank_mod([row[k:] for row in A[:k]]))
        lower.append(rank_mod([row[:k] for row in A[k:]]))
    ev=list(range(0,n,2)); od=list(range(1,n,2))
    four={
      f'{label_r},{label_c}':rank_mod([[A[i][j] for j in c] for i in r])
      for label_r,r in [('E',ev),('O',od)]
      for label_c,c in [('E',ev),('O',od)]
    }
    assert max(upper+lower)<=1
    assert all(v==m for v in four.values())
    return {'N':n,'max_upper_cut_rank':max(upper), 'max_lower_cut_rank':max(lower),
            'four_arbitrary_half_ranks':four, 'half':m}

if __name__=='__main__':
    import json
    print(json.dumps([witness(n) for n in (8,16,32,64)],indent=2))
