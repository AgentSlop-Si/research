  48	def eye(n): return [[int(i==j) for j in range(n)] for i in range(n)]
  49	def mm(a,b,p):
  50	    bt=list(zip(*b))
  51	    return [[sum(x*y for x,y in zip(row,col))%p for col in bt] for row in a]
  52	def mv(a,x,p): return [sum(v*w for v,w in zip(row,x))%p for row in a]
  53	def minus(a,b,p): return [[(x-y)%p for x,y in zip(ra,rb)] for ra,rb in zip(a,b)]
  54	def rank(a,p):
  55	    if not a: return 0
  56	    a=[list(row) for row in a];r=0
  57	    for c in range(len(a[0])):
  58	        k=next((k for k in range(r,len(a)) if a[k][c]%p),None)
  59	        if k is None: continue
  60	        a[k],a[r]=a[r],a[k]
  61	        inv=pow(a[r][c],-1,p)
  62	        a[r]=[v*inv%p for v in a[r]]
  63	        for k in range(r+1,len(a)):
  64	            scale=a[k][c]
  65	            a[k]=[(x-scale*y)%p for x,y in zip(a[k],a[r])]
  66	        r+=1
  67	        if r==len(a): break
  68	    return r

  70	def inverse(a,p):
  71	    n=len(a);m=[list(row)+e for row,e in zip(a,eye(n))]
  72	    for c in range(n):
  73	        k=next((k for k in range(c,n) if m[k][c]%p),None)
  74	        if k is None: return None
  75	        m[k],m[c]=m[c],m[k];v=pow(m[c][c],-1,p)
  76	        m[c]=[x*v%p for x in m[c]]
  77	        for k in range(n):
  78	            if k!=c:
  79	                v=m[k][c];m[k]=[(x-v*y)%p for x,y in zip(m[k],m[c])]
  80	    return [row[n:] for row in m]

  82	def root(n,p):
  83	    return next(w for w in range(2,p) if pow(w,n,p)==1 and len({pow(w,j,p) for j in range(n)})==n)

  85	def fourier(n,p,w): return [[pow(w,i*j,p) for j in range(n)] for i in range(n)]
  86	def perm(pi): return [[int(j==pi[i]) for j in range(len(pi))] for i in range(len(pi))]
  87	def orbit(pi):
  88	    o=[];j=0;seen=set()
  89	    while j not in seen: seen.add(j);o.append(j);j=pi[j]
  90	    assert j==0 and len(o)==len(pi)
  91	    return o

  93	def weighted_system(pi,coeff,p):
  94	    m=eye(len(pi))
  95	    for i,j in enumerate(pi): m[i][j]=(m[i][j]-coeff[i])%p
  96	    return m

  98	def closed_cycle(pi,d,rhs,p):
  99	    # Direct expanded numerator; recover forward by division. Zero edges use
 100	    # independent dense elimination, solely a reference for the zero case.
 101	    if not all(d):
 102	        inv=inverse(weighted_system(pi,d,p),p)
 103	        return None if inv is None else mv(inv,rhs,p)
 104	    order=orbit(pi);numerator=0;weight=1
 105	    for i in order:
 106	        numerator=(numerator+weight*rhs[i])%p
 107	        weight=weight*d[i]%p
 108	    if weight==1:return None
 109	    z=[0]*len(pi);z[0]=numerator*pow(1-weight,-1,p)%p
 110	    for i in order[:-1]:z[pi[i]]=(z[i]-rhs[i])*pow(d[i],-1,p)%p
 111	    assert (z[order[-1]]-d[order[-1]]*z[0]-rhs[order[-1]])%p==0
 112	    return z

 114	def fft_reference(x,w,p):
 115	    # Recursive decimation, independently coded.
 116	    n=len(x)
 117	    if n==1:return list(x)
 118	    e=fft_reference(x[::2],w*w%p,p);o=fft_reference(x[1::2],w*w%p,p)
 119	    lo=[];hi=[];tw=1
 120	    for a,b in zip(e,o):
 121	        v=tw*b%p;lo.append((a+v)%p);hi.append((a-v)%p);tw=tw*w%p
 122	    return lo+hi
