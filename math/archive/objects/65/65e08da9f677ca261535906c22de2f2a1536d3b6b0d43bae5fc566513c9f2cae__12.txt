   1 """Research prototype: two real FFTs per complex FFT, packed convolution, and cross-input DAG sharing.
   2 
   3 Uses numpy for accuracy / benchmark reference and optional optimized FFTs.
   4 The DAG portion counts radix-2 complex butterflies, not total time, and uses
   5 hash-consing of exact recursive subproblems. This does NOT beat n log n for
   6 arbitrary complex/real Fourier transforms.
   7 
   8 Run: python fft_packed_dag_research.py
   9 """

  22 def two_real_ffts_one_complex(a, b):
  23     """Recover the separate length-N complex FFTs of two real-valued length-N arrays."""
  24     a = np.asarray(a, dtype=np.float64)
  25     b = np.asarray(b, dtype=np.float64)
  26     if a.shape != b.shape or a.ndim != 1:
  27         raise ValueError('a and b must be real one-dimensional arrays of equal length')
  28     n = len(a)
  29     if not n:
  30         raise ValueError('n must be positive')
  31     z = np.fft.fft(a + 1j * b)
  32     partner = np.conj(z[(-np.arange(n)) % n])
  33     fa = (z + partner) / 2
  34     fb = (z - partner) / (2j)
  35     return fa, fb

  38 def packed_convolution(a, b):
  39     """Real linear convolution from one forward complex FFT and one inverse FFT.
  40 
  41     Zero-pad so circular convolution agrees with linear convolution.
  42     Imag(ifft(fft(a + 1j*b)**2))/2 = convolution(a,b).
  43     """
  44     a = np.asarray(a, dtype=np.float64)
  45     b = np.asarray(b, dtype=np.float64)
  46     if a.ndim != 1 or b.ndim != 1 or not len(a) or not len(b):
  47         raise ValueError('nonempty one-dimensional real arrays required')
  48     target = len(a) + len(b) - 1
  49     n = 1 << (target - 1).bit_length()
  50     a_padded = np.pad(a, (0, n - len(a)))
  51     b_padded = np.pad(b, (0, n - len(b)))
  52     z = np.fft.fft(a_padded + 1j * b_padded)
  53     return (np.fft.ifft(z * z).imag[:target]) / 2

  56 class JointFFTGraph:
  57     """Hash-conses identical FFT subproblems ACROSS arbitrary input arrays.
  58 
  59     Canonical keys are exact, not tolerant approximations. Two separate real
  60     FFTs can share subgraphs (and results) without packing into one complex
  61     array. For all-real inputs the same complex radix-2 butterfly code is used.
  62     """
  63     def __init__(self):
  64         self.intern = {}
  65         self.nodes = []
  66         self.values = {}
  67         self.twiddles = {}
  68         self.butterflies = 0
  69         self.build_visits = 0
  70 
  71     def build(self, a, off=0, stride=1, n=None):
  72         if n is None:
  73             n = len(a)
  74             if not n or n & (n - 1):
  75                 raise ValueError('length must be a positive power of two')
  76         self.build_visits += 1
  77         if n == 1:
  78             v = complex(a[off])
  79             key = ('leaf', v)
  80             node = (1, -1, -1, v)
  81         else:
  82             left = self.build(a, off, stride * 2, n // 2)
  83             right = self.build(a, off + stride, stride * 2, n // 2)
  84             key = ('internal', n, left, right)
  85             node = (n, left, right, None)
  86         ident = self.intern.get(key)
  87         if ident is None:
  88             ident = len(self.nodes)
  89             self.intern[key] = ident
  90             self.nodes.append(node)
  91         return ident
  92 
  93     def eval(self, ident):
  94         if ident in self.values:
  95             return self.values[ident]
  96         n, lid, rid, v = self.nodes[ident]
  97         if n == 1:
  98             out = (v,)
  99         else:
 100             ev = self.eval(lid)
 101             od = self.eval(rid)
 102             if n not in self.twiddles:
 103                 self.twiddles[n] = [cmath.exp(-2j * math.pi * k / n) for k in range(n//2)]
 104             out = [0j] * n
 105             for k in range(n // 2):
 106                 t = self.twiddles[n][k] * od[k]
 107                 out[k] = ev[k] + t
 108                 out[k+n//2] = ev[k] - t
 109                 self.butterflies += 1
 110             out = tuple(out)
 111         self.values[ident] = out
 112         return out
 113 
 114     def eval_inputs(self, a, b):
 115         ra = self.build(a)
 116         rb = self.build(b)
 117         return self.eval(ra), self.eval(rb)
 118 
 119     def summary(self):
 120         return {
 121             'butterflies': self.butterflies,
 122             'unique_nodes': len(self.nodes),
 123             'unique_by_size': dict(sorted(Counter(x[0] for x in self.nodes).items())),
 124             'cached_values': sum(len(x) for x in self.values.values()),
 125             'build_visits': self.build_visits,
 126         }

 129 def packed_dag(a, b):
 130     """One hash-consed complex FFT; split two real spectra in O(N)."""
 131     if len(a) != len(b):
 132         raise ValueError('same input lengths required')
 133     z_input = [complex(x, y) for x, y in zip(a, b)]
 134     graph = JointFFTGraph()
 135     root = graph.build(z_input)
 136     z = graph.eval(root)
 137     n = len(a)
 138     out_a, out_b = [], []
 139     for k in range(n):
 140         mate = z[(-k) % n].conjugate()
 141         out_a.append((z[k] + mate) / 2)
 142         out_b.append((z[k] - mate) / (2j))
 143     return tuple(out_a), tuple(out_b), graph.summary()

 146 def find_cyclic_shift(a, b):
 147     """Return right-shift s such that b[j] == a[(j-s)%n], or None.
 148 
 149     KMP search, O(n) comparisons and O(n) extra memory. Exact equality.
 150     """
 151     if len(a) != len(b) or not a:
 152         return None
 153     n = len(a)
 154     pi = [0] * n
 155     j = 0
 156     for i in range(1, n):
 157         while j and b[i] != b[j]:
 158             j = pi[j-1]
 159         if b[i] == b[j]:
 160             j += 1
 161         pi[i] = j
 162     j = 0
 163     for i in range(2*n-1):
 164         value = a[i % n]
 165         while j and value != b[j]:
 166             j = pi[j-1]
 167         if value == b[j]:
 168             j += 1
 169         if j == n:
 170             start = i-n+1
 171             return (-start) % n
 172     return None

 175 def shifted_fft_from_reference(fa, shift):
 176     """FFT(np.roll(a, shift)) from FFT(a), with O(n) arithmetic."""
 177     n = len(fa)
 178     phase_step = cmath.exp(-2j * math.pi * shift / n)
 179     phase = 1+0j
 180     result = []
 181     for val in fa:
 182         result.append(val * phase)
 183         phase *= phase_step
 184     return tuple(result)

 187 def tests():
 188     rng = np.random.default_rng(20261008)
 189     comparisons = 0
 190     for n in (1, 2, 3, 4, 8, 16, 32, 64, 256):
 191         for _ in range(7):
 192             a = rng.integers(-9, 10, n, dtype=np.int64)
 193             b = rng.integers(-9, 10, n, dtype=np.int64)
 194             fa, fb = two_real_ffts_one_complex(a, b)
 195             assert np.allclose(fa, np.fft.fft(a), rtol=1e-12, atol=1e-10)
 196             assert np.allclose(fb, np.fft.fft(b), rtol=1e-12, atol=1e-10)
 197             comparisons += 1
 198     convchecks = 0
 199     for la, lb in [(1,1),(1,7),(2,3),(4,4),(7,9),(16,16),(31,5),(39,63)]:
 200         for _ in range(4):
 201             a = rng.integers(-9, 10, la)
 202             b = rng.integers(-9, 10, lb)
 203             actual = packed_convolution(a,b)
 204             expected = np.convolve(a, b)
 205             assert np.allclose(actual, expected, atol=1e-8), (a,b,actual,expected)
 206             convchecks += 1
 207     assert np.rint(packed_convolution([4,3,2,1], [8,7,6,5])).astype(int).tolist() == [32,52,61,60,34,16,5]
 208     assert sum(j*10**i for i,j in enumerate(np.rint(packed_convolution([4,3,2,1],[8,7,6,5])).astype(int))) == 7006652
 209     graphchecks = 0
 210     for n in (2, 4, 8, 16, 32, 64, 128):
 211         for _ in range(5):
 212             a = rng.integers(0,2,n).tolist()
 213             b = rng.integers(0,2,n).tolist()
 214             joint = JointFFTGraph()
 215             fa, fb = joint.eval_inputs(a,b)
 216             fpa, fpb, _ = packed_dag(a,b)
 217             for actual in (fa, fpa):
 218                 assert np.allclose(actual,np.fft.fft(a),atol=1e-8)
 219             for actual in (fb, fpb):
 220                 assert np.allclose(actual,np.fft.fft(b),atol=1e-8)
 221             graphchecks += 1
 222     # Verify autodetected low-rank basis and rank>2 rejection.
 223     b0=rng.integers(-4,5,67).astype(float)
 224     b1=rng.integers(-4,5,67).astype(float)
 225     co=rng.integers(-5,6,(11,2)).astype(float)
 226     family=co[:,0,None]*b0+co[:,1,None]*b1
 227     spectra,mode=adaptive_rank_two_fft(family)
 228     assert mode=='rank2' and np.allclose(spectra, np.fft.fft(family,axis=1),atol=1e-8)
 229     assert detect_rank_two_real_family(np.stack([rng.standard_normal(67) for _ in range(3)])) is None
 230     shifts = 0
 231     for n in (1, 2, 4, 8, 16, 32, 127, 128):
 232         for _ in range(12):
 233             a=rng.integers(-50,51,n).tolist()
 234             s=int(rng.integers(0,n))
 235             b=np.roll(a,s).tolist()
 236             recovered=find_cyclic_shift(a,b)
 237             assert recovered is not None and np.array_equal(np.roll(a,recovered), b), (n,s,recovered)
 238             assert np.allclose(shifted_fft_from_reference(np.fft.fft(a), recovered),np.fft.fft(b),atol=1e-8)
 239             shifts+=1
 240     print(f'TESTS PASSED: packed spectra={comparisons}, packed convolutions={convchecks}, graph FFT pairs={graphchecks}, KMP shifts={shifts}, digit product 1234*5678=7006652')

 267 def run_timing(fn, reps=9):
 268     for _ in range(3): fn()
 269     times=[]
 270     for _ in range(reps):
 271         t=time.perf_counter_ns(); fn(); times.append((time.perf_counter_ns()-t)/1e6)
 272     return statistics.median(times)

 275 def benchmark_numpy_convolutions():
 276     print('NUMPY_TIMES_MS: N two_complex_fft_plus_ifft one_packed_complex_fft_plus_ifft two_real_rffts_plus_irfft')

 318 def detect_rank_two_real_family(signals, tol=1e-9):
 319     """Discover 0/1/2 real-valued basis vectors, and recover coefficients.
 320 
 321     Return (basis, coeffs) if rank <=2, else None. O(M*N) numeric work
 322     for this fixed rank, including verification. Not numerically robust
 323     for extremely ill-conditioned inputs; intended as a research demo.
 324     """
 325     X = np.asarray(signals, dtype=np.float64)
 326     if X.ndim != 2 or not np.isfinite(X).all():
 327         raise ValueError('must be finite real 2D data')
 328     m, n = X.shape
 329     bases=[]
 330     coeffs=[]
 331     for row in X:
 332         if not bases:
 333             if np.max(np.abs(row)) <= tol:
 334                 coeffs.append([])
 335                 continue
 336             bases=[row.copy()]
 337             # retroactively initialize earlier zero rows
 338             coeffs=[[0.] for _ in coeffs]
 339             p0=int(np.argmax(np.abs(bases[0])))
 340             coeffs.append([1.])
 341             continue
 342         if len(bases) == 1:
 343             alpha=row[p0]/bases[0][p0]
 344             if np.max(np.abs(row-alpha*bases[0])) <= tol*(1+np.max(np.abs(row))):
 345                 coeffs.append([float(alpha)])
 346                 continue
 347             bases.append(row.copy())
 348             coeffs=[c+[0.] for c in coeffs]
 349             coeffs.append([0.,1.])
 350             determinants=bases[0][p0]*bases[1]-bases[0]*bases[1][p0]
 351             p1=int(np.argmax(np.abs(determinants)))
 352             det=determinants[p1]
 353             if abs(det) <= tol:
 354                 return None
 355             continue
 356         det = bases[0][p0]*bases[1][p1]-bases[0][p1]*bases[1][p0]
 357         alpha=(row[p0]*bases[1][p1]-row[p1]*bases[1][p0])/det
 358         beta=(bases[0][p0]*row[p1]-bases[0][p1]*row[p0])/det
 359         predicted=alpha*bases[0]+beta*bases[1]
 360         if np.max(np.abs(predicted-row)) > tol*(1+np.max(np.abs(row))):
 361             return None
 362         coeffs.append([float(alpha),float(beta)])
 363     if not bases:
 364         return np.zeros((0,n)),np.zeros((m,0))
 365     return np.stack(bases),np.asarray(coeffs,dtype=np.float64)

 368 def adaptive_rank_two_fft(signals):
 369     """Find 2-basis structure and reuse spectra, or fall back to regular FFT."""
 370     found=detect_rank_two_real_family(signals)
 371     if found is None:
 372         return np.fft.fft(signals,axis=1), 'fallback'
 373     basis, coeffs=found
 374     if len(basis)==0:
 375         return np.zeros_like(signals,dtype=np.complex128),'zero'
 376     return coeffs @ np.fft.fft(basis,axis=1), f'rank{len(basis)}'

 379 def low_rank_family_experiment():
 380     """If M arrays live in span of r fixed vectors, use r FFTs + O(M r N) output.
 381 
 382     Signals' low-rank representation is known; discovery/verification is NOT timed.
 383     """
 384     print('LOW_RANK_MS: N M r independent_batched_fft known_basis_2_ffts autodetect_verify_2_basis_ffts')
 385     rng=np.random.default_rng(1008)
 386     for n in (4096,16384,65536):
 387         m=32; r=2
 388         basis = rng.standard_normal((r,n))
 389         coeffs = rng.integers(-3,4,(m,r))
 390         signals = coeffs @ basis
 391         assert signals.shape == (m,n)
 392         def direct(): return np.fft.fft(signals,axis=1)
 393         def reconstructed():
 394             f = np.fft.fft(basis,axis=1)
 395             return coeffs @ f
 396         assert np.allclose(direct(),reconstructed(),rtol=1e-10,atol=1e-7)
 397         t1=run_timing(direct,7)
 398         t2=run_timing(reconstructed,7)
 399         t3=run_timing(lambda: adaptive_rank_two_fft(signals),7)
 400         print(f'{n} {m} {r} {t1:.4f} {t2:.4f} {t3:.4f}')
