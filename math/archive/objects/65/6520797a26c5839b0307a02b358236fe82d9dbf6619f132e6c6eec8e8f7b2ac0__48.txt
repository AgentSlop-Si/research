  37 class UnitStepChirpTests(unittest.TestCase):
  38     def test_prime_power_validation(self) -> None:
  39         self.assertEqual(prime_power(1024), (2, 10))
  40         self.assertEqual(prime_power(729), (3, 6))
  41         self.assertEqual(prime_power(625), (5, 4))
  42         with self.assertRaises(ValueError):
  43             prime_power(12)
  44         self.assertEqual(validate_dimensions(256, 8), (2, 32, 8))
  45         self.assertEqual(validate_dimensions(729, 9), (3, 81, 9))

  47     def test_balanced_scan(self) -> None:
  48         rng = np.random.default_rng(1)
  49         for n in [1, 2, 3, 8, 31, 64, 257]:
  50             x = rng.normal(size=n) + 1j * rng.normal(size=n)
  51             got = balanced_exclusive_scan(x)
  52             ref = np.concatenate([[0.0 + 0.0j], np.cumsum(x)[:-1]])
  53             self.assertLess(float(np.max(np.abs(got - ref))), 3e-12)
  54 
  55     def test_decoder_matches_materialized_projection(self) -> None:
  56         rng = np.random.default_rng(2)
  57         for n, m in [(64, 4), (81, 3), (125, 5), (256, 8), (729, 9)]:
  58             for _ in range(3):
  59                 x = rng.normal(size=n) + 1j * rng.normal(size=n)
  60                 state = draw_state(n, rng)
  61                 d = compress_blocks(x, m, state)
  62                 ref = direct_row_estimate(x, m, state)
  63                 bal = decode_balanced_scan(d, n, m, state)
  64                 seq = decode_sequential(d, n, m, state)
  65                 self.assertLess(float(np.max(np.abs(bal - ref))), 3e-10)
  66                 self.assertLess(float(np.max(np.abs(seq - ref))), 3e-10)

  68     def test_unit_character_identity(self) -> None:
  69         for n in [8, 9, 25, 27]:
  70             p, _ = prime_power(n)
  71             units = [a for a in range(n) if math.gcd(a, n) == 1]
  72             zeta = np.exp(2j * np.pi / n)
  73             for c in range(n):
  74                 empirical = sum(zeta ** (a * c) for a in units) / len(units)
  75                 theory = unit_character_mean_prime_power(n, c)
  76                 self.assertLess(abs(empirical - theory), 2e-12)

  78     def test_exhaustive_covariance_small(self) -> None:
  79         # Exhaustively enumerate every unit a and every beta and compare the
  80         # resulting output-0 covariance with the closed formula.
  81         for n, m in [(8, 2), (9, 3)]:
  82             units = [a for a in range(1, n) if math.gcd(a, n) == 1]
  83             q = np.zeros((n, n), dtype=np.complex128)
  84             eye = np.eye(n, dtype=np.complex128)
  85             exact = unitary_fft(eye[:, 0])  # only used for shape/convention
  86             del exact
  87             for a in units:
  88                 for beta in range(n):
  89                     state = ChirpState(a, beta)
  90                     h = np.empty(n, dtype=np.complex128)
  91                     for j in range(n):
  92                         row = direct_row_estimate(eye[:, j], m, state)
  93                         h[j] = row[0] - 1.0 / math.sqrt(n)
  94                     q += np.outer(np.conj(h), h)
  95             q /= len(units) * n
  96             ref = covariance_matrix_formula(n, m, 0)
  97             self.assertLess(float(np.max(np.abs(q - ref))), 3e-12)

  99     def test_covariance_is_scaled_projector(self) -> None:
 100         for n, m in [(16, 2), (16, 4), (27, 3), (81, 9), (125, 5)]:
 101             q = covariance_matrix_formula(n, m)
 102             eig = np.linalg.eigvalsh(q)
 103             zero_mult, nonzero_mult, lam = covariance_projector_spectrum(n, m)
 104             self.assertEqual(zero_mult + nonzero_mult, n)
 105             self.assertLess(float(np.max(np.abs(eig[:zero_mult]))), 2e-11)
 106             self.assertLess(float(np.max(np.abs(eig[zero_mult:] - lam))), 2e-11)
 107             self.assertLess(float(np.linalg.norm(q @ q - lam * q)), 2e-10)
 108             self.assertAlmostEqual(float(np.trace(q).real), m - 1, places=10)
 109 
 110     def test_output_covariance_spectrum_invariant(self) -> None:
 111         n, m = 81, 9
 112         ref = np.linalg.eigvalsh(covariance_matrix_formula(n, m, 0))
 113         for output in [1, 2, 17, 80]:
 114             got = np.linalg.eigvalsh(covariance_matrix_formula(n, m, output))
 115             self.assertLess(float(np.max(np.abs(got - ref))), 2e-11)

 117     def test_explicit_extremizer(self) -> None:
 118         for n, m in [(64, 4), (81, 3), (125, 5)]:
 119             q = covariance_matrix_formula(n, m)
 120             x = worst_case_signal(n, m)
 121             value = float(np.vdot(x, q @ x).real)
 122             self.assertAlmostEqual(value, m / n, places=11)

 124     def test_unbiasedness_empirical(self) -> None:
 125         rng = np.random.default_rng(3)
 126         n, m = 81, 3
 127         x = rng.normal(size=n) + 1j * rng.normal(size=n)
 128         target = unitary_fft(x)
 129         acc = np.zeros(n, dtype=np.complex128)
 130         trials = 3000
 131         for _ in range(trials):
 132             acc += one_row_estimate(x, m, draw_state(n, rng))
 133         # Monte Carlo tolerance is deliberately loose; exact unbiasedness is
 134         # proved algebraically and the exhaustive covariance test is stronger.
 135         self.assertLess(float(np.max(np.abs(acc / trials - target))), 0.18 * np.linalg.norm(x))
 136 
 137     def test_finite_precision_bridge(self) -> None:
 138         rng = np.random.default_rng(4)
 139         x = rng.normal(size=4096) + 1j * rng.normal(size=4096)
 140         m, g = 16, 4.0
 141         u = np.finfo(np.float32).eps / 2
 142         eta = finite_precision_row_bound(x, m, u)
 143         exact = exact_arithmetic_uniform_bound(x, m, g)
 144         inflated = finite_precision_uniform_bound(x, m, g, u)
 145         self.assertGreater(eta, 0.0)
 146         self.assertGreater(inflated, exact)
 147         bits = sufficient_mantissa_bits(2**30, g)
 148         self.assertLess(bits, 64)
 149         self.assertGreater(bits, 30)

 153     def test_seed_from_cycle_closure_all_prime_powers(self) -> None:
 154         rng = np.random.default_rng(2026)
 155         for n, m in [(64, 4), (81, 3), (125, 5), (256, 8), (729, 9)]:
 156             x = rng.normal(size=n) + 1j * rng.normal(size=n)
 157             state = draw_state(n, rng)
 158             d = compress_blocks(x, m, state)
 159             d_fft = np.fft.fft(d)
 160             from prototype import _cycle_arrays, pairwise_sum, seed_value
 161             _, gauge, terms = _cycle_arrays(d_fft, n, m, state)
 162             closure_seed = pairwise_sum(terms) / 2.0
 163             direct_seed = seed_value(d, n, m, state)
 164             self.assertLess(abs(closure_seed - direct_seed), 2e-10)
 165             # A_N/A_0 is the omitted endpoint of the gauge cycle and equals -1.
 166             h = n & 1
 167             endpoint_phase_2n = (2 * (state.a * (n * (n - 1) // 2) + state.beta * n) + h * n) % (2*n)
 168             endpoint = np.exp(1j * np.pi * endpoint_phase_2n / n)
 169             self.assertLess(abs(endpoint + 1), 2e-12)

 171     def test_exact_combinatorial_certificate(self) -> None:

 177     def test_composite_projector_shortcut_is_false(self) -> None:
 178         n, m = 12, 3
 179         eig = np.linalg.eigvalsh(composite_covariance(n, m))
 180         positive = eig[eig > 1e-10]
 181         self.assertGreater(float(positive.max()), m / n + 0.1)
 182         self.assertGreater(float(positive.max() - positive.min()), 0.2)

 184     def test_probability_and_asymptotics(self) -> None:
 185         n, g, delta = 2**20, 4.0, 1e-3
 186         rows = required_rows(n, g, delta)
 187         self.assertEqual(rows % 2, 1)
 188         self.assertLessEqual(majority_failure_bound(n, rows, g), delta)
 189         # One row is linear when M is a power of p comparable to log N.
 190         for n, m in [(2**24, 32), (3**15, 27), (5**10, 25)]:
 191             _, k, _ = validate_dimensions(n, m)
 192             ratio = (n + k * math.log2(k)) / (n * math.log2(n))
 193             self.assertLess(ratio, 0.2)
