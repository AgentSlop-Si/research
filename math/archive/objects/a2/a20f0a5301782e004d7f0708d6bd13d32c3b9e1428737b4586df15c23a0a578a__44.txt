   3 class Bits:

   5     def __init__(self,words,n,w):

   7         self.mask=(1<<w)-1

  10     @classmethod

  13     def cut(self,start,n):

  15         for p in range(start,start+n,w):
  16             j,r=divmod(p,w);z=(self.a[j]>>r) if j<len(self.a) else 0
  17             if r and j+1<len(self.a):z|=(self.a[j+1]<<(w-r))&self.mask
  18             out.append(z);Bits.work+=4
  19         if n%w:out[-1]&=(1<<(n%w))-1
  20         return Bits(out,n,w)
  21     def put(self,start,b):
  22         # Each b word enters at most two output words, masking each output.
  23         w=self.w
  24         for i,z in enumerate(b.a):
  25             length=min(w,b.n-i*w);j,r=divmod(start+i*w,w)
  26             lo=min(length,w-r);lm=((1<<lo)-1)<<r

  29                 hm=(1<<(length-lo))-1
  30                 self.a[j+1]=(self.a[j+1]&~hm)|((z>>lo)&hm)
  31             Bits.work+=8
  32     def xor(self,b):

  35     def right(self,k):

  38         return z
  39     def left(self,k):

  44 def to_y(f,m,k,h):
  45     if m==k:return f
  46     H=1<<(m-1);R=1<<(m-1-k)
  47     low=f.cut(0,H*h);high=f.cut(H*h,H*h)
  48     q=high.xor(high.right((H-R)*h));rem=low.xor(q.left(R*h))
  49     out=Bits([0]*len(f.a),f.n,f.w)
  50     out.put(0,to_y(rem,m-1,k,h));out.put(H*h,to_y(q,m-1,k,h))
  51     return out

  53 def convert(f,m,h=1):
  54     if m<=1:return f
  55     k=1<<((m-1).bit_length()-1);D=1<<k
  56     upper=convert(to_y(f,m,k,h),m-k,h*D)
  57     out=Bits([0]*len(f.a),f.n,f.w)
  58     for i in range(1<<(m-k)):
  59         z=convert(upper.cut(i*h*D,h*D),k,h);out.put(i*h*D,z)
  60     return out

  62 def polymul(a,b):
  63     z=0
  64     while b:
  65         if b&1:z^=a
  66         a<<=1;b>>=1
  67     return z

  69 def basis_polys(m):
