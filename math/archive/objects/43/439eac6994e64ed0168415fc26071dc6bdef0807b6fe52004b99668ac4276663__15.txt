  14 def prime_trial(n):

  20 def det_fraction_free_mod(rows, p):
  21     """Bareiss recurrence evaluated in the prime field, with row pivoting."""
  22     a = [r[:] for r in rows]
  23     assert a and all(len(r) == len(a) for r in a)
  24     n = len(a)
  25     previous, sign = 1, 1
  26     for k in range(n - 1):
  27         pivot_row = next((i for i in range(k, n) if a[i][k]), None)
  28         if pivot_row is None:
  29             return 0
  30         if pivot_row != k:
  31             a[k], a[pivot_row] = a[pivot_row], a[k]
  32             sign *= -1
  33         pivot = a[k][k]
  34         inv_previous = pow(previous, -1, p)
  35         for i in range(k + 1, n):
  36             left = a[i][k]
  37             for j in range(k + 1, n):
  38                 a[i][j] = (pivot * a[i][j] - left * a[k][j]) * inv_previous % p
  39             a[i][k] = 0
  40         previous = pivot
  41     return sign * a[-1][-1] % p

  44 def certificate_check(data):
  45     results = []
  46     for key, source in data['conjugate_pair_cubic_certificates'].items():
  47         n = int(key)
  48         root = pow(3, (PRIME - 1) // n, PRIME)
  49         assert pow(root, n, PRIME) == 1 and pow(root, n // 2, PRIME) != 1
  50         known = [i for i in range(n) if i not in (1, n - 1)]
  51         monomials = [m for d in (1, 2, 3)
  52                      for m in combinations_with_replacement(known, d) if sum(m) % n == 1]
  53         supports = source['supports']
  54         assert len({tuple(s) for s in supports}) == len(supports)
  55         assert all(0 <= len(s) <= 3 and s == sorted(set(s)) and all(0 <= j < n for j in s)
  56                    for s in supports)
  57         assert len(supports) == len(monomials) + 1
  58         rows = []
  59         for support in supports:
  60             bins = [sum(pow(root, k*j, PRIME) for j in support) % PRIME for k in range(n)]

  62             for monomial in monomials:
  63                 value = 1

  66                 row.append(value)
  67             rows.append(row + [bins[1]])

  76 # Exact cyclotomic arithmetic for dyadic N: Z[z]/(z^(N/2)+1).
  77 def zpower(k, half):
  78     k %= 2 * half
  79     return tuple((1 if k < half else -1) if i == k % half else 0 for i in range(half))

  90 def mul(a, b):
  91     half = len(a)
  92     out = [0] * half
  93     for i, x in enumerate(a):
  94         for j, y in enumerate(b):
  95             k = i + j
  96             out[k % half] += x * y * (1 if k < half else -1)
  97     return tuple(out)

 100 def spectrum(x):
 101     n = len(x)
 102     out = []
 103     for k in range(n):
 104         s = (0,) * (n // 2)
 105         for j, bit in enumerate(x):
 106             if bit:
 107                 s = add(s, zpower(k*j, n//2))
 108         out.append(s)
 109     return out

 112 def cubic_control():
 113     count = 0
 114     for bits in product((0, 1), repeat=8):
 115         x = spectrum(bits)
 116         q = add(add(mul(x[1], x[6]), mul(x[2], x[5])), mul(x[3], x[4]))
 117         r = add(add(mul(mul(x[1], x[1]), x[5]), mul(x[1], mul(x[3], x[3]))),
 118                 mul(mul(x[2], x[2]), x[3]))
 119         rhs = add(scale(-1, mul(add(x[0], (-4, 0, 0, 0)), q)), scale(-1, r))
 120         rhs = add(rhs, scale(-2, mul(mul(x[1], x[2]), x[4])))
 121         assert scale(4, x[7]) == rhs
 122         count += 1
 123     return {'N': 8, 'all_binary_inputs_verified': count, 'ring': 'Z[z]/(z^4+1)'}

 144 def lagrange_coeffs(omit, coordinate):
 145     coefs = [F(0)] * 3
 146     for t in range(4):
 147         if t == omit:
 148             continue
 149         value = F((t % 2) if coordinate == 0 else (t // 2))
 150         others = [u for u in range(4) if u not in (omit, t)]
 151         a, b = others
 152         denom = (t-a)*(t-b)
 153         for degree, c in enumerate((a*b, -a-b, 1)):
 154             coefs[degree] += value*c/denom
 155     return coefs

 158 def sharp_quadratic_gate():
