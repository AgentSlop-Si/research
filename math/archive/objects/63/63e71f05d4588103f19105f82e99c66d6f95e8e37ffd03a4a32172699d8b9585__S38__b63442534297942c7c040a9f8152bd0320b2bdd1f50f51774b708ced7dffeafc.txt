  11 def rank_mod(rows,p=P):
  12     if not rows:return 0
  13     m=[list(row) for row in rows]
  14     h=len(m);w=len(m[0]);i=0
  15     for j in range(w):
  16         piv=next((t for t in range(i,h) if m[t][j]%p),None)
  17         if piv is None:continue
  18         m[i],m[piv]=m[piv],m[i]
  19         z=pow(m[i][j],p-2,p)
  20         for jj in range(j,w):m[i][jj]=m[i][jj]*z%p
  21         for t in range(i+1,h):
  22             a=m[t][j]
  23             if a:
  24                 for jj in range(j,w):m[t][jj]=(m[t][jj]-a*m[i][jj])%p
  25         i+=1
  26         if i==h:break
  27     return i

  30 class Phase10Tests(unittest.TestCase):
  31     def test_ntt_standalone_against_direct_dft(self):
  32         r=random.Random(11)
  33         for n in (1,2,4,8,16,32,64):
  34             for _ in range(3):
  35                 x=[r.randrange(P) for _ in range(n)]
  36                 self.assertEqual(ntt(x,True),direct_full_dft(x))
  37                 self.assertEqual(ntt(ntt(x,True),False),x)

  39     def test_transpose_circulant_fourier_identity(self):
  40         r=random.Random(73)
  41         for k in (4,8,16,32):
  42             for _ in range(3):
  43                 lam=[r.randrange(P) for _ in range(k)]
  44                 w=[r.randrange(P) for _ in range(k)]
  45                 phi=transpose_circulant_action(w,lam)
  46                 fw=ntt(w,True)
  47                 self.assertEqual(ntt(phi,True),[fw[i]*lam[-i%k]%P for i in range(k)])
  48                 # bilinear transpose identity checked independently
  49                 v=[r.randrange(P) for _ in range(k)]
  50                 Cv=ntt([a*b%P for a,b in zip(ntt(v,True),lam)],False)
  51                 self.assertEqual(sum(a*b for a,b in zip(w,Cv))%P,
  52                                  sum(a*b for a,b in zip(phi,v))%P)

  54     def test_berlekamp_massey_order_and_zero(self):
  55         self.assertEqual(berlekamp_massey([0]*14),[1])
  56         r=random.Random(39)
  57         for d in (1,2,3,4):
  58             for _ in range(12):
  59                 q=[1]+[r.randrange(1,P) for _ in range(d)]
  60                 f=[r.randrange(P) for _ in range(d)]
  61                 for j in range(d,3*d+5):
  62                     f.append(-sum(q[i]*f[j-i] for i in range(1,d+1))%P)
  63                 got=berlekamp_massey(f[:2*d])
  64                 self.assertEqual(len(got),d+1)
  65                 self.assertEqual(got,q)

 106     def test_half_width_generic_fit_does_not_save_work(self):
 107         # With d=M/2 the training step itself transforms ALL M columns.
 108         # Generic sequences usually have recurrence order d on these 2d
 109         # samples. The identity remains exact but savings disappear.
 110         r=random.Random(998)
 111         for k,m,d in ((16,8,4),(32,8,4),(64,16,8)):
 112             x=[r.randrange(P) for _ in range(k*m)]
 113             y,cert,_=phase10_transform(x,m,d,
 114                         rng=random.Random(77),return_certificate=True)
 115             self.assertTrue(cert['accepted'],(k,m,d,cert))
 116             self.assertEqual(y,reference_full_ntt(x))
 117             self.assertEqual(2*d,m)

 119     def test_random_certificate_can_be_fooled_if_vector_known(self):
 120         # A one-vector fingerprint is not deterministic. Construct u!=0 with
 121         # w dot u=0 for the known seeded first vector and corrupt late t=7.
 122         k,m,d=32,8,2
 123         x,_,_=make_spectral_recurrent_example(k,m,d,21)
 124         seed=123
 125         w_rng=random.Random(seed)
 126         w=[w_rng.randrange(P) for _ in range(k)]
 127         i=0
 128         while not w[i]:i+=1
 129         j=i+1
 130         u=[0]*k
 131         u[i]=w[j]
 132         u[j]=(-w[i])%P
 133         for b in range(k):x[b*m+(m-1)]=(x[b*m+(m-1)]+u[b])%P
 134         y,cert,_=phase10_transform(x,m,d,rounds=1,
 135                     rng=random.Random(seed),return_certificate=True)
 136         self.assertTrue(cert['accepted'])
 137         self.assertNotEqual(y,reference_full_ntt(x))
 138         y2,cert2,_=phase10_transform(x,m,d,rounds=2,
 139                     rng=random.Random(seed),return_certificate=True)
 140         self.assertFalse(cert2['accepted'])
 141         self.assertIsNone(y2)

 143     def test_rational_numerator_denominator_direct_equality(self):
 144         # Independently evaluate finite S at all roots (not using fast output).
 145         for k,m,d in ((8,4,2),(16,8,2),(32,8,3)):
 146             x,_,f=make_spectral_recurrent_example(k,m,d,67,force_roots=True)
 147             cols=extract_columns(x,m)
 148             train,q=learn_spectral_recurrences(cols,d)
 149             self.assertTrue(verify_raw_recurrence(cols,q,d,2,random.Random(55)))
 150             y,_=synthesize_from_recurrences(train,q,m,k*m)
 151             ztab=root_table(k*m)
 152             for ell in range(k):
 153                 for j in range(m):
 154                     ix=ell+j*k
 155                     z=ztab[(-ix)%(k*m)]
 156                     expect=sum(f[t][ell]*pow(z,t,P) for t in range(m))%P
 157                     self.assertEqual(y[ix],expect)

 159     def test_polynomial_identity_over_entire_coefficient_range(self):
 160         # Check Q*S has no interior coefficients and boundary coefficients
 161         # equal the head/tail decomposition before evaluating any root.
 162         for k,m,d in ((8,4,2),(16,8,2),(32,8,3),(32,16,4)):
 163             x,_,F=make_spectral_recurrent_example(k,m,d,17,force_roots=True)
 164             train,q=learn_spectral_recurrences(extract_columns(x,m),d)
 165             for ell in range(k):
 166                 f=[F[t][ell] for t in range(m)]
 167                 coeff=q[ell];r=len(coeff)-1
 168                 prod=[0]*(m+r)
 169                 for t in range(m):
 170                     for i in range(r+1):
 171                         prod[t+i]=(prod[t+i]+f[t]*coeff[i])%P
 172                 self.assertEqual(prod[r:m],[0]*(m-r))
 173                 head=[sum(coeff[i]*f[j-i] for i in range(j+1))%P for j in range(r)]
 174                 tail=[sum(coeff[i]*f[m+j-i] for i in range(j+1,r+1))%P
 175                       for j in range(r)]
 176                 self.assertEqual(prod[:r],head)
 177                 self.assertEqual(prod[m:],tail)

 179     def test_phase9_first_order_derived_innovations_are_dense(self):
 180         # Our order-2 inputs have generic, all-frequency, all-position
 181         # innovations relative to Phase9's one-step fitted controller.
 182         k,m,d=32,8,2
 183         x,_,F=make_spectral_recurrent_example(k,m,d,204)
 184         lam=[F[1][ell]*pow(F[0][ell],P-2,P)%P for ell in range(k)]
 185         for t in range(1,m-1):
 186             E=[(F[t+1][ell]-lam[ell]*F[t][ell])%P for ell in range(k)]
 187             e=ntt(E,False)
 188             self.assertTrue(all(z for z in E))
 189             self.assertTrue(all(z for z in e))
 190         self.assertEqual(rank_mod([[F[t][ell] for t in range(m)] for ell in range(k)]),m)
