"""Self-contained prior-checkpoint loader for extracted research package."""
from pathlib import Path
import sys,zipfile
ROOT=Path(__file__).resolve().parent
TARGET=ROOT/'_previous_extracted'/'fft_spectral_determinant_gate_2026-10-09'
if not (TARGET/'adjacent_tail.py').is_file():
 archive=ROOT/'prior_checkpoint_full.zip'
 if not archive.is_file():
  # Development fallback, not used in the distributable research ZIP.
  TARGET=Path('fft_spectral_determinant_gate_2026-10-09')
 else:
  with zipfile.ZipFile(archive) as z:z.extractall(ROOT/'_previous_extracted')
assert (TARGET/'adjacent_tail.py').is_file(),f'checkpoint missing: {TARGET}'
sys.path.insert(0,str(TARGET))
