  11 def log10_probability_sparse(k,m,s):
  12     # Exact probability, conditional on DFT(v0) all nonzero, for uniform input:
  13     # after v0/v1 determine C, each next v[t+1] is uniform; residual uniform.
  14     logs=[log10(comb(k,i))+i*log10(P-1)-k*log10(P) for i in range(min(s,k)+1)]
  15     highest=max(logs)
  16     return (m-2)*(highest+log10(sum(10**(u-highest) for u in logs)))

  19 def witness():
  20     k,m=16,8;n=k*m
  21     rng=random.Random(7427)
  22     innovations={t:[((3*t+2)%k,rng.randrange(1,P))] for t in range(1,m-1)}
  23     x,meta=make_dense_cycle(k,m,innovations,seed=721)
  24     answer,cert,c=exact_dense_cycle_with_innovations(
  25         x,m,1,3,3,Costs(),random.Random(727),True)
  26     assert answer==exact_reference_dft(x)
  27     poly=[[x[b*m+t] for t in range(m)] for b in range(k)]
  28     w=root_of_unity(n)
  29     tab=[pow(w,i,P) for i in range(n)]
  30     dechirped=[[x[b*m+t]*tab[(t*(t-1)//2+m*b*t)%n]%P
  31                  for t in range(m)] for b in range(k)]
  32     mode_occupied=[]
  33     for r in range(m):
  34         exists=False
  35         for b in range(k):
  36             value=sum(dechirped[b][t]*tab[(k*r*t)%n] for t in range(m))%P
  37             if value:exists=True;break
  38         mode_occupied.append(exists)
  39     # H_64 = [x[i+j]], indices 0..126
  40     half=n//2
  41     hankel=[[x[i+j] for j in range(half)] for i in range(half)]
  42     evidence={
  43         'n':n,'k':k,'m':m,'field_prime':P,'seed':721,
  44         'innovation_seed':7427,'all_input_nonzero':all(x),
  45         'kernel_support':sum(v!=0 for v in meta['kernel_c']),
  46         'distinct_eigenvalues':len(set(meta['lambda'])),
  47         'polyphase_column_rank':matrix_rank_mod(poly),
  48         'dechirped_column_rank':matrix_rank_mod(dechirped),
  49         'hankel_rank':matrix_rank_mod(hankel),
  50         'green_phase_modes_present':sum(mode_occupied),
  51         'exact_output_mismatches':sum(a!=b for a,b in zip(answer,exact_reference_dft(x))),
  52         'detected_innovations':cert['detected_innovations'],
  53         'operations':c.total(),
  54         'input_sha256':hashlib.sha256(','.join(map(str,x)).encode()).hexdigest(),
  55         'output_sha256':hashlib.sha256(','.join(map(str,answer)).encode()).hexdigest()
  56     }
  57     assert evidence['polyphase_column_rank']==m
  58     assert evidence['dechirped_column_rank']==m
  59     assert evidence['hankel_rank']==half
  60     assert evidence['green_phase_modes_present']==m
  61     assert evidence['kernel_support']==k
  62     assert evidence['all_input_nonzero']
  63     return evidence

  66 def scaling():
  67     records=[]
  68     for exponent in (8,9,10,11,12,13,14,15,16):
  69         n=1<<exponent
  70         m=1<<int(floor(log2(exponent)))
  71         k=n//m
  72         if k<m:continue
  73         for per_step in (0,1,3):
  74             rng=random.Random(57+n+per_step)
  75             events={t:[(b,rng.randrange(1,P)) for b in rng.sample(range(k),per_step)]
  76                     for t in range(1,m-1)} if per_step else {}
  77             x,_=make_dense_cycle(k,m,events,seed=72+per_step)
  78             start=time.perf_counter()
  79             y,cert,c=exact_dense_cycle_with_innovations(
  80                 x,m,per_step,3,3,Costs(),random.Random(27+per_step),True)
  81             elapsed=time.perf_counter()-start
  82             assert y is not None,(n,m,cert)
  83             assert y==exact_reference_dft(x)
  84             records.append({'n':n,'k':k,'m':m,'errors_per_transition':per_step,
  85                             'total_errors':cert['innovation_total'],
  86                             'mode':cert['mode'],'counted_ops':c.total(),
  87                             'ops_per_n':round(c.total()/n,3),
  88                             'seconds_python':round(elapsed,4),
  89                             'inversions':c.inversions,'ntt_butterflies':c.ntt_butterflies})
  90     return records

 101             for k,m,s in [(16,8,0),(16,8,1),(16,8,2),(32,8,2),
 102                           (128,16,2),(512,16,3)]],

 113     print('PASS: all generated exact output vectors match independent length-N NTT')
