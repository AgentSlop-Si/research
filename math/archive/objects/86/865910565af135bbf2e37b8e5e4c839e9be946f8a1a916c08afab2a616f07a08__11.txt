  13 def rr(A,p,with_transform=False):
  14     A=np.asarray(A,dtype=np.int64).copy()%p
  15     nr,nc=A.shape; T=np.eye(nr,dtype=np.int64); piv=[]; r=0
  16     for c in range(nc):
  17         if r==nr: break
  18         q=next((k for k in range(r,nr) if A[k,c]),None)
  19         if q is None: continue
  20         A[[r,q]]=A[[q,r]]; T[[r,q]]=T[[q,r]]
  21         z=pow(int(A[r,c]),-1,p); A[r]=A[r]*z%p; T[r]=T[r]*z%p
  22         for i in range(nr):
  23             if i!=r and A[i,c]:
  24                 z=int(A[i,c]); A[i]=(A[i]-z*A[r])%p; T[i]=(T[i]-z*T[r])%p
  25         piv.append(c); r+=1
  26     return A,piv,T

  34 def det(A,p):
