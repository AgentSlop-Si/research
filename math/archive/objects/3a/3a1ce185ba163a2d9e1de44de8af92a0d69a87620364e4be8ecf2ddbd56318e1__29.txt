  17 def params(h,q=3):
  18     t=(q-1)//2; v=comb(h,q); rho=comb(h,t)
  19     d=sum(comb(q,s)*comb(h-q,q-s) for s in range(0,q,2) if q-s<=h-q)

  91 # Formal noncommutative-polynomial verification of all eight scalar rows.
  92 # A word ('R','G','x') represents R G x; the dictionary is an exact
  93 # integer linear combination of such words. No numeric values assumed.
  94 def add(a,b,scale=1):
  95     out=dict(a)
  96     for w,c in b.items():
  97         out[w]=out.get(w,0)+scale*c
  98         if not out[w]: del out[w]
  99     return out
 100 def op(symbol,a): return {(symbol,)+w:c for w,c in a.items()}
 101 x0={('x',):1}; y0={('y',):1}; A0={('A',):1}; c0={('c',):1}

 104 y=add(y,op('R',c),-1)
