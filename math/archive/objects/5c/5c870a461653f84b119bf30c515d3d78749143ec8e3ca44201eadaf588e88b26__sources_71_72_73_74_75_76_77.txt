  68 def fft(c, plan, values, inverse=False, traffic=None):

  70     if len(values) != m:
  71         raise ValueError('FFT input length must be 2N')

  76         traffic['permutation_value_writes'] += m

 102 def prepare_orbit(field, n, eta):
 103     c = Counted(field)
 104     plan = fft_plan(c, n, eta)
 105     factors = [c.sub(c.one, root) for root in plan.powers]
 106     v = [c.zero] * (2*n)
 107     # The even orbit's zero coefficient removes dependence on v[0].
 108     v[2] = plan.inverse_n
 109     for r in range(1, n):
 110         v[(2*r+2) % (2*n)] = c.add(
 111             c.mul(factors[2*r], v[2*r]), plan.inverse_n)
 112     # Odd orbit: composition y -> 2*y + b; closure gives y = -b.
 113     b = c.zero
 114     for r in range(n):
 115         b = c.sub(c.mul(factors[2*r+1], b), plan.inverse_n)
 116     v[1] = c.neg(b)
 117     for r in range(n-1):
 118         v[2*r+3] = c.sub(
 119             c.mul(factors[2*r+1], v[2*r+1]), plan.inverse_n)
 120     return KernelPlan(plan, v, dict(c.count), 'orbit')

  39 def transform(x, q, p):
