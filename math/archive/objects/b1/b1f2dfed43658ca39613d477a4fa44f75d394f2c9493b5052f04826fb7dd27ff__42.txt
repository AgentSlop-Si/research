   1 """Reproducible stress test of the exact cut-rank lower bound after endpoint shuffles.
   2 The test does NOT decide existence of low-channel factorizations.
   3 """
   4 import json
   5 from pathlib import Path
   6 from cut_budget import endpoint_certificate,endpoint_patterns,fourier,permute
   7 
   8 def run():
   9     results=[]
  10     for n in (16,32,64):
  11         f=fourier(n);orders=endpoint_patterns(n)
  12         for left,right in [('natural','natural'),('natural','bit_reversal'),
  13                            ('bit_reversal','natural'),('parity','parity'),
  14                            ('bit_reversal','bit_reversal')]:
  15             if n==64 and (left,right) != ('bit_reversal','bit_reversal'):
  16                 continue
  17             c=endpoint_certificate(permute(f,orders[left],orders[right]),998244353)
  18             results.append({'n':n,'output_order':left,'input_order':right,**c})
  19             if left=='natural' or right=='natural':
  20                 assert c['necessary_rank_sum']>=n//4
  21     Path(__file__).with_name('endpoint_stress_results.json').write_text(json.dumps(results,indent=2)+'\n')
  22     for x in results:
  23         print(x['n'], x['output_order'],x['input_order'],
  24               x['necessary_rank_sum'], x['surplus'])
  25     return results
  26 if __name__=='__main__':run()
