  11 def fft(a, inverse=False):
  12     n=len(a); out=list(a); j=0
  13     for i in range(1,n):
  14         bit=n>>1
  15         while j&bit: j^=bit; bit>>=1
  16         j^=bit
  17         if i<j: out[i],out[j]=out[j],out[i]
  18     size=2
  19     while size<=n:
  20         w=pow(PRIMITIVE,(P-1)//size,P)
  21         if inverse:w=pow(w,-1,P)
  22         for start in range(0,n,size):
  23             phase=1
  24             for k in range(size//2):
  25                 u=out[start+k];v=out[start+k+size//2]*phase%P
  26                 out[start+k]=(u+v)%P;out[start+k+size//2]=(u-v)%P
  27                 phase=phase*w%P
  28         size*=2
  29     if inverse:out=[v*pow(n,-1,P)%P for v in out]
  30     return out

  32 def cycles(n,b):
  33     seen=set(); result=[]
  34     for k in range(n):
  35         if k in seen:continue
  36         cyc=[];j=k
  37         while j not in seen:
  38             seen.add(j);cyc.append(j);j=j*(1+b)%n
  39         assert j==k
  40         result.append(cyc)
  41     return result

  43 def geometry(n,b):
  44     l=n//b;records=[]
  45     for cyc in cycles(n,b):
  46         d=len(cyc);g=l//d;k=cyc[0]
  47         assert k%g==0
  48         u=k//g;assert 0<=u<b
  49         if d>1:assert u%2
  50         for e in range(d):
  51             a=(pow(u,-1,d)*e)%d if d>1 else 0
  52             c=(u*a-e)//d
  53             records.append((cyc,e,d,g,u,a,c))
  54     assert len(records)==n
  55     return records

  57 def fold_spectra(top,n,b):
  58     l=n//b;levels={l:[list(v) for v in top]};d=l
  59     wb=pow(PRIMITIVE,(P-1)//b,P);count=0
  60     while d>1:
  61         prev=levels[d];now=[[0]*b for _ in range(d//2)]
  62         for a in range(d//2):
  63             for q in range(b//2):
  64                 u=prev[a][2*q];v=pow(wb,q,P)*prev[a+d//2][2*q]%P
  65                 now[a][q]=(u+v)%P;now[a][q+b//2]=(u-v)%P;count+=1
  66         levels[d//2]=now;d//=2
  67     assert count==(n-b)//2
  68     return levels

  70 def extract(levels,n,b):
  71     wb=pow(PRIMITIVE,(P-1)//b,P)
  72     return [pow(wb,c%(b),P)*levels[d][a][u]%P for cyc,e,d,g,u,a,c in geometry(n,b)]

  74 def unfold(values,n,b):
  75     l=n//b;wb=pow(PRIMITIVE,(P-1)//b,P)
  76     levels={d:[[None]*b for _ in range(d)] for d in [1<<t for t in range(l.bit_length())]}
  77     for val,(cyc,e,d,g,u,a,c) in zip(values,geometry(n,b)):
  78         assert levels[d][a][u] is None
  79         levels[d][a][u]=val*pow(wb,(-c)%b,P)%P
  80     d=2;inv2=pow(2,-1,P);count=0
  81     while d<=l:
  82         low=levels[d//2];high=levels[d]
  83         for a in range(d//2):
  84             for q in range(b//2):
  85                 u=low[a][q];v=low[a][q+b//2]
  86                 assert u is not None and v is not None
  87                 high[a][2*q]=(u+v)*inv2%P
  88                 high[a+d//2][2*q]=(u-v)*inv2*pow(wb,-q,P)%P;count+=1
  89         d*=2
  90     assert count==(n-b)//2 and all(v is not None for row in levels[l] for v in row)
  91     return levels[l]

  93 def direct_residual(x,n,b):
  94     l=n//b;wb=pow(PRIMITIVE,(P-1)//b,P);ans=[]
  95     for cyc,e,d,g,u,a,c in geometry(n,b):
  96         k=cyc[0];r=g*e
  97         ans.append(sum(x[j]*pow(wb,((k*j-r)//l)%b,P) for j in range(n) if (k*j-r)%l==0)%P)
  98     return ans

 100 def full_output(ax,n,b):
 101     wn=pow(PRIMITIVE,(P-1)//n,P);out=[None]*n;index=0
 102     for cyc in cycles(n,b):
 103         d=len(cyc);g=(n//b)//d
 104         row=[ax[index+e]*pow(wn,g*e,P)%P for e in range(d)]
 105         index+=d
 106         for k,v in zip(cyc,fft(row)):out[k]=v
 107     assert all(v is not None for v in out)

 110 def numerical_r(n,b):
 111     l=n//b;wb=cmath.exp(2j*math.pi/b)
 112     levels={l:np.eye(n,dtype=complex).reshape((l,b,n))};d=l
 113     while d>1:
 114         old=levels[d];new=np.zeros((d//2,b,n),dtype=complex)
 115         for a in range(d//2):
 116             for q in range(b//2):
 117                 v=wb**q*old[a+d//2,2*q]
 118                 new[a,q]=old[a,2*q]+v;new[a,q+b//2]=old[a,2*q]-v
 119         levels[d//2]=new;d//=2
 120     records=geometry(n,b)
 121     r=np.array([wb**c*levels[d][a,u] for cyc,e,d,g,u,a,c in records])
 122     g=np.array([v[3] for v in records])
 123     rhat=r/np.sqrt(g[:,None]);gram=rhat@rhat.conj().T
 124     assert np.max(np.abs(gram-np.eye(n)))<2e-12
 125     support=np.count_nonzero(np.abs(r)>1e-9,axis=1)
 126     assert np.array_equal(support,g)
 127     h=l.bit_length()-1
 128     assert int(support.sum())==n+n*h//2
 129     probs=abs(rhat)**2
 130     entropy=-sum(float(p)*math.log2(float(p)) for p in probs.flat if p>1e-15)
 131     assert abs(entropy-(n-b))<1e-9
 132     return {'N':n,'B':b,'gram_error':float(np.max(abs(gram-np.eye(n)))),'nnz_R':int(support.sum()),'entropy_R':entropy,'predicted_entropy_R':n-b}
