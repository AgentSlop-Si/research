1: """Exact finite-field multi-stage Jacobian experiments and independent checks."""
2: import json,random,subprocess,sys,tempfile
3: from pathlib import Path
4: import numpy as np
5: from multistage_jacobian import materialize,check_differences,rank_minor
6: 
7: ROOT=Path(__file__).resolve().parent
8: BIN=ROOT/'rankmod'
9: 
10: def field_case(n,t,p,seed=99017):
11:     rng=random.Random(seed+n*31+t*997)
12:     perms=[]
13:     for _ in range(t-1):
14:         a=list(range(n));rng.shuffle(a);perms.append(a)
15:     base,J,pars,mats=materialize(n,t,perms,seed,p)
16:     check_differences(n,t,perms,pars,base,J,seed+299,p,num=30)
17:     with tempfile.TemporaryDirectory() as temp:
18:         path=Path(temp)/'jac.int32'
19:         J.astype(np.int32).tofile(path)
20:         stdout=subprocess.check_output([str(BIN),str(path),str(n*n),str(5*n*t),str(p)],text=True)
21:     head,cols=stdout.strip().splitlines()
22:     parts=dict(tok.split('=') for tok in head.split())
23:     rk=int(parts['rank']);det=int(parts['det'])
24:     pivots=[int(x) for x in cols.removeprefix('pivots=').split(',')]
25:     bound=min(n*n,t*(5*n-6)-(t-1)*n)
26:     assert rk<=bound,(n,t,rk,bound)
27:     if n<=16:
28:         rk2,p2,det2=rank_minor(J,p)
29:         assert rk2==rk and p2==pivots
30:         if rk==n*n: assert det2==det
31:     return {'n':n,'t':t,'p':p,'seed':seed,'permutations':perms,
32:             'rank':rk,'gauge_upper_bound':bound,'codimension':n*n-rk,
33:             'full_minor_det':det if rk==n*n else None,
34:             'full_minor_columns':pivots if rk==n*n else None,
35:             'rank_equivalent_over_two_routines':n<=16,
36:             'verified_coordinate_derivatives':30}
37: 
38: if __name__=='__main__':
39:     cases=[]

41:                 (32,3),(32,4),(32,5),(32,6),(32,7),(32,8),(32,9)]:
42:         for p in [12289,65537]:
