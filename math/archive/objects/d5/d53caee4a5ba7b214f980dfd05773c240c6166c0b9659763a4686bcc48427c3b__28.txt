  15 def bj(r,z):
  16     """Fresh bounded-argument factorial series, not SciPy or uploaded code."""
  17     sign = (-1)**(-r) if r < 0 else 1
  18     r = abs(r)
  19     z = np.asarray(z, dtype=float)
  20     term = np.ones_like(z)
  21     for j in range(1,r+1): term *= z/(2*j)
  22     acc = term.copy()
  23     for j in range(1,40):
  24         term = term * (-z*z/(4*j*(r+j)))
  25         acc = acc + term
  26     return sign*acc

  28 def grid(n):
  29     q=1<<(n-1).bit_length()
  30     j=np.array([(2*k*q+n)//(2*n) for k in range(n)],dtype=int)
  31     d=np.array([k*q-int(j[k])*n for k in range(n)],dtype=int)
  32     return q,j,d/n

  34 def tail(R):
  35     a=math.pi/2
  36     return 2*math.exp(math.pi**2/(4*(R+1)))*a**R/math.factorial(R)/(1-a/(R+1))
  37 
  38 def local_tail(delta,H):
  39     a=delta/2; r=H+1
  40     return 2*math.exp(delta*delta/(4*(r+1)))*a**r/math.factorial(r)/(1-a/(r+1))

  42 def cheb_coeff(theta,R):
  43     # Independent cosine definition, rather than Chebyshev recurrence.
  44     x=np.clip(2*theta,-1,1)
  45     c=np.array([np.cos(r*np.arccos(x)) for r in range(R)],dtype=complex)
  46     for r in range(1,R): c[r] *= 2*(-1j)**r
  47     return c
  48 
  49 def clenshaw(c,theta):
  50     n=c.shape[1]; u=np.zeros(n,dtype=complex); v=u.copy(); node=2*theta
  51     for r in reversed(range(1,len(c))):
  52         w=c[r]+2*node*u-v
  53         v,u=u,w
  54     return c[0]+node*u-v

  56 def phase_matrix(q,m,a,beta):
  57     k=q//m
  58     return np.array([[np.exp(2j*math.pi*((a*t*(t-1)//2+(a*m*b+beta)*t)%q)/q) for t in range(m)] for b in range(k)])
  59 
  60 def compress_matrix(q,m,a,beta):
  61     S=phase_matrix(q,m,a,beta); C=np.zeros((q//m,q),complex)
  62     for b in range(q//m): C[b,b*m:(b+1)*m]=S[b]
  63     return C
  64 
  65 def dft(q):
  66     return np.exp(-2j*np.pi*np.outer(np.arange(q),np.arange(q))/q)/math.sqrt(q)
  67 
  68 def recurrence_decoder(q,m,a,beta,d,selected):
  69     """Closed-form cycle recurrence; direct DFT and cumsum are independent controls."""
  70     k=q//m
  71     f=np.exp(-2j*np.pi*np.outer(np.arange(k),np.arange(k))/k) @ d
  72     cycle=np.array([(a*t)%q for t in range(q)])
  73     gauge=np.array([np.exp(2j*np.pi*((a*t*(t-1)//2+beta*t)%q)/q) for t in range(q)])
  74     boundary=np.array([np.exp(-2j*np.pi*((m*(int(v)+beta)+a*m*(m-1)//2)%q)/q) for v in cycle])
  75     forcing=(f[cycle%k]-boundary*f[(cycle+a*m)%k])/math.sqrt(q)
  76     terms=forcing/gauge
  77     prefix=np.r_[0,np.cumsum(terms[:-1])]
  78     y=np.empty(q,complex); y[cycle]=gauge*(sum(terms)/2-prefix)
  79     return y[selected]
  80 
  81 def weights(n,m,R,H):
  82     q,selected,theta=grid(n); k=q//m; coeff=cheb_coeff(theta,R)
  83     z=math.pi*np.arange(k)/k; delta=math.pi*np.arange(m)/q
  84     exact=np.array([bj(r,math.pi*np.arange(n)/q) for r in range(R)])
  85     approx=[]
  86     for r in range(R):
  87         a=np.zeros((k,m))
  88         for h in range(-H,H+1): a+=bj(r-h,z)[:,None]*bj(h,delta)[None,:]
  89         approx.append(a.ravel()[:n])
  90     approx=np.array(approx)
  91     return coeff.T @ exact, coeff.T @ approx, approx
