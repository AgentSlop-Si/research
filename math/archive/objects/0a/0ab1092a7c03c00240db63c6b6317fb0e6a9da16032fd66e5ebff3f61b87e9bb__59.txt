9: def select(a, k):
10:     if not 0 <= k < len(a):

22:         elif v > pivot: high.append(v)
23:         else: equal += 1
24:     if k < len(low): return select(low, k)
25:     if k < len(low) + equal: return pivot
26:     return select(high, k-len(low)-equal)
