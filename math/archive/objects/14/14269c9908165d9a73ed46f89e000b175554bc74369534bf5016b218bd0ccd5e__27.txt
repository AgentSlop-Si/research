0012 def sigma(M, K, a, b, d):
0013     t = np.arange(M, dtype=np.int64)
0014     r = np.arange(K, dtype=np.int64)
0015     phase = (a * t[None,:]**2 + (b + d*r[:,None])*t[None,:]) % M
0016     return np.exp(2j*np.pi*phase/M)
0017 
0018 def reconstruct(x, s):
0019     y = np.sum(s*x, axis=1)
0020     return np.fft.fft((np.conj(s)*y[:,None]).ravel(), norm='ortho')
0021 
0022 def signals(M,K):
0023     t=np.arange(M); r=np.arange(K)
0024     flat=np.ones((K,M),complex)
0025     mode=np.broadcast_to(np.exp(2j*np.pi*7*t/M),(K,M)).copy()
0026     phase=np.exp(2j*np.pi*rng.random((K,M)))
0027     spike=np.zeros((K,M),complex); spike[K//2,M//3]=1
0028     adversary=np.zeros((K,M),complex)
0029     # Fixed before drawing sketches: three coherent blocks aligned to seed a=1,b=0,d=1.
0030     adversary[:min(3,K)] = np.conj(sigma(M,K,1,0,1)[:min(3,K)])
0031     return {'flat':flat,'repeated_local_mode':mode,'random_phase':phase,
0032             'spike':spike,'fixed_coherent_three_blocks':adversary}

0034 def resonance_checks():

0040         h=pow(d,-1,M); q=(2*a*h)%M
0041         out=reconstruct(np.ones((K,M)),sigma(M,K,a,b,d))
0042         actual=out[q*K]/np.sqrt(K*M)
0043         active=np.arange(K)[(b+d*np.arange(K))%2==0]
0044         predicted=(2/K)*np.exp(2j*np.pi*((a*h*h+b*h)%M)/M)*np.sum(np.exp(2j*np.pi*active/M))
0045         predicted_abs=2*np.sin(np.pi*K/M)/(K*np.sin(2*np.pi/M))
0046         checks.append({'M':M,'K':K,'a':a,'b':b,'d':d,'alias_q':q,
0047                        'complex_error':float(abs(actual-predicted)),
0048                        'magnitude_error':float(abs(abs(actual)-predicted_abs))})
0049     assert max(v['complex_error'] for v in checks)<1e-11
0050     assert max(v['magnitude_error'] for v in checks)<1e-11
0051     return checks
