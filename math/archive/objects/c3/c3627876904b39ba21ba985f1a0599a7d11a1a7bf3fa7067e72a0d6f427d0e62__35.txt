  22 def certificate(n):
  23     P=bitrev(n);R,C=get_witness(n);G,big=make_graph(n,[1,1,1],[P,P]);source=('SRC',);sink=('SINK',)
  24     for j in C:G.add_edge(source,(0,'x',j,'i'),capacity=big)
  25     for j in R:G.add_edge((2,'y',j,'o'),sink,capacity=big)
  26     val,(source_part,sink_part)=nx.minimum_cut(G,source,sink,capacity='capacity',flow_func=nx.algorithms.flow.preflow_push)

  34     assert len(R)==len(C)
