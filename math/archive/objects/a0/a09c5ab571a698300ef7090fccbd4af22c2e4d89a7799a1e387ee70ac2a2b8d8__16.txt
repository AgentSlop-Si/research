   9 def root_sum(exponents,order,weights=None):
  10     """Exact integer coefficients in Z[z]/(z^(order/2)+1), order power of 2."""
  11     assert order>=2 and not order&(order-1)
  12     out=[0]*(order//2)
  13     if weights is None:
  14         weights=[1]*len(exponents)
  15     for exponent,weight in zip(exponents,weights):
  16         e=exponent%order
  17         if e>=order//2:
  18             out[e-order//2]-=weight
  19         else:
  20             out[e]+=weight
  21     return out

  24 def slope(r,coefficients,M):
  25     value=0
  26     for coefficient in reversed(coefficients):
  27         value=(value*r+coefficient)%M
  28     return value

  31 def dyadic_case(M,K,a,coefficients):
  32     assert M%4==0 and K%2==0 and coefficients[0]%2==1
  33     for r in range(0,K,2):
  34         c=slope(r,coefficients,M)
  35         assert c%2==1
  36         exponents=[a*t*t+c*t for t in range(M)]
  37         assert root_sum(exponents,M)==[0]*(M//2)
  38         for t in range(M//2):
  39             assert (exponents[t+M//2]-exponents[t])%M==M//2

  42 def prime_character_variance(M,K):
  43     """Error monomials have orthogonal seed characters iff labels differ.
  44 
  45     Labels correspond to the independently uniform seeds (a,alpha,b).
  46     For prime M and K<=M every off-diagonal error term has a distinct label.
  47     """
  48     labels={}
  49     multiplicities=Counter()
  50     for r in range(K):
  51         for u in range(M):
  52             for t in range(M):
  53                 if u==t:
  54                     continue
  55                 label=((u*u-t*t)%M,r*(u-t)%M,(u-t)%M)
  56                 assert label not in labels
  57                 labels[label]=(r,u,t)
  58                 multiplicities[(r,u)]+=1
  59     assert set(multiplicities.values())=={M-1}
  60     return {'M':M,'K':K,'orthogonal_error_characters':len(labels),
  61             'variance_coefficient':f'{M-1}/{K*M}',
  62             'statement':'Var[estimate_m]=(M-1)/N * ||x||_2^2 for every m and x'}
