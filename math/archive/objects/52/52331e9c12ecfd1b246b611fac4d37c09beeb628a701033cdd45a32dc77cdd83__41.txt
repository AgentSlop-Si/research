   1 from __future__ import annotations
   2 import json
   3 import unittest
   4 from pathlib import Path
   5 
   6 from two_line_weyl import (
   7     all_nonzero_vectors, canonical_partner, certify_pair, compatibility_kappa,
   8     dephased_fourier_tensor, det2, exact_two_line_factorization,
   9     field_for_length, fourier_matrix, fourier_weyl_coefficient,
  10     fourier_weyl_coefficient_direct, line_operator, mat_eq, mat_mul,
  11     matrix_rank, predicted_line_operator, qnorm,
  12 )
  13 
  14 
  15 class TwoLineWeylTests(unittest.TestCase):
  16     def test_weyl_coefficient_formula(self):
  17         for n in (5,7,11,13):
  18             p,w=field_for_length(n)
  19             for a,b in all_nonzero_vectors(n):
  20                 self.assertEqual(
  21                     fourier_weyl_coefficient(n,p,w,a,b),
  22                     fourier_weyl_coefficient_direct(n,p,w,a,b),
  23                 )
  24 
  25     def test_rank_dichotomy_exhaustive_n5_n7(self):
  26         for n in (5,7):
  27             for u in all_nonzero_vectors(n):
  28                 for v in all_nonzero_vectors(n):
  29                     if det2(u,v,n)==0:
  30                         continue
  31                     cert=certify_pair(n,u,v)
  32                     self.assertEqual(cert.tensor_rank,1 if cert.kappa==0 else n)
  33 
  34     def test_partner_parameterization(self):
  35         for n in (5,7,11,13,17):
  36             for u in all_nonzero_vectors(n):
  37                 if qnorm(u,n)==0:
  38                     continue
  39                 for t in (1,2,n-1):
  40                     v=canonical_partner(u,n,t)
  41                     self.assertNotEqual(det2(u,v,n),0)
  42                     self.assertEqual(compatibility_kappa(u,v,n),0)
  43 
  44     def test_exact_factorizations(self):
  45         cases={7:[(1,0),(1,2),(2,3)],11:[(1,0),(1,2),(3,4)],13:[(1,0),(0,1),(2,3)],17:[(1,0),(0,1),(3,5)]}
  46         for n,us in cases.items():
  47             p,w=field_for_length(n);F=fourier_matrix(n,p,w)
  48             for u in us:
  49                 if qnorm(u,n)==0:continue
  50                 for t in (1,2):
  51                     v,A,B=exact_two_line_factorization(n,p,w,u,t)
  52                     self.assertTrue(mat_eq(mat_mul(A,B,p),F,p),(n,u,v))
  53 
  54     def test_line_closed_forms(self):
  55         for n in (5,7,11,13,17):
  56             p,w=field_for_length(n)
  57             for u in all_nonzero_vectors(n):
  58                 if qnorm(u,n)==0:continue
  59                 self.assertTrue(mat_eq(line_operator(n,p,w,u),predicted_line_operator(n,p,w,u),p),(n,u))
  60 
  61     def test_nonvertical_factors_are_full_rank(self):
  62         for n in (5,7,11,13,17):
  63             p,w=field_for_length(n)
  64             for u in all_nonzero_vectors(n):
  65                 A=line_operator(n,p,w,u)
  66                 if qnorm(u,n)==0:
  67                     self.assertEqual(matrix_rank(A,p),1)
  68                 else:
  69                     self.assertEqual(matrix_rank(A,p),n)
  70 
  71     def test_bad_pair_requires_full_tensor_rank(self):
  72         cases=[(7,(1,0),(0,1)),(11,(1,1),(1,3)),(13,(2,1),(4,3)),(17,(1,4),(3,2))]
  73         for n,u,v in cases:
  74             self.assertNotEqual(det2(u,v,n),0)
  75             self.assertNotEqual(compatibility_kappa(u,v,n),0)
  76             p,w=field_for_length(n)
  77             self.assertEqual(matrix_rank(dephased_fourier_tensor(n,p,w,u,v),p),n)
  78 
  79     def test_predecessor_present(self):
  80         self.assertTrue(Path(__file__).with_name('PRIOR_v21_COMPLETE.zip').is_file())
