  31         raise ValueError('symbolic phase exponents must be integers in [0,N)')

  40     # Raw flows lie in [-b,b]. Counting selects a median in linear word work.

   6 def counted_power(counter, base, exponent):
   7     """Nonnegative binary powering with every field multiplication charged."""

   9         raise ValueError('nonnegative exponent required')

  13             out = counter.mul(out, base)
  14         exponent >>= 1
  15         if exponent:
  16             base = counter.mul(base, base)
  17     return out

  20 def apply_factor_powers(field, x, factor_powers, q):
  21     """Apply product_e (I-q^e S)^f[e], modulo the length-N nilpotent shift.
  22 
  23 q must have order N=len(x). Positive powers use y_i=x_i-c*x_(i-1).
  24 Negative powers use y_i=x_i+c*y_(i-1); no data-dependent division occurs.
  25 The dictionary is expected to contain a compiled signed flow. This routine
  26     counts all coefficient generation, copies, reads, writes, and field operations.
  27     A single deterministic power table uses N-1 multiplications. Every positive
  28     factor updates in descending order; negative factors update ascending, so
  29     only the output copy and O(N) coefficient table are materialized.
  30 It does not hide the cost of compiling the input operator graph into that flow.
  31 """
  32     n = len(x)
  33     if n < 1:
  34         raise ValueError('nonempty vector required')
  35     canonical = Counter()
  36     for exponent, multiplicity in factor_powers.items():

  38             raise ValueError('integer exponents and multiplicities required')
  39         canonical[exponent % n] += multiplicity
  40     root_counter, operation_counter = Counted(field), Counted(field)
  41     traffic = Counter({'initial_copy_reads': n, 'initial_copy_writes': n})
  42     value = list(x)
  43     powers = [field.one]
  44     for _ in range(1, n):
  45         powers.append(root_counter.mul(powers[-1], q))
  46     traffic['coefficient_generation_reads'] = n-1
  47     traffic['coefficient_table_writes'] = n
