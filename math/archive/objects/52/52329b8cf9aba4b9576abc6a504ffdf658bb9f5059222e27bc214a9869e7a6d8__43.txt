   1 """Finite-field probes of the prime-length, both-endpoint permutation corollary.
   2 
   3 Chebotarev's all-minors theorem is over characteristic zero. Modular tests can
   4 corroborate selected minors but CANNOT prove the characteristic-zero theorem.
   5 """
   6 from __future__ import annotations
   7 import json
   8 from itertools import combinations
   9 from pathlib import Path
  10 from random import Random
  11 from cut_budget import fourier, rank, permute, endpoint_certificate
  12 
  13 
  14 def is_prime(n):
  15     if n<2:return False
  16     if n%2==0:return n==2
  17     a=3
  18     while a*a<=n:
  19         if n%a==0:return False
  20         a+=2
  21     return True
  22 
  23 
  24 def field(n,seed=1):
  25     # Small/medium suitable primes with p == 1 (mod n). Primality checked.
  26     k=(1000000+seed)//n
  27     while True:
  28         p=k*n+1
  29         if is_prime(p):
  30             for a in range(2,100):
  31                 z=pow(a,k,p)
  32                 if z!=1 and pow(z,n,p)==1:
  33                     # n is prime: nonunity nth root has order exactly n
  34                     return p,a
  35         k+=1
  36 
  37 
  38 def run():
  39     rng=Random(20261009)
  40     out=[]
  41     for n in (5,7,11,13,17,31):
  42         m=n//2
  43         for seed in (1,41):
  44             p,g=field(n,seed)
  45             f=fourier(n,p,g)
  46             # Exhaustively test every m-by-m minor at small prime lengths.
  47             if n<=7:
  48                 subsets=list(combinations(range(n),m))
  49                 pairs=[(r,c) for r in subsets for c in subsets]
  50             else:
  51                 pairs=[]
  52                 for _ in range(60):
  53                     pairs.append((sorted(rng.sample(range(n),m)),sorted(rng.sample(range(n),m))))
  54             for rr,cc in pairs:
  55                 r=rank([[f[i][j] for j in cc] for i in rr],p)
  56                 assert r==m,(n,p,rr,cc,r)
  57             # Even if both endpoints are shuffled, prime Fourier half-blocks
  58             # have full row rank by Chebotarev (proven over C, sampled here).
  59             for _ in range(8):
  60                 ro=list(range(n));co=list(range(n))
  61                 rng.shuffle(ro);rng.shuffle(co)
  62                 w=permute(f,ro,co)
  63                 assert rank([row[:m] for row in w[:m]],p)==m
  64                 assert rank([row[m:] for row in w[:m]],p)==m
  65             # The standard general-cut transfer lemma works for odd N as well.
  66             cert=endpoint_certificate(permute(f,ro,co),p) if n<=17 else None
  67             if cert is not None:assert cert['necessary_rank_sum']>= (m+1)//2
  68             out.append({'length':n,'modulus':p,'primitive_generator':g,'minor_size':m,
  69                         'checked_square_minors':len(pairs),'both_endpoint_orders':8,
  70                         'channel_rank_sum_lower_bound':(m+1)//2,
  71                         'per_stage_equal_rank_bound':(m+3)//4})
  72     Path(__file__).with_name('prime_endpoint_results.json').write_text(json.dumps(out,indent=2)+'\n')
  73     print('PASS: prime Fourier minor samples:',sum(e['checked_square_minors'] for e in out))
  74     print('PASS: both-endpoint permutations:',sum(e['both_endpoint_orders'] for e in out))
  75     for x in out:
  76         if x['modulus']==field(x['length'],1)[0]:
  77             print('prime N=',x['length'],'rank sum >=',x['channel_rank_sum_lower_bound'])
  78     return out
  79 
  80 if __name__=='__main__':run()
