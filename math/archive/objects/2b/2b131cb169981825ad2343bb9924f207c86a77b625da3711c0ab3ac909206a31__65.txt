27: def generated_input(field,n,q,filters,lambdas,residual,use_gauss=False):
28:     current=list(residual)
29:     for descriptor in reversed(filters):
30:         a,b,shift,index=descriptor;parameter=lambdas[index]
31:         if use_gauss:
32:             columns=[]
33:             for j in range(n):
34:                 basis=[field.zero]*n;basis[j]=field.one
35:                 columns.append(independent_filter(field,basis,q,descriptor,lambdas))
36:             matrix=[list(row) for row in zip(*columns)]
37:             current=mv(field,inverse(field,matrix),current)
38:         else:
39:             # Independent forward recurrence y_next=(y-rhs)/edge with affine closure.
40:             order=[];j=0

42:             assert j==0 and len(set(order))==n
43:             coefficient=[field.mul(parameter,field.power(q,b*j%n)) for j in range(n)]
