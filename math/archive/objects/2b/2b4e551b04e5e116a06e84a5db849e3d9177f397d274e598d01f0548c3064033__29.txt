   9 def compose(a,b):return tuple(a[b[i]] for i in range(len(a)))
  10 def inverse(p):
  11     out=[0]*len(p)
  12     for i,j in enumerate(p):out[j]=i
  13     return tuple(out)
  14 def cycles(p):
  15     seen=set();out=[]
  16     for v in range(len(p)):
  17         if v in seen:continue
  18         c=[];w=v
  19         while w not in seen:seen.add(w);c.append(w);w=p[w]
  20         assert w==v;out.append(c)

  23 def closure(gens):
  24     group={ID};todo=[ID]
  25     while todo:
  26         p=todo.pop()

  28             q=compose(g,p)
  29             if q not in group:group.add(q);todo.append(q)
  30     return group

  34 def dimensions(n):

  38 def cover(a,b,label):
  39     axis,c,s=label
  40     return [((x+s*(2*y+c))%a)*b+y if axis=='H' else x*b+(y+s*(2*x+c))%b for x in range(a) for y in range(b)]
  41 def splice(p,tails):
  42     q=p[:]
  43     for j,t in enumerate(tails):q[t]=p[tails[(j+1)%len(tails)]]
  44     return q

  46 def init(a,b):return [splice(p,[min(c) for c in cycles(p)]) for p in (cover(a,b,label) for label in LABELS)]
  47 def physical(a,b,v,e):

  49 def sheet(a,b,v):

  52 def make_lift(a,b,small,label,require_even=True):
  53     n=a*b;ts=cover(a,b,label);tl=cover(2*a,2*b,label);eta=[0]*n
  54     for v in range(n):eta[ts[v]]=small[v]
  55     pre=[]
  56     for v in tl:
  57         x,y=divmod(v,2*b);xx,yy=divmod(eta[(x%a)*b+y%b],b)
  58         pre.append((xx+a*(x//a))*2*b+yy+b*(y//b))
  59     proj=lambda v:((v//(2*b))%a)*b+(v%(2*b))%b
  60     assert all(proj(pre[v])==small[proj(v)] for v in range(4*n))
  61     maps=[tuple(sheet(a,b,pre[physical(a,b,v,e)]) for e in range(4)) for v in range(n)]
  62     axis,c,sgn=label;total=0
  63     for x in range(a):
  64         for y in range(b):
  65             carry=(x+sgn*(2*y+c))//a if axis=='H' else (y+sgn*(2*x+c))//b
  66             total+=carry
  67             expected=tuple(2*((e//2+carry+(2*b//a)*(e%2))%2)+e%2 if axis=='H' else 2*(e//2)+(e%2+carry)%2 for e in range(4))
  68             assert maps[b*x+y]==expected and maps[b*x+y] in D8
  69     assert total==sgn*(b*(b-1+c) if axis=='H' else a*(a-1+c))
  70     monodromy=ID
  71     for v in cycles(small)[0]:monodromy=compose(maps[v],monodromy)
  72     cc=cycles(pre)
  73     if require_even:
  74         assert monodromy==ID and len(cc)==4 and all(len(c)==n for c in cc)
  75         tails=sorted(next(v for v in c if proj(v)==0) for c in cc)
  76         assert tails==[physical(a,b,0,e) for e in range(4)]
  77     else:tails=sorted(min(v for v in c if proj(v)==0) for c in cc)
  78     lifted=splice(pre,tails);assert len(cycles(lifted))==1
  79     post=[tuple(sheet(a,b,lifted[physical(a,b,v,e)]) for e in range(4)) for v in range(n)]
  80     final=ID
  81     for v in cycles(small)[0]:final=compose(post[v],final)
  82     if require_even:assert final==ROT and post[0]==compose(maps[0],ROT)
  83     if require_even:
  84         assert all(p in D8 and compose(p,SWAPX)==compose(SWAPX,p) for p in post)
  85         assert all(lifted[(v+2*n)%(4*n)]==(lifted[v]+2*n)%(4*n) for v in range(4*n))
  86     return lifted,post,{'carry_sum':total,'premerge_monodromy':monodromy,'premerge_cycle_lengths':list(map(len,cc)),'postmerge_monodromy':final,'all_postmerge_maps_in_D8':all(p in D8 for p in post)}

  88 def bank_holonomy(a,b,small,expected=None):
  89     large=[];edge=[];route_records=[]
  90     for p,label in zip(small,LABELS):

  92     n=a*b;transport=[None]*n;transport[0]=ID

  94     assert all(t is not None for t in transport)

 128 def masks_and_banks():
 129     W=[[(-1)**((c&e).bit_count()%2) for e in range(4)] for c in range(4)];groups=[0,1,2,2];out=[]
 130     for mask in itertools.product((0,1),repeat=4):
 131         mat=[[Fraction(sum(W[c][e]*mask[e]*W[d][e] for e in range(4)),4) for d in range(4)] for c in range(4)]

 136     p=998244353;banks=[]

 138         a,b=dimensions(n);length=4*n;w=pow(3,(p-1)//length,p);rho=pow(3,(p-1)//(2*b),p)
 139         active=set(range(a//2))
