#!/usr/bin/env python3
"""Bounded exact refinement: classify the W-identically-zero base fibers.
Uses the independently rebuilt polynomial data, not uploaded code.
"""
import json,time
from pathlib import Path
import sympy as sp
from sympy.polys.rings import ring
HERE=Path(__file__).resolve().parents[1]
zsym=sp.Symbol('z');K=sp.QQ.alg_field_from_poly(sp.Poly(zsym**4+1,zsym),alias='z')
R,s,a,b,x=ring('h0,h1,h2,h6',K);Z=R.zero;O=R.one
D=json.loads((HERE/'independent_polynomials.json').read_text())
def load(name):
    result=Z
    for mon,cs in D[name]['terms']:
        ex=[0]*4
        for j in mon:ex[j]+=1
        result+=R.from_dict({tuple(ex):sum((K.unit**j*int(v) for j,v in enumerate(cs)),K.zero)})
    return result
W=load('W');Wc=[R.from_dict({m[:3]+(0,):c for m,c in W.items() if m[3]==i}) for i in range(3)]
q1,r1=Wc[1].div(Wc[2]);q0,r0=Wc[0].div(Wc[2])
print('W2',Wc[2],flush=True);print('r1 factor',r1.factor_list(),flush=True);print('r0 factor',r0.factor_list(),flush=True)
g=r1.gcd(r0);print('gcd',g,flush=True)
print('W factor',W.factor_list(),flush=True)
print('DB factor',load('DB').factor_list(),flush=True)
def rec(P):
    out=[]
    for m,c in sorted(P.items()):
        co=list(reversed(c.to_list()))+[0]*4
        out.append({'exponents':list(m),'coefficient_rationals':[str(co[j]) for j in range(4)]})
    return out
result={'W2':rec(Wc[2]),'W1_remainder':rec(r1),'W0_remainder':rec(r0),'remainder_gcd':rec(g),'remainder_gcd_degree':max(map(sum,g),default=0),'W_factor_degrees':[{'degree':max(map(sum,p)),'multiplicity':e} for p,e in W.factor_list()[1]],'DB_factor_degrees':[{'degree':max(map(sum,p)),'multiplicity':e} for p,e in load('DB').factor_list()[1]]}
(HERE/'refinement_results.json').write_text(json.dumps(result,indent=2)+'\n')
