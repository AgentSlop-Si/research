import math, random, json, time
import torch

torch.set_default_dtype(torch.float64)
cdtype=torch.complex128

a=(1+1j)/2
b=(1-1j)/2
C=torch.tensor([[a,b],[b,a]],dtype=cdtype)
T=C
for _ in range(3):
    T=torch.kron(T,C)
n=16
Tnorm=(T.abs()**2).sum().item()

# Optimal 2n-4 gossip schedule: outsiders tell one of four hubs, hubs gossip, outsiders recall.
def base_schedule():
    hubs=[0,1,2,3]
    assign={v:hubs[(v-4)%4] for v in range(4,16)}
    first=[(v,assign[v]) for v in range(4,16)]
    core=[(0,1),(2,3),(0,2),(1,3)]
    second=[(v,assign[v]) for v in range(4,16)]
    return first+core+second

# Verify dependency gossip reaches all coordinates.
def support_ok(schedule):
    S=[{i} for i in range(n)]
    for p,q in schedule:
        u=S[p]|S[q]
        S[p]=set(u); S[q]=set(u)
    return all(len(s)==n for s in S)

# Add extra calls at random positions to 28-call support-complete schedule.
def make_schedule(g, rng):
    s=base_schedule()
    assert len(s)==28 and support_ok(s)
    while len(s)<g:
        p,q=rng.sample(range(n),2)
        pos=rng.randrange(len(s)+1)
        s.insert(pos,(p,q))
    return s

# matrix for embedded C call
def gate(pair):
    p,q=pair
    G=torch.eye(n,dtype=cdtype)
    G[p,p]=C[0,0]; G[p,q]=C[0,1]
    G[q,p]=C[1,0]; G[q,q]=C[1,1]
    return G

def optimize(schedule, seed, adam_steps=2500, lbfgs_steps=250):
    torch.manual_seed(seed)
    gs=[gate(x) for x in schedule]
    g=len(gs)
    # log magnitude + phase for every inter-gate diagonal. Small random symmetry break.
    x=torch.zeros((g+1,n,2),requires_grad=True)
    with torch.no_grad():
        x[:,:,0].normal_(0,0.01)
        x[:,:,1].normal_(0,0.01)
    opt=torch.optim.Adam([x], lr=0.025)
    best=1e99; bestx=None
    def build():
        z=torch.exp(x[:,:,0]+1j*x[:,:,1])
        M=torch.diag(z[0])
        for j,G in enumerate(gs):
            M=G@M
            M=torch.diag(z[j+1])@M
        return M
    for it in range(adam_steps):
        opt.zero_grad()
        M=build()
        loss=((M-T).abs()**2).sum()/Tnorm
        loss.backward()
        torch.nn.utils.clip_grad_norm_([x],10.0)
        opt.step()
        v=loss.item()
        if v<best:
            best=v; bestx=x.detach().clone()
        if best<1e-20: break
        if it in (1200,2000):
            for pg in opt.param_groups: pg['lr']*=0.35
    with torch.no_grad(): x.copy_(bestx)
    # LBFGS polish
    opt2=torch.optim.LBFGS([x],lr=0.4,max_iter=lbfgs_steps,line_search_fn='strong_wolfe',tolerance_grad=1e-12,tolerance_change=1e-15)
    def closure():
        opt2.zero_grad(); M=build(); loss=((M-T).abs()**2).sum()/Tnorm; loss.backward(); return loss
    try: opt2.step(closure)
    except Exception: pass
    with torch.no_grad():
        M=build(); rel=torch.linalg.norm(M-T)/torch.linalg.norm(T)
        sv=[]
        # condition of cumulative intermediates, including diagonal stages+gates
        z=torch.exp(x[:,:,0]+1j*x[:,:,1]); P=torch.diag(z[0]); maxcond=1.0
        for j,G in enumerate(gs):
            P=G@P; s=torch.linalg.svdvals(P); maxcond=max(maxcond,(s[0]/s[-1]).item())
            P=torch.diag(z[j+1])@P; s=torch.linalg.svdvals(P); maxcond=max(maxcond,(s[0]/s[-1]).item())
        mag=torch.exp(x[:,:,0])
        return rel.item(), maxcond, mag.min().item(), mag.max().item(), x.detach().cpu()

results=[]
start=time.time()
for g in [28,29,30,31]:
    for sched_seed in range(4):
        rng=random.Random(10000+g*100+sched_seed)
        schedule=make_schedule(g,rng)
        for opt_seed in range(2):
            rel,cond,mn,mx,params=optimize(schedule, seed=50000+g*100+sched_seed*10+opt_seed)
            rec=dict(g=g,sched_seed=sched_seed,opt_seed=opt_seed,rel_error=rel,max_intermediate_condition=cond,min_diag_mag=mn,max_diag_mag=mx,schedule=schedule)
            results.append(rec)
            print(json.dumps({k:v for k,v in rec.items() if k!='schedule'}),flush=True)

# exact standard tensor schedule = 32 calls, no diagonals needed.
standard=[]
for bit in range(4):
    step=1<<bit
    for block in range(0,n,2*step):
        for off in range(step): standard.append((block+off,block+off+step))
# Depending on kron ordering, reverse bit order still commutes as tensor factors.
M=torch.eye(n,dtype=cdtype)
for pr in standard: M=gate(pr)@M
std_err=(torch.linalg.norm(M-T)/torch.linalg.norm(T)).item()
summary={'elapsed_s':time.time()-start,'target':'C^tensor4','C':[[str(a),str(b)],[str(b),str(a)]], 'standard_calls':32,'standard_rel_error':std_err,'results':results}
with open('adaptive_fft_cpp/small_finite_win_search.json','w') as f: json.dump(summary,f,indent=2)
print('STANDARD',std_err,'support28',support_ok(base_schedule()))
