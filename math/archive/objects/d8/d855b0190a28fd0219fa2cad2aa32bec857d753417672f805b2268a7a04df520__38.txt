   5 def bareiss_det(A):
   6     n=len(A)
   7     if not n:return 1
   8     B=[list(map(int,row)) for row in A]
   9     s=1;prev=1
  10     for k in range(n-1):
  11         if not B[k][k]:
  12             r=next((i for i in range(k+1,n) if B[i][k]),None)
  13             if r is None:return 0
  14             B[k],B[r]=B[r],B[k];s=-s
  15         z=B[k][k]
  16         for i in range(k+1,n):
  17             for j in range(k+1,n):
  18                 num=B[i][j]*z-B[i][k]*B[k][j]
  19                 assert num%prev==0
  20                 B[i][j]=num//prev
  21         for i in range(k+1,n):B[i][k]=0
  22         prev=z
  23     return s*B[-1][-1]
