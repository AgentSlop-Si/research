"""Exact certificate: an 8x8 two-scan network is an input-output
superconcentrator while its separated-generator matrix family has codimension 4.

All 12,870 equal-size subset pairs are checked by integer maximum flow.
A nonzero 60x60 Jacobian minor mod 12289 witnesses characteristic-zero
image dimension >=60. An independent gauge/dimension argument gives <=60.
Therefore ALL cut rank inequalities hold vacuously while the realization
family satisfies additional polynomial identities.

This does NOT exclude the Fourier matrix or all degenerate quasiseparable charts.
"""
import json
import math
import random
import sys
from itertools import combinations
from pathlib import Path
import numpy as np
from jacobian_gate import MOD,basis,rank_mod,check_jacobian
ROOT=Path(__file__).resolve().parent/'prior_checkpoint'
sys.path.insert(0,str(ROOT))
from fast_mincut import ScanFlowOracle
from scan_flow import make_graph,mincut_bound

N=8
SHUFFLE=[7,0,6,3,4,1,2,5]
SEED=1011

def determinant_mod(M,mod=MOD):
    A=np.array(M,dtype=np.int64,copy=True)%mod
    assert A.shape[0]==A.shape[1]
    d=1
    for k in range(len(A)):
        loc=np.flatnonzero(A[k:,k]);assert loc.size,'singular minor'
        j=k+int(loc[0])
        if j!=k: A[[k,j]]=A[[j,k]];d=(-d)%mod
        pv=int(A[k,k]);d=d*pv%mod
        A[k,k:]=A[k,k:]*pow(pv,-1,mod)%mod
        for i in range(k+1,len(A)):
            if A[i,k]:
                A[i,k:]=(A[i,k:]-int(A[i,k])*A[k,k:])%mod
    return int(d)

def pivots(J,p=MOD):
    A=np.array(J,dtype=np.int64,copy=True)%p
    rowids=np.arange(A.shape[0]);r=0;cc=[]
    for c in range(A.shape[1]):
        if r==len(rowids):break
        ids=np.flatnonzero(A[r:,c])
        if not ids.size:continue
        j=r+int(ids[0]);A[[r,j]]=A[[j,r]];rowids[[r,j]]=rowids[[j,r]]
        A[r,c:]=A[r,c:]*pow(int(A[r,c]),-1,p)%p
        idx=np.flatnonzero(A[r+1:,c])+r+1
        if idx.size:A[idx,c:]=(A[idx,c:]-A[idx,c:c+1]*A[r:r+1,c:])%p
        cc.append(c);r+=1
    return [int(v) for v in rowids[:r]],[int(v) for v in cc]

def certify():
    o=ScanFlowOracle(N,[1,1],[SHUFFLE]);tested=0
    count_by_size={}
    for k in range(1,N+1):
        subsets=list(combinations(range(N),k));count_by_size[str(k)]=0
        for R in subsets:
            for C in subsets:
                got=o.bound(R,C)
                assert got==k,('counterexample!',R,C,got)
                tested+=1;count_by_size[str(k)]+=1
    assert tested==math.comb(2*N,N)-1 # omit k=0 empty case
    # Cross-check a subset of independently computed NetworkX graph cuts.
    G,inf=make_graph(N,[1,1],[SHUFFLE])
    rng=random.Random(8777)
    for _ in range(32):
        R=sorted(rng.sample(range(N),rng.randrange(1,N+1)))
        C=sorted(rng.sample(range(N),rng.randrange(1,N+1)))
        assert o.bound(R,C)==mincut_bound(G,inf,N,2,R,C)[0]
    M,J,a,b=basis(N,1,SHUFFLE,SEED)
    check_jacobian(M,J,a,b,SHUFFLE,nchecks=40)
    assert rank_mod(J)==60
    rows,cols=pivots(J)
    assert len(rows)==len(cols)==60
    det=determinant_mod(J[np.ix_(rows,cols)])
    assert det!=0
    certificate={'n':N,'stage_ranks':[1,1],'fixed_middle_shuffle':SHUFFLE,
        'endpoint_permutations':'identity (arbitrary relabeling does not alter result)',
        'checked_equal_cardinality_subset_pairs':tested,
        'including_empty_pair_count':tested+1,
        'flow_counts_by_size':count_by_size,
        'all_input_output_submatrix_rank_bounds':'min(|R|,|C|) for every R,C',
        'independent_networkx_cut_checks':32,
        'generator_parameter_seed':SEED,
        'field_prime':MOD,'number_chart_parameters':J.shape[1],
        'jacobian_rank':60,'jacobian_minor_rows':rows,'jacobian_minor_columns':cols,
        'jacobian_minor_determinant_mod_prime':det,
        'chart_dim_upper_from_gauge':60,'image_dim_over_complex':60,
        'ambient_matrix_dim':64,'algebraic_codimension':4,
        'scope':'separated-generator semiseparable two-stage family; no Fourier exclusion'}
    return certificate

def verify(cert):
    assert cert['n']==8 and cert['stage_ranks']==[1,1]
    assert cert['fixed_middle_shuffle']==SHUFFLE
    oracle=ScanFlowOracle(8,[1,1],[SHUFFLE])
    cnt=0
    for k in range(1,9):
        subs=list(combinations(range(8),k))
        for R in subs:
            for C in subs:
                assert oracle.bound(R,C)==k
                cnt+=1
    assert cnt==cert['checked_equal_cardinality_subset_pairs']
    M,J,*_=basis(8,1,SHUFFLE,cert['generator_parameter_seed'])
    rows=cert['jacobian_minor_rows'];cols=cert['jacobian_minor_columns']
    det=determinant_mod(J[np.ix_(rows,cols)])
    assert det==cert['jacobian_minor_determinant_mod_prime'] and det!=0
    assert rank_mod(J)==60
    return f'PASS all {cnt} nonempty equal-size cuts; Jacobian rank 60; 60x60 determinant {det} mod {MOD}; codimension 4'

if __name__=='__main__':
    z=certify()
    file=Path(__file__).with_name('superconcentrator_N8_certificate.json')
    file.write_text(json.dumps(z,indent=2)+'\n')
    print(verify(json.loads(file.read_text())))
    print('flow by k',z['flow_counts_by_size'])
