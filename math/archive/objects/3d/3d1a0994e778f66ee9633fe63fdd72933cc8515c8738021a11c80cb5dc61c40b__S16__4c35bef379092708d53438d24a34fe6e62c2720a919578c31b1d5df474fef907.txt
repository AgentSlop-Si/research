  17 def pmul(a,b,p):
  18     c=[0]*(len(a)+len(b)-1)
  19     for i,x in enumerate(a):
  20         for j,y in enumerate(b):c[i+j]=(c[i+j]+x*y)%p

  22 def padd(a,b,p):
  23     c=[0]*max(len(a),len(b))
  24     for i,v in enumerate(a):c[i]=v

  27 def peval(q,x,p):

  32 def premainder(a,b,p):

  40 def pgcd(a,b,p):

  55 def determinant(M,p):
  56     A=[row[:] for row in M];value=1
  57     for k in range(len(A)):
  58         j=next((j for j in range(k,len(A)) if A[j][k]%p),None)
  59         if j is None:return 0
  60         if j!=k:A[k],A[j]=A[j],A[k];value=-value
  61         pivot=A[k][k];value=value*pivot%p;iv=pow(pivot,-1,p)
  62         for i in range(k+1,len(A)):
  63             q=A[i][k]*iv%p
  64             for j in range(k,len(A)):A[i][j]=(A[i][j]-q*A[k][j])%p
  65     return value%p

  67 def resultant(R,w,p):

  69     P=([*P]+[0]*(4-len(P)))[::-1];Q=([*Q]+[0]*(4-len(Q)))[::-1]
  70     M=[]
  71     for poly in (P,Q):
  72         for j in range(3):M.append([0]*j+poly+[0]*(2-j))

  76 def dft(r,w,p,count=None):
