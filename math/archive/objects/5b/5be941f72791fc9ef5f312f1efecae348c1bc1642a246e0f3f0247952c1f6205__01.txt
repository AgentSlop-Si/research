   3 import itertools,json,math,random,sys
   4 from pathlib import Path
   5 RNG=random.Random(90107)
   6 def poly(nodes,p):
   7  a=[1]
   8  for x in nodes:
   9   b=[0]*(len(a)+1)
  10   for i,c in enumerate(a): b[i]=(b[i]-x*c)%p;b[i+1]=(b[i+1]+c)%p
  11   a=b
  12  return a
  13 def ev(a,x,p):
  14  s=0
  15  for c in reversed(a): s=(s*x+c)%p
  16  return s
  17 def rem(a,b,p):
  18  a=a[:]
  19  while len(a)>=len(b):
  20   c=a[-1]*pow(b[-1],-1,p)%p;k=len(a)-len(b)
  21   for j,t in enumerate(b):a[k+j]=(a[k+j]-c*t)%p
  22   a.pop()
  23  return a+[0]*(len(b)-1-len(a))
  24 def transpose(a):return [list(x) for x in zip(*a)] if a else []
  25 def mm(a,b,p):return [[sum(x*y for x,y in zip(row,col))%p for col in zip(*b)] for row in a]
  26 def mv(a,v,p):return [sum(x*y for x,y in zip(row,v))%p for row in a]
  27 def add(a,b,p,sgn=1):return [[(x+sgn*y)%p for x,y in zip(r,s)]for r,s in zip(a,b)]
  28 def eye(n):return [[int(i==j)for j in range(n)]for i in range(n)]
  29 def rank(a,p):
  30  if not a:return 0
  31  a=[r[:]for r in a];r=0
  32  for j in range(len(a[0])):
  33   k=next((k for k in range(r,len(a))if a[k][j]),None)
  34   if k is None:continue
  35   a[r],a[k]=a[k],a[r];q=pow(a[r][j],-1,p);a[r]=[v*q%p for v in a[r]]
  36   for k in range(r+1,len(a)):
  37    q=a[k][j];a[k]=[(x-q*y)%p for x,y in zip(a[k],a[r])]
  38   r+=1
  39   if r==len(a):break
  40  return r
  41 def inv(a,p):
  42  n=len(a)
  43  if not n:return []
  44  a=[row[:]+e for row,e in zip(a,eye(n))]
  45  for j in range(n):
  46   k=next(k for k in range(j,n)if a[k][j]);a[j],a[k]=a[k],a[j]
  47   v=pow(a[j][j],-1,p);a[j]=[x*v%p for x in a[j]]
  48   for i in range(n):
  49    if i!=j:
  50     v=a[i][j];a[i]=[(x-v*y)%p for x,y in zip(a[i],a[j])]
  51  return [r[n:]for r in a]
  52 def root(n,p):return next(w for w in range(1,p)if pow(w,n,p)==1 and len({pow(w,k,p)for k in range(n)})==n)
  53 def card(nodes,x,p):
  54  a=poly([z for z in nodes if z!=x],p);q=pow(ev(a,x,p),-1,p)
  55  return [c*q%p for c in a]
  56 def transfer(nodes,p):
  57  m=len(nodes);q=poly(nodes,p)
  58  return transpose([rem([0]*(m+j)+[1],q,p)for j in range(m)])
  59 def make_case(n,p,A,full=False):
  60  m=n//2;w=root(n,p);roots=[pow(w,-j,p)for j in range(n)];B=[i for i in range(n)if i not in A]
  61  e=[roots[i]for i in range(0,n,2)if i not in A];o=[roots[i]for i in A if i%2];t=len(e)
  62  ar=[roots[i]for i in A];br=[roots[i]for i in B];ra=transfer(ar,p);rb=transfer(br,p);I=eye(m)
  63  assert mm(ra,ra,p)==I and mm(rb,rb,p)==I
  64  ca=add(ra,I,p,-1);cb=add(rb,I,p)
  65  assert rank(ca,p)==rank(cb,p)==t
  66  assert rank(ca+cb,p)==min(m,2*t)
  67  assert rank(add(ra,rb,p,-1),p)==m
  68  r=2*min(t,m-t)
  69  assert rank(add(ra,rb,p),p)==r
  70  assert rank(add(mm(ra,rb,p),I,p),p)==r
  71  assert rank(add(mm(ra,rb,p),mm(rb,ra,p),p,-1),p)==r
  72  if not full:return {'n':n,'p':p,'t':t,'rank':r}
  73  ua=[card(ar,x,p)for x in o];ub=[card(br,x,p)for x in e]
  74  C=[[ev(c,x,p)for c in ub]for x in o];D=[[ev(c,x,p)for c in ua]for x in e]
  75  assert rank(C,p)==rank(D,p)==t
  76  PA,PB=poly(ar,p),poly(br,p)
  77  for i,x in enumerate(o):
  78   for j,y in enumerate(e):
  79    assert C[i][j]==ev(PB,x,p)*pow((x-y)*ev(poly([z for z in br if z!=y],p),y,p),-1,p)%p
  80    assert D[j][i]==ev(PA,y,p)*pow((y-x)*ev(poly([z for z in ar if z!=x],p),x,p),-1,p)%p
  81  low=[RNG.randrange(p)for _ in range(m)];high=[RNG.randrange(p)for _ in range(m)]
  82  a=[(x+y)%p for x,y in zip(low,high)];b=[(x-y)%p for x,y in zip(low,high)]
  83  for x,c in zip(o,ua):a=[(v-2*ev(high,x,p)*z)%p for v,z in zip(a,c)]
  84  for x,c in zip(e,ub):b=[(v+2*ev(high,x,p)*z)%p for v,z in zip(b,c)]
  85  assert a==rem(low+high,PA,p) and b==rem(low+high,PB,p)
  86  y=[(x-z)*pow(2,-1,p)%p for x,z in zip(a,b)]
  87  zo=[-v%p for v in mv(inv(D,p),[ev(y,x,p)for x in e],p)]
  88  ze=[-v%p for v in mv(inv(C,p),[ev(y,x,p)for x in o],p)]
  89  hh=y[:]
  90  for v,c in list(zip(zo,ua))+list(zip(ze,ub)):hh=[(z+v*q)%p for z,q in zip(hh,c)]
  91  ll=[(x-z)%p for x,z in zip(a,hh)]
  92  for v,c in zip(zo,ua):ll=[(z+2*v*q)%p for z,q in zip(ll,c)]
  93  assert ll==low and hh==high
  94  for j in A:assert ev(a,roots[j],p)==ev(low+high,roots[j],p)
  95  for j in B:assert ev(b,roots[j],p)==ev(low+high,roots[j],p)
  96  # Different fixed common coefficient bases leave all three defect ranks unchanged.
  97  G=eye(m)
  98  for j in range(m):G[j][j]=RNG.randrange(1,p)
  99  for j in range(m-1):G[j][j+1]=RNG.randrange(p)
 100  Gi=inv(G,p);raa=mm(mm(G,ra,p),Gi,p);rbb=mm(mm(G,rb,p),Gi,p)
 101  assert rank(add(mm(raa,rbb,p),mm(rbb,raa,p),p,-1),p)==r
 102  # Actual full child evaluator: ordinary coset DFT values then direct newcomer replacement.
 103  out=[None]*n
 104  for j in range(0,n,2):out[j]=ev(a,roots[j],p)
 105  for j in range(1,n,2):out[j]=ev(b,roots[j],p)
 106  for j in A:
 107   if j%2:out[j]=ev(a,roots[j],p)
 108  for j in B:
 109   if not j%2:out[j]=ev(b,roots[j],p)
 110  assert out==[ev(low+high,x,p)for x in roots]
 111  wrong=False
 112  if t:
 113   bad=y[:]
 114   for v,c in list(zip(zo,ua))+list(zip(ze,ub)):bad=[(z-v*q)%p for z,q in zip(bad,c)]
 115   wrong=bad!=high
 116  if t==1:
 117   ge=[pow(e[0],m-1-i,p)for i in range(m)];go=[pow(o[0],m-1-i,p)for i in range(m)]
 118   assert a==[(x+y+(o[0]-e[0])*ev(high,o[0],p)*z)%p for x,y,z in zip(low,high,ge)]
 119   assert b==[(x-y+(e[0]-o[0])*ev(high,e[0],p)*z)%p for x,y,z in zip(low,high,go)]
 120  return {'n':n,'p':p,'t':t,'rank':r,'wrong_sign_detected':wrong,'dense_counts':[sum(v!=0 for row in M for v in row)for M in (ra,rb)]}
