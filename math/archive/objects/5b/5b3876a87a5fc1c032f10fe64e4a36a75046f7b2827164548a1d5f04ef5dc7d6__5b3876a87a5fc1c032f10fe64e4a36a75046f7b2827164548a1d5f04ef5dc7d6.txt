"""Multi-Shear Green-Union: exact O_g(N) recognized sparse innovations.

N=Q*M; first 2g raw fibers are generated by g unknown dense shear orbits:
    y_t[u] = sum_j mu_j**t * b_j[(u-a_j*t) mod M], t < 2g.
Subsequent raw fibers obey the gth-order operator recurrence
    [prod_j(E - mu_j*S_(a_j))] y_t = r_t,
where total M-DFT support of r_t for t>=2g is R=O(Q).

Recover orbit shifts/scales and baselines solely from first 2g raw rows
using the Hankel trace DFT; recognize every innovation from full raw data;
fully reconstruct and check all raw samples. Evaluate all N exact DFT outputs
with the partial-fraction Green impulse of g modes, O_g(N+MQ+MR+QR).

This is an input-restricted theorem with a clean pilot and nondegenerate
Hankel condition, not an arbitrary-input faster FFT theorem.
"""
from __future__ import annotations
from dataclasses import dataclass
from itertools import combinations
from typing import Optional,Sequence
import phase_union_general as phase
from delta_graph import balanced_q,direct_row_dft
from fast_vandermonde import solve_moments
from delta_graph_budget import modes_with_budget
from multishear_union import Orbit,solve_square
from shear_gauge_union import inverse_for_scaled_roots


@dataclass(frozen=True)
class Innovation:
    t:int
    h:int
    c:int  # r_t[u] has c * eta^(h*u); unnormalized spectral impulse is M*c.


@dataclass(frozen=True)
class GreenCertificate:
    n:int
    q:int
    m:int
    p:int
    omega:int
    g:int
    orbits:tuple[Orbit,...]
    events:tuple[Innovation,...]
    clean_pilot_rows:int
    raw_checked:int


def recurrence_shifts(orbits,p,m):
    """Coefficient/shift representation of prod_j (E - mu_j S_a_j)."""
    result=[]
    for lag in range(1,len(orbits)+1):
        for subset in combinations(orbits,lag):
            scale=(-1)**lag
            offset=0
            for o in subset:
                scale=scale*o.scalar%p
                offset=(offset+o.shift)%m
            result.append((lag,offset,scale%p))
    return result


def recover_clean_pilot(x,p,omega,g,q,roots,subroots):
    """O(g M^2 + g^3 M) recovery of unknown g-component orbit pilot.

    Does NOT presume the remaining raw rows are unaltered orbit sums.
    Assumes the first 2g rows are clean and checks all 2g of them.
    """
    n=len(x);m=n//q
    spectra=[]
    for t in range(2*g):
        row=[x[t+q*u]%p for u in range(m)]
        spectra.append(direct_row_dft(row,p,subroots))
    trace=[0]*m
    for h in range(m):
        moments=[spectra[t][h] for t in range(2*g)]
        H=[[moments[i+j] for j in range(g)] for i in range(g)]
        coeff=solve_square(H,[-moments[i+g] for i in range(g)],p)
        if coeff is None:return None
        trace[h]=(-coeff[-1])%p
    freq=direct_row_dft(trace,p,subroots)
    inv_m=pow(m,-1,p)
    params=[((-ell)%m,val*inv_m%p) for ell,val in enumerate(freq) if val]
    if len(params)!=g or len({a for a,mu in params})!=g:return None
    amplitudes=[[0]*m for _ in range(g)]
    for h in range(m):
        nodes=[mu*subroots[(-a*h)%m]%p for a,mu in params]
        weights=solve_moments(nodes,[spectra[t][h] for t in range(g)],p)
        if weights is None:return None
        for j,val in enumerate(weights):amplitudes[j][h]=val
    baselines=[]
    for weights in amplitudes:
        baselines.append([inv_m*sum(weights[h]*subroots[(h*u)%m]
                                    for h in range(m))%p for u in range(m)])
    orbits=tuple(Orbit(a,mu,tuple(amplitudes[j]),tuple(baselines[j]))
                 for j,(a,mu) in enumerate(params))
    # Pilot truly clean; the rest of the input is intentionally not tested yet.
    scalars=[1]*g
    for t in range(2*g):
        shifts=[(o.shift*t)%m for o in orbits]
        for u in range(m):
            predicted=sum(scalars[j]*orbits[j].base[(u-shifts[j])%m]
                          for j in range(g))%p
            if predicted!=x[t+q*u]%p:return None
        for j,o in enumerate(orbits):scalars[j]=scalars[j]*o.scalar%p
    return orbits


def recognize(x:Sequence[int],p:int,omega:int,g:int=2,total_modes:Optional[int]=None,
              q:Optional[int]=None,balance_bound:int=4)->Optional[GreenCertificate]:
    n=len(x)
    q=balanced_q(n,balance_bound) if q is None else q
    if (q is None or q<2*g or n%q or g<1 or n//q<g or
        q*q>balance_bound*n or (n//q)**2>balance_bound*n):return None
    m=n//q
    budget=q if total_modes is None else total_modes
    if budget<0:raise ValueError('nonnegative total_modes required')
    roots=phase.phase_table(n,p,omega)
    subroots=phase.phase_table(m,p,roots[q%n])
    orbits=recover_clean_pilot(x,p,omega,g,q,roots,subroots)
    if orbits is None:return None
    operators=recurrence_shifts(orbits,p,m)
    events=[];spent=0
    by_t={}
    for t in range(2*g,q):
        residual=[]
        for u in range(m):
            value=x[t+q*u]%p
            for lag,shift,coef in operators:
                value+=coef*x[t-lag+q*((u-shift)%m)]
            residual.append(value%p)
        modes=modes_with_budget(residual,subroots,p,budget-spent)
        if modes is None:return None
        by_t[t]=modes
        for h,c in modes:events.append(Innovation(t,h,c))
        spent+=len(modes)

    # EXTRA independent time-domain certificate: reconstruct all Q raw rows
    # from the first g fitted components and the recognized r_t modes.
    reconstructed=[]
    scalars=[1]*g
    for t in range(q):
        if t<g:
            row=[sum(scalars[j]*o.base[(u-o.shift*t)%m]
                     for j,o in enumerate(orbits))%p for u in range(m)]
        else:
            modes=by_t.get(t,())
            row=[]
            for u in range(m):
                r=sum(c*subroots[(h*u)%m] for h,c in modes)
                recurrence=-sum(coef*reconstructed[t-lag][(u-shift)%m]
                                for lag,shift,coef in operators)
                row.append((r+recurrence)%p)
        if any(row[u]!=x[t+q*u]%p for u in range(m)):return None
        reconstructed.append(row)
        for j,o in enumerate(orbits):scalars[j]=scalars[j]*o.scalar%p
    return GreenCertificate(n,q,m,p,omega,g,orbits,tuple(events),2*g,n)


def transform_certified(cert:GreenCertificate,with_counters=False):
    n,q,m,p,g=cert.n,cert.q,cert.m,cert.p,cert.g
    roots=phase.phase_table(n,p,cert.omega)
    all_inv=[inverse_for_scaled_roots(roots,o.scalar,p) for o in cert.orbits]
    all_mu_pow=[]
    for o in cert.orbits:
        power=[1]*(q+1)
        for t in range(q):power[t+1]=power[t]*o.scalar%p
        all_mu_pow.append(power)

    # h-dependent impulse Green weights: h_v(lambda_1,...,lambda_g)
    #   = sum_j w_j(h) * lambda_j(h)^v,
    # w_j = lambda_j^(g-1) / prod_{i!=j}(lambda_j-lambda_i).
    subroots=phase.phase_table(m,p,roots[q%n])
    weight=[[0]*g for _ in range(m)]
    for h in range(m):
        lambdas=[o.scalar*subroots[(-o.shift*h)%m]%p for o in cert.orbits]
        for j,lam in enumerate(lambdas):
            denom=1
            for i,other in enumerate(lambdas):
                if i!=j:denom=denom*(lam-other)%p
            if not denom:raise ValueError('degenerate certified orbit eigenvalues')
            weight[h][j]=pow(lam,g-1,p)*pow(denom,-1,p)%p

    def G(j:int,ell:int,L:int):
        o=cert.orbits[j]
        z=o.scalar*roots[-ell%n]%p
        if z==1:return L%p
        zL=all_mu_pow[j][L]*roots[(-ell*L)%n]%p
        return (1-zL)*all_inv[j][ell]%p

    out=[0]*n
    for k in range(n):
        h=k%m
        val=0
        for j,o in enumerate(cert.orbits):
            ell=(k+q*o.shift*h)%n
            val+=o.spectrum[h]*G(j,ell,q)
        out[k]=val%p

    for e in cert.events:
        L=q-e.t
        for s in range(q):
            k=e.h+m*s
            zt=roots[(-k*e.t)%n]
            projected=0
            for j,o in enumerate(cert.orbits):
                ell=(k+q*o.shift*e.h)%n
                projected+=weight[e.h][j]*G(j,ell,L)
            out[k]=(out[k]+m*e.c*zt*projected)%p

    if with_counters:
        return out,{'n':n,'q':q,'m':m,'g':g,'events':len(cert.events),
                    'pilot_dft_terms':2*g*m*m,'trace_dft_terms':m*m,
                    'inverse_baseline_terms':g*m*m,
                    'raw_verification_terms':n,
                    'baseline_geometric_terms':g*n,
                    'innovation_geometric_terms':g*q*len(cert.events),
                    'batch_inverse_tables':g}
    return out


def transform_if_recognized(x,p,omega,g=2,total_modes=None,q=None,balance_bound=4):
    cert=recognize(x,p,omega,g,total_modes,q,balance_bound)
    return (None,None) if cert is None else (transform_certified(cert),cert)


def build_green_fixture(n,p,omega,q,orbits,innovations):
    """Test-only synthetic raw input. innovations[t]=list[(h,c)]."""
    m=n//q;g=len(orbits)
    subroots=phase.phase_table(m,p,pow(omega,q,p))
    ops=recurrence_shifts(orbits,p,m)
    rows=[]
    for t in range(q):
        if t<2*g:
            row=[sum(pow(o.scalar,t,p)*o.base[(u-o.shift*t)%m]
                     for o in orbits)%p for u in range(m)]
        else:
            modes=innovations.get(t,())
            row=[]
            for u in range(m):
                r=sum(c*subroots[(h*u)%m] for h,c in modes)
                recurrence=-sum(coef*rows[t-lag][(u-shift)%m]
                                for lag,shift,coef in ops)
                row.append((r+recurrence)%p)
        rows.append(row)
    x=[0]*n
    for t,row in enumerate(rows):
        for u,z in enumerate(row):x[t+q*u]=z
    return x
