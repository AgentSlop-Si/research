   1 #!/usr/bin/env python3
   2 """Independent permutation-screen checks. Imports no source-upload programs."""
   3 from pathlib import Path
   4 from itertools import permutations
   5 from math import factorial,comb
   6 from fractions import Fraction
   7 import json
   8 BASE=Path(__file__).resolve().parent
   9 
  10 def every_cut(pi,r,side):
  11     n=len(pi)
  12     for t in range(1,n):
  13         v=n-1-max(pi[-t:]) if side=='suffix' else min(pi[:t])
  14         if min(t,v)>r:return False
  15     return True
  16 
  17 def boundary(pi,r,side):
  18     n=len(pi)
  19     if r>=n-1:return True
  20     k=r+1
  21     return max(pi[-k:])>=n-k if side=='suffix' else min(pi[:k])<k
  22 
  23 def count_passing(n,k):
  24     assert n>=2*k
  25     total=factorial(n)
  26     one_bad=factorial(n-k)//factorial(n-2*k)*factorial(n-k)
  27     both_bad=sum((-1)**(a+b)*comb(k,a)**2*factorial(a)*comb(k,b)**2*factorial(b)*factorial(n-a-b)
  28                  for a in range(k+1) for b in range(k+1))
  29     return total-2*one_bad+both_bad
  30 
  31 cases=0; exhaustive=[]
  32 for n in range(2,9):
  33     counts={k:0 for k in range(1,n//2+1)}
  34     for pi in permutations(range(n)):
  35         for side in ['prefix','suffix']:
  36             for r in range(n):
  37                 assert every_cut(pi,r,side)==boundary(pi,r,side),(pi,r,side)
  38                 cases+=1
  39         for k in counts:
  40             counts[k]+=boundary(pi,k-1,'prefix') and boundary(pi,k-1,'suffix')
  41     for k,c in counts.items():
  42         assert c==count_passing(n,k),(n,k,c,count_passing(n,k))
  43         exhaustive.append({'N':n,'boundary_size':k,'rank_sum':k-1,'passing':c,'total':factorial(n)})
  44 large=[]
  45 for n in [8,9,16,32,64,128,256,1024]:
  46     c=count_passing(n,3);p=Fraction(c,factorial(n))
  47     assert 0<c<=factorial(n)
  48     assert p<=Fraction(81,n*(n-1))
  49     large.append({'N':n,'rank_sum':2,'passing':str(c),'total':str(factorial(n)),
  50                   'probability_exact':str(p),'probability_approx':float(p),
  51                   'union_bound':float(Fraction(81,n*(n-1)))})
  52 result={'every_cut_boundary_equivalence_checks':cases,'all_passed':True,
  53         'exact_count_comparisons':len(exhaustive),'exhaustive':exhaustive,'rank_sum_two':large,
  54         'uploaded_programs_executed':False}
  55 (BASE/'CHECK_RESULTS.json').write_text(json.dumps(result,indent=2)+'\n')
  56 print(json.dumps({k:v for k,v in result.items() if k not in ['exhaustive','rank_sum_two']},indent=2))
  57 print('N=8, rank sum 2:',count_passing(8,3),'/',factorial(8))
