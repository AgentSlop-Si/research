0005 def field(n):
0006  p=n+1
0007  while any(p%d==0 for d in range(2,math.isqrt(p)+1)):p+=n
0008  proper=[d for d in range(1,n) if n%d==0]
0009  for w in range(2,p):
0010   if pow(w,n,p)==1 and all(pow(w,d,p)!=1 for d in proper):return p,w
0011  raise AssertionError
0012 def solve(A,b,p):
0013  a=[row[:]+[v] for row,v in zip(A,b)];n=len(A)
0014  for j in range(n):
0015   i=next(i for i in range(j,n) if a[i][j]);a[i],a[j]=a[j],a[i]
0016   iv=pow(a[j][j],-1,p);a[j]=[v*iv%p for v in a[j]]
0017   for i in range(n):
0018    if i==j:continue
0019    z=a[i][j];a[i]=[(x-z*y)%p for x,y in zip(a[i],a[j])]
0020  return [a[i][-1] for i in range(n)]
0021 def rank(A,p):
0022  a=[row[:] for row in A];r=0
0023  for j in range(len(a[0])):
0024   i=next((i for i in range(r,len(a)) if a[i][j]),None)
0025   if i is None:continue
0026   a[i],a[r]=a[r],a[i];iv=pow(a[r][j],-1,p);a[r]=[v*iv%p for v in a[r]]
0027   for i in range(r+1,len(a)):
0028    z=a[i][j];a[i]=[(x-z*y)%p for x,y in zip(a[i],a[r])]
0029   r+=1
0030   if r==len(a):break
0031  return r
0032 def dft(x,w,p):return [sum(v*pow(w,-k*t,p) for t,v in enumerate(x))%p for k in range(len(x))]
0033 def mst_prim(m):
0034  seen={0};dist=[0]+[m//math.gcd(m,j) for j in range(1,m)];cost=0
0035  while len(seen)<m:
0036   j=min((j for j in range(m) if j not in seen),key=lambda j:dist[j]);cost+=dist[j];seen.add(j)
0037   for k in range(m):
0038    if k not in seen:dist[k]=min(dist[k],m//math.gcd(m,k-j))
0039  return cost
0040 def mst_formula(m):
0041  L=1;ans=0
0042  for d in range(2,m+1):
0043   if m%d==0:
0044    nxt=math.lcm(L,d);ans+=d*(m//L-m//nxt);L=nxt
0045  return ans
