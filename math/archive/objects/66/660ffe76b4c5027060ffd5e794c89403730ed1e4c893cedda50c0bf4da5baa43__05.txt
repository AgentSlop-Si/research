   1 """Auditor-authored exact finite-field helpers. No constructor/source imports."""
   2 def eye(n):return [[int(i==j) for j in range(n)] for i in range(n)]
   3 def mm(a,b,p):return [[sum(x*y for x,y in zip(r,c))%p for c in zip(*b)] for r in a]
   4 def mv(a,x,p):return [sum(u*v for u,v in zip(r,x))%p for r in a]
   5 def add(a,b,p,t=1):return [[(x+t*y)%p for x,y in zip(r,s)] for r,s in zip(a,b)]
   6 def scale(a,t,p):return [[v*t%p for v in r] for r in a]
   7 def block(a,rr,cc):return [[a[i][j] for j in cc] for i in rr]
   8 def inverse(a,p):
   9  n=len(a);z=[r[:]+e for r,e in zip(a,eye(n))]
  10  for j in range(n):
  11   u=next((i for i in range(j,n) if z[i][j]),None)
  12   if u is None:return None
  13   z[j],z[u]=z[u],z[j];iv=pow(z[j][j],-1,p);z[j]=[v*iv%p for v in z[j]]
  14   for i in range(n):
  15    if i!=j:
  16     t=z[i][j];z[i]=[(x-t*y)%p for x,y in zip(z[i],z[j])]
  17  return [r[n:] for r in z]
  18 def det(a,p):
  19  z=[r[:] for r in a];answer=1
  20  for j in range(len(z)):
  21   u=next((i for i in range(j,len(z)) if z[i][j]),None)
  22   if u is None:return 0
  23   if u!=j:z[j],z[u]=z[u],z[j];answer=-answer
  24   v=z[j][j];answer=answer*v%p;iv=pow(v,-1,p)
  25   for i in range(j+1,len(z)):
  26    t=z[i][j]*iv%p;z[i]=[(x-t*y)%p for x,y in zip(z[i],z[j])]
  27  return answer%p
  28 def rank(a,p):
  29  z=[r[:] for r in a];k=0
  30  for j in range(len(z[0]) if z else 0):
  31   u=next((i for i in range(k,len(z)) if z[i][j]),None)
  32   if u is None:continue
  33   z[k],z[u]=z[u],z[k];iv=pow(z[k][j],-1,p)
  34   for i in range(k+1,len(z)):
  35    t=z[i][j]*iv%p;z[i]=[(x-t*y)%p for x,y in zip(z[i],z[k])]
  36   k+=1
  37   if k==len(z):break
  38  return k
  39 def root_table(n,p):
  40  w=pow(5 if p==97 else 3,(p-1)//n,p);assert pow(w,n,p)==1 and pow(w,n//2,p)!=1
  41  out=[1]*n
