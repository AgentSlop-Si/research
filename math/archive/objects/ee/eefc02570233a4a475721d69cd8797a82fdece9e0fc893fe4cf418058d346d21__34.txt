   1 """Permutation-endpoint independent DFT subspace uncertainty gate.
   2 For H of length h and spectral output I of t positions, the intersection
   3 E_H intersect F^{-1}(E_I) has dimension d <= floor(h*t/n), by trace of
   4 orthogonal projection products for normalized complex DFT.
   5 Hence t-floor(ht/n) <= relevant QS rank budget.
   6 """
   7 from itertools import permutations

   9 
  10 def uncertainty_violations(pi,cup=1,cdown=1,bup=1,bdown=1):
  11  n=len(pi); fail=[]
  12  for u in range(n):
  13   a=n;b=-1
  14   for v in range(u,n):
  15    t=v-u+1
  16    a=min(a,pi[v]);b=max(b,pi[v]); h=b-a+1
  17    bcost=(u>0)*bup+(v<n-1)*bdown
  18    budget=bcost+(a>0)*cup+(b<n-1)*cdown
  19    necessary=t-(h*t)//n
  20    if necessary>budget:fail.append((u,v,a,b,t,h,necessary,budget))
  21  return fail
