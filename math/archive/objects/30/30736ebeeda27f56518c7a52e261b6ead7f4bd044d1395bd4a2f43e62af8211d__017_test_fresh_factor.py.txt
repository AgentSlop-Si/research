import random
import unittest
from fresh_factor import (
    filters, filter_signal, coeff_recover, sparse_product_and_recover,
    choose_plan, naive, ntt_convolution, exact_crt, convolve,
    multiply_int, integer_digits,
)

P=998244353

class TestFreshFactor(unittest.TestCase):
    def test_filter_inverse(self):
        rng=random.Random(1)
        for n in [1,2,3,5,8,19,37]:
            a=[rng.randrange(-5,6) for _ in range(n)]
            for q in filters():
                d=filter_signal(a,q)[:n]
                self.assertEqual(coeff_recover(d,q),a)
                dmod=filter_signal(a,q,P)[:n]
                self.assertEqual(coeff_recover(dmod,q,P),[x%P for x in a])

    def test_exact_integer_convolution_all_filters(self):
        rng=random.Random(2)
        for n,m in [(1,1),(2,3),(5,5),(7,11),(19,17),(43,34)]:
            a=[rng.randrange(-5,6) for _ in range(n)]
            b=[rng.randrange(-5,6) for _ in range(m)]
            truth=naive(a,b)
            ff=filters()
            for qa in ff:
                for qb in ff:
                    got=sparse_product_and_recover(a,b,qa,qb)
                    self.assertEqual(got,truth,(n,m,qa,qb))

    def test_modular_convolution_all_filters(self):
        rng=random.Random(3)
        for n,m in [(1,5),(6,7),(17,22),(53,49)]:
            a=[rng.randrange(P) for _ in range(n)]
            b=[rng.randrange(P) for _ in range(m)]
            truth=naive(a,b,P)
            ff=filters()
            for qa in ff:
                for qb in ff:
                    got=sparse_product_and_recover(a,b,qa,qb,P)
                    self.assertEqual(got,truth,(n,m,qa,qb))

    def test_ntt_crt_random_exact(self):
        rng=random.Random(4)
        for n,m in [(1,1),(2,3),(7,10),(33,65),(114,99),(128,151)]:
            a=[rng.randrange(-1000,1000) for _ in range(n)]
            b=[rng.randrange(-1000,1000) for _ in range(m)]
            truth=naive(a,b)
            self.assertEqual(exact_crt(a,b),truth)
            self.assertEqual(ntt_convolution(a,b),[x%P for x in truth])

    def test_force_structured_plan(self):
        rng=random.Random(5)
        b=[rng.randrange(-30,31) for _ in range(150)]
        signals=[
            [7]*160,
            [j+1 for j in range(160)],
            [j*j+3*j+2 for j in range(160)],
            [(-1)**j*4 for j in range(160)],
            [9]*57+[2]*63+[-5]*40,
        ]
        f=[1,1]
        for j in range(2,160): f.append(f[-1]+f[-2])
        signals.append(f)
        for a in signals:
            ans,plan=convolve(a,b,mode='structured')
            self.assertEqual(ans,naive(a,b))
            self.assertLess(plan.nnz_a,len(a)//4,plan.qa)

    def test_auto_dense_ntt(self):
        rng=random.Random(6)
        a=[rng.randrange(P) for _ in range(1024)]
        b=[rng.randrange(P) for _ in range(1023)]
        got,plan=convolve(a,b,mod=P,mode='auto')
        self.assertEqual(got,naive(a,b,P))
        self.assertEqual(plan.method,'ntt')

    def test_auto_structure(self):
        rng=random.Random(7)
        a=[4]*400
        b=[rng.randrange(-20,21) for _ in range(390)]
        result,plan=convolve(a,b,mode='auto')
        self.assertEqual(plan.method,'structured')
        self.assertEqual(result,naive(a,b))

    def test_integer_multiply(self):
        # 400-digit 777...7 x arbitrary 320-digit integer
        rng=random.Random(8)
        for a,b in [(1234,5678),(123456,789001),
                    (int('7'*400),int(''.join(str(rng.randrange(10)) for _ in range(319))+'1'))]:
            got,_=multiply_int(a,b)
            self.assertEqual(got,a*b)

    def test_boundaries_error(self):
        with self.assertRaises(ValueError): filter_signal([], [1,-1])
        with self.assertRaises(ValueError): filter_signal([1], [0,1])
        with self.assertRaises(ValueError): convolve([1],[2],mod=7)
        with self.assertRaises(ValueError): choose_plan([], [1])

    def test_hard_random_case_mod2(self):
        # Demonstrate the invertible triangular map preserves the number of vectors.
        # Exhaustively for n=6 and all binary Q of degree <=2.
        for q in ([1], [1,1], [1,0,1], [1,1,1]):
            observed=set()
            for mask in range(1<<6):
                a=[(mask>>j)&1 for j in range(6)]
                y=filter_signal(a,q,mod=2)[:6]
                observed.add(tuple(y))
            self.assertEqual(len(observed),1<<6)

if __name__=='__main__':unittest.main()

class TestRecurrenceDiscovery(unittest.TestCase):
    def test_fibonacci_mod(self):
        from fresh_factor import find_recurrence
        seq=[1,1]
        for _ in range(350):seq.append((seq[-1]+seq[-2])%P)
        self.assertEqual(find_recurrence(seq,3,p=P),(1,P-1,P-1))
        rng=random.Random(31)
        other=[rng.randrange(P) for _ in range(330)]
        ans,plan=convolve(seq,other,mod=P)
        self.assertEqual(ans,naive(seq,other,P))
        self.assertEqual(plan.method,'structured')
        self.assertLessEqual(plan.nnz_a,4)

    def test_geometric_mod(self):
        from fresh_factor import find_recurrence
        seq=[(17*pow(345678, j, P))%P for j in range(100)]
        self.assertEqual(find_recurrence(seq,2,p=P),(1,(-345678)%P))
        rng=random.Random(37)
        b=[rng.randrange(P) for _ in range(85)]
        out,plan=convolve(seq,b,mod=P)
        self.assertEqual(out,naive(seq,b,P))
        self.assertLessEqual(plan.nnz_a,2)

    def test_integer_lift_and_random_abort(self):
        from fresh_factor import find_recurrence
        seq=[2,3]
        for _ in range(30):seq.append(2*seq[-1]+seq[-2])
        self.assertEqual(find_recurrence(seq,3,integer_lift=True),(1,-2,-1))
        rng=random.Random(52)
        noise=[rng.randrange(2) for _ in range(400)]
        self.assertIsNone(find_recurrence(noise,3,p=P))
        self.assertIsNone(find_recurrence(noise,3,integer_lift=True))

class TestIntegerNearPower(unittest.TestCase):
    def test_near_power_and_generic(self):
        from fresh_factor import multiply_near_power
        rng=random.Random(90210)
        for bits in [1,2,64,512,2048,10000]:
            b=rng.getrandbits(bits)+1
            for a in [(1<<bits)-1,(1<<bits)+3,(1<<bits)-7 if bits>=4 else 3]:
                got,mode=multiply_near_power(a,b)
                self.assertEqual(got,a*b)
                self.assertNotEqual(mode,'general multiplication fallback')
        for i in range(50):
            a=rng.getrandbits(400)+1;b=rng.getrandbits(450)+1
            got,_=multiply_near_power(a,b)
            self.assertEqual(got,a*b)
    def test_invalid(self):
        from fresh_factor import multiply_near_power
        with self.assertRaises(ValueError):multiply_near_power(-1,2)
