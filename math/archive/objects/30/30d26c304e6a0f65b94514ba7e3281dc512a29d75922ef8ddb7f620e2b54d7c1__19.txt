   1 #!/usr/bin/env python3
   2 """Independent exact finite-instance checks. Imports no uploaded implementation.
   3 
   4 Run from any directory. The script uses only the Python standard library and
   5 writes its JSON result next to itself. No source-bundle files are needed.
   6 """
   7 from itertools import product
   8 from pathlib import Path
   9 import json
  10 import math
  11 import random
  12 import time
  13 
  14 P = 998244353
  15 G = 3
  16 
  17 
  18 def dft(v, root):
  19     n = len(v)
  20     return [sum(v[j] * pow(root, (-m*j) % n, P) for j in range(n)) % P
  21             for m in range(n)]
  22 
  23 
  24 def exclusive_scan(v):
  25     """Own balanced integer scan, used modulo P and for exact sums."""
  26     a = list(v)
  27     n = len(a)
  28     stride = 2
  29     while stride <= n:
  30         for right in range(stride-1, n, stride):
  31             a[right] += a[right-stride//2]
  32         stride *= 2
  33     a[-1] = 0
  34     stride = n
  35     while stride >= 2:
  36         for right in range(stride-1, n, stride):
  37             left = right-stride//2
  38             saved = a[left]
  39             a[left] = a[right]
  40             a[right] += saved
  41         stride //= 2
  42     return a
  43

  45 def modular_row(n, m, a, beta):
  46     k = n//m
  47     z = pow(G, (P-1)//n, P)
  48     x = [(j*j*19+23*j+71) % P for j in range(n)]
  49     phase = [[pow(z, (a*t*t+(2*a*m*r+beta)*t) % n, P)
  50               for t in range(m)] for r in range(k)]
  51     d = [sum(phase[r][t]*x[r*m+t] for t in range(m)) % P
  52          for r in range(k)]
  53     projected = [(d[r]*pow(phase[r][t], P-2, P)) % P
  54                  for r in range(k) for t in range(m)]
  55     target = dft(projected, z)
  56     D = dft(d, pow(z, m, P))
  57     out = [None]*n
  58     # This is the unnormalized exact recurrence; omit common 1/sqrt(N).
  59     for seed in (0, 1):
  60         ids = [(seed+2*a*j) % n for j in range(n//2)]
  61         gauges = [pow(z, (a*j*j+(beta+seed)*j) % n, P)
  62                   for j in range(n//2)]
  63         for j in range(len(ids)-1):
  64             assert gauges[j+1] == (gauges[j]*pow(z,(a+beta+ids[j]) % n,P)) % P
  65         forcings = [(D[idx % k]-pow(z,(-a*m*m-idx*m-beta*m) % n,P)
  66                      *D[(idx+2*a*m) % k]) % P for idx in ids]
  67         q = [(f*pow(gauge,P-2,P)) % P for f,gauge in zip(forcings,gauges)]
  68         prefixes = exclusive_scan(q)
  69         seed_value = sum(d[r]*pow(z,(-a*t*t-(2*a*m*r+beta+seed)*t-seed*m*r)%n,P)
  70                          for r in range(k) for t in range(m)) % P
  71         assert seed_value == target[seed]
  72         for idx, gauge, pref in zip(ids,gauges,prefixes):
  73             out[idx] = gauge*(seed_value-pref) % P
  74     assert out == target
  75     return {'N':n,'M':m,'a':a,'beta':beta,'all_outputs_exact':True}

  78 def grid_estimate_twice(word):
  79     """tau=3, W=21, offsets 0,7,14. Return twice the center median."""
  80     centers2 = []
  81     for k in range(3):
  82         cand = 0
  83         count = 0
  84         for z in word:
  85             cell = (z-7*k)//21  # Python // is signed floor, exactly.
  86             if count == 0:
  87                 cand, count = cell, 1
  88             elif cell == cand:
  89                 count += 1
  90             else:
  91                 count -= 1
  92         centers2.append(42*cand+14*k+21)
  93     return sorted(centers2)[1]

  96 def ledger_ratio(L, Lm, t):
  97     c = math.expm1(2*math.log1p(t))
  98     e = c+(1+c)*(c+t*(2+c))
  99     a = math.expm1(math.log1p(c)+Lm*math.log1p(t))
 100     b = math.expm1((L-Lm)*math.log1p(e))
 101     h = math.expm1(math.log1p(a)+math.log1p(b)+math.log1p(c))
 102     s = math.expm1(math.log1p(a)+2*math.log1p(c)+L*math.log1p(t))
 103     f = 2*h+c*(1+h)+t*(2+c)*(1+h)
 104     q = (1+c)*f+2*c
 105     gamma = math.expm1(2*L*math.log1p(t))
 106     pref = q+gamma*(2+q)
 107     M_over_sqrtNM = 2.0**((Lm-L)/2)
 108     coefficient = (1+c)*(1+t)*(M_over_sqrtNM*s+pref)+(3*t*(1+c)+c)*M_over_sqrtNM
 109     return coefficient/(L*t)

 112 def run():
 113     started = time.time()
 114     exact = []
 115     for n,m in [(8,2),(16,2),(32,4),(64,4),(128,8),(256,8)]:
 116         for a,beta in [(1,0),(3,1),(n-1,n-1)]:
 117             exact.append(modular_row(n,m,a,beta))
 118     scan_cases = 0
 119     for ell in range(1,10):
 120         word = [((j*71)%103)-51 for j in range(2**ell)]
 121         actual = exclusive_scan(word)
 122         acc = 0
 123         for value, result in zip(word, actual):
 124             assert result == acc
 125             acc += value
 126         scan_cases += 1
 127     grid_all = grid_majority = endpoint_majority = 0
 128     for mu in range(21):
 129         alphabet = [mu-30,mu-3,mu-2,mu,mu+2,mu+3,mu+30]
 130         for length in (1,3,5):
 131             for word in product(alphabet,repeat=length):
 132                 grid_all += 1
 133                 good = sum(abs(z-mu)<=3 for z in word)
 134                 if 2*good > length:
 135                     grid_majority += 1
 136                     endpoint_majority += any(abs(z-mu)==3 for z in word)
 137                     assert abs(grid_estimate_twice(word)-2*mu) <= 15
 138     rng = random.Random(813257)
 139     random_cases = 3000
 140     for _ in range(random_cases):
 141         mu = rng.randrange(-500,501)
 142         length = rng.choice([7,11,21,51,101])
 143         good_count = rng.randrange(length//2+1,length+1)
 144         word = [mu+rng.randrange(-3,4) for _ in range(good_count)]
 145         word += [mu+rng.choice([-1,1])*rng.randrange(4,10**6) for _ in range(length-good_count)]
 146         rng.shuffle(word)
 147         assert abs(grid_estimate_twice(word)-2*mu) <= 15
 148     largest = (0,None)
 149     ledger_cases = 0
 150     for L in list(range(3,257))+[512,1024,4096,10000,100000,1000000]:
 151         for Lm in sorted(set([1,max(1,(L-1)//4),max(1,(L-1)//2)])):
 152             assert 2*Lm+1 <= L
 153             for frac in [1e-12,1e-9,1e-6,0.001,0.01,0.1,0.5,1.0]:
 154                 t = .01*frac/L
 155                 ratio = ledger_ratio(L,Lm,t)
 156                 assert ratio < 100
 157                 ledger_cases += 1
 158                 if ratio > largest[0]:
 159                     largest = (ratio, {'L':L,'log2_M':Lm,'t':t})
 160     return {
 161         'status':'all independent checks passed',
 162         'uploaded_code_executed':False,
 163         'exact_modular_gauge_cases':exact,
 164         'exact_balanced_scan_cases':scan_cases,
 165         'exact_grid_words_tested':grid_all,
 166         'exact_grid_strict_majority_words':grid_majority,
 167         'majority_words_containing_interval_endpoints':endpoint_majority,
 168         'random_adversarial_grid_cases':random_cases,
 169         'finite_precision_ledger_cases':ledger_cases,
 170         'maximum_ledger_ratio_to_sqrtNM_L_t':largest,
 171         'caveat':'Finite tests supplement the written proofs. They do not certify NumPy/IEEE phase generation or establish general numerical bounds by experiment.',
 172         'elapsed_seconds':time.time()-started,
 173     }
 174 
 175 if __name__ == '__main__':
 176     result = run()
 177     target = Path(__file__).resolve().with_name('independent_v8_results.json')
 178     target.write_text(json.dumps(result,indent=2)+'\n')
 179     print(json.dumps({k:v for k,v in result.items() if k!='exact_modular_gauge_cases'},indent=2))
