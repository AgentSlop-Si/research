   1 #!/usr/bin/env python3
   2 """New audit code: treat all uploaded archive bytes as inert data."""
   3 import hashlib, io, json, re, stat, zipfile
   4 from pathlib import Path, PurePosixPath
   5 
   6 ROOT = Path(__file__).resolve().parent
   7 prior = json.loads((ROOT/'PRIOR_HASH_REFERENCE.json').read_text())
   8 known_base = set(prior['baseline']['hashes'])
   9 known_prev = set(prior['previous_additions']['hashes'])
  10 known = known_base | known_prev
  11 records, manifests, archives = [], [], []
  12 bytes_seen = 0
  13 
  14 def walk(blob, chain, depth=0):
  15     global bytes_seen
  16     assert depth < 12, 'archive nesting depth cap'
  17     with zipfile.ZipFile(io.BytesIO(blob)) as z:
  18         infos = z.infolist()
  19         assert len(infos) < 10000
  20         assert sum(i.file_size for i in infos) < 100_000_000
  21         content = {}
  22         for i in infos:
  23             p = PurePosixPath(i.filename)
  24             assert not p.is_absolute() and '..' not in p.parts and '\\' not in i.filename
  25             assert not stat.S_ISLNK(i.external_attr >> 16)
  26             if i.is_dir():
  27                 continue
  28             assert i.filename not in content, 'duplicate archive path'
  29             b = z.read(i)
  30             bytes_seen += len(b)
  31             assert bytes_seen < 500_000_000
  32             content[i.filename] = b
  33             digest = hashlib.sha256(b).hexdigest()
  34             records.append({'chain':chain,'member_path':i.filename,'sha256':digest,'bytes':len(b),
  35                             'known_in_base_texts':digest in known_base,
  36                             'known_in_prior_additions':digest in known_prev,
  37                             'new_to_combined_prior':digest not in known})
  38         for name, b in content.items():
  39             if PurePosixPath(name).name.lower() in ('sha256sums.txt','sha256sums','manifest.sha256'):
  40                 for line in b.decode('utf-8').splitlines():
  41                     if not line.strip(): continue
  42                     m = re.fullmatch(r'([0-9a-fA-F]{64})\s+\*?(.+)',line)
  43                     if not m:
  44                         manifests.append({'chain':chain,'manifest':name,'line':line,'status':'unparsed'})
  45                         continue
  46                     expected, ref = m.groups()
  47                     candidates = [str(PurePosixPath(name).parent / ref), ref]
  48                     matches = list(dict.fromkeys(p for p in candidates if p in content))
  49                     good = [p for p in matches if hashlib.sha256(content[p]).hexdigest()==expected.lower()]
  50                     manifests.append({'chain':chain,'manifest':name,'referenced_path':ref,
  51                                       'resolved_path':good[0] if good else (matches[0] if matches else None),
  52                                       'expected':expected.lower(),
  53                                       'status':'pass' if good else ('mismatch' if matches else 'missing')})
  54             if name.lower().endswith('.zip'):
  55                 walk(b,chain+'!/'+name,depth+1)
  56 
  57 for p in sorted((ROOT/'source_archives').glob('*.zip')):
  58     b=p.read_bytes(); before=len(records)
  59     walk(b,p.name)
  60     sub=records[before:]
  61     archives.append({'archive':p.name,'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest(),
  62                      'member_occurrences':len(sub),'unique_hashes':len({r['sha256'] for r in sub}),
  63                      'new_unique_hashes':len({r['sha256'] for r in sub if r['new_to_combined_prior']})})
  64 
  65 summary={'member_occurrences':len(records),'unique_hashes':len({r['sha256'] for r in records}),
  66          'known_occurrences':sum(not r['new_to_combined_prior'] for r in records),
  67          'new_unique_hashes':len({r['sha256'] for r in records if r['new_to_combined_prior']}),
  68          'total_recursive_uncompressed_bytes':bytes_seen,'by_archive':archives,
  69          'source_manifest_entries':len(manifests),'source_manifest_passed':sum(r['status']=='pass' for r in manifests),
  70          'source_manifest_failures':[r for r in manifests if r['status']!='pass'],
  71          'comparison_scope':'528 baseline canonical text contents plus all 417 member occurrences of 14:50 additions; inventories remain distinct.'}
  72 (ROOT/'SOURCE_MEMBER_INVENTORY.json').write_text(json.dumps(records,indent=2)+'\n')
  73 (ROOT/'source_manifest_verification.json').write_text(json.dumps(manifests,indent=2)+'\n')
  74 (ROOT/'recursive_summary.json').write_text(json.dumps(summary,indent=2)+'\n')
  75 print(json.dumps(summary,indent=2))
