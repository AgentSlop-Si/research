   1 #!/usr/bin/env python3
   2 """Independent construction checks for the composite bounds and dyadic proof."""
   3 from fractions import Fraction
   4 import math,json
   5 import numpy as np
   6 from independent_composite_probe import phi,full_exact,block_norm
   7 
   8 def layer_masks(m):
   9     p=1<<(m-1).bit_length()
  10     d=np.arange(1,m)[:,None];u=np.arange(m)[None,:]
  11     layers=[d==u]
  12     width=2
  13     while width<=p:
  14         layers.append((d//width==u//width)&(d%width<width//2)&(u%width>=width//2))
  15         width*=2
  16     return layers
  17 
  18 def exact_count_bound(n,m):
  19     shell={g:[0]*m for g in range(1,m) if n%g==0}
  20     for u in range(m):
  21         for d in range(u-m+1,u+1):
  22             if d:shell[math.gcd(n,d)][u]+=1
  23     c=sum((Fraction(max(v),phi(n//g)) for g,v in shell.items()),Fraction(0))
  24     for g,v in shell.items():
  25         assert max(v)<=(m-1)//g
  26         assert Fraction(phi(n),phi(n//g))<=g
  27     coarse=Fraction((m-1)*len(shell),phi(n))
  28     assert c<=coarse
  29     return c,coarse,len(shell)
  30 
  31 def test_explicit_analysis(n,m):
  32     ns=np.arange(n);unitary=np.exp(-2j*np.pi*np.outer(ns,ns)/n)/np.sqrt(n)
  33     rows=[];ds=[]
  34     for d in range(1,m):
  35         g=math.gcd(n,d)
  36         factor=math.sqrt(phi(n)/phi(n//g))
  37         for k in range(n):
  38             if math.gcd(k,n)==g:rows.append(factor*unitary[k]);ds.append(d)
  39     t=np.array(rows);ds=np.array(ds);u=ns%m
  40     tp=t*(ds[:,None]<=u[None,:]);tn=t*(ds[:,None]<=m-1-u[None,:])
  41     q=np.array(full_exact(n,m),float)
  42     residual=np.max(np.abs((tp.conj().T@tp+tn.conj().T@tn)/phi(n)-q))
  43     assert residual<5e-13,(n,m,residual)
  44     opnorm=np.linalg.norm(t,2)
  45     assert opnorm**2<=m-1+1e-11
  46     layers=layer_masks(m);parts=[]
  47     for layer in layers:
  48         a=t*layer[ds-1][:,u]
  49         assert np.linalg.norm(a,2)<=opnorm+1e-11
  50         parts.append(a)
  51     assert np.max(np.abs(sum(parts)-tp))<1e-13
  52     assert np.linalg.norm(tp,2)<=len(layers)*opnorm+1e-11
  53     c,coarse,shells=exact_count_bound(n,m)
  54     norm=block_norm(n,m)
  55     dyadic=Fraction(2*(m-1)*len(layers)**2,phi(n))
  56     assert norm<=float(c)+1e-11
  57     assert norm<=float(dyadic)+1e-11
  58     return dict(N=n,M=m,gram_residual=float(residual),exact_C=str(c),divisor_bound=str(coarse),dyadic_bound=str(dyadic),covariance_norm=norm,unmasked_norm_squared=float(opnorm**2),number_of_layers=len(layers),shells=shells)
  59 
  60 def main():
  61     for m in range(2,513):
  62         masks=layer_masks(m)
  63         s=sum(a.astype(int) for a in masks)
  64         expected=np.arange(1,m)[:,None]<=np.arange(m)[None,:]
  65         assert np.array_equal(s,expected)
  66     count=0
  67     for n in range(4,1001):
  68         for m in range(2,math.isqrt(n)+1):
  69             if n%m==0:exact_count_bound(n,m);count+=1
  70     records=[test_explicit_analysis(n,m) for n,m in [(12,2),(12,3),(18,3),(30,5),(36,6),(60,5),(60,6),(64,8),(72,8),(100,10),(120,10)]]
  71     result=dict(dyadic_partition_checks=511,exact_count_checks=count,explicit_analysis_checks=records,status='passed')
  72     with open('bound_certificate_results.json','w') as f:json.dump(result,f,indent=2)
  73     print(json.dumps(result,indent=2))
  74 if __name__=='__main__':main()
