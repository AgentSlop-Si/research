1: """Exact root-of-unity q-Pascal factorization, research oracle not fast FFT.
2: 
3: All vector outputs are materialized; immutable preparation is charged separately.
4: The input consists of a field, N, and a primitive Nth root q. The field's root
5: finding and exact field representation costs are not unit arithmetic costs.
6: """
7: from collections import Counter
8: from dataclasses import dataclass
9: 
10: 
11: class Counted:
12:     def __init__(self, field):
13:         self.field, self.count = field, Counter()
14:     def __getattr__(self, name):
15:         return getattr(self.field, name)
16:     def add(self, a, b):
17:         self.count['add'] += 1
18:         return self.field.add(a, b)
19:     def sub(self, a, b):
20:         self.count['sub'] += 1
21:         return self.field.sub(a, b)
22:     def mul(self, a, b):
23:         self.count['mul'] += 1
24:         return self.field.mul(a, b)
25:     def neg(self, a):
26:         self.count['neg'] += 1
27:         return self.field.neg(a)
28:     def inv(self, a):
29:         self.count['constant_inverse'] += 1
30:         return self.field.inv(a)
31: 
32: 
33: @dataclass
34: class Plan:
35:     n: int
36:     q: object
37:     factorial: list
38:     kernel: list
39:     middle: list
40:     preparation: dict
41: 
42: 
43: def prepare(field, n, q):
44:     """O(N) scalar preparation; no factorial inversions or lookup table.
45: 
46: One inversion obtains the constant 1/2. In a model supplying 1/2 this is
47: omitted. The power-of-two rational 1/N is then generated by log2(N) multiplies.
48: No input-dependent division occurs. Primitive-root validity is a precondition.
49: """
50:     if n < 2 or n & (n-1):
51:         raise ValueError('N must be a power of two at least 2')
52:     c = Counted(field)
53:     powers = [field.one]
54:     for _ in range(1, n):
55:         powers.append(c.mul(powers[-1], q))
56:     factors = [field.one] + [c.sub(field.one, powers[j]) for j in range(1, n)]
57:     f = [field.one]
58:     for j in range(1, n):
59:         f.append(c.mul(f[-1], factors[j]))
60:     half = c.inv(field.integer(2))
61:     inverse_n = field.one
62:     for _ in range(n.bit_length()-1):
63:         inverse_n = c.mul(inverse_n, half)
64:     suffix, t = field.one, [field.zero] * n
65:     for j in reversed(range(n)):
66:         t[j] = c.mul(suffix, inverse_n)
67:         if j:
68:             suffix = c.mul(suffix, factors[j])
69:     chirp, middle = field.one, []
70:     for j in range(n):
71:         middle.append(c.mul(chirp, t[j]))
72:         if j+1 < n:
73:             chirp = c.neg(c.mul(chirp, powers[j]))
74:     return Plan(n, q, f, t, middle, dict(c.count))
75: 
76: 
77: def lower_convolution(field, kernel, x, traffic=None):
78:     """Direct truncated convolution: quadratic, deliberately not hidden FFT."""
79:     out = []
80:     for i in range(len(x)):
81:         value = field.mul(kernel[i], x[0])
82:         for j in range(1, i+1):
83:             value = field.add(value, field.mul(kernel[i-j], x[j]))
84:         out.append(value)
85:     if traffic is not None:
86:         n = len(x)
87:         traffic['kernel_reads'] += n*(n+1)//2
88:         traffic['input_reads'] += n*(n+1)//2
89:         traffic['convolution_output_writes'] += n
90:     return out

93: def apply(field, plan, x):
94:     """Return F_N x and charged direct-evaluator counts.
95: 
96: F = S T diag(d) T^T S. The same kernel applies T and T^T via reversal.
97: """
98:     n = plan.n
99:     if len(x) != n:
100:         raise ValueError('wrong input length')
101:     c, traffic = Counted(field), Counter()
102:     v = [c.mul(a, b) for a, b in zip(plan.factorial, x)]
103:     v = list(reversed(lower_convolution(c, plan.kernel, list(reversed(v)), traffic)))
104:     v = [c.mul(a, b) for a, b in zip(plan.middle, v)]
105:     v = lower_convolution(c, plan.kernel, v, traffic)
106:     v = [c.mul(a, b) for a, b in zip(plan.factorial, v)]
107:     traffic['reversal_element_copies'] = 2*n
108:     traffic['diagonal_operand_reads'] = 6*n
109:     traffic['diagonal_output_writes'] = 3*n
110:     return v, {'field_operations': dict(c.count), 'array_traffic': dict(traffic)}

113: def hankel(field, plan, r, shift=0):
114:     if shift < 0 or shift+2*r-2 >= plan.n:
115:         raise ValueError('kernel indices outside available coefficients')
116:     return [[plan.kernel[shift+i+j] for j in range(r)] for i in range(r)]

118: 
119: def hankel_determinant_formula(field, plan, r, shift=0):
120:     """All nonzero for shift+2r-2 < N; proof in README.
121: 
122: det[t_(shift+i+j)] = q^(sum_{s=1}^{r-1} s^2)
123:   * product_(i<j)(q^(shift+j)-q^(shift+i))
124:   / product_i f_(shift+i+r-1).
125: """
126:     if shift < 0 or shift+2*r-2 >= plan.n:
127:         raise ValueError('kernel indices outside available coefficients')
128:     out = field.power(plan.q, sum(s*s for s in range(1, r)))
129:     points = [field.power(plan.q, shift+i) for i in range(r)]
130:     for i in range(r):
131:         for j in range(i+1, r):
132:             out = field.mul(out, field.sub(points[j], points[i]))
133:         out = field.mul(out, plan.kernel[shift+i+r-1])
134:     return out
