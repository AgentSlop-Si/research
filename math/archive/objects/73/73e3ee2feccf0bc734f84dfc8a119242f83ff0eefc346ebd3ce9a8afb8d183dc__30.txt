1: """Independent rational-number check for midpoint and cubic gauge theorems.
2: 
3: Does not import degree_boundary; uses exact fractions rather than modular code.
4: """
5: from fractions import Fraction as F
6: from itertools import product
7: from random import Random
8: 
9: 
10: def qpoly(x,A,alpha):
11:     # Degree-two Boolean-correct polynomial, arbitrary target linear map A.
12:     return [sum(A[k][i]*x[i]+alpha[k][i]*(x[i]*x[i]-x[i])
13:                 for i in range(len(x))) for k in range(len(A))]
14: 
15: 
16: def cpoly(x,A,gauge):
17:     n=len(x)
18:     def pairs():
19:         for i in range(n):
20:             for j in range(i+1,n):
21:                 yield (i,j,x[i]*x[i]*x[j]-x[i]*x[j]*x[j])
22:     values=list(pairs())
23:     return [sum(A[k][i]*x[i]**3 for i in range(n))+
24:             sum(gauge[k][t]*v for t,(_,__,v) in enumerate(values))
25:             for k in range(len(A))]
26: 
27: 
28: def cjac(a,h,A,gauge):
29:     n=len(a)
30:     out=[sum(A[k][i]*3*a[i]*a[i]*h[i] for i in range(n))
31:          for k in range(len(A))]
32:     t=0
33:     for i in range(n):
34:         for j in range(i+1,n):
35:             d=2*a[i]*h[i]*a[j]+a[i]**2*h[j]-h[i]*a[j]**2-2*a[i]*a[j]*h[j]
36:             for k in range(len(A)):
37:                 out[k]+=gauge[k][t]*d
38:             t+=1
39:     return out

41: 
42: def det_bareiss(M):
43:     A=[[F(v) for v in row] for row in M]
44:     n=len(A);p=F(1)
45:     for k in range(n-1):
46:         if A[k][k]==0:
47:             swap=next((i for i in range(k+1,n) if A[i][k]),None)
48:             if swap is None:return F(0)
49:             A[k],A[swap]=A[swap],A[k]
50:             p=-p
51:         for i in range(k+1,n):
52:             for j in range(k+1,n):
53:                 A[i][j]=(A[k][k]*A[i][j]-A[i][k]*A[k][j])/p
54:         p=A[k][k]
55:         for i in range(k+1,n):A[i][k]=0
56:     return A[-1][-1]

59: def main():
60:     rng=Random(20261009)
61:     r={'quadratic_boolean_cases':0,'quadratic_general_midpoint_cases':0,
62:        'cubic_boolean_cases':0,'cubic_nplus1_cases':0,'moment_determinant_cases':0}
63:     for n in (2,3,4,5,6):
64:         A=[[F(rng.randrange(-4,5)) for i in range(n)] for k in range(n)]
65:         alpha=[[F(rng.randrange(-4,5)) for i in range(n)] for k in range(n)]
66:         gauge=[[F(rng.randrange(-4,5)) for t in range(n*(n-1)//2)] for k in range(n)]
67:         for bits in product((F(0),F(1)),repeat=n):
68:             ref=[sum(A[k][i]*bits[i] for i in range(n)) for k in range(n)]
69:             assert qpoly(bits,A,alpha)==ref
70:             assert cpoly(bits,A,gauge)==ref
71:             r['quadratic_boolean_cases']+=1;r['cubic_boolean_cases']+=1
72:         for __ in range(8):
73:             h=[F(rng.randrange(-10,11),rng.randrange(1,8)) for i in range(n)]
74:             midpoint=[F(1,2)]*n
75:             vp=qpoly([v+x for v,x in zip(midpoint,h)],A,alpha)
76:             vm=qpoly([v-x for v,x in zip(midpoint,h)],A,alpha)
77:             ref=[sum(A[k][i]*h[i] for i in range(n)) for k in range(n)]
78:             assert [(x-y)/2 for x,y in zip(vp,vm)]==ref
79:             r['quadratic_general_midpoint_cases']+=1
80:             y=cjac([F(1)]*n,h,A,gauge)
81:             for j in range(n):
82:                 e=[F(int(i==j)) for i in range(n)]
83:                 z=cjac(e,h,A,gauge)
84:                 y=[t+u for t,u in zip(y,z)]
85:             assert [v/6 for v in y]==ref
86:             r['cubic_nplus1_cases']+=1
87:         # determinant of M=(I+11^T)/6 equals (n+1)/6^n
88:         M=[[F(2 if i==j else 1,6) for j in range(n)] for i in range(n)]
89:         assert det_bareiss(M)==F(n+1,6**n)
90:         r['moment_determinant_cases']+=1
91:     return r
