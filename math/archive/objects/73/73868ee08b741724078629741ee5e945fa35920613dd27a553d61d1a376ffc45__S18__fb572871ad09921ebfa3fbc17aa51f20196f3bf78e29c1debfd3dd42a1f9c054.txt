   1 """Deterministic manifest and ZIP packaging for Phase23. No external deps."""
   2 from pathlib import Path
   3 from zipfile import ZipFile,ZIP_DEFLATED
   4 import hashlib
   5 ROOT=Path(__file__).resolve().parent
   6 OUT=ROOT.parent/'fft_hero1_phase23_20261010.zip'
   7 
   8 def valid(path):
   9     if not path.is_file() or '__pycache__' in path.parts:return False
  10     if path.name=='MANIFEST.sha256':return False
  11     if path.suffix in ('.pyc','.pyo'):return False
  12     return True
  13 
  14 files=sorted(x for x in ROOT.rglob('*') if valid(x))
  15 lines=[]
  16 for path in files:
  17     rel=path.relative_to(ROOT)
  18     sha=hashlib.sha256(path.read_bytes()).hexdigest()
  19     lines.append(f'{sha}  {rel.as_posix()}')
  20 (ROOT/'MANIFEST.sha256').write_text('\n'.join(lines)+'\n')
  21 with ZipFile(OUT,'w',ZIP_DEFLATED,compresslevel=6) as zipout:
  22     for path in sorted(files+[ROOT/'MANIFEST.sha256']):
  23         zipout.write(path,arcname=str(Path(ROOT.name)/path.relative_to(ROOT)))
  24 print('archive',OUT,'files',len(files)+1,'bytes',OUT.stat().st_size)
