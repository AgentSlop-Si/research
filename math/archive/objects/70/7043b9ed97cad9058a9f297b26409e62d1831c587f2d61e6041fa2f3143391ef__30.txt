9: def increment(plan,fb,injections,initial):
10:     n,p,R=plan['n'],plan['p'],fb['batches']
11:     forcing=[[0]*n for _ in range(R)]
12:     for vertex,source in injections:
13:         value=apply_t(plan,initial[source*n:(source+1)*n])
14:         forcing[vertex]=[(a+b)%p for a,b in zip(forcing[vertex],value)]
15:     chain=[forcing[0][:]]
16:     for r in range(1,R):
17:         chain.append([(a+b)%p for a,b in zip(forcing[r],apply_j(plan,chain[-1]))])
18:     scalar=chain[-1][fb['observed']]*pow(fb['kappa'],-1,p)%p
19:     correction=[scalar*u%p for u in fb['u']]
20:     out=[]
21:     for r in range(R):
22:         out.append([(a+b)%p for a,b in zip(chain[r],correction)])
23:         correction=apply_j(plan,correction)
24:     assert out[0]==[(a+u*out[-1][fb['observed']])%p for a,u in zip(forcing[0],fb['u'])]
25:     for r in range(1,R):
26:         assert out[r]==[(a+b)%p for a,b in zip(forcing[r],apply_j(plan,out[r-1]))]
27:     return out
