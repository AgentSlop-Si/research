   5 def masks(G,q):
   6     assert G>0 and G&(G-1)==0
   7     out=[]
   8     for j in range(G.bit_length()-1):
   9         mask=0
  10         for r in range(G):
  11             for c in range(G):
  12                 if (r>>j)&1==0 and (c>>j)&1==1:
  13                     mask |= ((1<<q)-1) << ((r*G+c)*q)
  14         out.append((q*(1<<j)*(G-1),mask))
  15     return out
  16 
  17 def packed_transpose(x,G,q,ms):
  18     for d,m in ms:
  19         t=((x>>d)^x)&m
  20         x ^= t^(t<<d)
  21     return x
  22 
  23 def oracle(x,G,q):
  24     y=0
  25     for r in range(G):
  26         for c in range(G):
  27             y |= ((x>>((r*G+c)*q))&((1<<q)-1)) << ((c*G+r)*q)
  28     return y
