0005 def field(n):
0006  p=n+1
0007  while any(p%d==0 for d in range(2,math.isqrt(p)+1)):p+=n
0008  for w in range(2,p):
0009   if pow(w,n,p)==1 and pow(w,n//2,p)!=1:return p,w

0011 def dft(x,w,p):return [sum(v*pow(w,-j*t,p) for t,v in enumerate(x))%p for j in range(len(x))]
0012 def fit(features,targets,p):
0013  if not features:return []
0014  s=len(features[0]);A=[list(f)+[y%p] for f,y in zip(features,targets)];piv=[];r=0
0015  for j in range(s):
0016   q=next((i for i in range(r,len(A)) if A[i][j]%p),None)
0017   if q is None:continue
0018   A[q],A[r]=A[r],A[q];v=pow(A[r][j],-1,p);A[r]=[x*v%p for x in A[r]]
0019   for i in range(len(A)):
0020    if i!=r:
0021     z=A[i][j];A[i]=[(x-z*y)%p for x,y in zip(A[i],A[r])]
0022   piv.append(j);r+=1
0023  if any(not any(row[:s]) and row[s] for row in A):return None
0024  ans=[0]*s
0025  for i,j in enumerate(piv):ans[j]=A[i][s]
0026  assert all(sum(a*b for a,b in zip(f,ans))%p==y%p for f,y in zip(features,targets))
0027  return ans
0028 def rank(A,p):
0029  A=[row[:] for row in A];r=0
0030  for j in range(len(A[0])):
0031   i=next((i for i in range(r,len(A)) if A[i][j]%p),None)
0032   if i is None:continue
0033   A[r],A[i]=A[i],A[r];iv=pow(A[r][j],-1,p);A[r]=[x*iv%p for x in A[r]]
0034   for i in range(r+1,len(A)):
0035    z=A[i][j];A[i]=[(x-z*y)%p for x,y in zip(A[i],A[r])]
0036   r+=1
0037   if r==len(A):break
0038  return r
0039 def columns_to_raw(cols):return [cols[t][b] for b in range(len(cols[0])) for t in range(len(cols))]

0044 def rational(cols,r,shifts,w,p):

0046  features=[];targets=[]
0047  for t in range(r,m):
0048   for b in range(k):features.append([cols[t-h][(b+d)%k] for h in range(1,r+1) for d in shifts]);targets.append(cols[t][b])
0049  co=fit(features,targets,p) if features else [0]*(r*len(shifts))

0051  init=[dft(cols[t],wk,p) for t in range(r)];ans=[0]*n;sing=0
0052  for ell in range(k):
0053   coeff=[sum(co[h*len(shifts)+i]*pow(wk,ell*d,p) for i,d in enumerate(shifts))%p for h in range(r)]
0054   seq=[init[t][ell] for t in range(r)]
0055   for t in range(r,m):seq.append(sum(coeff[h-1]*seq[t-h] for h in range(1,r+1))%p)
0056   front=[(seq[u]-sum(coeff[h-1]*seq[u-h] for h in range(1,u+1)))%p for u in range(r)]
0057   back=[-sum(coeff[h-1]*seq[m+u-h] for h in range(u+1,r+1))%p for u in range(r)]
0058   local_sing=0

0061    direct=sum(v*pow(z,t,p) for t,v in enumerate(seq))%p
0062    top=sum((front[u]+pow(wk,-ell,p)*back[u])*pow(z,u,p) for u in range(r))%p
0063    assert den*direct%p==top
0064    if den:got=top*pow(den,-1,p)%p

0067   assert local_sing<=r;sing+=local_sing

0069 def infer_shift(cols,w,p):
0070  k=len(cols[0]);m=len(cols);wk=pow(w,m,p)
0071  a=[sum(cols[0][b]*pow(wk,-ell*b,p) for b in range(k))%p for ell in range(3)]

0075  if not dif:return None

0077  return next((d for d in range(1,k) if pow(wk,d,p)==q),None)

0098  if not any(f):return not any(v)
