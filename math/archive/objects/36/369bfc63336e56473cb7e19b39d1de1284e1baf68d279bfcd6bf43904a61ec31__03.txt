  40 struct Timings {
  41     double fingerprint_ms=0, lookup_ms=0, relation_ms=0, fft_ms=0, reconstruct_ms=0, cache_ms=0;
  42     double total() const { return fingerprint_ms+lookup_ms+relation_ms+fft_ms+reconstruct_ms+cache_ms; }
  43     void add(const Timings& o){ fingerprint_ms+=o.fingerprint_ms; lookup_ms+=o.lookup_ms; relation_ms+=o.relation_ms; fft_ms+=o.fft_ms; reconstruct_ms+=o.reconstruct_ms; cache_ms+=o.cache_ms; }

  46 class FFTWBackend {
  47     struct PlanEntry {
  48         int n{};
  49         Vec scratch_in;
  50         std::vector<std::array<double,2>> scratch_out;
  51         fftw_plan plan{};
  52         explicit PlanEntry(int n_) : n(n_), scratch_in(n_), scratch_out(n_/2+1) {
  53             plan = fftw_plan_dft_r2c_1d(n, scratch_in.data(), reinterpret_cast<fftw_complex*>(scratch_out.data()), FFTW_ESTIMATE_);
  54             if(!plan) throw std::runtime_error("FFTW plan creation failed");
  55         }
  56         ~PlanEntry(){ if(plan) fftw_destroy_plan(plan); }
  57         PlanEntry(const PlanEntry&)=delete; PlanEntry& operator=(const PlanEntry&)=delete;
  58     };
  59     std::unordered_map<int, PlanEntry*> plans_;
  60 public:
  61     ~FFTWBackend(){ for(auto &kv:plans_) delete kv.second; }
  62     Spec rfft(const Vec& x) {
  63         int n=(int)x.size();
  64         auto it=plans_.find(n);
  65         if(it==plans_.end()) it=plans_.emplace(n,new PlanEntry(n)).first;
  66         Spec out(n/2+1);
  67         fftw_execute_dft_r2c(it->second->plan, const_cast<double*>(x.data()), reinterpret_cast<fftw_complex*>(out.data()));
  68         return out;
  69     }
  70 };

  72 static inline uint64_t mix64(uint64_t z){
  73     z ^= z >> 30; z *= 0xbf58476d1ce4e5b9ULL;
  74     z ^= z >> 27; z *= 0x94d049bb133111ebULL;
  75     return z ^ (z >> 31);
  76 }
  77 static uint64_t hash_bytes(const Vec& x) {
  78     // Word-at-a-time non-cryptographic fingerprint. Exact hits are still
  79     // verified with memcmp, so collisions affect speed only, never correctness.
  80     const uint64_t* p=reinterpret_cast<const uint64_t*>(x.data());
  81     uint64_t h=0x9e3779b97f4a7c15ULL ^ mix64((uint64_t)x.size());
  82     for(size_t i=0;i<x.size();i+=4){
  83         uint64_t a=p[i];
  84         uint64_t b=(i+1<x.size()?p[i+1]:0x243f6a8885a308d3ULL);
  85         uint64_t c=(i+2<x.size()?p[i+2]:0x13198a2e03707344ULL);
  86         uint64_t d=(i+3<x.size()?p[i+3]:0xa4093822299f31d0ULL);
  87         h ^= mix64(a + 0x9e3779b97f4a7c15ULL);
  88         h ^= std::rotl(mix64(b),17); h ^= std::rotl(mix64(c),33); h ^= std::rotl(mix64(d),49);
  89         h = h*0x9ddfea08eb382d69ULL + 0x632be59bd9b4e019ULL;
  90     }
  91     return mix64(h);
  92 }

  94 static bool bit_equal(const Vec& a,const Vec& b){
  95     return a.size()==b.size() && std::memcmp(a.data(),b.data(),a.size()*sizeof(double))==0;

  98 static std::pair<double,double> signature(const Vec& a){
  99     long double s=0,q=0;
 100     for(double v:a){s+=v; q+=(long double)v*v;}
 101     return {(double)s,(double)q};
 102 }
 103 static bool similar_sig(std::pair<double,double>a,std::pair<double,double>b){
 104     auto close=[](double x,double y){return std::abs(x-y)<=1e-7+1e-10*std::max(std::abs(x),std::abs(y));};
 105     return close(a.first,b.first)&&close(a.second,b.second);
 106 }

 112 static int cyclic_right_shift(const Vec& base,const Vec& target){
 113     if(base.size()!=target.size()) return -1;
 114     const int n=(int)base.size(); if(n==0) return -1;
 115     auto a=as_bits(base), p=as_bits(target);
 116     std::vector<int> pi(n);
 117     for(int i=1,j=0;i<n;i++){ while(j&&p[i]!=p[j]) j=pi[j-1]; if(p[i]==p[j]) j++; pi[i]=j; }
 118     int j=0;
 119     for(int i=0;i<2*n-1;i++){
 120         uint64_t v=a[i%n];
 121         while(j&&v!=p[j]) j=pi[j-1];
 122         if(v==p[j]) j++;
 123         if(j==n){ int start=i-n+1; if(start<n) return (n-start)%n; j=pi[j-1]; }
 124     }
 125     return -1;
 126 }

 128 static Spec shifted_spectrum(const Spec& base,int n,int shift){
 129     Spec out(base.size());
 130     C omega=std::exp(C(0,-2.0*M_PI*shift/n)), phase(1,0);
 131     for(size_t k=0;k<out.size();k++){ out[k]=base[k]*phase; phase*=omega; }
 132     return out;
 133 }
 134 static Spec sparse_spectrum(const Spec& base,int n,const std::vector<std::pair<int,double>>& edits){
 135     Spec out=base;
 136     for(auto [pos,d]:edits){
 137         if(pos==0){ for(auto &z:out) z+=d; continue; }
 138         C omega=std::exp(C(0,-2.0*M_PI*pos/n)), phase(1,0);
 139         for(auto &z:out){ z += d*phase; phase*=omega; }
 140     }
 141     return out;
 142 }

 148 class AdaptiveFFT {
 149     FFTWBackend& fft_;
 150     std::deque<Entry> cache_;
 151     size_t max_refs_=3;
 152     int max_sparse_=1;
 153 public:
 154     Timings times;
 155     std::unordered_map<std::string,size_t> stats;
 156     explicit AdaptiveFFT(FFTWBackend& f,size_t refs=3,int sparse=1):fft_(f),max_refs_(refs),max_sparse_(sparse){}
 157     void clear(){cache_.clear(); times={}; stats.clear();}
 158     Spec run(const Vec& x,Mode mode){
 159         auto all0=Clock::now(); (void)all0;
 160         uint64_t h=0;
 161         if(mode!=Mode::HashOnly || true){ auto t=Clock::now(); h=hash_bytes(x); times.fingerprint_ms+=ms_since(t); }
 162         if(mode==Mode::HashOnly){
 163             auto t=Clock::now(); auto y=fft_.rfft(x); times.fft_ms+=ms_since(t); stats["fft"]++; return y;
 164         }
 165         {
 166             auto t=Clock::now();
 167             for(auto it=cache_.rbegin();it!=cache_.rend();++it){
 168                 if(it->h==h && bit_equal(it->input,x)){
 169                     Spec y=it->spectrum; times.lookup_ms+=ms_since(t); stats["exact"]++; return y;
 170                 }
 171             }
 172             times.lookup_ms+=ms_since(t);
 173         }
 174         if(mode==Mode::Structural){
 175             auto t=Clock::now();
 176             auto xs=signature(x);
 177             for(auto it=cache_.rbegin();it!=cache_.rend();++it){
 178                 if(it->input.size()!=x.size()) continue;
 179                 if(max_sparse_>0){
 180                     std::vector<std::pair<int,double>> edits; edits.reserve(max_sparse_+1);
 181                     // sampled reject matching Python engine's spirit
 182                     bool promising=true; int sampled=0;
 183                     for(int q=0;q<16 && q<(int)x.size();q++){
 184                         size_t idx=(size_t)q*(x.size()-1)/std::max(1,15);
 185                         if(x[idx]!=it->input[idx] && ++sampled>max_sparse_){promising=false;break;}
 186                     }
 187                     if(promising){
 188                         for(size_t i=0;i<x.size();i++) if(x[i]!=it->input[i]){
 189                             edits.push_back({(int)i,x[i]-it->input[i]});
 190                             if((int)edits.size()>max_sparse_) break;
 191                         }
 192                         if(!edits.empty() && (int)edits.size()<=max_sparse_){
 193                             times.relation_ms+=ms_since(t); auto r=Clock::now(); auto y=sparse_spectrum(it->spectrum,(int)x.size(),edits); times.reconstruct_ms+=ms_since(r); stats["sparse"]++; return y;
 194                         }
 195                     }
 196                 }
 197                 if(similar_sig(xs,it->sig)){
 198                     int shift=cyclic_right_shift(it->input,x);
 199                     if(shift>=0){ times.relation_ms+=ms_since(t); auto r=Clock::now(); auto y=shifted_spectrum(it->spectrum,(int)x.size(),shift); times.reconstruct_ms+=ms_since(r); stats["shift"]++; return y; }
 200                 }
 201             }
 202             times.relation_ms+=ms_since(t);
 203         }
 204         auto tf=Clock::now(); auto y=fft_.rfft(x); times.fft_ms+=ms_since(tf); stats["fft"]++;
 205         auto tc=Clock::now();
 206         Entry e{h,x,y,signature(x)}; cache_.push_back(std::move(e)); while(cache_.size()>max_refs_) cache_.pop_front();
 207         times.cache_ms+=ms_since(tc);
 208         return y;
 209     }
 210 };

 212 struct Rank2Result { bool ok=false; int rank=0; Vec b0,b1; std::vector<std::array<double,2>> c; };
 213 static bool close_val(double a,double b,double tol=2e-12){return std::abs(a-b)<=tol*(1+std::abs(b));}
 214 static Rank2Result detect_rank2(const std::vector<Vec>& rows,double tol=2e-12){
 215     Rank2Result r; if(rows.empty()) return r; size_t n=rows[0].size();
 216     int p0=-1,p1=-1;
 217     for(const auto& row:rows){
 218         if(row.size()!=n) return {};
 219         if(r.rank==0){
 220             auto it=std::max_element(row.begin(),row.end(),[](double a,double b){return std::abs(a)<std::abs(b);});
 221             if(it==row.end() || std::abs(*it)<=tol){ r.c.push_back({0,0}); continue; }
 222             r.b0=row; p0=(int)std::distance(row.begin(),it); r.rank=1; r.c.push_back({1,0}); continue;
 223         }
 224         if(r.rank==1){
 225             double alpha=row[p0]/r.b0[p0]; bool good=true;
 226             for(size_t j=0;j<n;j++) if(!close_val(alpha*r.b0[j],row[j],tol)){good=false;break;}
 227             if(good){r.c.push_back({alpha,0});continue;}
 228             r.b1=row; double best=0; for(size_t j=0;j<n;j++){double d=r.b0[p0]*r.b1[j]-r.b0[j]*r.b1[p0]; if(std::abs(d)>std::abs(best)){best=d;p1=(int)j;}}
 229             double scale=*std::max_element(r.b0.begin(),r.b0.end(),[](double a,double b){return std::abs(a)<std::abs(b);}) * *std::max_element(r.b1.begin(),r.b1.end(),[](double a,double b){return std::abs(a)<std::abs(b);});
 230             (void)scale; if(std::abs(best)<=1e-14) return {}; r.rank=2; r.c.push_back({0,1}); continue;
 231         }
 232         double det=r.b0[p0]*r.b1[p1]-r.b0[p1]*r.b1[p0];
 233         double a=(row[p0]*r.b1[p1]-row[p1]*r.b1[p0])/det;
 234         double b=(r.b0[p0]*row[p1]-r.b0[p1]*row[p0])/det;
 235         bool good=true; for(size_t j=0;j<n;j++) if(!close_val(a*r.b0[j]+b*r.b1[j],row[j],tol)){good=false;break;}
 236         if(!good) return {}; r.c.push_back({a,b});
 237     }
 238     r.ok=r.rank>0; return r;
 239 }

 241 static std::vector<Spec> reconstruct_rank2(FFTWBackend& fft,const Rank2Result& r,Timings& tm){
 242     auto tf=Clock::now(); Spec f0=fft.rfft(r.b0), f1; if(r.rank==2) f1=fft.rfft(r.b1); tm.fft_ms+=ms_since(tf);
 243     auto tr=Clock::now(); std::vector<Spec> out(r.c.size(),Spec(f0.size()));
 244     for(size_t i=0;i<out.size();i++) for(size_t k=0;k<f0.size();k++) out[i][k]=r.c[i][0]*f0[k]+(r.rank==2?r.c[i][1]*f1[k]:C(0));
 245     tm.reconstruct_ms+=ms_since(tr); return out;
 246 }
