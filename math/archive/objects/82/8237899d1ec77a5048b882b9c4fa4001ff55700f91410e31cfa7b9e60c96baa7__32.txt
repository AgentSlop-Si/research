7: // No fast-math assumptions. This is floating-point code, not an exact field FFT.
8: struct C { double r, i; };
9: inline C add(C a,C b) { return {a.r+b.r,a.i+b.i}; }
10: inline C sub(C a,C b) { return {a.r-b.r,a.i-b.i}; }
11: inline C mul(C a,C b) { return {a.r*b.r-a.i*b.i,a.r*b.i+a.i*b.r}; }
12: inline double energy(C a) { return a.r*a.r+a.i*a.i; }
13: inline bool zero(C a) { return a.r==0.0 && a.i==0.0; }

15: struct Stats {
16:     std::uint64_t butterflies=0,visited_nodes=0,threshold_tests=0;
17:     std::uint64_t pruned_nodes=0,pruned_outputs=0,computed_outputs=0;
18:     std::uint64_t norm_value_volume=0,zero_value_tests=0,root_pairs=0;
19:     std::uint64_t full_output_writes=0,max_stack=0;
20: };

22: template<bool Count> void dit(std::vector<C>& a,const std::vector<C>& roots,Stats& s) {
23:     const std::size_t n=a.size();
24:     for(std::size_t i=1,j=0;i<n;i++) {
25:         std::size_t bit=n>>1;
26:         for(;j&bit;bit>>=1) j^=bit;
27:         j^=bit;
28:         if(i<j) std::swap(a[i],a[j]);
29:     }
30:     for(std::size_t len=2;len<=n;len<<=1) {
31:         const std::size_t half=len>>1,step=n/len;
32:         for(std::size_t start=0;start<n;start+=len) {
33:             for(std::size_t j=0;j<half;j++) {
34:                 C x=a[start+j],y=mul(a[start+j+half],roots[j*step]);
35:                 a[start+j]=add(x,y);a[start+j+half]=sub(x,y);
36:             }
37:         }
38:         if constexpr(Count) s.butterflies+=n/2;
39:     }
40: }

42: template<bool Count> void dif(std::vector<C>& a,const std::vector<C>& roots,Stats& s) {
43:     const std::size_t n=a.size();
44:     for(std::size_t len=n;len>=2;len>>=1) {
45:         const std::size_t half=len>>1,step=n/len;
46:         for(std::size_t start=0;start<n;start+=len) {
47:             for(std::size_t j=0;j<half;j++) {
48:                 C x=a[start+j],y=a[start+j+half];
49:                 a[start+j]=add(x,y);a[start+j+half]=mul(sub(x,y),roots[j*step]);
50:             }
51:         }
52:         if constexpr(Count) s.butterflies+=n/2;
53:     }
54:     for(std::size_t i=1,j=0;i<n;i++) {
55:         std::size_t bit=n>>1;
56:         for(;j&bit;bit>>=1) j^=bit;
57:         j^=bit;
58:         if(i<j) std::swap(a[i],a[j]);
59:     }
60: }

62: struct Node { std::size_t start,len,offset,stride;double energy;bool all_zero; };
63: 
64: template<bool Count,bool Approx>
65: void adaptive(std::vector<C>& a,const std::vector<C>& roots,double* output,
66:               double root_energy,double epsilon,double inverse_sqrt_n,double scale,Stats& s) {
67:     const std::size_t n=a.size();
68:     std::fill(output,output+2*n,0.0);
69:     if constexpr(Count) s.full_output_writes=n;
70:     std::vector<Node> stack;stack.reserve(64);
71:     stack.push_back({0,n,0,1,root_energy,false});
72:     const double threshold=epsilon*epsilon*root_energy;
73:     while(!stack.empty()) {
74:         if constexpr(Count) s.max_stack=std::max<std::uint64_t>(s.max_stack,stack.size());
75:         const Node node=stack.back();stack.pop_back();
76:         if constexpr(Count) {s.visited_nodes++;s.threshold_tests++;}
77:         bool prune;
78:         if constexpr(Approx) prune=node.energy <= threshold*double(node.stride);
79:         else prune=node.all_zero;
80:         if(prune) {
81:             if constexpr(Count) {s.pruned_nodes++;s.pruned_outputs+=node.len;}
82:             continue;
83:         }
84:         if(node.len==1) {
85:             const C value=a[node.start];
86:             output[2*node.offset]=(value.r*inverse_sqrt_n)*scale;
87:             output[2*node.offset+1]=(value.i*inverse_sqrt_n)*scale;
88:             if constexpr(Count) s.computed_outputs++;
89:             continue;
90:         }
91:         const std::size_t half=node.len/2,step=n/node.len;
92:         double left_energy=0,right_energy=0;
93:         bool left_zero=true,right_zero=true;
94:         for(std::size_t j=0;j<half;j++) {
95:             const C x=a[node.start+j],y=a[node.start+j+half];
96:             const C left=add(x,y),right=mul(sub(x,y),roots[j*step]);
97:             a[node.start+j]=left;a[node.start+j+half]=right;
98:             if constexpr(Approx) {left_energy+=energy(left);right_energy+=energy(right);}
99:             else {left_zero=zero(left)&&left_zero;right_zero=zero(right)&&right_zero;}
100:         }
101:         if constexpr(Count) {
102:             s.butterflies+=half;
103:             if constexpr(Approx) s.norm_value_volume+=node.len;
104:             else s.zero_value_tests+=node.len;
105:         }
106:         stack.push_back({node.start+half,half,node.offset+node.stride,node.stride*2,right_energy,right_zero});
107:         stack.push_back({node.start,half,node.offset,node.stride*2,left_energy,left_zero});
108:     }
109: }

111: template<bool Count>
112: int transform(const double* input,double* output,std::size_t n,int algorithm,double epsilon,Stats& s) {
113:     if(n==0 || (n&(n-1)) || algorithm<0 || algorithm>2 || !std::isfinite(epsilon) || epsilon<0) return 1;
114:     double scale=0;
115:     for(std::size_t j=0;j<2*n;j++) {
116:         if(!std::isfinite(input[j])) return 2;
117:         scale=std::max(scale,std::abs(input[j]));
118:     }
119:     if(scale==0) {
120:         std::fill(output,output+2*n,0.0);
121:         if constexpr(Count) {s.full_output_writes=n;s.pruned_outputs=n;s.pruned_nodes=1;}
122:         return 0;
123:     }
124:     std::vector<C> a(n),roots(n/2);
125:     double root_energy=0;
126:     for(std::size_t j=0;j<n;j++) {
127:         a[j]={input[2*j]/scale,input[2*j+1]/scale};
128:         if(algorithm==2 && epsilon>0) root_energy+=energy(a[j]);
129:     }
130:     if constexpr(Count) if(algorithm==2 && epsilon>0) s.norm_value_volume+=n;
131:     const double angle=-2.0*std::acos(-1.0)/double(n);
132:     for(std::size_t j=0;j<n/2;j++) roots[j]={std::cos(angle*double(j)),std::sin(angle*double(j))};
133:     // Exactly representable special roots improve exact-zero recognition.
134:     if(n>=2) roots[0]={1,0};
135:     if(n>=4) roots[n/4]={0,-1};
136:     if constexpr(Count) s.root_pairs=n/2;
137:     // Normalize before restoring scale. Computing scale/sqrt(N) first can
138:     // underflow for subnormal inputs even when the true output is representable.
139:     const double inverse_sqrt_n=1.0/std::sqrt(double(n));
140:     if(algorithm==2) {
141:         if(epsilon>0) adaptive<Count,true>(a,roots,output,root_energy,epsilon,inverse_sqrt_n,scale,s);
142:         else adaptive<Count,false>(a,roots,output,root_energy,epsilon,inverse_sqrt_n,scale,s);
143:     } else {
144:         if(algorithm==0) dit<Count>(a,roots,s);else dif<Count>(a,roots,s);
145:         for(std::size_t j=0;j<n;j++) {
146:             output[2*j]=(a[j].r*inverse_sqrt_n)*scale;
147:             output[2*j+1]=(a[j].i*inverse_sqrt_n)*scale;
148:         }
149:         if constexpr(Count) {s.full_output_writes=n;s.computed_outputs=n;}
150:     }
151:     for(std::size_t j=0;j<2*n;j++) if(!std::isfinite(output[j])) return 3;
152:     return 0;
153: }

155: extern "C" int native_fft(const double* input,double* output,std::size_t n,
156:                           int algorithm,double epsilon,Stats* stats) {
157:     Stats local;
158:     int status=stats ? transform<true>(input,output,n,algorithm,epsilon,local)
159:                      : transform<false>(input,output,n,algorithm,epsilon,local);
160:     if(stats) *stats=local;
161:     return status;
162: }

164: extern "C" double consume(const double* data,std::size_t n) {
165:     // Every component is read. The checksum does not overflow on large values.
166:     double scale=0;
167:     for(std::size_t j=0;j<2*n;j++) scale=std::max(scale,std::abs(data[j]));
168:     if(scale==0) return 0;
169:     double sum=0;
170:     for(std::size_t j=0;j<2*n;j++) sum+=(data[j]/scale)*double(1+j%13);
171:     return sum;
172: }
