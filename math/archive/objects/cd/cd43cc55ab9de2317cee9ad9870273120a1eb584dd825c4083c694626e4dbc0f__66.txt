  11 def echelon_rank(a, mod):
  12     a=[list(map(lambda z:z%mod,row)) for row in a]
  13     if not a:return 0
  14     m=len(a);cols=len(a[0]);pivot=0
  15     for col in range(cols):
  16         k=pivot
  17         while k<m and a[k][col]==0:k+=1
  18         if k==m:continue
  19         a[k],a[pivot]=a[pivot],a[k]
  20         inv=pow(a[pivot][col],-1,mod)
  21         for j in range(col,cols):a[pivot][j]=a[pivot][j]*inv%mod
  22         for k in range(pivot+1,m):
  23             mult=a[k][col]
  24             if mult:
  25                 for j in range(col,cols):
  26                     a[k][j]=(a[k][j]-mult*a[pivot][j])%mod
  27         pivot+=1
  28         if pivot==m:break
  29     return pivot

  32 def modular_inverse(x,mod):
  33     n=len(x)
  34     a=[[(v%mod) for v in row]+[int(j==i) for j in range(n)] for i,row in enumerate(x)]
  35     for i in range(n):
  36         k=next((k for k in range(i,n) if a[k][i]),None)
  37         if k is None:raise ValueError('not invertible')
  38         a[i],a[k]=a[k],a[i]
  39         fac=pow(a[i][i],-1,mod)
  40         a[i]=[v*fac%mod for v in a[i]]
  41         for k in range(n):
  42             if k!=i:
  43                 fac=a[k][i]
  44                 if fac:
  45                     a[k]=[(u-fac*v)%mod for u,v in zip(a[k],a[i])]
  46     return [row[n:] for row in a]

  49 def fourier_matrix(n,mod,gen):
  50     root=pow(gen,(mod-1)//n,mod)
  51     return [[pow(root,i*j,mod) for j in range(n)] for i in range(n)]

  54 def qsrank_independent(a,mod):
  55     n=len(a);v=0
  56     for j in range(1,n):
  57         v=max(v,echelon_rank([row[j:] for row in a[:j]],mod))
  58         v=max(v,echelon_rank([row[:j] for row in a[j:]],mod))
  59     return v

  62 def green(n,mod):
  63     def get(t):
  64         mat=[[0]*n for _ in range(n)]
  65         for i in range(n):
  66             mat[i][i]=t
  67             if i:
  68                 mat[i][i-1]=-1
  69                 mat[i-1][i]=-1
  70         return modular_inverse(mat,mod)
  71     a=get(3);b=get(4)
  72     return [[(a[i][j]+b[i][j])%mod for j in range(n)] for i in range(n)]

  75 def run():
