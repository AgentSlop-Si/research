   1	"""Projected relation-aware Fourier state with no O(r*N) spectrum storage.
   2	
   3	A consumer fixes q Fourier bins.  Instead of storing every basis spectrum, it
   4	stores only the r-by-q projection of the basis onto those bins.  After this
   5	preprocessing, N appears only in phase generation for sparse point edits.
   6	
   7	For state x = sum_i c_i b_i + e, where e is sparse and all updates arrive as
   8	trusted relation metadata, the view supports O(1) metadata updates and chooses
   9	at query time between:
  10	
  11	* replay H logged deltas: O(q H), or
  12	* rebuild from current relation: O(q (r+k)), k = current sparse support.
  13	
  14	Thus a query after H updates costs O(q * min(H, r+k)) in this model, provided
  15	the needed recent events are retained.  This is a structured/dynamic result;
  16	it does not compute the first arbitrary dense DFT or discover the relation.
  17	"""
  18	from __future__ import annotations
  19	from collections import deque
  20	from dataclasses import dataclass
  21	import math
  22	from typing import Iterable, Sequence
  23	import numpy as np
  24	
  25	
  26	@dataclass(frozen=True)
  27	class ProjectedStats:
  28	    strategy: str
  29	    backlog: int
  30	    q: int
  31	    r: int
  32	    k: int
  33	    retained_events: int
  34	
  35	
  36	class ProjectedFourierState:
  37	    def __init__(self, n: int, bins: Sequence[int], basis_projection: np.ndarray,
  38	                 coefficients: Sequence[complex] | None = None,
  39	                 sparse_residual: Iterable[tuple[int, complex]] = (),
  40	                 *, event_capacity: int | None = None,
  41	                 prepared_positions: Sequence[int] | None = None):
  42	        self.n=int(n)
  43	        if self.n<1: raise ValueError('n must be positive')
  44	        b=np.asarray(bins,dtype=np.int64)
  45	        if b.ndim!=1 or b.size<1 or np.any(b<0) or np.any(b>=self.n):
  46	            raise ValueError('bad frequency bins')
  47	        self.bins=b.astype(np.intp,copy=True); self.q=len(self.bins)
  48	        proj=np.asarray(basis_projection,dtype=np.complex128)
  49	        if proj.ndim!=2 or proj.shape[1]!=self.q:
  50	            raise ValueError('basis_projection must have shape (r,q)')
  51	        self.projection=np.ascontiguousarray(proj); self.r=proj.shape[0]
  52	        if coefficients is None: self.coefficients=np.zeros(self.r,dtype=np.complex128)
  53	        else:
  54	            c=np.asarray(coefficients,dtype=np.complex128)
  55	            if c.shape!=(self.r,): raise ValueError('bad coefficients')
  56	            self.coefficients=c.copy()
  57	        self.sparse={}
  58	        for j,d in sparse_residual:
  59	            self._point_update(int(j),complex(d),log=False)
  60	
  61	        self._prepared_row={}; self._prepared_phase=None
  62	        if prepared_positions is not None:
  63	            unique=list(dict.fromkeys(int(x) for x in prepared_positions))
  64	            if any(j<0 or j>=self.n for j in unique): raise ValueError('bad prepared position')
  65	            self._prepared_row={j:i for i,j in enumerate(unique)}
  66	            self._prepared_phase=np.exp((-2j*np.pi/self.n)*np.outer(np.asarray(unique,float),self.bins.astype(float)))
  67	
  68	        self.version=0; self.view_version=0
  69	        # None means keep enough history until explicit compact(); otherwise a
  70	        # bounded deque forces stale views to rebuild, keeping memory finite.
  71	        self.event_capacity=None if event_capacity is None else max(0,int(event_capacity))
  72	        self.events=deque()  # (version, kind, key, delta)
  73	        self.cached=self._rebuild_value()
  74	        self.last_stats=ProjectedStats('initial',0,self.q,self.r,len(self.sparse),0)
  75	
  76	    def _phase(self,j:int):
  77	        row=self._prepared_row.get(j)
  78	        if row is not None and self._prepared_phase is not None:
  79	            return self._prepared_phase[row]
  80	        return np.exp((-2j*np.pi*j/self.n)*self.bins.astype(float))
  81	
  82	    def _append(self,kind,key,delta):
  83	        self.version+=1
  84	        self.events.append((self.version,kind,int(key),complex(delta)))
  85	        if self.event_capacity is not None:
  86	            while len(self.events)>self.event_capacity: self.events.popleft()
  87	
  88	    def add_coefficient(self,i:int,delta:complex):
  89	        """Trusted metadata fast path; caller guarantees 0<=i<r and finiteness."""
  90	        z=complex(delta)
  91	        if z==0j:return
  92	        i=int(i); self.coefficients[i]+=z; self._append('c',i,z)
  93	
  94	    def _point_update(self,j:int,d:complex,*,log=True):
  95	        if not 0<=j<self.n: raise ValueError('point outside signal')
  96	        if d==0j:return
  97	        v=self.sparse.get(j,0j)+d
  98	        if v==0j:self.sparse.pop(j,None)
  99	        else:self.sparse[j]=v
 100	        if log:self._append('p',j,d)
 101	
 102	    def add_point(self,j:int,delta:complex):
 103	        """Trusted numeric delta with index validation."""
 104	        self._point_update(int(j),complex(delta),log=True)
 105	
 106	    def _rebuild_value(self):
 107	        y=np.asarray(self.coefficients@self.projection,dtype=np.complex128)
 108	        for j,d in self.sparse.items(): y += d*self._phase(j)
 109	        return y
 110	
 111	    def _history_available(self):
 112	        if self.view_version==self.version:return True
 113	        if not self.events:return False
 114	        return self.events[0][0] <= self.view_version+1
 115	
 116	    def query(self):
 117	        H=self.version-self.view_version
 118	        if H==0:
 119	            self.last_stats=ProjectedStats('cached',0,self.q,self.r,len(self.sparse),len(self.events))
 120	            return self.cached.copy()
 121	        d=self.r+len(self.sparse)
 122	        # The key theorem-level planner: do not even scan a long event log if
 123	        # rebuilding from current factorized state is cheaper.
 124	        if H>max(1,d) or not self._history_available():
 125	            self.cached=self._rebuild_value(); self.view_version=self.version
 126	            self.last_stats=ProjectedStats('rebuild',H,self.q,self.r,len(self.sparse),len(self.events))
 127	            return self.cached.copy()
 128	
 129	        # H <= r+k: replay exactly the chronological deltas.  No coalescing is
 130	        # required for the O(qH) guarantee and this path is friendly to a C++
 131	        # implementation consisting of q-wide AXPYs.
 132	        for ver,kind,key,delta in self.events:
 133	            if ver<=self.view_version: continue
 134	            if kind=='c': self.cached += delta*self.projection[key]
 135	            else: self.cached += delta*self._phase(key)
 136	        self.view_version=self.version
 137	        self.last_stats=ProjectedStats('replay',H,self.q,self.r,len(self.sparse),len(self.events))
 138	        return self.cached.copy()
 139	
 140	    def compact(self):
 141	        """Discard history through the last answered query."""
 142	        while self.events and self.events[0][0]<=self.view_version: self.events.popleft()
 143	
 144	    @property
 145	    def storage_bytes(self):
 146	        base=self.projection.nbytes+self.coefficients.nbytes+self.cached.nbytes
 147	        phase=0 if self._prepared_phase is None else self._prepared_phase.nbytes
 148	        # Python object/dict overhead is intentionally excluded; this is the
 149	        # numeric payload size used for asymptotic/storage comparisons.
 150	        return int(base+phase+len(self.sparse)*(8+16))
