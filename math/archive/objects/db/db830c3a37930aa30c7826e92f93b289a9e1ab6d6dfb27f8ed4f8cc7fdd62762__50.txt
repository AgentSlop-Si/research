2: """Independent exact checker: does not import topology or rank-cut search code.
3: Rational coefficient vectors prove identities for independent arbitrary inputs.
4: Rank witnesses are checked using graph reachability, not search's bitmask pass.
5: """

14: def baseline(n):
15:  pairs=[]
16:  size=1
17:  while size<n:
18:   for first in range(0,n,2*size):
19:    for j in range(size):pairs.append((first+j,first+j+size))
20:   size*=2
21:  return pairs
22: 
23: def target(n):return [[1 if (i&j).bit_count()%2==0 else -1 for j in range(n)] for i in range(n)]
24: 
25: def replay_word(w):
26:  n=w['width'];v=[[F(i==j) for j in range(n)] for i in range(n)]
27:  stats=Counter()
28:  for g in w['word']:
29:   if g['op']=='h2':
30:    i,j=g['pair'];assert 0<=i<n and 0<=j<n and i!=j
31:    a,b=v[i],v[j];v[i]=[x+y for x,y in zip(a,b)];v[j]=[x-y for x,y in zip(a,b)]
32:    stats['kernel_calls']+=1;stats['scalar_additions']+=2;stats['operand_reads']+=4;stats['scalar_writes']+=2
33:   elif g['op']=='diagonal':
34:    assert len(g['values'])==n
35:    for i,s in enumerate(g['values']):
36:     q=F(s);assert q
37:     if q!=1:v[i]=[q*x for x in v[i]];stats['scalar_multiplications']+=1;stats['operand_reads']+=1;stats['scalar_writes']+=1
38:   elif g['op']=='permutation':
39:    p=g['sources'];assert sorted(p)==list(range(n));v=[v[i] for i in p]
40:    stats['materialized_permutation_reads']+=n;stats['materialized_permutation_writes']+=n
41:   else:raise AssertionError('unknown operation')
42:  return v,dict(stats)

44: def verify_word(w,expected):
45:  val,cost=replay_word(w);assert val==expected,'exact full-state operator mismatch';return cost
46: 
47: def from_witness(n,w):
48:  s=baseline(n);assert 0<=w['delete']<len(s);del s[w['delete']]
49:  seen=set()
50:  for pos,a,b in w['replace']:
51:   assert pos not in seen and 0<=pos<len(s) and 0<=a<b<n and (a,b)!=s[pos]
52:   seen.add(pos);s[pos]=(a,b)
53:  return s
54: 
55: def graph(n,s):
56:  # Two fresh value nodes per call are the next versions of the same two
57:  # physical registers, NOT extra simultaneously live data registers.
58:  edges=[[] for _ in range(n+2*len(s))];current=list(range(n))
59:  for t,(i,j) in enumerate(s):
60:   a,b=current[i],current[j];u,v=n+2*t,n+2*t+1
61:   for x in [a,b]:edges[x].extend([u,v])
62:   current[i],current[j]=u,v
63:  return edges,current
64: 
65: def reachable(edges,source,blocked):
66:  seen=set();stack=[source]
67:  while stack:
68:   v=stack.pop()
69:   if v in seen or v in blocked:continue
70:   seen.add(v);stack.extend(edges[v])
71:  return seen

73: def check_cut(n,w,c):
74:  s=from_witness(n,w);edges,ends=graph(n,s)
75:  assert len(c['cut_nodes'])==1 and 0<=c['cut_nodes'][0]<len(edges)
76:  assert len(set(c['outputs']))==2 and all(0<=o<n for o in c['outputs'])
77:  assert len(set(c['inputs']))==len(c['inputs'])>n//2 and all(0<=i<n for i in c['inputs'])
78:  blocked=set(c['cut_nodes'])
79:  for inp in c['inputs']:
80:   visits=reachable(edges,inp,blocked)
81:   assert all(ends[o] not in visits for o in c['outputs']),'cut leaves a path'
82:  assert c['network_rank_upper_bound']==1 and c['target_rank_lower_bound']==2 and c['permutation_invariant']
83:  # Certify the target assertion for EVERY possible row-pair, not merely the
84:  # two named physical output positions. Column permutations retain class sizes.
85:  h=target(n)
86:  for a,b in combinations(range(n),2):
87:   ratio=Counter(h[a][j]*h[b][j] for j in range(n))
88:   assert max(ratio.values())==n//2
89:  return True
90: 
91: def full_graph(n,w):
92:  edges,ends=graph(n,from_witness(n,w))
93:  return all(all(out in reachable(edges,inp,set()) for out in ends) for inp in range(n))

95: def mutation_must_fail(f):
96:  try:f()
97:  except (AssertionError,IndexError,ValueError):return True
98:  raise AssertionError('corrupted certificate accepted')

127:  # Exact H2-to-C identity over Q(i), represented by rational pairs.
128:  def add(z,w):return(z[0]+w[0],z[1]+w[1])
129:  def mul(z,w):return(z[0]*w[0]-z[1]*w[1],z[0]*w[1]+z[1]*w[0])
130:  def mm(a,b):return [[add(mul(a[i][0],b[0][j]),mul(a[i][1],b[1][j])) for j in range(2)]for i in range(2)]

132:  ans=mm(mm(ss,[[a,b],[b,a]]),ss)
