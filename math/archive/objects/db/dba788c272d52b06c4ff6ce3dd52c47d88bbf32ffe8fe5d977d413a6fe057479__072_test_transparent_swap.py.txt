import unittest
import numpy as np
from transparent_swap_check import normalized_dft,generalized_swap,expected_swap

class TransparentSwapTests(unittest.TestCase):
    def test_generalized_swap_unitary(self):
        for B in (2,4,8):
            U=normalized_dft(B)
            W,_=generalized_swap(U)
            self.assertLess(np.linalg.norm(W-expected_swap(U)),1e-12)
            self.assertLess(np.linalg.norm(W.conj().T@W-np.eye(2*B)),1e-12)

if __name__=='__main__': unittest.main()
