   1 """A short readable demonstration of guaranteed sharing and lookup batching."""
   2 import numpy as np
   3 from structural_fft import FFTGraph, SharedLookupTable, fwht_mod, tabulated_fwht_mod
   4 
   5 
   6 def run():
   7     n=16
   8     plain=FFTGraph(n,'radix2')
   9     split=FFTGraph(n,'split_radix')
  10     rng=np.random.default_rng(20261009)
  11     a=rng.standard_normal(n)
  12     b=rng.standard_normal(n)
  13     xa,xb=split.execute_two_real(a,b)
  14     assert np.allclose(xa,np.fft.fft(a))
  15     assert np.allclose(xb,np.fft.fft(b))
  16     print('Two completely different real inputs -> one complex split-radix FFT')
  17     print('Radix-2 generic multiplications per FFT:',plain.stats()['nontrivial_complex_multiplies'])
  18     print('Split-radix generic multiplications per FFT:',split.stats()['nontrivial_complex_multiplies'])
  19     print('Two radix-2 transforms, generic multiplies:',2*plain.stats()['nontrivial_complex_multiplies'])
  20     print('Packed split-radix, generic multiplies:',split.stats()['nontrivial_complex_multiplies'])
  21     print('Plus O(N) arithmetic to decode both spectra.')
  22     print()
  23     x=[1,0,2,1,2,0,1,0,2,1,0,2,2,1,1,0]
  24     cache=SharedLookupTable(p=3)
  25     result, accounting=tabulated_fwht_mod(x,3,block=4,cache=cache)
  26     assert result==fwht_mod(x,3)
  27     print('Walsh-Hadamard transform over F3, 16 arbitrary input symbols:')
  28     print('Input:',x)
  29     print('Output:',result)
  30     print('Lookup passes:',accounting['passes'], 'table lookups:',accounting['lookups'])
  31     print('Full offline-table compilation:',cache.stats())
  32     print('\nNOTE: Finite-field WHT is not an ordinary complex FFT.')
  33 
  34 
  35 if __name__=='__main__':
  36     run()
