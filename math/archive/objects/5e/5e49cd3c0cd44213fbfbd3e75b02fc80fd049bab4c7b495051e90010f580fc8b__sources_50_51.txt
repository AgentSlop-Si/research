  41 def cycle_solve(c, order, coefficients, rhs, closure_inverse):

 123 def signed_cycle_forward(field, values):
 124     """K(C)x for C(x)=(x1,...,xN-1,-x0), including N=1.
 125

 128     """

 133     if two == c.zero: raise ValueError('characteristic two excluded')
 134     half = c.inv(two)

 136     total = values[0]
