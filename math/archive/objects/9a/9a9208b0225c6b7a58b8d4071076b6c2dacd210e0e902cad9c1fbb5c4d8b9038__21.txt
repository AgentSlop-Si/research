0020 def dft(x,w,p):

0027 def idft(x,w,p):return [z*pow(len(x),-1,p)%p for z in dft(x,pow(w,-1,p),p)]
0028 def solve(mat,b,p):

0030  for j in range(n):

0038 def polyprod(factors,p):

0046 def rows_to_x(rows):return [rows[t][u] for u in range(len(rows[0])) for t in range(len(rows))]
0047 def x_to_rows(x,q):return [[x[t+q*u] for u in range(len(x)//q)] for t in range(q)]
0048 def G(z,L,p):return (L%p if z%p==1 else (1-pow(z,L,p))*pow(1-z,-1,p)%p)
