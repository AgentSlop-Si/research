28: def direct(field, x, q):
29:     out = []
30:     for k in range(len(x)):
31:         value = field.zero
32:         for j, xj in enumerate(x):
33:             value = field.add(value, field.mul(xj, field.power(q, j*k)))
34:         out.append(value)
35:     return out

38: def gauss(field, matrix, rhs):
39:     n = len(rhs)
40:     rows = [list(row)+[value] for row, value in zip(matrix, rhs)]
41:     for j in range(n):
42:         pivot = next((i for i in range(j,n) if rows[i][j] != field.zero), None)
43:         if pivot is None:
44:             return None
45:         rows[j], rows[pivot] = rows[pivot], rows[j]
46:         scale = field.inv(rows[j][j])
47:         rows[j] = [field.mul(value, scale) for value in rows[j]]
48:         for i in range(n):
49:             if i != j:
50:                 scale = rows[i][j]
51:                 rows[i] = [field.sub(a, field.mul(scale,b)) for a,b in zip(rows[i],rows[j])]
52:     return [row[-1] for row in rows]

55: def source_matrix(field,n,q,lam,a,b,shift):
56:     matrix = [[field.one if i==j else field.zero for j in range(n)] for i in range(n)]
57:     for i in range(n):
58:         j = (a*i+shift)%n
59:         matrix[i][j] = field.sub(matrix[i][j], field.mul(lam, field.power(q,(b*i)%n)))
60:     return matrix
