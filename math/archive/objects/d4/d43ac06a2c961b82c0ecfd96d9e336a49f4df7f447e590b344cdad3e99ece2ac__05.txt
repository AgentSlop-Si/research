0012 def is_prime(p):

0016 def cycles(n):
0017     k = n.bit_length() - 1
0018     b = 1 << (k // 2)
0019     a = n // b
0020     answer = []
0021     for axis, c, sign in LABELS:
0022         p = []
0023         for v in range(n):
0024             x, y = divmod(v, b)
0025             if axis == "H":
0026                 x = (x + sign * (2 * y + c)) % a
0027             else:
0028                 y = (y + sign * (2 * x + c)) % b
0029             p.append(x * b + y)
0030         seen = bytearray(n)
0031         reps = []
0032         for v in range(n):
0033             if not seen[v]:
0034                 reps.append(v)
0035                 w = v
0036                 while not seen[w]:
0037                     seen[w] = 1
0038                     w = p[w]
0039                 assert w == v
0040         q = p.copy()
0041         for i, v in enumerate(reps):
0042             q[v] = p[reps[(i + 1) % len(reps)]]
0043         seen, v = set(), 0
0044         for _ in range(n):
0045             seen.add(v)
0046             v = q[v]
0047         assert len(seen) == n and v == 0
0048         answer.append(q)

0052 def green(p, denominator, prime=PRIME):
0053     n = len(p)
0054     tau = pow(denominator, -1, prime)
0055     wrap = (1 - pow(tau, n, prime)) % prime
0056     assert wrap
0057     result = [[0] * n for _ in range(n)]
0058     initial = pow(wrap, -1, prime)
0059     for column in range(n):
0060         row, coefficient = column, initial
0061         for _ in range(n):
0062             result[row][column] = coefficient
0063             row = p[row]
0064             coefficient = coefficient * tau % prime
0065     return result

0068 def multiply(a, b, prime=PRIME):

0079 def pivots(a, prime=PRIME):
0080     mat = [row[:] for row in a]
0081     rr, selected = 0, []
0082     for cc in range(len(mat[0])):
0083         pivot = next((i for i in range(rr, len(mat)) if mat[i][cc] % prime), None)
0084         if pivot is None:
0085             continue
0086         mat[rr], mat[pivot] = mat[pivot], mat[rr]
0087         inv = pow(mat[rr][cc], -1, prime)
0088         mat[rr] = [v * inv % prime for v in mat[rr]]
0089         for i in range(rr + 1, len(mat)):
0090             factor = mat[i][cc]
0091             mat[i] = [(u - factor * v) % prime for u, v in zip(mat[i], mat[rr])]
0092         selected.append(cc)
0093         rr += 1
0094         if rr == len(mat):
0095             break
0096     return selected

0099 def determinant(a, prime=PRIME):
0100     mat = [row[:] for row in a]
0101     answer = 1
0102     for j in range(len(mat)):
0103         pivot = next((i for i in range(j, len(mat)) if mat[i][j] % prime), None)
0104         if pivot is None:
0105             return 0
0106         if pivot != j:
0107             mat[j], mat[pivot] = mat[pivot], mat[j]
0108             answer = -answer
0109         factor = mat[j][j]
0110         answer = answer * factor % prime
0111         inverse = pow(factor, -1, prime)
0112         for i in range(j + 1, len(mat)):
0113             scale = mat[i][j] * inverse % prime
0114             for k in range(j + 1, len(mat)):
0115                 mat[i][k] = (mat[i][k] - scale * mat[j][k]) % prime
0116     return answer % prime

0119 def solve(a, rhs, prime=PRIME):
0120     n = len(rhs)
0121     mat = [row[:] + [v] for row, v in zip(a, rhs)]
0122     for j in range(n):
0123         pivot = next(i for i in range(j, n) if mat[i][j] % prime)
0124         mat[j], mat[pivot] = mat[pivot], mat[j]
0125         inv = pow(mat[j][j], -1, prime)
0126         mat[j] = [v * inv % prime for v in mat[j]]
0127         for i in range(n):
0128             if i != j:
0129                 scale = mat[i][j]
0130                 mat[i] = [(u - scale * v) % prime for u, v in zip(mat[i], mat[j])]
0131     return [row[-1] for row in mat]
