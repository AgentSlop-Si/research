#!/usr/bin/env python3
"""Independent exact controls; standard library only; no submitted code imported."""
import itertools, json, random, sys
from pathlib import Path

def ident(n): return [[int(i==j) for j in range(n)] for i in range(n)]
def zero(n,m=None): return [[0]*(n if m is None else m) for _ in range(n)]
def mul(a,b,p):
    out=zero(len(a),len(b[0]))
    for i,row in enumerate(a):
        for k,x in enumerate(row):
            if x:
                for j,y in enumerate(b[k]): out[i][j]=(out[i][j]+x*y)%p
    return out
def scale(a,c,p): return [[x*c%p for x in row] for row in a]
def add(a,b,p): return [[(x+y)%p for x,y in zip(r,s)] for r,s in zip(a,b)]
def sub(a,b,p): return add(a,scale(b,-1,p),p)
def trans(a): return list(map(list,zip(*a)))
def inv(a,p):
    n=len(a); u=[[(x%p) for x in a[i]]+ident(n)[i] for i in range(n)]
    for j in range(n):
        k=next(k for k in range(j,n) if u[k][j]); u[j],u[k]=u[k],u[j]
        u[j]=[x*pow(u[j][j],-1,p)%p for x in u[j]]
        for k in range(n):
            if k!=j:
                c=u[k][j]; u[k]=[(x-c*y)%p for x,y in zip(u[k],u[j])]
    return [r[n:] for r in u]
def powmat(a,n,p):
    z=ident(len(a))
    while n:
        if n&1:z=mul(z,a,p)
        a=mul(a,a,p);n//=2
    return z
def root(n,p):
    return next(w for w in range(1,p) if pow(w,n,p)==1 and all(pow(w,d,p)!=1 for d in range(1,n) if n%d==0))
def fourier(n,p,w=None):
    w=root(n,p) if w is None else w
    return [[pow(w,i*j,p) for j in range(n)] for i in range(n)]
def shift(n):
    z=zero(n)
    for j in range(n): z[(j+1)%n][j]=1
    return z
def diag(v): return [[x if i==j else 0 for j in range(len(v))] for i,x in enumerate(v)]
def blockdiag(blocks):
    n=sum(map(len,blocks)); z=zero(n); offset=0
    for b in blocks:
        for i,row in enumerate(b):z[offset+i][offset:offset+len(b)]=row
        offset+=len(b)
    return z
def kron(a,b): return [[x*y for x in ar for y in br] for ar in a for br in b]
def stat(a):
    rows=[sum(x!=0 for x in r) for r in a]; cols=[sum(x!=0 for x in r) for r in trans(a)]
    return dict(nnz=sum(rows),max_row=max(rows),max_column=max(cols))
def check(x,msg):
    if not x: raise AssertionError(msg)

def vector(a,v,p): return [sum(x*y for x,y in zip(row,v))%p for row in a]
def rank(a,p):
    z=[row[:] for row in a];r=0
    for j in range(len(z[0])):
        candidates=[i for i in range(r,len(z)) if z[i][j]%p]
        if not candidates:continue
        k=candidates[0];z[r],z[k]=z[k],z[r];c=pow(z[r][j],-1,p);z[r]=[x*c%p for x in z[r]]
        for i in range(len(z)):
            if i!=r:
                c=z[i][j];z[i]=[(x-c*y)%p for x,y in zip(z[i],z[r])]
        r+=1
        if r==len(z):break
    return r

def prime(p):return p>=2 and all(p%d for d in range(2,int(p**0.5)+1))
def field(n):return next(p for p in range(n+1,10000,n) if prime(p))
def data(q,m,p):
    n=q*m;w=root(n,p);fm=fourier(m,p,pow(w,q,p));fq=fourier(q,p,pow(w,m,p));L=kron(fq,ident(m));Delta=diag([pow(w,b*t,p) for b in range(q) for t in range(m)]);R=zero(n)
    for b in range(q):
        for s in range(m):R[b*m+s][b+q*s]=1
    return n,w,fm,fq,L,Delta,R

def bank(active,q,m,fm):return blockdiag([fm if b in active else ident(m) for b in range(q)])
def selector(active,q,m):return diag([int(b in active) for b in range(q) for _ in range(m)])
def batch_conversion(q,m,p,w,L):
    check(m%q==0,'fixed radix integer');r=m//q;n=q*m;inp=zero(n);out=zero(n);wm=pow(w,q,p);wr=pow(wm,q,p)
    for b in range(q):
        for h in range(r):
            for s in range(q):inp[s*m+(b*r+h)][b*m+(h+r*s)]=1
        for t in range(q):
            for u in range(r):
                for h in range(r):out[b*m+(t+q*u)][t*m+(b*r+h)]=pow(wm,h*t,p)*pow(wr,h*u,p)%p
    converted=mul(mul(out,L,p),inp,p)
    return converted,inp,out

def interval_case(q,m,p,j,start):
    n,w,fm,fq,L,Delta,R=data(q,m,p);I=ident(n);active={(start+t)%q for t in range(j)};B=bank(active,q,m,fm);C=bank(set(range(q))-active,q,m,fm);P=mul(mul(L,Delta,p),C,p);FF=fourier(n,p,w);di=inv(Delta,p)
    check(mul(mul(P,B,p),R,p)==FF,'CT exterior target');nnz=n*(j+(q-j)*m);check(stat(P)['nnz']==nnz,'exact exterior density')
    check(mul(mul(P,inv(C,p),p),di,p)==L,'correctly ordered inverse stripping')
    k=None;rep=None
    if j:
        k=(q+j-1)//j;rep=zero(n)
        for ell in range(k):
            interval=set(range(ell*j,min((ell+1)*j,q)));E=selector(interval,q,m);a=(start-ell*j)%q
            U=kron(powmat(shift(q),a,p),ident(m));mod=kron(diag([pow(pow(w,m,p),(-a*u)%q,p) for u in range(q)]),ident(m))
            shifted={(b+a)%q for b in interval};check(shifted<=active,'masked support routes into active interval')
            term=mul(mul(mul(mul(mod,P,p),di,p),U,p),E,p);rep=add(rep,term,p)
        check(rep==L,'support-cover batch reduction')
    fullbank=blockdiag([fm]*q);converted,inp,out=batch_conversion(q,m,p,w,L);check(converted==fullbank,'fixed radix bank conversion')
    if j:
        reconstructed_bank=mul(mul(out,rep,p),inp,p);E=selector(active,q,m);BJ=add(mul(E,reconstructed_bank,p),sub(I,E,p),p)
        reconstructed_F=mul(mul(P,BJ,p),R,p);check(reconstructed_F==FF,'k+1 P-call full Fourier reconstruction')
    else:check(mul(P,R,p)==FF,'empty active direct P reconstruction')
    # Square reverse reduction by two L calls; the same matrices check its ordering.
    reverse=False
    if m==q:
        E=selector(active,q,m);CC=add(E,mul(sub(I,E,p),converted,p),p)
        check(mul(mul(L,Delta,p),CC,p)==P,'square two-L reverse reduction');reverse=True
    return dict(q=q,m=m,N=n,p=p,active_size=j,interval_start=start,nnz_P=nnz,batch_P_calls=k,full_Fourier_P_calls=k+1 if j else 1,small_radix=m//q,small_radix_transforms=n//(m//q),square_reverse_checked=reverse)

def coefficient(A,Y,p):
    n=len(A);rows=[]
    for i in range(n):
        for j in range(n):
            v=[0]*(n*n)
            for k in range(n):v[i*n+k]=(v[i*n+k]+A[k][j])%p;v[k*n+j]=(v[k*n+j]-Y[i][k])%p
            rows.append(v)
    return rows

def uniqueness(q,m,active):
    n=q*m;p=field(n);n,w,fm,fq,L,Delta,R=data(q,m,p);B=bank(active,q,m,fm);V=mul(B,R,p);Vi=inv(V,p);P=mul(mul(L,Delta,p),bank(set(range(q))-active,q,m,fm),p);D=diag([pow(w,i,p) for i in range(n)]);S=shift(n)
    A1=mul(mul(V,S,p),Vi,p);A2=mul(mul(V,D,p),Vi,p);Y1=D;Y2=inv(S,p)
    c1=coefficient(A1,Y1,p);c2=coefficient(A2,Y2,p);cal=[0]*(n*n)
    for k in range(n):cal[k]=V[k][0]
    r1=rank(c1,p);r12=rank(c1+c2,p);r123=rank(c1+c2+[cal],p)
    check((r1,r12,r123)==(n*n-n,n*n-1,n*n),'linear uniqueness ranks')
    check(mul(P,V,p)==fourier(n,p,w),'calibrated exact solution')
    return dict(N=n,q=q,m=m,p=p,active=sorted(active),single_generator_rank=r1,two_generator_rank=r12,calibrated_rank=r123)

def wrong_order():
    q=m=4;p=17;n,w,fm,fq,L,D,R=data(q,m,p);C=bank({1,2,3},q,m,fm);P=mul(mul(L,D,p),C,p)
    wrong=mul(mul(P,inv(D,p),p),inv(C,p),p);check(wrong!=L,'noncommuting stripping negative control')
    return dict(N=n,p=p,wrong_order_rejected=True)

def main():
    configs=[(2,2),(3,3),(4,4),(5,5),(2,4),(3,6),(4,8),(3,9)]
    cases=[]
    for q,m in configs:
        p=field(q*m)
        for j in range(q+1):
            for start in ([0] if j in [0,q] else [0,q-1]):cases.append(interval_case(q,m,p,j,start))
    ranks=[uniqueness(q,m,active) for q,m in [(2,2),(2,3),(4,2)] for active in [set(),{0},set(range(q))]]
    result=dict(status='PASS',interval_and_radix_cases=cases,number_cases=len(cases),uniqueness_controls=ranks,wrong_order_control=wrong_order(),uploaded_or_constructor_code_executed=False)
    path=Path(sys.argv[1]) if len(sys.argv)>1 else Path(__file__).with_name('INDEPENDENT_RESULTS.json');path.write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result,indent=2))
if __name__=='__main__':main()
