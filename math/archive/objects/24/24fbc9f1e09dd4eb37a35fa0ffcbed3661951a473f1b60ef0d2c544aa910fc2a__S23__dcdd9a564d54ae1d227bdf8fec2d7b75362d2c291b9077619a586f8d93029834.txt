   1 #!/usr/bin/env python3
   2 import json,random,time
   3 from math import floor,log2
   4 from phase9_dense_cycle import P,Costs,exact_reference_dft
   5 from phase9_spectral_innovations import make_spectral_cycle,exact_spectral_innovation_cycle
   6 
   7 def main():
   8     rows=[]
   9     for exponent in (8,9,10,11,12,13,14,15,16):
  10         n=1<<exponent
  11         m=1<<int(floor(log2(exponent)))
  12         k=n//m
  13         for s in (0,1,3):
  14             rng=random.Random(735+n+s)
  15             events={t:[(ell,rng.randrange(1,P)) for ell in rng.sample(range(k),s)]
  16                     for t in range(1,m-1)} if s else {}
  17             x,meta=make_spectral_cycle(k,m,events,seed=73+s)
  18             start=time.perf_counter()
  19             y,cert,c=exact_spectral_innovation_cycle(x,m,s,3,Costs(),random.Random(1984),True)
  20             elapsed=time.perf_counter()-start
  21             assert y is not None and y==exact_reference_dft(x)
  22             rows.append({'n':n,'k':k,'m':m,'spectral_modes_per_step':s,
  23                 'innovation_modes_total':cert['total_innovation_modes'],
  24                 'field_operations':c.total(),'operations_per_input':round(c.total()/n,3),
  25                 'inversions':c.inversions,'ntt_butterflies':c.ntt_butterflies,
  26                 'seconds_python':round(elapsed,4)})
  27             print('N=',n,'M=',m,'spectral modes=',s,'ops/N=',rows[-1]['operations_per_input'])
  28     open('SPECTRAL_AUDIT_RESULTS.json','w').write(json.dumps({'rows':rows,'note':
  29          'Measured modular field-operation counts, not CPU superiority or general-input FFT speedup.'},indent=2))
  30     print('ALL DUAL SPECTRAL SCALING TESTS EXACT')
  31 if __name__=='__main__':main()
