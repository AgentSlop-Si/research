   1 #!/usr/bin/env python3
   2 """Dual Phase9 innovation decoder: unknown dense C, spectrally sparse innovations.
   3 
   4 If v_{t+1}=C v_t+e_t, first transition clean, C arbitrary circulant,
   5 and each innovation F_K(e_t) has <=s nonzero frequencies, then recover C,
   6 unknown innovation spectral support/values and ALL N Fourier outputs in
   7 O(N(s+1)+K log K+s M^2) exact field operations; for constant s, M=Theta(logN),
   8 this is O(N). Each innovation can be dense in time/position domain.
   9 
  10 A Freivalds check rejects invalid inputs except probability <=p^(-R); verified
  11 accepted promised inputs are evaluated with exact finite-field formulas.
  12 This is not a universal all-input FFT speedup, nor a bounded-bit algorithm.
  13 """
  14 from __future__ import annotations
  15 import random
  16 import secrets
  17 from typing import Optional
  18 from phase8_weyl_cycle import P,Costs,exact_ntt,roots,batch_inverse_nonzero,add,mul,inv
  19 from phase9_dense_cycle import (learn_eigenvalues,decode_sparse_syndromes,
  20                                 ntt_inverse)
  21 from phase6_exact import root_of_unity

  24 def recover_spectral_innovations(cols,lam,tab,max_modes:int,c:Costs):
  25     m=len(cols);k=len(cols[0]);n=k*m
  26     if max_modes==0:return [[] for _ in range(m-1)]
  27     if 2*max_modes>k:raise ValueError('2s<=K required')
  28     wk=tab[m]
  29     # lambda_ell = sum_d c_d omega_K^(ell*d), so c = DFT_minus(lambda)/K.
  30     kernel=ntt_inverse(lam,wk,c)
  31     innovations=[[] for _ in range(m-1)]
  32     for t in range(1,m-1):
  33         probe=[]
  34         for b in range(2*max_modes):
  35             cv=0
  36             for d in range(k):
  37                 cv=add(cv,mul(kernel[d],cols[t][(b+d)%k],c),c)
  38             probe.append(add(cols[t+1][b],-cv,c))
  39         # e_t[b] = sum_ell (E_t[ell]/K)*omega_K^(ell*b).
  40         # Prony recovers ell and amplitude E/K from 2s consecutive b.
  41         recovered=decode_sparse_syndromes(probe,k,wk,max_modes,c)
  42         if recovered is None:return None
  43         innovations[t]=[(ell,mul(amp,k%P,c)) for ell,amp in recovered]
  44     return innovations

  47 def verify_spectral_innovations(cols,lam,innovations,tab,m,c:Costs,rounds=3,rng=None):
  48     """Randomized all-transition Freivalds check with spectral errors."""
  49     if rng is None:rng=secrets.SystemRandom()
  50     k=len(cols[0]);wk=tab[m];wk_inv=tab[-m]
  51     invk=inv(k,c)
  52     for _ in range(rounds):
  53         w=[rng.randrange(P) for _ in range(k)]
  54         fw=exact_ntt(w,wk_inv,c)
  55         fphi=[mul(fw[ell],lam[-ell%k],c) for ell in range(k)]
  56         phi=ntt_inverse(fphi,wk_inv,c)
  57         for t in range(m-1):
  58             lhs=rhs=0
  59             for b in range(k):
  60                 lhs=add(lhs,mul(w[b],cols[t+1][b],c),c)
  61                 rhs=add(rhs,mul(phi[b],cols[t][b],c),c)
  62             # w^T e = (1/K) sum_ell E[ell]*F_minus(w)[-ell]
  63             for ell,ev in innovations[t]:
  64                 rhs=add(rhs,mul(invk,mul(ev,fw[-ell%k],c),c),c)
  65             if lhs!=rhs:return False
  66     return True

  69 def exact_spectral_innovation_cycle(x:list[int],m:int,max_modes_per_step:int,
  70                                      rounds:int=3,c:Optional[Costs]=None,rng=None,
  71                                      return_certificate:bool=False):
  72     c=c or Costs()
  73     n=len(x)
  74     if m<2 or (m&(m-1)) or n&(n-1) or n%m:raise ValueError
  75     k=n//m
  76     if k<m or k&(k-1) or (P-1)%n or max_modes_per_step<0 or 2*max_modes_per_step>k:
  77         raise ValueError
  78     if rounds<1:raise ValueError('at least one Freivalds round required')
  79     c.input_reads+=n
  80     cols=[[x[b*m+t]%P for b in range(k)] for t in range(m)]
  81     f0,lam,tab=learn_eigenvalues(cols,n,c)
  82     if f0 is None:
  83         return (None,{"accepted":False,"reason":"clean first transition impossible"},c) if return_certificate else None
  84     innovations=recover_spectral_innovations(cols,lam,tab,max_modes_per_step,c)
  85     if innovations is None:
  86         return (None,{"accepted":False,"reason":"spectral sparse decoder failed"},c) if return_certificate else None
  87     if not verify_spectral_innovations(cols,lam,innovations,tab,m,c,rounds,rng):
  88         return (None,{"accepted":False,"reason":"independent recurrence fingerprint failed"},c) if return_certificate else None
  89     events_by_ell=[[] for _ in range(k)]
  90     for t,changes in enumerate(innovations):
  91         for ell,E in changes:
  92             length=m-1-t
  93             # rho(ell,j) = lambda_ell*z(ell,0)*omega_M^(-j).
  94             # For a fixed event, generate all M rho^length values by one
  95             # initial exponentiation and a geometric walk: O(M) per event.
  96             rho0=mul(lam[ell],tab[(-ell)%n],c)
  97             power=pow(rho0,length,P)
  98             c.mults+=2*max(1,length.bit_length())
  99             ratio=tab[(-k*length)%n]
 100             powers=[]
 101             for j in range(m):
 102                 powers.append(power)
 103                 power=mul(power,ratio,c)
 104             events_by_ell[ell].append((t,E,powers))
 105     # One output sweep + O(#events * M) exact correction work.
 106     answer=[0]*n
 107     nonsingular=[];indices=[];singular=[]
 108     for ell in range(k):
 109         eigenpower=1
 110         for _ in range(m):eigenpower=mul(eigenpower,lam[ell],c)
 111         # rho^M = lambda^M * omega_K^(-ell), independent of j.
 112         base_num=mul(f0[ell],add(1,-mul(eigenpower,tab[(-ell*m)%n],c),c),c)
 113         for j in range(m):
 114             index=ell+j*k
 115             z=tab[(-index)%n]
 116             rho=mul(lam[ell],z,c)
 117             den=add(1,-rho,c)
 118             if den:
 119                 # stage all N reciprocal denominators for one batch inversion
 120                 numer=base_num
 121                 for t,E,powers in events_by_ell[ell]:
 122                     z_t=tab[(-index*(t+1))%n]
 123                     numer=add(numer,mul(mul(E,z_t,c),add(1,-powers[j],c),c),c)
 124                 indices.append((index,numer));nonsingular.append(den)
 125             else:
 126                 y=mul(m%P,f0[ell],c)
 127                 for t,E,_ in events_by_ell[ell]:
 128                     y=add(y,mul(mul(E,tab[(-index*(t+1))%n],c),(m-1-t)%P,c),c)
 129                 singular.append((index,y))
 130     for inverse,(index,num) in zip(batch_inverse_nonzero(nonsingular,c),indices):
 131         answer[index]=mul(num,inverse,c)
 132     for index,y in singular:answer[index]=y
 133     cert={"accepted":True,"n":n,"k":k,"m":m,"mode":"dual-spectral-innovation",
 134           "max_modes_per_step":max_modes_per_step,
 135           "total_innovation_modes":sum(map(len,innovations)),
 136           "error_vectors_dense_in_position_domain":True,
 137           "freivalds_rounds":rounds,"false_acceptance_upper_bound":f"{P}^-{rounds}",
 138           "counted_ops":c.total(),"counts":c.as_dict(),
 139           "detected_innovations":innovations}
 140     return (answer,cert,c) if return_certificate else answer

 143 def make_spectral_cycle(k:int,m:int,innovations:dict[int,list[tuple[int,int]]],seed=0,
 144                         lam_override=None):
 145     rng=random.Random(seed);n=k*m
 146     if n&(n-1) or k<m or (P-1)%n:raise ValueError
 147     wn=root_of_unity(n);wk=pow(wn,m,P);wk_inv=pow(wk,P-2,P)
 148     v0=[rng.randrange(1,P) for _ in range(k)]
 149     f0=exact_ntt(v0,wk_inv)
 150     if any(a==0 for a in f0):raise AssertionError
 151     lam=[rng.randrange(1,P) for _ in range(k)] if lam_override is None else list(lam_override)
 152     spectrum=f0[:];cols=[v0]
 153     for t in range(m-1):
 154         spectrum=[lam[ell]*spectrum[ell]%P for ell in range(k)]
 155         for ell,E in innovations.get(t,[]):
 156             spectrum[ell]=(spectrum[ell]+E)%P
 157         cols.append(ntt_inverse(spectrum,wk_inv))
 158     x=[cols[t][b] for b in range(k) for t in range(m)]
 159     kernel=ntt_inverse(lam,wk)
 160     return x,{"lambda":lam,"kernel":kernel,"innovations":innovations}
