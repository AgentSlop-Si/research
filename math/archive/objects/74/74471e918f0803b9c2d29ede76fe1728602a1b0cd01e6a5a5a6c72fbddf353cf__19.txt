 115 def givens_qr(a):
 116     n=len(a);r=a.copy();q=np.eye(n,dtype=complex);count=0
 117     for k in range(n-1):
 118         x,y=r[k,k],r[k+1,k]
 119         if abs(y)==0:continue
 120         rho=math.sqrt(abs(x)**2+abs(y)**2)
 121         g=np.array([[x.conjugate(),y.conjugate()],[-y,x]])/rho
 122         r[k:k+2]=g@r[k:k+2]
 123         q[:,k:k+2]=q[:,k:k+2]@g.conj().T
 124         count+=1
 125     phases=np.diag(r)/abs(np.diag(r))
 126     r=phases.conjugate()[:,None]*r;q=q*phases[None,:]
 127     return q,r,count

 129 def haar_gates(m):
 130     h=np.eye(m);count=0;leaves=[]
