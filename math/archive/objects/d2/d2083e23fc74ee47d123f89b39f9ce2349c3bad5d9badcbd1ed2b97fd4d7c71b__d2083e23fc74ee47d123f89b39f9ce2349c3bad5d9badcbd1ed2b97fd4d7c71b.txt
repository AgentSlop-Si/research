"""Re-run Phase12-16 tests plus Phase16 independent audit scripts."""
import os
import json
import re
import subprocess
import sys
from pathlib import Path

HERE=Path(__file__).resolve().parent
ROOT=HERE.parent
PHASE_FOLDERS=[
  ROOT/'fft_green_phase12_unitary_escape',
  ROOT/'fft_green_phase13_phase_coherence',
  ROOT/'fft_green_phase14_galois_coherence',
  ROOT/'fft_green_phase15_folding',
  HERE,
]
SCRIPTS=['entropy_gate_audit.py','component_entropy.py','green_givens_compiler.py',
         'haar_green_givens.py','haar_entropy.py']

def run(cmd,cwd,env):
    p=subprocess.run(cmd,cwd=cwd,env=env,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE,check=True)
    return p.stdout,p.stderr

def main():
    env=os.environ.copy()
    env['OPENBLAS_NUM_THREADS']='1'
    env['PYTHONPATH']=os.pathsep.join(str(p) for p in PHASE_FOLDERS)
    results=[]
    summary_lines=[]
    for phase,cwd in zip(range(12,17), PHASE_FOLDERS):
        out,err=run([sys.executable,'-m','unittest','discover','-s','tests','-v'],cwd,env)
        (HERE/f'phase{phase}_test.log').write_text(out+err)
        match=re.search(r'Ran (\d+) tests',err)
        assert match,(phase,err)
        count=int(match.group(1))
        results.append(dict(phase=phase,tests=count,status='PASS'))
        summary_lines.append(f'Phase {phase}: {count} tests passed')
    independent={}
    for script in SCRIPTS:
        out,err=run([sys.executable,script],HERE,env)
        (HERE/f'{script[:-3]}.log').write_text(out+err)
        records=[json.loads(line) for line in out.splitlines() if line.strip()]
        independent[script]=dict(records=len(records),first=records[0],last=records[-1])
        summary_lines.append(f'{script}: {len(records)} audited cases passed')
    assert sum(r['tests'] for r in results)==55
    results.append(dict(total_tests=55,status='PASS'))
    report=dict(phases=results,independent_audits=independent,
        status='All inherited and Phase16 tests passed; no faster FFT claimed.')
    (HERE/'validation_results.json').write_text(json.dumps(report,indent=2))
    (HERE/'validation.log').write_text('\n'.join(summary_lines)+'\n')
    print('\n'.join(summary_lines))

if __name__=='__main__':main()
