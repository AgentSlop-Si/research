  11 def invperm(p):
  12  q=[None]*len(p)
  13  for i,j in enumerate(p):q[j]=i
  14  return q

  16 def pmatrix(p):return [[int(p[j]==i) for j in range(len(p))] for i in range(len(p))]
  17 def transpose(a):return list(map(list,zip(*a)))
  18 def add(a,b,f,k=1):return [[(x+k*y)%f.p for x,y in zip(r,s)] for r,s in zip(a,b)]
  19 def scale(a,t,f):return [[x*t%f.p for x in r] for r in a]
  20 def left(a,d,f):return [[v*x%f.p for x in r] for r,v in zip(a,d)]
  21 def right(a,d,f):return [[x*v%f.p for x,v in zip(r,d)] for r in a]
  22 def green(p,t,f):
  23  a=add(f.eye(len(p)),pmatrix(p),f,-t);g=f.inverse(a)
  24  assert f.mul(a,g)==f.eye(len(p)) and f.mul(g,a)==f.eye(len(p))
  25  return g

  27 def block(a,sig,tau,f):
  28  n=len(a)//2;h=f.inv(2)
  29  return [[h*(a[i][j]+tau*a[i][j+n]+sig*a[i+n][j]+sig*tau*a[i+n][j+n])%f.p for j in range(n)] for i in range(n)]

  31 def reflect(p):
  32  order=[];v=0
  33  for k in range(len(p)):order.append(v);v=p[v]
  34  assert v==0 and len(set(order))==len(p)
  35  r=[None]*len(p)
  36  for k,v in enumerate(order):r[v]=order[-k%len(p)]
  37  return r

  39 def paircheck(p0,p1,h,f,t,weighted=False):
  40  N=len(p0);n=N//2;I=f.eye(N);zero=[[0]*N for _ in range(N)]
  41  assert t and (1-pow(t,N,f.p))%f.p
  42  for p in (p0,p1):
  43   assert len(cycles(p))==1
  44   assert all(p[v+n]==(p[v]+n)%N for v in range(n))
  45   for v in range(N):
  46    w=v
  47    for _ in range(n):w=p[w]
  48    assert w==(v+n)%N
  49  assert [v%n for v in p0]==[v%n for v in p1]

  51  g0,g1=green(p0,t,f),green(p1,t,f);k=add(g0,g1,f,-1)
  52  assert f.rank(k)==h and all(k[i+n][j]==-k[i][j]%f.p and k[j][i+n]==-k[j][i]%f.p for i in range(n) for j in range(N))
  53  C=block(k,-1,-1,f);assert f.rank(C)==h
  54  a=list(range(1,n+1)) if weighted else [1]*n
  55  b=list(range(n+1,2*n+1)) if weighted else [1]*n
  56  aa=a+[-x for x in a];bb=b+[-x for x in b]
  57  m=left(k,aa,f);assert f.mul(m,m)==zero
  58  results=[]
  59  for tr in (False,True):
  60   rev=right(transpose(k) if tr else k,bb,f);assert f.mul(rev,rev)==zero
  61   s=f.mul(add(I,m,f),add(I,rev,f));si=f.mul(add(I,rev,f,-1),add(I,m,f,-1))
  62   assert f.mul(s,si)==I and f.mul(si,s)==I and f.det(s)==1
  63   A=left(C,a,f);B=right(transpose(C) if tr else C,b,f)
  64   assert block(s,1,1,f)==add(f.eye(n),f.mul(A,B),f)
  65   assert block(s,1,-1,f)==A and block(s,-1,1,f)==B and block(s,-1,-1,f)==f.eye(n)
  66   ranks=[f.rank(A),f.rank(B)];rankdiff=f.rank(add(s,I,f,-1));assert ranks==[h,h] and rankdiff==2*h
  67   results.append({'transpose_reverse':tr,'cross_ranks':ranks,'rank_S_minus_I':rankdiff,'determinant':1,'both_inverse_products':True})
  68  # Reciprocal and reflection implementations, checked in full physical coordinates.
  69  for p,g in ((p0,g0),(p1,g1)):
  70   ip=invperm(p);gi=green(ip,t,f);gr=green(p,f.inv(t),f)
  71   assert gi==transpose(g)==add(I,gr,f,-1)
  72   r=reflect(p);assert all(r[r[v]]==v and r[p[r[v]]]==ip[v] for v in range(N))
  73   assert gi==[[g[r[i]][r[j]] for j in range(N)] for i in range(N)]
  74  if h==n:
  75   assert p1==[(v+n)%N for v in p0]
  76   T=block(pmatrix(p0),-1,-1,f)
  77   formula=scale(f.mul(T,f.inverse(add(f.eye(n),f.mul(T,T),f,-t*t))),2*t,f)
  78   assert C==formula and f.det(C)
  79  return {'N':N,'h':h,'relative_cycles':cc,'weighted':weighted,'shears':results}

  81 def makepair(q,epsilon,flip):
  82  n=len(q)
  83  return [[q[k]+n*(s^epsilon[k]^(int(k in flip) if which else 0)) for s in range(2) for k in range(n)] for which in (0,1)]
