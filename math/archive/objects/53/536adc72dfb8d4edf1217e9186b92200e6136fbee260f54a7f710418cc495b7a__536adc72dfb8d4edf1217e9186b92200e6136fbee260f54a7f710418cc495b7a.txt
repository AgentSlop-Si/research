import unittest
from numeric_audit import (sharp_gap_control,sharp_sparse_weyl_gap,random_sparse_covariance,
                          inverse_sparse_floor,near_resonance_real)

class NumericTests(unittest.TestCase):
    def test_sharp_gap_8(self):self.assertAlmostEqual(sharp_gap_control(8)['ratio'],1,places=11)
    def test_sharp_gap_32(self):self.assertAlmostEqual(sharp_gap_control(32)['ratio'],1,places=11)
    def test_sharp_gap_128(self):self.assertAlmostEqual(sharp_gap_control(128)['ratio'],1,places=10)
    def test_sparse_gap_s1(self):
        r=sharp_sparse_weyl_gap(64,1,3);self.assertAlmostEqual(r['ratio'],r['exact_gap'],places=12)
    def test_sparse_gap_s3(self):
        r=sharp_sparse_weyl_gap(64,3,3);self.assertAlmostEqual(r['ratio'],r['exact_gap'],places=12)
    def test_sparse_gap_s15(self):
        r=sharp_sparse_weyl_gap(64,15,3);self.assertAlmostEqual(r['ratio'],r['exact_gap'],places=12)
    def test_random_sparse_gap(self):
        r=random_sparse_covariance(64);self.assertGreaterEqual(r['actual']+1e-12,r['guaranteed'])
    def test_inverse_sparse_floor_16(self):
        r=inverse_sparse_floor(16,3);self.assertGreaterEqual(r['actual_operator_error']+1e-10,r['guaranteed_floor'])
    def test_inverse_sparse_floor_64(self):
        r=inverse_sparse_floor(64,3);self.assertGreaterEqual(r['actual_operator_error']+1e-10,r['guaranteed_floor'])
    def test_two_inverse_sparse_floor_32(self):
        r=inverse_sparse_floor(32,3,t=2);self.assertGreaterEqual(r['actual_operator_error']+1e-10,r['guaranteed_floor'])
    def test_two_inverse_sparse_floor_64(self):
        r=inverse_sparse_floor(64,3,t=2);self.assertGreaterEqual(r['actual_operator_error']+1e-10,r['guaranteed_floor'])
    def test_near_resonance_not_phase_coherent(self):
        a=near_resonance_real(32);b=near_resonance_real(64)
        self.assertGreater(b['condition'],a['condition'])
        self.assertGreater(b['relative_first_defect'],0.1)
        self.assertGreater(b['relative_second_defect'],0.1)

if __name__=='__main__':unittest.main()
