   6 def zeros(a,b=None):return [[0]*(a if b is None else b) for _ in range(a)]
   7 def I(n):return [[int(i==j) for j in range(n)] for i in range(n)]
   8 def add(A,B,p,k=1):return [[(a+k*b)%p for a,b in zip(x,y)] for x,y in zip(A,B)]
   9 def scale(A,k,p):return [[k*x%p for x in row] for row in A]
  10 def mm(A,B,p):
  11     R=zeros(len(A),len(B[0]))
  12     for i,row in enumerate(A):
  13         for k,x in enumerate(row):
  14             if x:
  15                 for j,y in enumerate(B[k]):
  16                     if y:R[i][j]=(R[i][j]+x*y)%p
  17     return R
  18 def mv(A,x,p):return [sum(a*b for a,b in zip(row,x))%p for row in A]
  19 def vv(x,y,p,k=1):return [(a+k*b)%p for a,b in zip(x,y)]
  20 def dot(x,y,p):return sum(a*b for a,b in zip(x,y))%p
  21 def inv(A,p):

  24         k=next((k for k in range(j,n) if R[k][j]),None)
  25         if k is None:return None
  26         R[j],R[k]=R[k],R[j];a=pow(R[j][j],-1,p);R[j]=[x*a%p for x in R[j]]
  27         for k in range(n):
  28             if k!=j:
  29                 a=R[k][j];R[k]=[(x-a*y)%p for x,y in zip(R[k],R[j])]
  30     return [r[n:] for r in R]
  31 def det(A,p):
  32     R=[r[:] for r in A];d=1
  33     for j in range(len(R)):
  34         k=next((k for k in range(j,len(R)) if R[k][j]),None)
  35         if k is None:return 0
  36         if k!=j:R[j],R[k]=R[k],R[j];d=-d
  37         d=d*R[j][j]%p;a=pow(R[j][j],-1,p)
  38         for k in range(j+1,len(R)):
  39             z=R[k][j]*a%p;R[k]=[(x-z*y)%p for x,y in zip(R[k],R[j])]
  40     return d%p
  41 def rank(A,p):
  42     R=[r[:] for r in A];h=0
  43     for j in range(len(R[0])):
  44         k=next((k for k in range(h,len(R)) if R[k][j]),None)
  45         if k is None:continue

  47         for k in range(h+1,len(R)):
  48             a=R[k][j];R[k]=[(x-a*y)%p for x,y in zip(R[k],R[h])]
  49         h+=1
  50         if h==len(R):break

  52 def sub(A,rows,cols):return [[A[i][j] for j in cols] for i in rows]
  53 def tr(A):return [list(r) for r in zip(*A)]
  54 def function(n,m,f,weights):

  57     return A
  58 def border(A,a,b,c):return [r+[a[i]] for i,r in enumerate(A)]+[b+[c]]
  59 def disp(X,A,Y,p):return add(mm(X,A,p),mm(A,Y,p),p,-1)
  60 def poly_eval(Cs,z,p):
