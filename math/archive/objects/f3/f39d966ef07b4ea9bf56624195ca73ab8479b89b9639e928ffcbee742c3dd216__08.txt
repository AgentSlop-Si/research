   8 def boundary(a,n):return sum((j in a)!=((j+1) in a) for j in range(n-1))
   9 def mask_set(n,m):return {j for j in range(n) if (j&m).bit_count()%2==0}
  10 def bformula(k,m):return sum(2**(k-1-s) for s in range(k) if (m&((1<<(s+1))-1)).bit_count()%2)
  11 def primitive(n,p):
  12     for a in range(2,p):
  13         w=pow(a,(p-1)//n,p)
  14         if pow(w,n,p)==1 and (n==1 or pow(w,n//2,p)!=1):return w
  15     raise AssertionError

  36 def parity_and_twists():

  73 def stage_products():

  75     for trial in range(200):

  78         for layer,r in enumerate(ranks):

  82             stage=[[rng.randrange(p) if i==j else sum(factors[0 if i>j else 2][i][k]*factors[1 if i>j else 3][j][k] for k in range(r))%p for j in range(n)] for i in range(n)]
