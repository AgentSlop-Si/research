   1	"""Relation-aware FFT dispatcher — research implementation, real 1D inputs only.
   2	
   3	Supported relations are provided by the producer, not guessed from arbitrary
   4	unrelated data: known sparse edits, known shifts, and known linear combinations.
   5	Output can remain lazy (selected bins) or be materialized by several algorithms.
   6	
   7	SciPy (or NumPy) implements the full numerical FFT. This does NOT implement
   8	OpenAI's exact #130 algorithm and is not a new worst-case FFT theorem.
   9	"""

  29	def _signal(x):
  30	    a = np.asarray(x)
  31	    if a.ndim != 1 or a.size < 1 or np.iscomplexobj(a):
  32	        raise ValueError('expected nonempty 1-D real input')
  33	    if not np.issubdtype(a.dtype, np.number):
  34	        raise ValueError('expected numeric real input')
  35	    a = np.array(a, dtype=np.float64, order='C', copy=True)
  36	    if not np.isfinite(a).all():
  37	        raise ValueError('NaN/Inf not supported')
  38	    return a

  41	def _validate_spectrum(z, n):
  42	    # Real-input rFFT must have real DC and (if N even) real Nyquist.
  43	    z[0] = complex(z[0].real)
  44	    if n % 2 == 0:
  45	        z[-1] = complex(z[-1].real)
  46	    return z

  49	def _retained_array_bytes(array):
  50	    """Count an ndarray's retained backing allocation, including view padding."""
  51	    owner = array
  52	    while isinstance(owner.base, np.ndarray):
  53	        owner = owner.base
  54	    return owner.nbytes

  57	class Base:
  58	    """Immutable raw input and rFFT shared by multiple state versions."""
  59	    def __init__(self, signal, spectrum):
  60	        self.signal = signal
  61	        self.spectrum = spectrum
  62	        self.signal.flags.writeable = False
  63	        self.spectrum.flags.writeable = False
  64	
  65	    @property
  66	    def memory_bytes(self):
  67	        return self.signal.nbytes + self.spectrum.nbytes

  70	class SpectralState:
  71	    """Linear combination of shifted cached bases plus sparse input edits.
  72	
  73	    State never implicitly creates a new dense spectrum after a known edit.
  74	    Editing and forking are cheap in the number of distinct terms/edits.
  75	    Note: users must not mutate dictionary internals; public methods validate.
  76	    """
  77	    def __init__(self, engine, n, terms, edits=None):
  78	        self.engine = engine
  79	        self.n = int(n)
  80	        self.terms = dict(terms)        # {(Base, right_shift): real scalar}
  81	        self.edits = dict(edits or {})  # {input_index: real delta}
  82	        self._cached_output = None
  83	        self._cached_bytes = 0
  84	        self._mutations = 0
  85	        engine._states.add(self)
  86	
  87	    @property
  88	    def distinct_edits(self):
  89	        return len(self.edits)
  90	
  91	    @property
  92	    def term_count(self):
  93	        return len(self.terms)
  94	
  95	    def fork(self):
  96	        return SpectralState(self.engine, self.n, self.terms, self.edits)
  97	
  98	    def edit(self, position, delta):
  99	        p = int(position)
 100	        d = float(delta)
 101	        if not (0 <= p < self.n) or not math.isfinite(d):
 102	            raise ValueError('invalid edit position/delta')
 103	        value = self.edits.get(p, 0.0) + d
 104	        if not math.isfinite(value):
 105	            raise ValueError('edit overflow')
 106	        if value == 0.0:
 107	            self.edits.pop(p, None)
 108	        else:
 109	            self.edits[p] = value
 110	        self._invalidate()
 111	        self.engine.stats['edits'] += 1
 112	        return self
 113	
 114	    def _invalidate(self):
 115	        self._cached_output = None
 116	        self._cached_bytes = 0
 117	        self._mutations += 1
 118	
 119	    def shift(self, amount):
 120	        """Return a new right-circular-shifted state without moving N samples."""
 121	        s = int(amount) % self.n
 122	        terms = {(base, (offset + s) % self.n): value
 123	                 for (base, offset), value in self.terms.items()}
 124	        edits = {(p+s) % self.n: value for p, value in self.edits.items()}
 125	        return SpectralState(self.engine, self.n, terms, edits)
 126	
 127	    def bin(self, frequency):
 128	        return self.engine.bin(self, frequency)
 129	
 130	    def bins(self, frequencies):
 131	        return self.engine.bins(self, frequencies)
 132	
 133	    def spectrum(self, method='auto', cache_output=False, copy=True):
 134	        return self.engine.materialize(self, method, cache_output, copy)
 135	
 136	    def signal(self):
 137	        return self.engine.signal(self)

 150	class RelationFFT:
 151	    """Cost-aware dispatcher with bounded *internal* caches.
 152	
 153	    max_cache_bytes limits strong references held for implicit exact lookup.
 154	    max_phase_bytes limits cached phase-vector bytes. Explicitly held states
 155	    and bases belong to the caller; they are separately reported as live bytes.
 156	    max_temp_bytes is a conservative pruned work-array eligibility budget,
 157	    not a hard allocator/RSS limit. Backend/runtime hidden allocations and
 158	    separately budgeted persistent phase-cache storage are excluded.
 159	
 160	    The conservative fallback avoids expensive data-driven relation searches.
 161	    Relation metadata (edit index/shift/basis coefficients) is passed explicitly.
 162	    Optional calibrate(state) benchmarks methods on actual state and selects a
 163	    method for future states of the same size/shape.
 164	    """
 165	    def __init__(self, *, max_cache_bytes=64*1024**2,
 166	                 max_phase_bytes=32*1024**2,
 167	                 max_temp_bytes=192*1024**2,
 168	                 small_native_cutoff=4096):
 169	        self.max_cache_bytes = int(max_cache_bytes)
 170	        self.max_phase_bytes = int(max_phase_bytes)
 171	        self.max_temp_bytes = int(max_temp_bytes)
 172	        self.small_native_cutoff = int(small_native_cutoff)
 173	        if min(self.max_cache_bytes, self.max_phase_bytes, self.max_temp_bytes) < 0:
 174	            raise ValueError('byte budgets must be nonnegative')
 175	        self._exact: OrderedDict[tuple, Base] = OrderedDict()
 176	        self._exact_bytes = 0
 177	        self._phase: OrderedDict[tuple, np.ndarray] = OrderedDict()
 178	        self._phase_bytes = 0
 179	        self._states = weakref.WeakSet()
 180	        self._bases = weakref.WeakSet()
 181	        self._profiles: dict[tuple, Profile] = {}
 182	        self._auto_cache_misses = 0
 183	        self._auto_cache_skips = 0
 184	        self.stats = Counter()
 185	        self.last_method = None
 186	
 187	    @staticmethod
 188	    def _key(a):
 189	        # Digest then mandatory full-array verification: hashes alone NEVER prove equality.
 190	        return (int(a.size), hashlib.blake2b(a.tobytes(), digest_size=16).digest())
 191	
 192	    def _fetch(self, a, *, populate):
 193	        key = self._key(a)
 194	        base = self._exact.get(key)
 195	        if base is not None and np.array_equal(a, base.signal):
 196	            self._exact.move_to_end(key)
 197	            self.stats['exact_hits'] += 1
 198	            return base
 199	        self.stats['exact_misses'] += 1
 200	        z = np.asarray(_fft.rfft(a), dtype=np.complex128)
 201	        base = Base(a, z)
 202	        self._bases.add(base)
 203	        if populate and base.memory_bytes <= self.max_cache_bytes:
 204	            prior = self._exact.pop(key, None)
 205	            if prior is not None:
 206	                self._exact_bytes -= prior.memory_bytes
 207	            self._exact[key] = base
 208	            self._exact_bytes += base.memory_bytes
 209	            while self._exact_bytes > self.max_cache_bytes:
 210	                _, victim = self._exact.popitem(last=False)
 211	                self._exact_bytes -= victim.memory_bytes
 212	                self.stats['cache_evictions'] += 1
 213	        return base
 214	
 215	    def register(self, signal, *, cache_exact=False):
 216	        x = _signal(signal)
 217	        base = self._fetch(x, populate=cache_exact) if cache_exact else Base(x, np.asarray(_fft.rfft(x), dtype=np.complex128))
 218	        self._bases.add(base)
 219	        self.stats['reference_ffts'] += 1
 220	        return SpectralState(self, len(x), {(base, 0): 1.0})
 221	
 222	    def fft(self, signal, *, cache='auto', copy=True):
 223	        """Unannotated input: conservative exact cache or native FFT.
 224	
 225	        cache=True: always hash, exact-verify hits, store when within budget.
 226	        cache=False: always invoke native FFT.
 227	        cache='auto' (default): stop hashing after three consecutive misses;
 228	            periodically probe (1 of every 8 later requests) so it can resume.
 229	        Explicitly supplied relations never need this slow discovery route.
 230	        """
 231	        if cache not in (True, False, 'auto'):
 232	            raise ValueError("cache must be True, False, or 'auto'")
 233	        x = _signal(signal)
 234	        should_try = cache is True or (cache == 'auto' and self._auto_cache_misses < 3)
 235	        if cache == 'auto' and not should_try:
 236	            self._auto_cache_skips += 1
 237	            should_try = self._auto_cache_skips % 8 == 0
 238	        if not should_try or self.max_cache_bytes == 0:
 239	            self.last_method = 'native'
 240	            self.stats['native_ffts'] += 1
 241	            if cache == 'auto': self.stats['hash_bypasses'] += 1
 242	            return np.asarray(_fft.rfft(x), dtype=np.complex128)
 243	        prev_hits=self.stats['exact_hits']
 244	        base = self._fetch(x, populate=True)
 245	        hit=self.stats['exact_hits']>prev_hits
 246	        if cache == 'auto':
 247	            self._auto_cache_misses = 0 if hit else self._auto_cache_misses + 1
 248	        self.last_method = 'exact_hit' if hit else 'native_cached'
 249	        z = base.spectrum.copy() if copy else base.spectrum.view()
 250	        if not copy: z.flags.writeable = False
 251	        return z
 252	
 253	    def combine(self, states: Sequence[SpectralState], coefficients: Sequence[float]):
 254	        """Build x=sum c_i * x_i, without forming any signal or spectrum."""
 255	        if len(states) != len(coefficients) or not states:
 256	            raise ValueError('states and coefficients must have same positive length')
 257	        n = states[0].n
 258	        terms, edits = {}, {}
 259	        for state, raw_c in zip(states, coefficients):
 260	            if state.engine is not self or state.n != n:
 261	                raise ValueError('states must belong to this engine and share length')
 262	            c = float(raw_c)
 263	            if not math.isfinite(c):
 264	                raise ValueError('nonfinite combination coefficient')
 265	            for key, a in state.terms.items():
 266	                terms[key] = terms.get(key, 0.0) + c*a
 267	            for p, delta in state.edits.items():
 268	                edits[p] = edits.get(p, 0.0) + c*delta
 269	        terms = {key: val for key, val in terms.items() if val}
 270	        edits = {key: val for key, val in edits.items() if val}
 271	        return SpectralState(self, n, terms, edits)
 272	
 273	    def states_from_bases(self, basis_signals, coefficient_matrix):
 274	        """M different signals described by M x r supplied coefficients.
 275	
 276	        Cost of forming spectra after preparation: O(M*r*N).
 277	        """
 278	        matrix = np.asarray(coefficient_matrix, dtype=np.float64)
 279	        if matrix.ndim != 2 or not np.isfinite(matrix).all():
 280	            raise ValueError('coefficient_matrix must be finite 2D')
 281	        bases = [self.register(x) for x in basis_signals]
 282	        if len(bases) != matrix.shape[1]:
 283	            raise ValueError('number of bases != coefficient columns')
 284	        return [self.combine(bases, row) for row in matrix]
 285	
 286	    def _phase_vector(self, n, position, count):
 287	        position = int(position) % n
 288	        key = (int(n), position, int(count))
 289	        existing = self._phase.get(key)
 290	        if existing is not None:
 291	            self._phase.move_to_end(key)
 292	            self.stats['phase_hits'] += 1
 293	            return existing
 294	        self.stats['phase_misses'] += 1
 295	        phase = phases_tiled(n, position, count=count)
 296	        phase.flags.writeable = False
 297	        retained_bytes = _retained_array_bytes(phase)
 298	        if retained_bytes <= self.max_phase_bytes:
 299	            self._phase[key] = phase
 300	            self._phase_bytes += retained_bytes
 301	            while self._phase_bytes > self.max_phase_bytes:
 302	                _, old = self._phase.popitem(last=False)
 303	                self._phase_bytes -= _retained_array_bytes(old)
 304	        return phase

 306	    @staticmethod
 307	    def _profile_key(state):
 308	        return (state.n, state.distinct_edits, state.term_count,
 309	                any(shift for (_,shift) in state.terms))
 310	
 311	    def recommend(self, state):
 312	        self._check_state(state)
 313	        k, r, n = state.distinct_edits, state.term_count, state.n
 314	        if r == 1 and k == 0:
 315	            ((base,shift), c), = state.terms.items()
 316	            if shift == 0 and c == 1.0:
 317	                return 'reuse'
 318	        if r == 0 and k == 0:
 319	            return 'zero'
 320	        key = self._profile_key(state)
 321	        if key in self._profiles:
 322	            return self._profiles[key].best_method
 323	        # Deliberately conservative: SciPy is far faster than Python bookkeeping
 324	        # on modest arrays. Only inexpensive known relationships are preferred.
 325	        if n < self.small_native_cutoff:
 326	            return 'native'
 327	        if not k and r <= 2:
 328	            return 'linear'
 329	        if k == 1 and r <= 2 and n >= 16384:
 330	            return 'direct'
 331	        if k <= 2 and r <= 2 and n >= 65536:
 332	            return 'direct'
 333	        return 'native'

 335	    def _check_state(self, state):
 336	        if not isinstance(state, SpectralState) or state.engine is not self:
 337	            raise TypeError('state must belong to this RelationFFT instance')

 339	    def signal(self, state):
 340	        self._check_state(state)
 341	        out = np.zeros(state.n, dtype=np.float64)
 342	        for (base, shift), c in state.terms.items():
 343	            if shift:
 344	                out += c*np.roll(base.signal, shift)
 345	            else:
 346	                out += c*base.signal
 347	        for position, delta in state.edits.items():
 348	            out[position] += delta
 349	        return out
 350	
 351	    def bin(self, state, frequency):
 352	        self._check_state(state)
 353	        q = int(frequency)
 354	        if not 0 <= q <= state.n//2:
 355	            raise IndexError('rfft frequency index out of range')
 356	        val = 0j
 357	        for (base, shift), c in state.terms.items():
 358	            angle = -2*math.pi*((q*shift)%state.n)/state.n
 359	            val += c*base.spectrum[q]*complex(math.cos(angle), math.sin(angle))
 360	        for p, d in state.edits.items():
 361	            angle = -2*math.pi*((q*p)%state.n)/state.n
 362	            val += d*complex(math.cos(angle),math.sin(angle))
 363	        if q == 0 or (state.n % 2 == 0 and q == state.n//2):
 364	            return complex(val.real)
 365	        return val
 366	
 367	    def bins(self, state, frequencies: Iterable[int]):
 368	        self._check_state(state)
 369	        q = np.asarray(list(frequencies), dtype=np.int64)
 370	        if (q < 0).any() or (q > state.n//2).any():
 371	            raise IndexError('rfft frequency index out of range')
 372	        out = np.zeros(q.size, dtype=np.complex128)
 373	        for (base, shift), c in state.terms.items():
 374	            if shift:
 375	                phase = np.exp(-2j*np.pi*((q*shift) % state.n)/state.n)
 376	                out += c*base.spectrum[q]*phase
 377	            else:
 378	                out += c*base.spectrum[q]
 379	        for p,d in state.edits.items():
 380	            out += d*np.exp(-2j*np.pi*((q*p) % state.n)/state.n)
 381	        mask = (q==0) | ((state.n%2==0) & (q==state.n//2))
 382	        out[mask] = out[mask].real
 383	        return out
 384	
 385	    def _linear_spectrum(self, state):
 386	        n=state.n
 387	        out = np.zeros(n//2+1, dtype=np.complex128)
 388	        for (base, shift), c in state.terms.items():
 389	            if shift:
 390	                phase = self._phase_vector(n, shift, out.size)
 391	                out += c*base.spectrum*phase
 392	            else:
 393	                out += c*base.spectrum
 394	        return out
 395	
 396	    def _direct_spectrum(self, state):
 397	        out = self._linear_spectrum(state)
 398	        for p,d in state.edits.items():
 399	            out += d*self._phase_vector(state.n,p,out.size)
 400	        return _validate_spectrum(out, state.n)

 402	    @staticmethod
 403	    def _pruned_required_bytes(state):
 404	        """Conservative visible-array estimate shared by dispatch and execution.
 405	
 406	        Charge six length-N complex buffers for rows, FFT output, transpose
 407	        copy, half-spectrum output/addition, phase/multiplication staging and
 408	        workspace reserve; also six length-m float buffers and 64 KiB for
 409	        tile padding/small allocations. This intentionally overestimates the
 410	        normal array peak. Unspecified FFT/runtime hidden allocations are
 411	        not bounded; this is not a hard process-memory guarantee.
 412	        """
 413	        n, k = state.n, state.distinct_edits
 414	        if not k:
 415	            return 0  # no sparse/pruned kernel is needed
 416	        if n & (n-1):
 417	            raise ValueError('pruned mode requires power-of-two length')
 418	        b = 1 << (k-1).bit_length()
 419	        if b > n:
 420	            raise ValueError('more edited coordinates than signal positions')
 421	        return 96*n + 48*(n//b) + 64*1024
 422	
 423	    def _pruned_spectrum(self,state):
 424	        """Residue-class decomposition for known sparse changes.
 425	
 426	        b=2^ceil(log2 k), m=n/b. For k edits at t=r+b*l,
 427	        B[q,r] += delta * exp(-2*pi*i*q*t/n), q in [0,m).
 428	        Then FFT_b(B[q,:])[j] = D[q+j*m]. Only O(N) input preparation
 429	        (because m*k<=N), plus N/b length-b FFTs and O(N) output.
 430	        Fully explicit complex temporary array takes 16*N bytes.
 431	        This implementation is NUMERICAL, not OpenAI #130.
 432	        """
 433	        n = state.n
 434	        k = state.distinct_edits
 435	        if not k:
 436	            return _validate_spectrum(self._linear_spectrum(state),n)
 437	        if n & (n-1):
 438	            raise ValueError('pruned mode requires power-of-two length')
 439	        b=1 << (k-1).bit_length()
 440	        if self._pruned_required_bytes(state) > self.max_temp_bytes:
 441	            raise MemoryError('pruned work-array estimate exceeds temporary budget')
 442	        m = n//b
 443	        rows = np.zeros((m,b),dtype=np.complex128)
 444	        q = np.arange(m,dtype=np.float64)
 445	        for t,d in state.edits.items():
 446	            # Floating phase in bounded range; avoid huge multipliers where possible.
 447	            angle = -2*np.pi*(np.remainder(q*t,n))/n
 448	            rows[:,t%b] += d*np.exp(1j*angle)
 449	        transformed = _fft.fft(rows,axis=1)
 450	        # DFT[k] is indexed as k=q+m*j.
 451	        delta = transformed.T.ravel()[:n//2+1]
 452	        return _validate_spectrum(self._linear_spectrum(state)+delta,n)
 453	
 454	    def materialize(self, state, method='auto', cache_output=False, copy=True):
 455	        self._check_state(state)
 456	        if method not in ('auto','reuse','zero','linear','direct','pruned','native'):
 457	            raise ValueError('unknown materialization strategy')
 458	        if state._cached_output is not None:
 459	            self.last_method='stored_output'
 460	            out=state._cached_output.copy() if copy else state._cached_output.view()
 461	            if not copy: out.flags.writeable=False
 462	            return out
 463	        if method=='auto':
 464	            method=self.recommend(state)
 465	        self.last_method=method
 466	        self.stats['strategy_'+method]+=1
 467	        if method=='reuse':
 468	            if state.distinct_edits!=0 or state.term_count!=1:
 469	                raise ValueError('reuse only valid for one unedited base')
 470	            ((base,shift),c),=state.terms.items()
 471	            if shift or c!=1.0:
 472	                raise ValueError('reuse requires unshifted unit coefficient')
 473	            out=base.spectrum.copy()  # protect cache from client modification
 474	        elif method=='zero':
 475	            if state.term_count or state.distinct_edits:
 476	                raise ValueError('zero strategy only for identically zero state')
 477	            out=np.zeros(state.n//2+1,dtype=np.complex128)
 478	        elif method=='linear':
 479	            if state.distinct_edits:
 480	                raise ValueError('linear strategy requires no pending edits')
 481	            out=_validate_spectrum(self._linear_spectrum(state),state.n)
 482	        elif method=='direct':
 483	            out=self._direct_spectrum(state)
 484	        elif method=='pruned':
 485	            out=self._pruned_spectrum(state)
 486	        elif method=='native':
 487	            out=np.asarray(_fft.rfft(self.signal(state)),dtype=np.complex128)
 488	        if cache_output:
 489	            # Explicit opt-in. Live output memory is caller-held, not an LRU item.
 490	            state._cached_output=out.copy()
 491	            state._cached_output.flags.writeable=False
 492	            state._cached_bytes=state._cached_output.nbytes
 493	        if not copy:
 494	            out.flags.writeable=False
 495	        return out

 497	    def calibrate(self, state, *, repeats=4, include_pruned=True):
 498	        """Measure true end-to-end reconstruction of this state.
 499	
 500	        The first trial includes cold setup for a given method, the median
 501	        represents repeat-query throughput (warm-cache). No input relationship
 502	        discovery is assumed; it is supplied by the producer.
 503	        """
 504	        self._check_state(state)
 505	        if repeats < 2:
 506	            raise ValueError('repeats must be >=2')
 507	        if state._cached_output is not None:
 508	            raise ValueError('cannot calibrate a state with a stored dense result; use a fork()')
 509	        methods=['native']
 510	        if state.distinct_edits:
 511	            methods+=['direct']
 512	            if (include_pruned and state.n & (state.n-1)==0
 513	                    and self._pruned_required_bytes(state) <= self.max_temp_bytes):
 514	                methods+=['pruned']
 515	        else:
 516	            methods+=['linear']
 517	        benchmark={}
 518	        reference=np.asarray(_fft.rfft(self.signal(state)),dtype=np.complex128)
 519	        for method in methods:
 520	            runs=[]
 521	            for run in range(repeats):
 522	                t0=time.perf_counter_ns()
 523	                try:
 524	                    out=self.materialize(state,method=method,cache_output=False)
 525	                except MemoryError:
 526	                    if method != 'pruned':
 527	                        raise
 528	                    # A backend allocation may fail despite the estimate.
 529	                    # Do not select a candidate that did not finish all trials.
 530	                    runs=[]
 531	                    self.stats['calibration_pruned_memory_skips'] += 1
 532	                    break
 533	                elapsed=(time.perf_counter_ns()-t0)*1e-9
 534	                np.testing.assert_allclose(out,reference,rtol=2e-8,atol=2e-7)
 535	                runs.append(elapsed)
 536	            if len(runs) == repeats:
 537	                benchmark[method] = float(np.median(runs))
 538	        best=min(benchmark,key=benchmark.get)
 539	        prof=Profile(state.n,state.distinct_edits,state.term_count,
 540	                     any(shift for (_,shift) in state.terms),benchmark,best)
 541	        self._profiles[self._profile_key(state)]=prof
 542	        return prof

 544	    def memory_report(self):
 545	        # These live-state holdings are not subject to the internal LRU budget;
 546	        # they remain resident because the caller owns a handle to the state.
 547	        live_bases=list(self._bases)
 548	        states=list(self._states)
 549	        return {
 550	            'internal_exact_cache_bytes':self._exact_bytes,
 551	            'internal_phase_cache_bytes':self._phase_bytes,
 552	            'live_base_bytes':sum(base.memory_bytes for base in live_bases),
 553	            'live_state_dense_output_bytes':sum(s._cached_bytes for s in states),
 554	            'live_states':len(states),
 555	            'live_bases':len(live_bases),
 556	            'internal_cache_budget':self.max_cache_bytes,
 557	            'phase_cache_budget':self.max_phase_bytes,
 558	            'max_temp_bytes':self.max_temp_bytes,
 559	        }

 562	def sparse_from_known_changes(base_spectrum, n, changes, method='direct'):
 563	    """Reference rFFT plus known edits, with no stored time-domain reference.
 564	
 565	    'auto' conservatively means 'direct' in this spectrum-only interface.
 566	    Native recomputation cannot be used without reconstructing missing input;
 567	    no inverse FFT or other hidden reconstruction is performed.
 568	    """
 569	    e=RelationFFT(max_cache_bytes=0)
 570	    original=np.zeros(n)
 571	    spectrum=np.array(base_spectrum,dtype=np.complex128,copy=True)
 572	    if spectrum.shape != (n//2+1,):
 573	        raise ValueError('wrong spectrum shape')
 574	    b=Base(original,spectrum)
 575	    state=SpectralState(e,n,{(b,0):1.0})
 576	    for p,d in changes:
 577	        state.edit(p,d)
 578	    if method=='native':
 579	        raise ValueError('reference spectrum alone insufficient for native recompute')
 580	    if method=='auto':
 581	        method='direct'
 582	    return e.materialize(state,method=method)

 584	class TrackedBins:
 585	    """Maintain Q selected frequencies under known edits in O(Q) per event.
 586	
 587	    This is a live subscription to a FORK of a SpectralState. Edits must be
 588	    performed via this object's .edit()/.edit_many() methods; external edits of
 589	    tracked.state cause a RuntimeError instead of silently returning stale bins.
 590	    It does not construct a new N-element FFT after each edit.
 591	    """
 592	    def __init__(self, state: SpectralState, frequencies: Iterable[int],
 593	                 max_phase_entries=16):
 594	        self.state=state.fork()
 595	        self.n=self.state.n
 596	        q=np.asarray(list(frequencies),dtype=np.int64)
 597	        if (q<0).any() or (q>self.n//2).any():
 598	            raise IndexError('rfft frequency index out of range')
 599	        self.frequencies=q
 600	        self._values=self.state.bins(q)
 601	        self._version=self.state._mutations
 602	        self.max_phase_entries=max(0,int(max_phase_entries))
 603	        self._phases: OrderedDict[int,np.ndarray]=OrderedDict()
 604	        self.events=0
 605	
 606	    def _check(self):
 607	        if self.state._mutations!=self._version:
 608	            raise RuntimeError('TrackedBins.state mutated outside the tracked edit API')
 609	
 610	    def _phase(self,p):
 611	        phase=self._phases.get(p)
 612	        if phase is not None:
 613	            self._phases.move_to_end(p)
 614	            return phase
 615	        phase=np.exp(-2j*np.pi*((self.frequencies*p)%self.n)/self.n)
 616	        if self.max_phase_entries:
 617	            self._phases[p]=phase
 618	            while len(self._phases)>self.max_phase_entries:
 619	                self._phases.popitem(last=False)
 620	        return phase
 621	
 622	    def edit(self,position,delta):
 623	        self._check()
 624	        p=int(position); d=float(delta)
 625	        self.state.edit(p,d)
 626	        self._values += d*self._phase(p)
 627	        self._version=self.state._mutations
 628	        self.events += 1
 629	        return self
 630	
 631	    def edit_many(self,changes):
 632	        # Merge repeated positions within an event packet before updating bins.
 633	        self._check()
 634	        merged={}
 635	        for p,d in changes:
 636	            p=int(p); d=float(d)
 637	            if not 0<=p<self.n or not math.isfinite(d):
 638	                raise ValueError('invalid edit')
 639	            merged[p]=merged.get(p,0.0)+d
 640	        for p,d in merged.items():
 641	            if d: self.edit(p,d)
 642	        return self
 643	
 644	    def values(self,copy=True):
 645	        self._check()
 646	        out=self._values.copy() if copy else self._values.view()
 647	        if not copy:
 648	            out.flags.writeable=False
 649	        return out
 650	
 651	    def current_state(self):
 652	        self._check()
 653	        return self.state.fork()
 654	
 655	
 656	def subscribe(state: SpectralState, frequencies, **kwargs):
 657	    return TrackedBins(state,frequencies,**kwargs)
