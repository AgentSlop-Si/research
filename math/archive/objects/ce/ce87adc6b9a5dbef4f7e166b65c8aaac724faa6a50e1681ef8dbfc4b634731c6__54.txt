1: #!/usr/bin/env python3
2: """Frequency-shift median Fourier sketch (v5 research prototype).
3: 
4: This module implements a randomized approximation to the *full* unitary DFT for
5: length N = K*M.  One sketch uses one length-K FFT and K length-M FFTs.  The
6: random frequency shifts make the estimator unbiased and give an exact per-bin
7: variance formula.
8: 
9: The implementation is intentionally readable and independently checkable.  It
10: is a research prototype, not a replacement for an optimized FFT library.
11: """
12: from __future__ import annotations

24: ComplexArray = npt.NDArray[np.complex128]
25: IntArray = npt.NDArray[np.int64]
26: 
27: FOURTH_ROOTS: ComplexArray = np.asarray([1.0, -1.0, 1.0j, -1.0j], dtype=np.complex128)

29: 
30: def unitary_fft(x: ComplexArray, axis: int = -1) -> ComplexArray:
31:     """Return the unitary DFT along ``axis``."""
32:     n = x.shape[axis]
33:     if n <= 0:
34:         raise ValueError("FFT axis must be non-empty")
35:     return np.fft.fft(x, axis=axis) / math.sqrt(n)
36: 
37: 
38: def exact_unitary_dft(x: ComplexArray) -> ComplexArray:
39:     """Reference full unitary DFT."""
40:     x = np.asarray(x, dtype=np.complex128)
41:     if x.ndim != 1 or x.size == 0:
42:         raise ValueError("x must be a non-empty one-dimensional array")
43:     return unitary_fft(x)

45: 
46: @dataclass(frozen=True)
47: class SketchRandomness:
48:     """Random state for one frequency-shift sketch."""
49: 
50:     shifts: IntArray
51:     phases: ComplexArray
52: 
53:     def validate(self, *, k: int, m: int) -> None:
54:         if self.shifts.shape != (m,):
55:             raise ValueError(f"shifts must have shape ({m},)")
56:         if self.phases.shape != (m,):
57:             raise ValueError(f"phases must have shape ({m},)")
58:         if np.any(self.shifts < 0) or np.any(self.shifts >= k):
59:             raise ValueError("all shifts must lie in [0, K)")
60:         if not np.allclose(np.abs(self.phases), 1.0):
61:             raise ValueError("all phases must have unit magnitude")
62: 
63: 
64: def draw_randomness(k: int, m: int, rng: np.random.Generator) -> SketchRandomness:
65:     """Draw independent q_t in Z_K and fourth-root phases xi_t."""
66:     if k <= 0 or m <= 0:
67:         raise ValueError("k and m must be positive")
68:     shifts = rng.integers(0, k, size=m, dtype=np.int64)
69:     phases = FOURTH_ROOTS[rng.integers(0, 4, size=m)]
70:     return SketchRandomness(shifts=shifts, phases=phases)

73: def frequency_shift_sketch_with_state(
74:     x: ComplexArray,
75:     m: int,
76:     state: SketchRandomness,
77: ) -> ComplexArray:
78:     """Evaluate one randomized full-spectrum estimate.
79: 
80:     The input is reshaped as x[r, t] = x[r*M + t], with N = K*M.  For q_t and
81:     xi_t from ``state`` we form
82: 
83:         z_r = sum_t conj(xi_t) exp(-2*pi*i*r*q_t/K) x[r,t],
84: 
85:     compute its length-K unitary FFT Z, then for each p compute a length-M
86:     unitary FFT of
87: 
88:         xi_t Z[(p-q_t) mod K] exp(-2*pi*i*p*t/N).
89: 
90:     The output is ordered in the ordinary DFT order m_out = p + K*v.
91:     """
92:     x = np.asarray(x, dtype=np.complex128)
93:     if x.ndim != 1 or x.size == 0:
94:         raise ValueError("x must be a non-empty one-dimensional array")
95:     if m <= 0 or x.size % m != 0:
96:         raise ValueError("m must be a positive divisor of len(x)")
97: 
98:     n = int(x.size)
99:     k = n // m
100:     state.validate(k=k, m=m)
101:     q = state.shifts
102:     xi = state.phases
103:     x2 = x.reshape(k, m)
104: 
105:     # O(N) sketch construction.  The matrix is materialized only to keep the
106:     # prototype transparent; a streaming implementation can accumulate z using
107:     # O(K+M) auxiliary storage.
108:     r = np.arange(k, dtype=np.float64)[:, None]
109:     modulation = np.conj(xi)[None, :] * np.exp(-2j * np.pi * r * q[None, :] / k)
110:     z = np.sum(x2 * modulation, axis=1)
111:     z_hat = unitary_fft(z)
112: 
113:     # K independent length-M FFTs.  Vectorization changes no arithmetic count.
114:     p = np.arange(k, dtype=np.int64)[:, None]
115:     t = np.arange(m, dtype=np.float64)[None, :]
116:     gather = z_hat[(p - q[None, :]) % k]
117:     local = xi[None, :] * gather * np.exp(-2j * np.pi * p * t / n)
118:     values = unitary_fft(local, axis=1)  # values[p, v]
119: 
120:     # Standard frequency order: output index is p + K*v.
121:     return np.ascontiguousarray(values.T.reshape(n))

124: def balanced_bucket_sketch_with_state(
125:     x: ComplexArray,
126:     m: int,
127:     bucket_count: int,
128:     state: SketchRandomness,
129: ) -> ComplexArray:
130:     """Generalization with equally sized deterministic column buckets.
131: 
132:     Columns are assigned to ``bucket = t mod bucket_count``.  Only columns in
133:     the same bucket collide.  If ``c = M / bucket_count``, the exact per-bin
134:     variance becomes ``(c-1)||x||_2^2/N``.  The price is ``bucket_count``
135:     length-K FFTs.
136:     """
137:     x = np.asarray(x, dtype=np.complex128)
138:     if x.ndim != 1 or x.size == 0:
139:         raise ValueError("x must be a non-empty one-dimensional array")
140:     if m <= 0 or x.size % m != 0:
141:         raise ValueError("m must be a positive divisor of len(x)")
142:     if bucket_count <= 0 or m % bucket_count != 0:
143:         raise ValueError("bucket_count must be a positive divisor of m")
144: 
145:     n = int(x.size)
146:     k = n // m
147:     state.validate(k=k, m=m)
148:     q = state.shifts
149:     xi = state.phases
150:     x2 = x.reshape(k, m)
151:     buckets = np.arange(m, dtype=np.int64) % bucket_count
152: 
153:     r = np.arange(k, dtype=np.float64)[:, None]
154:     modulation = np.conj(xi)[None, :] * np.exp(-2j * np.pi * r * q[None, :] / k)
155:     mixed = x2 * modulation
156:     z = np.zeros((k, bucket_count), dtype=np.complex128)
157:     for b in range(bucket_count):
158:         z[:, b] = np.sum(mixed[:, buckets == b], axis=1)
159:     z_hat = unitary_fft(z, axis=0)
160: 
161:     p = np.arange(k, dtype=np.int64)[:, None]
162:     t = np.arange(m, dtype=np.float64)[None, :]
163:     gather = z_hat[(p - q[None, :]) % k, buckets[None, :]]
164:     local = xi[None, :] * gather * np.exp(-2j * np.pi * p * t / n)
165:     values = unitary_fft(local, axis=1)
166:     return np.ascontiguousarray(values.T.reshape(n))

169: def balanced_bucket_sketch(
170:     x: ComplexArray,
171:     m: int,
172:     bucket_count: int,
173:     rng: np.random.Generator,
174: ) -> ComplexArray:
175:     """Draw fresh randomness and evaluate a balanced-bucket sketch."""
176:     n = int(np.asarray(x).size)
177:     if m <= 0 or n % m != 0:
178:         raise ValueError("m must be a positive divisor of len(x)")
179:     return balanced_bucket_sketch_with_state(
180:         x, m, bucket_count, draw_randomness(n // m, m, rng)
181:     )

184: def balanced_bucket_per_bin_variance(
185:     x: ComplexArray, m: int, bucket_count: int
186: ) -> float:
187:     """Exact per-bin variance for equal buckets of size c=M/bucket_count."""
188:     x = np.asarray(x, dtype=np.complex128)
189:     if (
190:         x.ndim != 1
191:         or x.size == 0
192:         or m <= 0
193:         or x.size % m != 0
194:         or bucket_count <= 0
195:         or m % bucket_count != 0
196:     ):
197:         raise ValueError("invalid x, m, or bucket_count")
198:     collision_size = m // bucket_count
199:     return float((collision_size - 1) * np.vdot(x, x).real / x.size)

202: def frequency_shift_sketch(
203:     x: ComplexArray,
204:     m: int,
205:     rng: np.random.Generator,
206: ) -> ComplexArray:
207:     """Draw fresh randomness and evaluate one sketch."""
208:     n = int(np.asarray(x).size)
209:     if m <= 0 or n % m != 0:
210:         raise ValueError("m must be a positive divisor of len(x)")
211:     return frequency_shift_sketch_with_state(x, m, draw_randomness(n // m, m, rng))

214: def direct_expansion_estimate(
215:     x: ComplexArray,
216:     m: int,
217:     state: SketchRandomness,
218: ) -> ComplexArray:
219:     """Slow independent evaluation of the algebraic expansion.
220: 
221:     This routine first computes every column transform
222: 
223:         A_t[p] = K^{-1/2} sum_r x[r,t] exp(-2*pi*i*r*p/K)
224: 
225:     and then evaluates the double sum directly.  It is used only as a verifier
226:     for the fast separable implementation.
227:     """
228:     x = np.asarray(x, dtype=np.complex128)
229:     if x.ndim != 1 or x.size == 0 or m <= 0 or x.size % m != 0:
230:         raise ValueError("x must be one-dimensional and m must divide len(x)")
231:     n = int(x.size)
232:     k = n // m
233:     state.validate(k=k, m=m)
234:     q = state.shifts
235:     xi = state.phases
236: 
237:     x2 = x.reshape(k, m)
238:     a = unitary_fft(x2, axis=0)  # a[p, t]
239:     t_idx = np.arange(m, dtype=np.float64)
240:     u_idx = np.arange(m, dtype=np.int64)
241:     values = np.empty((k, m), dtype=np.complex128)
242: 
243:     for p in range(k):
244:         pre_fft = np.empty(m, dtype=np.complex128)
245:         for t in range(m):
246:             freq = (p + q[u_idx] - q[t]) % k
247:             mixed = np.sum(np.conj(xi) * a[freq, u_idx])
248:             pre_fft[t] = xi[t] * mixed * np.exp(-2j * np.pi * p * t / n)
249:         values[p, :] = unitary_fft(pre_fft)
250: 
251:     return np.ascontiguousarray(values.T.reshape(n))

254: def coordinatewise_complex_median(estimates: ComplexArray) -> ComplexArray:
255:     """Median real and imaginary components separately."""
256:     estimates = np.asarray(estimates, dtype=np.complex128)
257:     if estimates.ndim != 2 or estimates.shape[0] % 2 != 1:
258:         raise ValueError("estimates must have shape (odd_rows, N)")
259:     return np.median(estimates.real, axis=0) + 1j * np.median(estimates.imag, axis=0)

262: def median_frequency_shift_fft(
263:     x: ComplexArray,
264:     m: int,
265:     rows: int,
266:     rng: np.random.Generator,
267: ) -> ComplexArray:
268:     """Return the coordinatewise median of an odd number of independent sketches."""
269:     if rows <= 0 or rows % 2 != 1:
270:         raise ValueError("rows must be a positive odd integer")
271:     estimates = np.stack([frequency_shift_sketch(x, m, rng) for _ in range(rows)])
272:     return coordinatewise_complex_median(estimates)

275: def per_bin_variance(x: ComplexArray, m: int) -> float:
276:     """Exact E|estimate[j]-DFT(x)[j]|^2 for every output bin j."""
277:     x = np.asarray(x, dtype=np.complex128)
278:     if x.ndim != 1 or x.size == 0 or m <= 0 or x.size % m != 0:
279:         raise ValueError("x must be one-dimensional and m must divide len(x)")
280:     return float((m - 1) * np.vdot(x, x).real / x.size)

283: def median_uniform_bound(
284:     x: ComplexArray,
285:     m: int,
286:     rows: int,
287:     delta: float,
288: ) -> float:
289:     """The proved all-output absolute error bound for coordinatewise medians.
290: 
291:     If rows = 2*t-1 and C_t = binom(2*t-1, t), then with probability at least
292:     1-delta,
293: 
294:       max_j |estimate[j]-DFT(x)[j]|
295:         <= sqrt(2) * (2*N*C_t/delta)^(1/(2*t))
296:            * sqrt((M-1)/N) * ||x||_2.
297:     """
298:     x = np.asarray(x, dtype=np.complex128)
299:     if rows <= 0 or rows % 2 != 1:
300:         raise ValueError("rows must be a positive odd integer")
301:     if not (0.0 < delta < 1.0):
302:         raise ValueError("delta must lie in (0,1)")
303:     if m <= 0 or x.size % m != 0:
304:         raise ValueError("m must divide len(x)")
305:     t = (rows + 1) // 2
306:     c_t = math.comb(2 * t - 1, t)
307:     sigma = math.sqrt(per_bin_variance(x, m))
308:     return math.sqrt(2.0) * (2.0 * x.size * c_t / delta) ** (1.0 / (2.0 * t)) * sigma

311: def arithmetic_cost_proxy(n: int, m: int, rows: int = 1) -> float:
312:     """Asymptotic operation proxy R*N*(log2 M + log2(N/M)/M + 1)."""
313:     if n <= 1 or m <= 0 or n % m != 0 or rows <= 0:
314:         raise ValueError("require n>1, positive divisor m, and positive rows")
315:     k = n // m
316:     return rows * n * (1.0 + math.log2(m) + math.log2(k) / m)

319: def balanced_bucket_cost_proxy(
320:     n: int, m: int, bucket_count: int, rows: int = 1
321: ) -> float:
322:     """Cost proxy for the equal-bucket extension.
323: 
324:     With collision size c=M/bucket_count, the proxy is
325:     R*N*(log2 M + log2(N/M)/c + 1).
326:     """
327:     if (
328:         n <= 1
329:         or m <= 0
330:         or n % m != 0
331:         or bucket_count <= 0
332:         or m % bucket_count != 0
333:         or rows <= 0
334:     ):
335:         raise ValueError("invalid dimensions")
336:     k = n // m
337:     c = m // bucket_count
338:     return rows * n * (1.0 + math.log2(m) + math.log2(k) / c)

341: def _case_signals(k: int, m: int, rng: np.random.Generator) -> dict[str, ComplexArray]:
342:     n = k * m
343:     r = np.arange(k)
344:     tone = max(1, k // 5)
345:     cases: dict[str, ComplexArray] = {}
346:     cases["flat"] = np.ones(n, dtype=np.complex128)
347:     cases["random_phase"] = np.exp(2j * np.pi * rng.random(n))
348:     cases["gaussian"] = rng.normal(size=n) + 1j * rng.normal(size=n)
349: 
350:     local = np.zeros((k, m), dtype=np.complex128)
351:     local[:, m // 3] = np.exp(2j * np.pi * tone * r / k)
352:     cases["local_spike_tone"] = local.reshape(n)
353: 
354:     pattern = np.exp(2j * np.pi * rng.random(m))
355:     repeated = np.exp(2j * np.pi * tone * r[:, None] / k) * pattern[None, :]
356:     cases["repeated_pattern_tone"] = repeated.reshape(n)
357: 
358:     spike = np.zeros(n, dtype=np.complex128)
359:     spike[n // 3] = 1.0
360:     cases["single_spike"] = spike
361:     return cases

364: def run_reproducible_experiments(seed: int = 20261009) -> dict[str, Any]:
365:     """Run compact algebra, variance, median, and cost checks."""
366:     rng = np.random.default_rng(seed)
367: 
368:     # 1. Fast evaluator versus a structurally independent double-sum evaluator.
369:     expansion_rows: list[dict[str, Any]] = []
370:     for k, m in [(5, 3), (8, 4), (9, 5), (16, 8)]:
371:         n = k * m
372:         x = rng.normal(size=n) + 1j * rng.normal(size=n)
373:         state = draw_randomness(k, m, rng)
374:         fast = frequency_shift_sketch_with_state(x, m, state)
375:         slow = direct_expansion_estimate(x, m, state)
376:         expansion_rows.append(
377:             {
378:                 "K": k,
379:                 "M": m,
380:                 "N": n,
381:                 "max_abs_difference": float(np.max(np.abs(fast - slow))),
382:             }
383:         )
384: 
385:     # 2. Exact variance prediction checked empirically.  We average over every
386:     # output bin as well as trials, so modest trial counts are informative.
387:     variance_rows: list[dict[str, Any]] = []
388:     for k, m in [(64, 8), (64, 16), (128, 16), (128, 32)]:
389:         for name, x in _case_signals(k, m, rng).items():
390:             exact = exact_unitary_dft(x)
391:             predicted = per_bin_variance(x, m)
392:             sq_error_sum = 0.0
393:             trials = 40
394:             for _ in range(trials):
395:                 estimate = frequency_shift_sketch(x, m, rng)
396:                 sq_error_sum += float(np.sum(np.abs(estimate - exact) ** 2))
397:             empirical = sq_error_sum / (trials * x.size)
398:             variance_rows.append(
399:                 {
400:                     "K": k,
401:                     "M": m,
402:                     "N": k * m,
403:                     "case": name,
404:                     "trials": trials,
405:                     "predicted_per_bin_mse": predicted,
406:                     "empirical_per_bin_mse": empirical,
407:                     "empirical_over_predicted": empirical / predicted if predicted else 1.0,
408:                 }
409:             )
410: 
411:     # 3. Median behavior, normalized by the coefficient energy scale ||x||/sqrt(N).
412:     median_rows: list[dict[str, Any]] = []
413:     for k, m in [(128, 16), (128, 32), (256, 32)]:
414:         for name, x in _case_signals(k, m, rng).items():
415:             exact = exact_unitary_dft(x)
416:             scale = float(np.linalg.norm(x) / math.sqrt(x.size))
417:             for rows in [1, 3, 5, 9]:
418:                 max_errors: list[float] = []
419:                 rms_errors: list[float] = []
420:                 for _ in range(8):
421:                     estimate = median_frequency_shift_fft(x, m, rows, rng)
422:                     error = np.abs(estimate - exact)
423:                     max_errors.append(float(np.max(error) / scale))
424:                     rms_errors.append(float(np.sqrt(np.mean(error**2)) / scale))
425:                 median_rows.append(
426:                     {
427:                         "K": k,
428:                         "M": m,
429:                         "N": k * m,
430:                         "case": name,
431:                         "rows": rows,
432:                         "normalized_max_median": float(np.median(max_errors)),
433:                         "normalized_max_p90": float(np.quantile(max_errors, 0.9)),
434:                         "normalized_rms_median": float(np.median(rms_errors)),
435:                         "single_sketch_predicted_normalized_rms": math.sqrt(m - 1),
436:                     }
437:                 )
438: 
439:     # 4. Balanced-bucket extension: exact variance c-1, with c=M/d.
440:     bucket_rows: list[dict[str, Any]] = []
441:     k, m = 64, 16
442:     for name, x in _case_signals(k, m, rng).items():
443:         exact = exact_unitary_dft(x)
444:         for bucket_count in [1, 2, 4, 8, 16]:
445:             predicted = balanced_bucket_per_bin_variance(x, m, bucket_count)
446:             trials = 40
447:             observed = 0.0
448:             for _ in range(trials):
449:                 estimate = balanced_bucket_sketch(x, m, bucket_count, rng)
450:                 observed += float(np.mean(np.abs(estimate - exact) ** 2))
451:             observed /= trials
452:             bucket_rows.append(
453:                 {
454:                     "K": k,
455:                     "M": m,
456:                     "N": k * m,
457:                     "case": name,
458:                     "bucket_count": bucket_count,
459:                     "collision_size": m // bucket_count,
460:                     "predicted_per_bin_mse": predicted,
461:                     "empirical_per_bin_mse": observed,
462:                     "empirical_over_predicted": (
463:                         observed / predicted if predicted else 1.0
464:                     ),
465:                 }
466:             )
467: 
468:     # 5. Symbolic scaling proxies for two useful parameter regimes.
469:     scaling_rows: list[dict[str, Any]] = []
470:     for log2_n in [20, 30, 40, 60, 80, 120, 160, 240]:
471:         # Use real-valued proxies rather than requiring astronomically large integer N.
472:         l = float(log2_n)
473:         m_fast = max(2.0, l)
474:         m_trade = max(2.0, math.log2(l))
475:         for label, m_value in [("M~logN", m_fast), ("M~loglogN", m_trade)]:
476:             per_n = 1.0 + math.log2(m_value) + (l - math.log2(m_value)) / m_value
477:             scaling_rows.append(
478:                 {
479:                     "log2_N": log2_n,
480:                     "regime": label,
481:                     "M_proxy": m_value,
482:                     "cost_over_N": per_n,
483:                     "cost_over_Nlog2N": per_n / l,
484:                     "single_sketch_std_over_energy_scale": math.sqrt(max(0.0, m_value - 1.0)),
485:                 }
486:             )
487: 
488:     max_expansion_error = max(row["max_abs_difference"] for row in expansion_rows)
489:     variance_ratios = [row["empirical_over_predicted"] for row in variance_rows]
490: 
491:     return {
492:         "metadata": {
493:             "seed": seed,
494:             "normalization": "unitary DFT",
495:             "algorithm": "frequency-shift median Fourier sketch",
496:         },
497:         "summary": {
498:             "max_fast_vs_expansion_error": max_expansion_error,
499:             "variance_ratio_min": min(variance_ratios),
500:             "variance_ratio_max": max(variance_ratios),
501:             "variance_ratio_median": float(np.median(variance_ratios)),
502:             "number_of_expansion_checks": len(expansion_rows),
503:             "number_of_variance_checks": len(variance_rows),
504:             "number_of_median_checks": len(median_rows),
505:             "number_of_balanced_bucket_checks": len(bucket_rows),
506:         },
507:         "fast_vs_direct_expansion": expansion_rows,
508:         "variance_checks": variance_rows,
509:         "median_checks": median_rows,
510:         "balanced_bucket_checks": bucket_rows,
511:         "scaling_proxies": scaling_rows,
512:     }
