   9 def rank_formula(k,a,b):return (1<<k)//(4 if a.bit_length()+b.bit_length()<=k else 2)
  10 def boundary(k,a):return sum(1<<(k-j-1) for j in range(k) if (a&((1<<(j+1))-1)).bit_count()%2)
  11 def pull(mask,columns):return sum(((mask&v).bit_count()%2)<<j for j,v in enumerate(columns))
  12 def binary_rank(rows):
  13     piv={}
  14     for r in rows:
  15         while r:
  16             j=r.bit_length()-1
  17             if j in piv:r^=piv[j]
  18             else:piv[j]=r;break
  19     return len(piv)
  20 def random_columns(k,rng):
  21     rows=[1<<j for j in range(k)]
  22     for _ in range(2*k*k):
  23         i,j=rng.sample(range(k),2)
  24         if rng.randrange(4)==0:rows[i],rows[j]=rows[j],rows[i]
  25         else:rows[i]^=rows[j]
  26     return [sum(((rows[i]>>j)&1)<<i for i in range(k)) for j in range(k)]
  27 def transform_table(cols):
  28     k=len(cols);basis=[pull(1<<j,cols) for j in range(k)];out=[0]*(1<<k)
  29     for m in range(1,1<<k):low=m&-m;out[m]=out[m^low]^basis[low.bit_length()-1]
  30     return out
  31 
  32 def trace(k,cols,a,weights=None):
  33     masks=[a]
  34     for col in cols:masks.append(pull(masks[-1],col))
  35     ds=[boundary(k,m) for m in masks];weights=weights or [1]*len(masks);cap=sum(r*d for r,d in zip(weights,ds));rank=rank_formula(k,a,masks[-1])
  36     return {'N':1<<k,'input_mask':a,'output_mask':masks[-1],'masks':masks,'stage_capacities':ds,'capacity':cap,'complex_fourier_rank':rank,'gap':rank-cap}
  37 
  38 def best(k,cols):
  39     tables=[transform_table(c) for c in cols];bs=[0]+[boundary(k,m) for m in range(1,1<<k)];winner=None;key=None
  40     for a in range(1,1<<k):
  41         m=a;cap=bs[m]
  42         for table in tables:m=table[m];cap+=bs[m]
  43         gap=rank_formula(k,a,m)-cap;candidate=(gap,-cap)
  44         if key is None or candidate>key:winner=a;key=candidate
  45     return trace(k,cols,winner)

  47 def primitive(n,p):
  48     for g in range(2,p):
  49         w=pow(g,(p-1)//n,p)
  50         if pow(w,n,p)==1 and pow(w,n//2,p)!=1:return w
  51     raise AssertionError

  57         sets={(a,s):[j for j in range(n) if (a&j).bit_count()%2==s] for a in range(1,n) for s in range(2)}
