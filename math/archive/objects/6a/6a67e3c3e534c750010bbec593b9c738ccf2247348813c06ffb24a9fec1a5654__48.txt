0010 def counted_route(M):
0011     # Explicit O(N) stable integer counting sort; no comparison sorting.
0012     N=M*M;areas=[(i//M+1)*(i%M+1) for i in range(N)]
0013     count=[0]*(N+1)
0014     for a in areas:count[a]+=1
0015     start=[0]*(N+1);total=0
0016     for a in range(1,N+1):start[a]=total;total+=count[a]
0017     pos=start[:];order=[0]*N
0018     for i,a in enumerate(areas):order[pos[a]]=i;pos[a]+=1
0019     sigma=[0]*N
0020     for r,i in enumerate(order):sigma[i]=order[N-1-r]
0021     assert all(sigma[sigma[i]]==i for i in range(N))
0022     return areas,sigma
