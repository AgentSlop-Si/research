72:     def test_exhaustive_small_local_variance(self) -> None:
73:         from itertools import product
74:         from prototype import SketchRandomness, compress_one_row
75: 
76:         k, m = 3, 2
77:         x = np.asarray([1+2j, -1+0.5j, 0.25-0.75j, 2-1j, -0.5+0.25j, 1.5+0j])
78:         ref = exact_local_spectra(x, m)
79:         accum = np.zeros((k, m), dtype=float)
80:         count = 0
81:         roots = [1.0, -1.0, 1.0j, -1.0j]
82:         for q in product(range(k), repeat=m):
83:             for phases in product(roots, repeat=m):
84:                 state = SketchRandomness(
85:                     shifts=np.asarray(q, dtype=np.int64),
86:                     phases=np.asarray(phases, dtype=np.complex128),
87:                 )
88:                 est = local_estimate_from_row(compress_one_row(x, m, state), state)
89:                 accum += np.abs(est - ref) ** 2
90:                 count += 1
91:         exact_average = accum / count
92:         predicted = np.broadcast_to(local_variance_vector(x, m), (k, m))
93:         self.assertLess(float(np.max(np.abs(exact_average - predicted))), 1e-12)

133:     def test_asymptotic_ratio_decreases(self) -> None:
134:         ratios = []
135:         memory = []
136:         for l in [64, 128, 256, 512, 1024, 2048, 4096]:
137:             ll = math.log2(l)
138:             rows = int(math.ceil(l / ll)) | 1
139:             m = 1 << int(math.floor(math.log2(l)))
140:             per_n = rows * (2.0 + (l - math.log2(m)) / m) + math.log2(m)
141:             ratios.append(per_n / l)
142:             # N itself is symbolic; R*K/N = R/M.
143:             memory.append(1.0 + rows / m)
144:         self.assertLess(ratios[-1], ratios[0])
145:         self.assertLess(memory[-1], 2.0)
