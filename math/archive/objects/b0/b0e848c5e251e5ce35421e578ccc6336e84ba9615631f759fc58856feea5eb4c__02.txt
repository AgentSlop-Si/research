  10 def prime(p): return p>=2 and all(p%d for d in range(2,math.isqrt(p)+1))
  11 def field(n):
  12     k=4
  13     while not prime(k*n+1): k+=1
  14     p=k*n+1
  15     factors=[q for q in range(2,n+1) if n%q==0 and prime(q)]
  16     for g in range(2,p):
  17         w=pow(g,k,p)
  18         if pow(w,n,p)==1 and all(pow(w,n//q,p)!=1 for q in factors): return p,w
  19     raise AssertionError('root')
  20 def dft(r,w,p,count=None):
  21     n=len(r)
  22     return [sum(v*pow(w,j*k%n,p) for j,v in enumerate(r))%p for k in range(n if count is None else count)]
  23 def idft(R,w,p): return [v*pow(len(R),-1,p)%p for v in dft(R,pow(w,-1,p),p)]
  24 def conv(C,r,p): return [sum(c*r[(j-e)%len(r)] for e,c in C.items())%p for j in range(len(r))]
  25 def spectrum(C,n,w,p): return [sum(c*pow(w,e*k%n,p) for e,c in C.items())%p for k in range(n)]
  26 def residual_from_inverse(C,n,w,p):
  27     values=spectrum(C,n,w,p)
  28     if not all(values): return None
  29     r=idft([pow(v,-1,p) for v in values],w,p)
  30     assert conv(C,r,p)==[1]+[0]*(n-1)
  31     return r
  32 
  33 def solve(A,b,p):
  34     m=len(b); M=[list(row)+[value] for row,value in zip(A,b)]
  35     for col in range(m):
  36         pivot=next((i for i in range(col,m) if M[i][col]%p),None)
  37         if pivot is None: return None
  38         M[col],M[pivot]=M[pivot],M[col]
  39         inv=pow(M[col][col],-1,p)
  40         M[col]=[v*inv%p for v in M[col]]
  41         for i in range(m):
  42             if i!=col:
  43                 factor=M[i][col]
  44                 M[i]=[(a-factor*b)%p for a,b in zip(M[i],M[col])]
  45     return [row[-1] for row in M]
  46 
  47 def inverse_proposal(r,s,w,p):
  48     R=dft(r,w,p,2*s)
  49     if not all(R): return None,'zero-probe'
  50     b=[pow(v,-1,p) for v in R]
  51     recurrence=None
  52     for degree in range(1,s+1):
  53         H=[[b[i+j] for j in range(degree)] for i in range(degree)]
  54         a=solve(H,[-b[i+degree]%p for i in range(degree)],p)
  55         if a is not None and all((b[i+degree]+sum(a[j]*b[i+j] for j in range(degree)))%p==0 for i in range(len(b)-degree)):
  56             recurrence=a+[1];break
  57     if recurrence is None:return None,'no-recurrence'
  58     nodes=[(e,pow(w,e,p)) for e in range(len(r)) if sum(c*pow(w,e*j,p) for j,c in enumerate(recurrence))%p==0]
  59     if len(nodes)!=degree:return None,'not-subgroup-roots'
  60     a=solve([[pow(z,k,p) for _,z in nodes] for k in range(degree)],b[:degree],p)
  61     if a is None or not all(a):return None,'degenerate'
  62     return {e:c for (e,z),c in zip(nodes,a)},'proposal'
  63 def recover(r,s,w,p):
  64     C,reason=inverse_proposal(r,s,w,p)
  65     if C is None:return None,reason
  66     if conv(C,r,p)!=[1]+[0]*(len(r)-1):return None,'certificate-failed'
  67     return C,'accepted'
  68 
  69 def graph_detect(r,p,K):
  70     n=len(r)
  71     if n<2*K+1 or not all(r):return None
  72     ratios=collections.Counter(r[j]*pow(r[j-1],-1,p)%p for j in range(n))
  73     if len(ratios)>2*K*K:return None
  74     possible={0}
  75     for count in ratios.values():possible|={x+count for x in tuple(possible)}
  76     candidates=[1] if len(ratios)==1 else [pow(u,-1,n) for u in sorted(possible) if 0<u<n and math.gcd(u,n)==1]
  77     for delta in candidates:
  78         sample=collections.Counter(r[j]*pow(r[(j-delta)%n],-1,p)%p for j in range(2*K+1))
  79         alpha=max(sample,key=sample.get)
  80         h=[(r[j]-alpha*r[(j-delta)%n])%p for j in range(n)]
  81         if sum(bool(v) for v in h)<=K:return delta,alpha,h
  82     return None
  83 
  84 def binomial_signal(n,d,a,j,c,p):
  85     assert math.gcd(n,d)==1 and pow(a,n,p)!=1
  86     r=[0]*n
  87     scale=c*pow((1-pow(a,n,p))%p,-1,p)%p
  88     for t in range(n):r[(j+t*d)%n]=scale*pow(a,t,p)%p
  89     return r
  90 
  91 def graph_decode(r,candidate,w,p):
  92     d,a,h=candidate;n=len(r);H=dft(h,w,p);R=[]
  93     for k in range(n):
  94         den=(1-a*pow(w,k*d%n,p))%p
  95         R.append(H[k]*pow(den,-1,p)%p if den else sum(r[j]*pow(w,k*j%n,p) for j in range(n))%p)
  96     return R
