   9 def affine(cols):

  70 def primitive(p):
  71  assert prime_by_trial_division(p) and (p-1)%N==0
  72  for g in range(2,p):

  88 def upper_sum(M,p):
  89  out=[];acc=[0]*len(M[0])
  90  for row in reversed(M):acc=[(a+b)%p for a,b in zip(acc,row)];out.append(acc[:])
  91  return out[::-1]
  92 def cumulative_columns(M,p):
  93  out=[]
  94  for row in M:
  95   acc=0;r=[]
  96   for v in reversed(row):acc=(acc+v)%p;r.append(acc)
  97   out.append(r[::-1])
  98  return out
