import unittest
from audit import positive,one_equation_trap,dual_ambiguity
from harper_monomial import allowed_symmetries,no_single_harper_sandwich
from stable_green import example
from spectral_green import complex_nonnormal_control,complex_nonnormal_nondominant_control
from green_product_audit import reject_fixed_green_product
from near_resonance import audit as resonant_control

class Phase19(unittest.TestCase):
    def test_joint_positive_8(self):self.assertTrue(positive(8)['same_mask_solves_both_dualities'])
    def test_joint_positive_16(self):self.assertTrue(positive(16)['same_mask_solves_both_dualities'])
    def test_joint_positive_32(self):self.assertTrue(positive(32)['same_mask_solves_both_dualities'])
    def test_one_intertwiner_insufficient_8(self):self.assertTrue(one_equation_trap(8)['joint_gate_excludes_every_monomial'])
    def test_one_intertwiner_insufficient_16(self):self.assertTrue(one_equation_trap(16)['joint_gate_excludes_every_monomial'])
    def test_one_intertwiner_insufficient_32(self):self.assertTrue(one_equation_trap(32)['joint_gate_excludes_every_monomial'])
    def test_dual_alone_has_parity_ambiguity_8(self):self.assertTrue(dual_ambiguity(8)['column_one_calibration_rejects'])
    def test_dual_alone_has_parity_ambiguity_16(self):self.assertTrue(dual_ambiguity(16)['column_one_calibration_rejects'])
    def test_dual_alone_has_parity_ambiguity_32(self):self.assertTrue(dual_ambiguity(32)['column_one_calibration_rejects'])
    def test_fixed_green_product_joint_N8(self):self.assertTrue(reject_fixed_green_product(8,0)['all_middle_monomials_excluded_for_fixed_stages'])
    def test_fixed_green_product_joint_N16(self):self.assertTrue(reject_fixed_green_product(16,1)['all_middle_monomials_excluded_for_fixed_stages'])
    def test_fixed_green_product_joint_N32(self):self.assertTrue(reject_fixed_green_product(32,2)['all_middle_monomials_excluded_for_fixed_stages'])
    def test_near_resonance_n16(self):self.assertTrue(resonant_control(16)['DFT_phase_coherence_failed'])
    def test_near_resonance_n64(self):self.assertTrue(resonant_control(64)['DFT_phase_coherence_failed'])
    def test_near_resonance_n128(self):self.assertTrue(resonant_control(128)['DFT_phase_coherence_failed'])
    def test_harper_centralizer_4(self):self.assertTrue(allowed_symmetries(4)['only_identity_reflection'])
    def test_harper_centralizer_8(self):self.assertTrue(allowed_symmetries(8)['only_identity_reflection'])
    def test_harper_sandwich_8(self):self.assertTrue(no_single_harper_sandwich(8)['middle_monomial_impossible_for_pair'])
    def test_harper_sandwich_16(self):self.assertTrue(no_single_harper_sandwich(16)['middle_monomial_impossible_for_pair'])
    def test_harper_sandwich_32(self):self.assertTrue(no_single_harper_sandwich(32)['middle_monomial_impossible_for_pair'])
    def test_spectral_green_non_DD_8(self):self.assertTrue(complex_nonnormal_control(8)['column_diagonal_dominance_fails'])
    def test_spectral_green_non_DD_64(self):self.assertTrue(complex_nonnormal_control(64)['column_diagonal_dominance_fails'])
    def test_spectral_green_non_DD_256(self):self.assertTrue(complex_nonnormal_control(256)['column_diagonal_dominance_fails'])
    def test_spectral_nonnormal_non_DD_8(self):self.assertTrue(complex_nonnormal_nondominant_control(8)['nonnormal'])
    def test_spectral_nonnormal_non_DD_64(self):self.assertTrue(complex_nonnormal_nondominant_control(64)['nonnormal'])
    def test_spectral_nonnormal_non_DD_256(self):self.assertTrue(complex_nonnormal_nondominant_control(256)['nonnormal'])
    def test_stable_green_16(self):self.assertGreater(example(16)['actual_spectral_error'],example(16)['rigorous_spectral_error_floor'])
    def test_stable_green_32(self):self.assertGreater(example(32)['actual_spectral_error'],example(32)['rigorous_spectral_error_floor'])
    def test_stable_green_64(self):self.assertGreater(example(64)['actual_spectral_error'],example(64)['rigorous_spectral_error_floor'])

if __name__=='__main__':unittest.main()
