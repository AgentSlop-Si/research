   1 #!/usr/bin/env python3
   2 """Standalone exact nonzero-mask butterfly controls."""
   3 import json
   4 from pathlib import Path
   5 
   6 P = 65537
   7 ALPHA = pow(3, (P - 1) // 4, P)
   8 assert ALPHA not in (0, 1, P - 1)
   9 
  10 
  11 def eye(n):
  12     return [[int(i == j) for j in range(n)] for i in range(n)]
  13 
  14 
  15 def mm(a, b):
  16     n = len(a)
  17     return [[sum(a[i][k] * b[k][j] for k in range(n)) % P for j in range(n)] for i in range(n)]
  18 
  19 
  20 def masked_cycle(n, order, destinations, weight):
  21     assert sorted(order) == list(range(n))
  22     w = [[0] * n for _ in range(n)]
  23     masks = [0] * n
  24     for i, src in enumerate(order):
  25         dst = order[(i + 1) % n]
  26         if dst in destinations:
  27             masks[dst] = weight
  28             w[dst][src] = weight
  29     product = 1
  30     for value in masks:
  31         product = product * value % P
  32     assert product == 0
  33     u = [[(int(i == j) - w[i][j]) % P for j in range(n)] for i in range(n)]
  34     ui = [[(int(i == j) + w[i][j]) % P for j in range(n)] for i in range(n)]
  35     assert mm(w, w) == [[0] * n for _ in range(n)]
  36     assert mm(u, ui) == mm(ui, u) == eye(n)
  37     return ui, sum(bool(v) for v in masks)
  38 
  39 
  40 def check(n, kind):
  41     pairs = [(i, i + 1) for i in range(0, n, 2)] if kind == 'adjacent' else [(i, i + n // 2) for i in range(n // 2)]
  42     top, bottom = {a for a, _ in pairs}, {b for _, b in pairs}
  43     upper_order = [v for a, b in pairs for v in (b, a)]
  44     lower_order = [v for a, b in pairs for v in (a, b)]
  45     upper, unz = masked_cycle(n, upper_order, top, pow(ALPHA, -1, P))
  46     beta = (-1 - ALPHA) % P
  47     lower, lnz = masked_cycle(n, lower_order, bottom, pow(beta, -1, P))
  48     endpoint = [(1 - ALPHA) % P if i in top else beta for i in range(n)]
  49     assert all(endpoint) and ALPHA
  50     h = [[(ALPHA * upper[i][j] + endpoint[i] * lower[i][j]) % P for j in range(n)] for i in range(n)]
  51     expected = [[0] * n for _ in range(n)]
  52     for a, b in pairs:
  53         expected[a][a] = expected[a][b] = expected[b][a] = 1
  54         expected[b][b] = P - 1
  55     assert h == expected
  56     return {'N': n, 'pairing': kind, 'nonzero_edges_per_green': [unz, lnz],
  57             'hamiltonian_cycles': 2, 'denominator_guards': [1, 1],
  58             'exact_butterfly_identity': True}
  59 
  60 
  61 def main():
  62     result = {'field': P, 'alpha': ALPHA,
  63               'cases': [check(n, kind) for n in (4, 16, 64) for kind in ('adjacent', 'halves')],
  64               'all_exact_assertions_passed': True,
  65               'scope': 'Known butterfly layers realized with nonzero matching masks; no growing-size Fourier compression.'}
  66     Path(__file__).with_name('NONZERO_BUTTERFLY_CONTROLS.json').write_text(json.dumps(result, indent=2) + '\n')
  67     print(json.dumps(result, indent=2))
  68 
  69 
  70 if __name__ == '__main__':
  71     main()
