0005 class OrbitTests(unittest.TestCase):
0006     def test_orbits_partition(self):
0007         for n in (2,4,8,16,32,64):
0008             os=orbits(n)
0009             self.assertEqual(set(sum([list(s) for s in os.values()],[])), set(range(n)))
0010             self.assertEqual(sum(map(len,os.values())),n)
0011 
0012     def test_minimum_representatives(self):
0013         for n in (2,4,8,16,32,64):
0014             self.assertEqual(len(representatives(n)),n.bit_length())
0015             self.assertEqual({frequency_conductor(n,k) for k in representatives(n)},set(orbits(n)))
0016 
0017     def test_formal_fourier(self):
0018         self.assertEqual(dft_bin_exact([1,0,0,0,0,0,0,0],3),(1,0,0,0))
0019         self.assertEqual(dft_bin_exact([0,0,0,0,1,0,0,0],3),(-1,0,0,0))
0020         self.assertEqual(dft_bin_exact([1]*8,0),(8,0,0,0))
0021         for k in range(1,8):self.assertEqual(dft_bin_exact([1]*8,k),(0,0,0,0))
0022 
0023     def test_single_missing_orbit(self):
0024         for n in (2,4,8,16,32):
0025             for m,frequencies in orbits(n).items():
0026                 a,b=binary_coset_collision_witness(n,m)
0027                 self.assertEqual(orbit_support_of_difference(a,b),frequencies)
0028 
0029     def test_exhaustive_binary_unique(self):
0030         n=8; q=representatives(n)
0031         sigs={dft_selected_exact([(t>>j)&1 for j in range(n)],q) for t in range(1<<n)}
0032         self.assertEqual(len(sigs),1<<n)
0033 
0034     def test_linear_independence_subset_sums(self):
0035         for n in (4,8,16):

0037             self.assertEqual(len(vals),1<<(n//2))
