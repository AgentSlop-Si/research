0008 def dense_sig(M,K,a,b,d):
0009  r=np.arange(K)[:,None]; t=np.arange(M)[None,:]
0010  return np.exp(2j*np.pi*((a*t*t+(b+d*r)*t)%M)/M)
0011 
0012 def table(M,K,a,b,p):
0013  N=M*K; t=np.arange(M)
0014  H=np.zeros(M,complex); step=(2*a)%M; g=math.gcd(step,M)
0015  twist=np.exp(-2j*np.pi*p/K)-1
0016  for seed in range(g):
0017   H[seed]=np.sum(np.exp(-2j*np.pi*((a*t*t+b*t)%M)/M-2j*np.pi*(p+K*seed)*t/N))
0018   q=seed
0019   for _ in range(M//g-1):
0020    qn=(q+step)%M
0021    H[qn]=np.exp(2j*np.pi*p/N+2j*np.pi*((a+b+q)%M)/M)*(H[q]+twist)
0022    q=qn
0023  return H
0024 
0025 def short_cyclic_conv(alpha,v):
0026  K=len(alpha); M=len(v); P=1<<(2*K-2).bit_length()
0027  fa=np.fft.fft(alpha,P); out=np.zeros(M,complex)
0028  for start in range(0,M,K):
0029   block=v[start:start+K]
0030   conv=np.fft.ifft(np.fft.fft(block,P)*fa)[:len(block)+K-1]
0031   np.add.at(out,(start+np.arange(len(conv)))%M,conv)
0032  return out
0033 
0034 def fast(c,M,K,a,b,d):
0035  out=np.empty(M*K,complex); gd=math.gcd(d,M); L=M//gd; js=np.arange(L)
0036  for p in range(K):
0037   H=table(M,K,a,b,p)
0038   alpha=c*np.exp(-2j*np.pi*p*np.arange(K)/K)
0039   if K>L:
0040    beta=np.zeros(L,complex)
0041    np.add.at(beta,np.arange(K)%L,alpha)
0042   for orbit in range(gd):
0043    indices=(orbit+d*js)%M
0044    h=H[indices]; v=h[(-js)%L]
0045    if K<=L: w=short_cyclic_conv(alpha,v)
0046    else: w=np.fft.ifft(np.fft.fft(beta)*np.fft.fft(v))
0047    out[p+K*indices]=w[(-js)%L]/np.sqrt(M*K)
0048  return out

0050 results=[]
0051 for M,K in [(8,2),(8,4),(16,4),(16,8),(32,4),(31,4),(31,8)]:
0052  for a,b,d in ([(a,b,d) for a,b in [(1,0),(3,5)] for d in range(M)] if M<=16 else [(1,0,0),(1,0,2),(3,5,4),(3,5,8),(M-1,M-2,M-3),(1,0,1),(3,5,7)]):
0053   if a%M==0: continue
0054   N=M*K; sig=dense_sig(M,K,a,b,d); t=np.arange(M)
0055   table_err=0
0056   for p in range(K):
0057    reference=np.array([np.sum(np.exp(-2j*np.pi*((a*t*t+b*t)%M)/M-2j*np.pi*(p+K*q)*t/N)) for q in range(M)])
0058    table_err=max(table_err,float(np.max(abs(table(M,K,a,b,p)-reference))))
0059   basis_err=0
0060   for pos in range(N):
0061    x=np.zeros(N,complex); x[pos]=1
0062    c=np.sum(sig*x.reshape(K,M),axis=1)
0063    ref=np.fft.fft((sig.conj()*c[:,None]).ravel(),norm='ortho')
0064    basis_err=max(basis_err,float(np.max(abs(fast(c,M,K,a,b,d)-ref))))
0065   random_err=0
0066   for _ in range(4):
0067    x=rng.normal(size=(K,M))+1j*rng.normal(size=(K,M))
0068    c=np.sum(sig*x,axis=1)
0069    ref=np.fft.fft((sig.conj()*c[:,None]).ravel(),norm='ortho')
0070    random_err=max(random_err,float(np.max(abs(fast(c,M,K,a,b,d)-ref))))
0071   results.append({'M':M,'K':K,'a':a,'b':b,'d':d,'basis_vectors_tested':N,'random_vectors_tested':4,'table_max_absolute_error':table_err,'all_basis_decoder_max_absolute_error':basis_err,'four_random_decoder_max_absolute_error':random_err})
0072 assert max(r['all_basis_decoder_max_absolute_error'] for r in results)<1e-10
0073 assert max(r['four_random_decoder_max_absolute_error'] for r in results)<1e-9
