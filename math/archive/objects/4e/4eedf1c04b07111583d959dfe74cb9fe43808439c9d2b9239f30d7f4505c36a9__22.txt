0021 def factors(n):
0022     out = []
0023     d = 2
0024     while d*d <= n:
0025         if n % d == 0:
0026             out.append(d)
0027             while n % d == 0:
0028                 n //= d
0029         d += 1
0030     if n > 1:
0031         out.append(n)
0032     return out

0035 def prime(n):
0036     return n >= 2 and all(n % d for d in range(2, math.isqrt(n)+1))

0039 def field(q):
0040     p = q+1
0041     while not prime(p):
0042         p += q
0043     for w in range(2, p):
0044         if pow(w, q, p) == 1 and all(pow(w, q//d, p) != 1 for d in factors(q)):
0045             return p, w
0046     raise AssertionError('No primitive root found')

0049 def transform(v, w, p):
0050     n = len(v)
0051     table = [pow(w, i, p) for i in range(n)]
0052     return [sum(v[t]*table[(t*r) % n] for t in range(n)) % p for r in range(n)]

0055 def chirp(q, m, a, beta, p, w):
0056     return [[pow(w, (a*t*(t-1)//2 + (a*m*b+beta)*t) % q, p)
0057              for t in range(m)] for b in range(q//m)]

0060 def analysis(x, m, a, beta, p, w):
0061     q = len(x)
0062     s = chirp(q, m, a, beta, p, w)
0063     return [transform([s[b][t]*x[b*m+t] % p for t in range(m)], pow(w, q//m, p), p)
0064             for b in range(q//m)]

0067 def synthesis(ds, m, a, beta, p, w):
0068     q = len(ds)*m
0069     s = chirp(q, m, a, beta, p, w)
0070     im = pow(m, -1, p)
0071     out = []
0072     for b, d in enumerate(ds):
0073         v = transform(d, pow(w, -(q//m), p), p)
0074         out.extend(v[t]*im*pow(s[b][t], -1, p) % p for t in range(m))
0075     return out

0078 def recognize(x, m, a, beta, cap, p, w):
0079     """Derive selected modes, certify full block, discover only upon failure."""
0080     q = len(x)
0081     s = chirp(q, m, a, beta, p, w)
0082     active = set()
0083     stored = []
0084     discovery = 0
0085     for b in range(q//m):
0086         v = [s[b][t]*x[b*m+t] % p for t in range(m)]
0087         d = {r: sum(v[t]*pow(w, (q//m)*r*t, p) for t in range(m)) % p for r in active}
0088         recon = [sum(d[r]*pow(w, -(q//m)*r*t, p) for r in active)*pow(m, -1, p) % p
0089                  for t in range(m)]
0090         if recon != v:
0091             discovery += 1
0092             full = transform(v, pow(w, q//m, p), p)
0093             nonzero = {r for r, value in enumerate(full) if value}
0094             assert nonzero-active
0095             active |= nonzero
0096             if len(active) > cap:
0097                 assert discovery <= cap+1
0098                 return None, sorted(active), discovery
0099             d = {r: full[r] for r in active}
0100         stored.append(d)
0101     assert discovery <= len(active)
0102     return [[row.get(r, 0) for r in range(m)] for row in stored], sorted(active), discovery

0105 def green_row(d, q, m, a, beta, p, w):
0106     """Solve y[k]-w^(-k-beta)y[k+a]=forcing[k] around one cycle."""
0107     k = q//m
0108     bhat = transform(d, pow(w, -m, p), p)
0109     forcing = [(bhat[j % k] - pow(w, -(m*(j+beta)+a*m*(m-1)//2), p)
0110                *bhat[(j+a*m) % k]) % p for j in range(q)]
0111     order = [(a*j) % q for j in range(q)]
0112     gauges = [pow(w, a*j*(j-1)//2+beta*j, p) for j in range(q+1)]
0113     assert gauges[-1] == p-1
0114     increments = [forcing[order[j]]*pow(gauges[j], -1, p) % p for j in range(q)]
0115     initial = sum(increments)*pow(2, -1, p) % p
0116     value = initial
0117     y = [0]*q
0118     for j in range(q):
0119         y[order[j]] = gauges[j]*value % p
0120         value = (value-increments[j]) % p
0121     assert gauges[-1]*value % p == initial
0122     assert all((y[j]-pow(w, -j-beta, p)*y[(j+a) % q]-forcing[j]) % p == 0 for j in range(q))
0123     return y

0126 def decode(ds, m, a, beta, p, w):
0127     q = len(ds)*m
0128     y = [0]*q
0129     for r in range(m):
0130         d = [row[r] for row in ds]
0131         if any(d):
0132             z = green_row(d, q, m, a, (beta+r*(q//m)) % q, p, w)
0133             y = [(u+v) % p for u, v in zip(y, z)]
0134     return [v*pow(m, -1, p) % p for v in y]

0137 def det_mod(mat, p):
0138     arr = [row[:] for row in mat]
0139     result = 1
0140     for j in range(len(arr)):
0141         pivot = next((i for i in range(j, len(arr)) if arr[i][j]), None)
0142         if pivot is None:
0143             return 0
0144         if pivot != j:
0145             arr[j], arr[pivot] = arr[pivot], arr[j]
0146             result = -result
0147         value = arr[j][j]
0148         result = result*value % p
0149         for i in range(j+1, len(arr)):
0150             mult = arr[i][j]*pow(value, -1, p) % p
0151             for k in range(j, len(arr)):
0152                 arr[i][k] = (arr[i][k]-mult*arr[j][k]) % p
0153     return result % p

0156 def check_case(x, m, a, beta, cap, p, w):
0157     oracle = analysis(x, m, a, beta, p, w)
0158     support = [r for r in range(m) if any(row[r] for row in oracle)]
0159     ds, found, discoveries = recognize(x, m, a, beta, cap, p, w)
0160     assert (ds is not None) == (len(support) <= cap)
0161     if ds is not None:
0162         assert ds == oracle and found == support
0163         assert synthesis(ds, m, a, beta, p, w) == x
0164         y = decode(ds, m, a, beta, p, w)
0165         assert y == transform(x, pow(w, -1, p), p)
0166         return True, len(x), discoveries
0167     return False, 0, discoveries
