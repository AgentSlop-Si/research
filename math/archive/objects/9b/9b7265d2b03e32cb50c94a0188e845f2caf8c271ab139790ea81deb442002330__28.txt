   4 from pathlib import Path
   5 import independent_phase21_math as M
   6 trim,add,mul,ev,rem,gcd=M.trim,M.padd,M.pmul,M.peval,M.premainder,M.pgcd
   7 RNG=random.Random(220991)
   8 def sub(a,b,p):return add(a,[-c%p for c in b],p)
   9 def divlin(a,x,p):
  10  q=[0]*(len(a)-1);q[-1]=a[-1]
  11  for j in range(len(q)-2,-1,-1):q[j]=(a[j+1]+x*q[j+1])%p
  12  assert (a[0]+x*q[0])%p==0
  13  return trim(q)
  14 def remove_identity(v,p):
  15  v=trim([c%p for c in v]);k=0
  16  while len(v)>1 and ev(v,2,p)==0:v=divlin(v,2,p);k+=1
  17  return v,k
  18 def dickson_full(n,p):
  19  a=[2];b=[0,1]
  20  if n==0:return a
  21  for _ in range(1,n):a,b=b,sub(mul([0,1],b,p),a,p)
  22  return b
  23 # Independent 2x2 transfer-matrix powering; no source binary-pair recurrence.
  24 def dickson_quotient(n,v,p):
  25  def matmul(A,B):return [[rem(add(mul(A[i][0],B[0][j],p),mul(A[i][1],B[1][j],p),p),v,p)for j in range(2)]for i in range(2)]
  26  a=[[[1],[]],[[],[1]]];b=[[[0,1],[p-1]],[[1],[]]]
  27  while n:
  28   if n&1:a=matmul(a,b)
  29   n//=2
  30   if n:b=matmul(b,b)
  31  return rem(add(a[0][0],a[1][1],p),v,p)
  32 def sieve(v,n,p,full=None):
  33  v=trim([c%p for c in v]);assert v
  34  v,removed=remove_identity(v,p)
  35  if len(v)<2:return [1],removed
  36  if full is None:d=dickson_quotient(n,v,p)
  37  else:d=rem(full,v,p)
  38  return gcd(v,sub(d,[2],p),p),removed
  39 def bsgs(target,w,n,p):
  40  q=math.isqrt(n)+1;table={};z=1
  41  for j in range(q):table.setdefault(z,j);z=z*w%p
  42  step=pow(w,-q,p);z=target
  43  for i in range((n+q-1)//q+1):
  44   if z in table:
  45    d=(i*q+table[z])%n
  46    if pow(w,d,p)==target:return d
  47   z=z*step%p
  48  return None
  49 def enum_half(g,w,n,p):
  50  out=[]
  51  for d in range(1,n//2+1):
  52   t=(pow(w,d,p)+pow(w,-d,p))%p
  53   if ev(g,t,p)==0:out+= [d] if d==n-d else [d,n-d]
  54  return sorted(out)
  55 def enum_linear(g,w,n,p):
  56  t=-g[0]*pow(g[1],-1,p)%p
  57  # Independent exhaustive field oracle for quadratic roots, deliberately not a complexity claim.
  58  roots=[z for z in range(p)if (z*z-t*z+1)%p==0 and pow(z,n,p)==1 and z!=1]
  59  ds=[bsgs(z,w,n,p)for z in roots];assert all(d is not None for d in ds)
  60  return sorted(set(ds))
  61 def findroot(n,p):return next(w for w in range(1,p)if pow(w,n,p)==1 and len({pow(w,k,p)for k in range(n)})==n)
  62 def trace_from_U(U,b,p):
  63  r=U[:];V=[0]*(b+1)
  64  for j in range(b,-1,-1):
  65   c=r[b+j];V[j]=c
  66   for k in range(j+1):r[b+j-2*k]=(r[b+j-2*k]-c*math.comb(j,k))%p
  67  assert not any(r)
  68  return trim(V)
  69 def gate(r,w,p,s):
  70  n=len(r);R=M.dft(r,w,p,4*s);meta={'full_checks':0,'candidate_roots':0}
  71  if not all(R):return None,{**meta,'reason':'zero-probe'}
  72  T=[pow(x,-1,p)for x in R];U=M.compress(T,s,p)
  73  if not any(U):return None,{**meta,'reason':'zero-resultant'}
  74  b=s*(s+1)*(2*s+1)//6;V=trace_from_U(U,b,p);g,k=sieve(V,n,p)
  75  roots=enum_linear(g,w,n,p) if len(g)==2 else enum_half(g,w,n,p) if len(g)>1 else []
  76  old=[d for d in range(1,n)if ev(U,pow(w,d,p),p)==0];assert roots==old
  77  meta.update({'candidate_roots':len(roots),'trace_gcd_degree':len(g)-1,'identity_factors_removed':k})
  78  powers=[pow(w,k,p)for k in range(n)]
  79  for delta in roots:
  80   z=powers[delta];ga=gcd(M.pencil(T,z,0,s,p),M.pencil(T,z,1,s,p),p)
  81   if len(ga)!=2:continue
  82   alpha=-ga[0]%p
  83   if not alpha:continue
  84   Y=[T[k]*(1-alpha*pow(z,k,p))%p for k in range(4*s)];A=M.prony(Y,s,powers,p)
  85   if not A:continue
  86   Ah=[sum(v*pow(z,e,p)for e,v in A.items())%p for z in powers]
  87   if not all(Ah):continue
  88   meta['full_checks']+=1
  89   lhs=[sum(v*r[(j-e)%n]for e,v in A.items())%p for j in range(n)]
  90   rhs=[(int(j==0)-alpha*int(j==delta))%p for j in range(n)]
  91   if lhs!=rhs:return None,{**meta,'reason':'full-certificate-failed'}
  92   out=[(1-alpha*pow(w,k*delta,p))*pow(Ah[k],-1,p)%p for k in range(n)]
  93   assert out==M.dft(r,w,p)
  94   return (A,delta,alpha,out),{**meta,'reason':'accepted'}
  95  return None,{**meta,'reason':'no-candidate'}
