   1	"""Practical streaming update for a FIXED, KNOWN sample index.
   2	
   3	Useful complement to the conditional OpenAI-#130/sparse factoring result.
   4	This is the classical Fourier rank-one update, not a new theorem.
   5	"""
   6	import numpy as np
   7	
   8	class PreparedPointEdit:
   9	    """Maintain rFFT(x) for repeated known changes at one input position."""
  10	    def __init__(self,n:int,index:int):
  11	        if n<1 or index<0 or index>=n:
  12	            raise ValueError('invalid length or index')
  13	        self.n,self.index=n,index
  14	        k=np.arange(n//2+1,dtype=np.float64)
  15	        self.phase=np.exp((-2j*np.pi*index/n)*k)
  16	    def update(self,cached_rfft,delta):
  17	        arr=np.asarray(cached_rfft)
  18	        if arr.ndim!=1 or arr.size!=self.phase.size:
  19	            raise ValueError('incorrect spectrum')
  20	        if complex(delta).imag!=0:
  21	            raise ValueError('only real delta supported by rFFT')
  22	        return arr + delta*self.phase
