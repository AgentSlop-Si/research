   1 """Independent 64-bit bounded-word verification of the new uploaded guarded-sum idea.
   2 Tests the accumulation primitive; does not benchmark a complete FFT/table builder.
   3 """
   4 import random,json
   5 
   6 def layout(G,bound,word_bits):
   7     h=(2*bound*G).bit_length() # 2**h > max lane sum
   8     lanes=word_bits//h
   9     if not lanes:raise ValueError('no complete lane fits')
  10     return h,lanes
  11 
  12 def prepack(row,G,bound,W):
  13     h,t=layout(G,bound,W); out=[]
  14     for s in range(0,len(row),t):
  15         word=sum((v+bound)<<(h*j) for j,v in enumerate(row[s:s+t]))
  16         assert all(-bound<=v<=bound for v in row[s:s+t])
  17         assert word.bit_length()<=W
  18         out.append(word)
  19     return out
  20 
  21 def packed_sum(rows,B,G,bound,W):
  22     h,t=layout(G,bound,W); words=[0]*len(rows[0]); adds=0
  23     for row in rows:
  24         for j,v in enumerate(row):
  25             words[j]+=v; adds+=1
  26             assert words[j].bit_length()<=W # No hidden giant integer or word overflow.
  27     mask=(1<<h)-1
  28     out=[((words[j//t]>>(h*(j%t)))&mask)-G*bound for j in range(B)]
  29     return out,adds
  30 
  31 rng=random.Random(20261009);blocks=0;lanes=0;records=[]
  32 for G in [1,2,4,8,16,32,64,128,256]:
  33  for bound in [1,3,8,17]:
  34   for B in [1,7,32,64,129]:
  35    h,t=layout(G,bound,64)
  36    for mode in ['random','max','min','alternating']:
  37     raw=[]
  38     for i in range(G):
  39      raw.append([rng.randint(-bound,bound) if mode=='random' else bound if mode=='max' else -bound if mode=='min' else (bound if (i+j)%2 else -bound) for j in range(B)])
  40     packed=[prepack(row,G,bound,64) for row in raw]
  41     got,adds=packed_sum(packed,B,G,bound,64)
  42     assert got==[sum(row[j] for row in raw) for j in range(B)]
  43     assert adds==G*((B+t-1)//t)
  44     blocks+=1;lanes+=B
  45    records.append({'groups':G,'lane_bound':bound,'B':B,'guarded_lane_bits':h,'lanes_per_64bit_word':t,'words_per_table_result':(B+t-1)//t})
  46 print(json.dumps({'status':'passed','block_cases':blocks,'lane_sums_checked':lanes,'physical_word_bits':64,'records':records,'scope':'Every packed word and accumulated word stayed within64bits.
