0023 def primitive_root_of_order(n, p):
0024     # Audit fixture setup only; not part of an online algorithm.
0025     assert p > n and (p-1) % n == 0
0026     for a in range(1, p):
0027         w = pow(a, (p-1)//n, p)
0028         if pow(w,n,p) == 1 and all(pow(w,d,p) != 1 for d in range(1,n) if n%d == 0):
0029             return w
0030     raise AssertionError((n,p))

0033 def materialize(rows, q, m, p, w):
0034     eta = pow(w,q,p)
0035     return [sum(c*pow(eta,h*u,p) for h,c in rows[t].items()) % p
0036             for u in range(m) for t in range(q)]

0039 def raw_coefficients(x,q,m,p,w):
0040     eta=pow(w,q,p)
0041     inv=pow(m,-1,p)
0042     return [{h: a for h in range(m)
0043              if (a := inv*sum(x[t+q*u]*pow(eta,(-h*u)%m,p) for u in range(m))%p)}
0044             for t in range(q)]

0047 def direct_convolution(x,y,p):
0048     n=len(x)
0049     return [sum(x[j]*y[(k-j)%n] for j in range(n))%p for k in range(n)]

0052 def graph_coefficients(xrows,yrows,q,m,p,w):
0053     # Independent algebra implementation: enumerate row pairs, intersect labels,
0054     # keep only a dictionary accumulator. Uploaded code uses frequency buckets,
0055     # a dense array, and an event list instead.
0056     eta=pow(w,q,p)
0057     coeff={}
0058     edges=0
0059     events=[]
0060     for a in range(q):
0061         for b in range(q):
0062             t=(a+b)%q
0063             carry=(a+b)//q
0064             for h in xrows[a].keys() & yrows[b].keys():
0065                 term=m*xrows[a][h]*yrows[b][h]*pow(eta,(-h*carry)%m,p)%p
0066                 idx=(t,h)
0067                 old=coeff.get(idx,0)
0068                 coeff[idx]=(old+term)%p
0069                 events.append((idx,old,coeff[idx]))
0070                 edges+=1
0071     return {k:v for k,v in coeff.items() if v}, edges, events

0082 def rank_factor(matrix,p,bound=None):
0083     """Fresh fixed-rank elimination with an explicit reconstruction assertion."""
0084     q=len(matrix); m=len(matrix[0]) if q else 0
0085     basis=[]; pivots=[]; coefficients=[]
0086     for raw in matrix:
0087         residual=[v%p for v in raw]
0088         row_factors=[]
0089         for pivot,base in zip(pivots,basis):
0090             multiplier=residual[pivot]
0091             row_factors.append(multiplier)
0092             residual=[(a-multiplier*b)%p for a,b in zip(residual,base)]
0093         pivot=next((j for j,a in enumerate(residual) if a),None)
0094         if pivot is not None:
0095             if bound is not None and len(basis)==bound:
0096                 return None
0097             multiplier=residual[pivot]
0098             inverse=pow(multiplier,-1,p)
0099             basis.append([inverse*a%p for a in residual])
0100             pivots.append(pivot)
0101             row_factors.append(multiplier)
0102         coefficients.append(row_factors)
0103     for row in coefficients:
0104         row.extend([0]*(len(basis)-len(row)))
0105     assert all(sum(coefficients[t][l]*basis[l][h] for l in range(len(basis)))%p == matrix[t][h]%p
0106                for t in range(q) for h in range(m))
0107     return coefficients,basis

0110 def low_rank_reconstruct(matrix,q,m,p,w,bound):
0111     labels=[h for h in range(m) if any(matrix[t][h] for t in range(q))]
0112     compressed=[[matrix[t][h] for h in labels] for t in range(q)]
0113     factor=rank_factor(compressed,p,bound)
0114     if factor is None:
0115         return None
0116     left,right=factor
0117     eta=pow(w,q,p)
0118     powers=[1]
0119     for _ in range(1,m):
0120         powers.append(powers[-1]*eta%p)
0121     evaluated=[[sum(row[j]*powers[(h*u)%m] for j,h in enumerate(labels))%p
0122                 for u in range(m)] for row in right]
0123     return [sum(left[t][l]*evaluated[l][u] for l in range(len(right)))%p
0124             for u in range(m) for t in range(q)]

0127 def verify_quotient_identity(xrows,yrows,states,q,m,p,w):
0128     # The quotient gauge turns the carry-twisted length-Q product into a cyclic
0129     # convolution separately for every h; this does not give a free joint FFT.
0130     for h in range(m):
0131         alpha=[row.get(h,0)*pow(w,(-h*t)%(q*m),p)%p for t,row in enumerate(xrows)]
0132         beta=[row.get(h,0)*pow(w,(-h*t)%(q*m),p)%p for t,row in enumerate(yrows)]
0133         ordinary=direct_convolution(alpha,beta,p)
0134         for t in range(q):
0135             assert states.get((t,h),0)==m*pow(w,h*t,p)*ordinary[t]%p
0136             COUNTS['quotient_state_comparisons']+=1

0139 def verify_pair(xrows,yrows,q,m,p,w,check_raw=True):
0140     x=materialize(xrows,q,m,p,w); y=materialize(yrows,q,m,p,w)
0141     # Recover modes from raw arrays by a separately written full short DFT.
0142     if check_raw:
0143         assert raw_coefficients(x,q,m,p,w)==xrows
0144         assert raw_coefficients(y,q,m,p,w)==yrows
0145         COUNTS['raw_mode_recoveries']+=2
0146     states,edges,events=graph_coefficients(xrows,yrows,q,m,p,w)
0147     expected=direct_convolution(x,y,p)
0148     actual=reconstruct(states,q,m,p,w)
0149     assert actual==expected
0150     COUNTS['all_output_convolution_pairs']+=1
0151     COUNTS['all_output_convolution_bins']+=q*m
0152     ax=Counter(h for row in xrows for h in row)
0153     by=Counter(h for row in yrows for h in row)
0154     assert edges==sum(ax[h]*by[h] for h in ax)
0155     kx=sum(map(len,xrows)); ky=sum(map(len,yrows)); s=len(states)
0156     assert s<=q*m and s<=edges and s*s<=q*kx*ky
0157     COUNTS['support_bound_cases']+=1
