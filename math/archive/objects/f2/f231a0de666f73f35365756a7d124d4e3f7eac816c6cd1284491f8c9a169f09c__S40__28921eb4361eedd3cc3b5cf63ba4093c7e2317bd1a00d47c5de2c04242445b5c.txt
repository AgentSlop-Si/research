   1 """Exact fixed-graph Fourier research primitives, over a supplied prime field.
   2 
   3 The actual Fourier executor below is O(N log N), NOT a faster FFT.
   4 The O(N) auxiliary algorithms compute closed traces or tree Green operators,
   5 not the arbitrary-input Fourier vector. See THEORY.md for the missing step.
   6 
   7 Only Python's standard library is required. The finite fields are verification
   8 models: no claim that a fixed field supports unbounded transform lengths.
   9 """
  10 from __future__ import annotations
  11 
  12 from dataclasses import dataclass
  13 from typing import Sequence
  14 
  15 
  16 @dataclass
  17 class Counts:
  18     additions: int = 0  # Includes subtractions, not indexing.
  19     multiplications: int = 0
  20 
  21 
  22 @dataclass(frozen=True)
  23 class Field:
  24     p: int
  25     generator: int
  26 
  27     def inverse(self, value: int) -> int:
  28         value %= self.p
  29         if value == 0:
  30             raise ZeroDivisionError("Zero has no multiplicative inverse.")
  31         return pow(value, -1, self.p)
  32 
  33     def root(self, length: int) -> int:
  34         if length < 1 or (self.p - 1) % length:
  35             raise ValueError("The supplied field must contain the required root.")
  36         root = pow(self.generator, (self.p - 1) // length, self.p)
  37         if pow(root, length, self.p) != 1:
  38             raise ValueError("Invalid supplied generator/root.")
  39         if length > 1 and length & (length - 1) == 0:
  40             if pow(root, length // 2, self.p) == 1:
  41                 raise ValueError("The supplied dyadic root is not primitive.")
  42         return root
  43 
  44 
  45 def binary_reversals(bits: int) -> list[int]:
  46     if bits < 0:
  47         raise ValueError("bits must be nonnegative")
  48     result = [0] * (1 << bits)
  49     if bits:
  50         for j in range(1, len(result)):
  51             result[j] = (result[j >> 1] >> 1) | ((j & 1) << (bits - 1))
  52     return result
  53 
  54 
  55 class FixedGraph:
  56     """Plan for Q_N; full root, phase and address preparation is O(N)."""
  57 
  58     def __init__(self, length: int, field: Field):
  59         if length < 1 or length & (length - 1):
  60             raise ValueError("length must be a positive power of two")
  61         self.N = length
  62         self.n = length.bit_length() - 1
  63         self.field = field
  64         self.w = field.root(length)
  65         self.roots = [1] * length
  66         # Count exact field multiplications for root-table preparation separately.
  67         self.root_table_multiplications = max(0, length - 2)
  68         if length > 1:
  69             self.roots[1] = self.w
  70         for j in range(2, length):
  71             self.roots[j] = self.roots[j - 1] * self.w % field.p
  72         self.phi = [0] * length
  73         for j in range(1, length):
  74             self.phi[j] = self.phi[j >> 1] + (j & 1) * (j >> 1)
  75         self.reversal = binary_reversals(self.n)
  76 
  77     def _check(self, x: Sequence[int]) -> None:
  78         if len(x) != self.N:
  79             raise ValueError(f"Expected {self.N} coordinates, got {len(x)}")
  80 
  81     def apply(self, x: Sequence[int], counts: Counts | None = None) -> list[int]:
  82         """One Q application: N additions and N/2-1 multiplies (N >= 2)."""
  83         self._check(x)
  84         if self.N == 1:
  85             return [x[0] % self.field.p]
  86         p = self.field.p
  87         m = self.N // 2
  88         out = [0] * self.N
  89         for r in range(m):
  90             out[2 * r] = (x[r] + x[r + m]) % p
  91             difference = (x[r] - x[r + m]) % p
  92             out[2 * r + 1] = difference if r == 0 else self.roots[r] * difference % p
  93         if counts is not None:
  94             counts.additions += self.N
  95             counts.multiplications += m - 1
  96         return out
  97 
  98     def power(self, x: Sequence[int], exponent: int, counts: Counts | None = None) -> list[int]:
  99         if exponent < 0:
 100             raise ValueError("This evaluator accepts nonnegative powers.")
 101         self._check(x)
 102         result = [v % self.field.p for v in x]
 103         for _ in range(exponent):
 104             result = self.apply(result, counts)
 105         return result
 106 
 107     def fourier(self, x: Sequence[int], counts: Counts | None = None) -> list[int]:
 108         """Actual full Fourier executor. Its work is still Theta(N log N)."""
 109         v = self.power(x, self.n, counts)
 110         p = self.field.p
 111         out = [0] * self.N
 112         for k, r in enumerate(self.reversal):
 113             if self.phi[r] == 0:
 114                 out[k] = v[r]
 115             else:
 116                 out[k] = self.roots[(-self.phi[r]) % self.N] * v[r] % p
 117                 if counts is not None:
 118                     counts.multiplications += 1
 119         return out
 120 
 121     def trace_prefix(self, extra: int = 0) -> tuple[list[int], int]:
 122         """tr(Q^k), k=1..n+extra, by 2^(n+extra+1)-2 periodic states.
 123 
 124         Any FIXED extra preserves O(N) work. With extra=0 the count is 2N-2.
 125         Exponent/address integers are polynomially bounded for fixed extra.
 126         """
 127         if extra < 0:
 128             raise ValueError("extra must be nonnegative")
 129         phase_table = [0] * (1 << (self.n + extra))
 130         for j in range(1, len(phase_table)):
 131             phase_table[j] = phase_table[j >> 1] + (j & 1) * (j >> 1)
 132         traces = []
 133         states = 0
 134         for k in range(1, self.n + extra + 1):
 135             copies, remainder = divmod(self.n, k)
 136             repeat = ((1 << (copies * k)) - 1) // ((1 << k) - 1)
 137             reversals = binary_reversals(k)
 138             total = 0
 139             mask = (1 << k) - 1
 140             for a in range(1 << k):
 141                 s = (a * repeat) << remainder
 142                 if remainder:
 143                     s += a >> (k - remainder)
 144                 # Next k emitted bits: rotate the period word to position n.
 145                 v = (((a << remainder) & mask) | (a >> (k - remainder))) if remainder else a
 146                 exponent = (s * reversals[v] + phase_table[v]) & (self.N - 1)
 147                 total = (total + self.roots[exponent]) % self.field.p
 148             states += 1 << k
 149             traces.append(total)
 150         return traces, states
 151 
 152     def determinant_jet(self, extra: int = 0) -> list[int]:
 153         """det(I-zQ) mod z^(n+extra+1), not an arbitrary-input vector solve."""
 154         degree = self.n + extra
 155         if self.field.p <= degree:
 156             raise ValueError("Need characteristic greater than the truncation degree.")
 157         traces, _ = self.trace_prefix(extra)
 158         d = [1]
 159         for j in range(1, degree + 1):
 160             value = sum(d[j - k] * traces[k - 1] for k in range(1, j + 1))
 161             d.append(-value * self.field.inverse(j) % self.field.p)
 162         return d
 163 
 164     def branch_apply(self, x: Sequence[int], bit: int) -> list[int]:
 165         self._check(x)
 166         if self.N < 2 or bit not in (0, 1):
 167             raise ValueError("Branches require N >= 2 and bit in {0,1}.")
 168         p = self.field.p
 169         out = [0] * self.N
 170         for s, value in enumerate(x):
 171             target = (2 * s + bit) & (self.N - 1)
 172             out[target] = (out[target] + self.roots[bit * s] * value) % p
 173         return out
 174 
 175     def branch_solve(self, x: Sequence[int], bit: int, z: int) -> list[int]:
 176         """Solve (I-z A_bit)y=x by one tree accumulation and one root division.
 177 
 178         Topological preparation is included on every call in this implementation.
 179         It could instead be cached, but that is not needed for the O(N) bound.
 180         """
 181         self._check(x)
 182         if self.N < 2 or bit not in (0, 1):
 183             raise ValueError("Branches require N >= 2 and bit in {0,1}.")
 184         p = self.field.p
 185         root = 0 if bit == 0 else self.N - 1
 186         successor = [(2 * s + bit) & (self.N - 1) for s in range(self.N)]
 187         indegree = [0] * self.N
 188         for s in range(self.N):
 189             if s != root:
 190                 indegree[successor[s]] += 1
 191         stack = [s for s in range(self.N) if indegree[s] == 0]
 192         out = [v % p for v in x]
 193         processed = 0
 194         while stack:
 195             s = stack.pop()
 196             if s == root:
 197                 continue
 198             t = successor[s]
 199             out[t] = (out[t] + z * self.roots[bit * s] * out[s]) % p
 200             processed += 1
 201             indegree[t] -= 1
 202             if indegree[t] == 0:
 203                 stack.append(t)
 204         if processed != self.N - 1:
 205             raise AssertionError("The binary branch should be a rooted tree plus self-loop.")
 206         out[root] = out[root] * self.field.inverse(1 - z * self.roots[bit * root]) % p
 207         return out
 208 
 209     def haar_apply(self, x: Sequence[int], z: int) -> list[int]:
 210         """Apply H_N(z)=C(I-z A_0)^-1 J with an O(N)-work recursive Haar tree."""
 211         if self.N < 2 or len(x) != self.N // 2:
 212             raise ValueError("H input length must be N/2, with N >= 2.")
 213         p = self.field.p
 214         beta = (2 * z - 1) * self.field.inverse(1 - z) % p
 215 
 216         def rec(v: Sequence[int], scale: int) -> list[int]:
 217             m = len(v)
 218             if m == 1:
 219                 return [beta * v[0] % p]
 220             h = m // 2
 221             u = [(v[a] + v[a + h]) % p for a in range(h)]
 222             d = [(v[a] - v[a + h]) % p for a in range(h)]
 223             small = rec(u, scale * 2)
 224             out = [0] * m
 225             for a in range(h):
 226                 out[2 * a] = z * small[a] % p
 227                 out[2 * a + 1] = self.roots[scale * (2 * a + 1)] * d[a] % p
 228             return out
 229 
 230         return rec(x, 1)
 231 
 232     def haar_inverse(self, y: Sequence[int], z: int) -> list[int]:
 233         """Invert H itself in O(N); this does NOT invert I-zH."""
 234         if self.N < 2 or len(y) != self.N // 2:
 235             raise ValueError("H output length must be N/2, with N >= 2.")
 236         p = self.field.p
 237         iz, i2 = self.field.inverse(z), self.field.inverse(2)
 238         ibeta = (1 - z) * self.field.inverse(2 * z - 1) % p
 239 
 240         def rec(v: Sequence[int], scale: int) -> list[int]:
 241             m = len(v)
 242             if m == 1:
 243                 return [ibeta * v[0] % p]
 244             h = m // 2
 245             u = rec([v[2 * a] * iz % p for a in range(h)], scale * 2)
 246             d = [v[2 * a + 1] * self.roots[(-scale * (2 * a + 1)) % self.N] % p
 247                  for a in range(h)]
 248             return ([(u[a] + d[a]) * i2 % p for a in range(h)]
 249                     + [(u[a] - d[a]) * i2 % p for a in range(h)])
 250 
 251         return rec(y, 1)
