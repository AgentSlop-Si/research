  14	def determinant_monomials(entries):
  15	    """Subset DP for det([s_ij*t^q_ij]), producing exact integer coefficients."""
  16	    n=len(entries);state={0:{0:1}}
  17	    for r in range(n):
  18	        nxt={}
  19	        for mask,poly in state.items():
  20	            for c in range(n):
  21	                if mask>>c&1:continue
  22	                sign,exponent=entries[r][c]
  23	                if (mask>>(c+1)).bit_count()%2:sign=-sign
  24	                dest=nxt.setdefault(mask|1<<c,{})
  25	                for power,coefficient in poly.items():
  26	                    dest[power+exponent]=dest.get(power+exponent,0)+sign*coefficient
  27	        state={mask:{p:c for p,c in poly.items() if c} for mask,poly in nxt.items()}
  28	    return sp.Poly.from_dict({(p,):c for p,c in state[(1<<n)-1].items()},T,domain=sp.ZZ)

  30	def sparse(poly):
  31	    assert all(c.q == 1 for m,c in poly.terms())
  32	    return [[m[0],int(c)] for m,c in sorted(poly.terms())]

  34	def guard_factors(poly):
  35	    """Return exact decomposition into t-powers and factors dividing t^64-1."""
  36	    p=poly.monic();valuation=0;layers=[]
  37	    while p.degree()>0 and p.eval(0)==0:p=p.exquo(sp.Poly(T,T));valuation+=1
  38	    bad=sp.Poly(T**64-1,T)
  39	    while p.degree()>0:
  40	        f=sp.gcd(p,bad)
  41	        if f.degree()==0:return None
  42	        p=p.exquo(f);layers.append(sparse(f))
  43	    return {'zero_valuation':valuation,'root_of_unity_layers':layers}
