#!/usr/bin/env python3
"""Fresh formula implementation; never imports or executes uploaded source code."""
import math,json,argparse
from fractions import Fraction
import numpy as np
from functools import lru_cache

@lru_cache(None)
def phi(n):
    out=n; p=2
    while p*p<=n:
        if n%p==0:
            out-=out//p
            while n%p==0:n//=p
        p+=1
    if n>1:out-=out//n
    return out
@lru_cache(None)
def mu(n):
    out=1;p=2
    while p*p<=n:
        if n%p==0:
            n//=p;out=-out
            if n%p==0:return 0
        p+=1
    return -out if n>1 else out

def ramanujan(n,q):
    v=n//math.gcd(n,q)
    return phi(n)*mu(v)//phi(v)

def full_exact(n,m):
    ds=[set(range(u-m+1,u+1))-{0} for u in range(m)]
    return [[sum(Fraction(ramanujan(n,d*(j-i)),n*phi(n)) for d in ds[i%m]&ds[j%m]) for j in range(n)] for i in range(n)]

def masks(n,m):
    out={}
    u=np.arange(m)
    for d in range(1,m):
        g=math.gcd(n,d)
        if g not in out:out[g]=np.zeros((m,m),float)
        pos=(u>=d).astype(float);neg=(u<m-d).astype(float)
        out[g]+=np.outer(pos,pos)+np.outer(neg,neg)
    return out

def block_norm(n,m,return_block=False):
    k=n//m;out=masks(n,m)
    # Gauged block-Fourier basis removes exp(2 pi i ell u / N).
    diff=(np.arange(m)[:,None]-np.arange(m)[None,:])%m
    largest=-1;arg=-1;best=None
    seen=set()
    for ell in range(k):
        gs=tuple(math.gcd(ell+s*k,n) for s in range(m))
        if gs in seen:continue
        seen.add(gs)
        b=np.zeros((m,m),complex)
        for g,w in out.items():
            kernel=np.fft.ifft(np.array([a==g for a in gs],float))/phi(n//g)
            b+=w*kernel[diff]
        val=float(np.linalg.eigvalsh(b)[-1])
        if val>largest:largest=val;arg=ell;best=b
    if return_block:return largest,arg,best,len(seen)
    return largest

def verify():
    records=[]
    for n in range(4,101):
        for m in range(2,math.isqrt(n)+1):
            if n%m:continue
            exact=full_exact(n,m);a=np.array(exact,float)
            val=np.linalg.eigvalsh(a)[-1]
            block=block_norm(n,m)
            assert np.max(np.abs(a-a.T))==0
            assert abs(np.trace(a)-(m-1))<1e-12
            assert abs(val-block)<2e-12,(n,m,val,block)
            records.append((n,m))
    return len(records)

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--max-n',type=int,default=1000);ap.add_argument('--output',default='probe_results.json');ap.add_argument('--verify',action='store_true');args=ap.parse_args()
    checked=verify() if args.verify else 0
    records=[];fails=[];worst=(0,None)
    for n in range(4,args.max_n+1):
        for m in range(2,math.isqrt(n)+1):
            if n%m:continue
            val=block_norm(n,m);ratio=val*phi(n)/m
            rec=dict(N=n,M=m,K=n//m,phi_N=phi(n),norm=val,relative_to_M_over_N=val*n/m,relative_to_M_over_phi=ratio)
            records.append(rec)
            if ratio>worst[0]:worst=(ratio,rec);print('WORST',rec,flush=True)
            if ratio>1+1e-10:fails.append(rec);print('COUNTEREXAMPLE',rec,flush=True)
    data=dict(formula_checks=checked,parameter_pairs=len(records),max_n=args.max_n,worst=worst[1],counterexamples=fails,records=records)
    with open(args.output,'w') as f:json.dump(data,f,indent=2)
    print(json.dumps({k:v for k,v in data.items() if k!='records'},indent=2))
if __name__=='__main__':main()
