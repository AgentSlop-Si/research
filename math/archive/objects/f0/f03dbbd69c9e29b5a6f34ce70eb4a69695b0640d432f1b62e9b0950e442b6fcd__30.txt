0009 class TestHybrid(unittest.TestCase):
0010     def test_all_small_binary_sparsities(self):
0011         rng=np.random.default_rng(188)
0012         for n in [1,2,4,8,16,32,64,128,512]:
0013             for k in sorted(set([0,1,2,3,4,7,16,n//2,n])):
0014                 if k>n: continue
0015                 indices=rng.choice(n,size=k,replace=False)
0016                 vals=rng.normal(size=k)+1j*rng.normal(size=k)
0017                 edits=list(zip(indices.tolist(),vals.tolist()))
0018                 y,plan=sparse_dft(n,edits,return_plan=True)
0019                 x=np.zeros(n,dtype=np.complex128)
0020                 x[indices]=vals
0021                 self.assertTrue(np.allclose(y,np.fft.fft(x),rtol=2e-11,atol=2e-10), f'{n=} {k=} maxdiff={np.max(np.abs(y-np.fft.fft(x)))}')
0022                 self.assertEqual(plan.n,n)
0023                 self.assertLessEqual(plan.sparse_phase_terms,n)
0024                 self.assertGreaterEqual(plan.b,max(k,1))
0025                 if k:
0026                     self.assertLess(plan.b,2*k)

0028     def test_duplicates_cancel(self):
0029         d=normalize_changes(16,[(3,2),(3,-2),(8,1),(8,3)])
0030         self.assertEqual(d,{8:4+0j})
0031         y=sparse_dft(16,[(3,2),(3,-2),(8,1),(8,3)])
0032         x=np.zeros(16);x[8]=4
0033         self.assertTrue(np.allclose(y,np.fft.fft(x)))

0055     def test_low_rank_sparse(self):
0056         rng=np.random.default_rng(33)
0057         for n in [8,64,256]:
0058             basis=rng.normal(size=(2,n))
0059             coeff=np.array([2.5,-0.75])
0060             edits=[(0,0.4),(n//2,-0.65),(n-1,1.2)]
0061             y,plan=low_rank_plus_sparse(np.fft.fft(basis,axis=1),coeff,edits)
0062             x=coeff@basis
0063             for pos,val in edits:x[pos]+=val
0064             self.assertTrue(np.allclose(y,np.fft.fft(x),atol=2e-10))

0067     def test_pluggable_small_transform(self):
0068         def brute(a):
0069             m=len(a)
0070             omega=np.exp(-2j*np.pi/m)
0071             return np.array([sum(a[j]*omega**(j*k) for j in range(m)) for k in range(m)])
0072         for n in [4,8,16,64]:
0073             y=sparse_dft(n,[(0,1),(n//2,3)],small_transform_1d=brute)
0074             x=np.zeros(n);x[0]=1;x[n//2]=3
0075             self.assertTrue(np.allclose(y,np.fft.fft(x),atol=1e-10))
