  11 def rank(A,p):
  12     a=[[int(x)%p for x in row] for row in A]
  13     if not a:return 0
  14     r=0
  15     for j in range(len(a[0])):
  16         k=next((i for i in range(r,len(a)) if a[i][j]),None)
  17         if k is None:continue
  18         a[r],a[k]=a[k],a[r]
  19         v=pow(a[r][j],-1,p);a[r]=[x*v%p for x in a[r]]
  20         for i in range(r+1,len(a)):
  21             v=a[i][j]
  22             if v:a[i]=[(x-v*y)%p for x,y in zip(a[i],a[r])]
  23         r+=1
  24         if r==len(a):break
  25     return r

  28 def eye(n):return [[int(i==j) for j in range(n)] for i in range(n)]
  29 def add(A,B,p):return [[(x+y)%p for x,y in zip(a,b)] for a,b in zip(A,B)]
  30 def sub(A,B,p):return [[(x-y)%p for x,y in zip(a,b)] for a,b in zip(A,B)]
  31 def mul(A,B,p):return [[sum(x*y for x,y in zip(a,b))%p for b in zip(*B)] for a in A]
  32 def vec(A,v,p):return [sum(x*y for x,y in zip(a,v))%p for a in A]
  33 def inv(A,p):
  34     n=len(A);a=[list(row)+e for row,e in zip(A,eye(n))]
  35     for j in range(n):
  36         k=next(i for i in range(j,n) if a[i][j]%p)
  37         a[j],a[k]=a[k],a[j]
  38         u=pow(a[j][j],-1,p);a[j]=[x*u%p for x in a[j]]
  39         for i in range(n):
  40             if i!=j:
  41                 u=a[i][j];a[i]=[(x-u*y)%p for x,y in zip(a[i],a[j])]
  42     return [a[n:] for a in a]

  45 def prime_factors(n):
  46     a=[];q=2
  47     while q*q<=n:
  48         if n%q==0:
  49             a.append(q)
  50             while n%q==0:n//=q
  51         q+=1
  52     if n>1:a.append(n)
  53     return a

  56 def primitive_root(p):
  57     assert p>2 and all(p%d for d in range(2,math.isqrt(p)+1))
  58     fs=prime_factors(p-1)
  59     return next(g for g in range(2,p) if all(pow(g,(p-1)//q,p)!=1 for q in fs))

  62 def roots(n,p,g):
  63     assert (p-1)%n==0
  64     w=pow(g,-((p-1)//n),p);a=[1]
  65     for _ in range(n-1):a.append(a[-1]*w%p)
  66     assert len(set(a))==n and a[-1]*w%p==1
  67     return a

  70 def matrices(n,p,g,scale=1):
  71     a=roots(n,p,g)
  72     H=[[scale*a[i*j%n]%p for j in range(n)] for i in range(n)]
  73     Z=[[int(i==(j+1)%n) for j in range(n)] for i in range(n)]
  74     D=[[a[i] if i==j else 0 for j in range(n)] for i in range(n)]
  75     Zi=[[int(i==(j-1)%n) for j in range(n)] for i in range(n)]
  76     return H,Z,D,Zi

  79 def residuals(A,p,g):
  80     _,Z,D,Zi=matrices(len(A),p,g)
  81     return sub(mul(A,Z,p),mul(D,A,p),p),sub(mul(A,D,p),mul(Zi,A,p),p)

  84 def recognize(A,p,g):
  85     """Complete general-N monomial Fourier recognizer; returns a certificate.
  86     Primitive exponent means gcd(e,N)=1, also for non-prime-power lengths.
  87     Root table is prepared by successive multiplication, not repeated powers.
  88     """
  89     n=len(A)
  90     assert all(len(a)==n for a in A)
  91     if any(x%p==0 for a in A for x in a):return {'equivalent':False,'reason':'zero'}
  92     a=roots(n,p,g);lookup={x:i for i,x in enumerate(a)}
  93     ri=[A[0][0]*pow(A[i][0],-1,p)%p for i in range(n)]
  94     ci=[pow(A[0][j],-1,p) for j in range(n)]
  95     E=[];pivot=None
  96     for i in range(n):
  97         E.append([])
  98         for j in range(n):
  99             x=lookup.get(A[i][j]*ri[i]*ci[j]%p)
 100             if x is None:return {'equivalent':False,'reason':'non-root'}
 101             E[i].append(x)
 102             if pivot is None and math.gcd(x,n)==1:pivot=(i,j)
 103     if pivot is None:return {'equivalent':False,'reason':'no-unit-pivot'}
 104     i,j=pivot;v=pow(E[i][j],-1,n)
 105     alpha=[E[k][j]*v%n for k in range(n)];beta=E[i]
 106     if sorted(alpha)!=list(range(n)) or sorted(beta)!=list(range(n)):
 107         return {'equivalent':False,'reason':'not-permutations'}
 108     if any(E[i][j]!=alpha[i]*beta[j]%n for i in range(n) for j in range(n)):
 109         return {'equivalent':False,'reason':'nonseparable'}
 110     left=[A[i][0]%p for i in range(n)]
 111     right=[A[0][j]*pow(A[0][0],-1,p)%p for j in range(n)]
 112     assert alpha[0]==beta[0]==0
 113     assert all(A[i][j]%p==left[i]*a[alpha[i]*beta[j]%n]*right[j]%p
 114                for i in range(n) for j in range(n))
 115     return {'equivalent':True,'alpha':alpha,'beta':beta,'left':left,'right':right}

 118 def fingerprint(A,p,g,k=0,l=1):
 119     n=len(A);a=roots(n,p,g);logs={x:i for i,x in enumerate(a)}
 120     if any(A[j][k]%p==0 or A[j][l]%p==0 for j in range(n)):return None
 121     base=A[0][k]*pow(A[0][l],-1,p)%p;hist=[0]*n
 122     for j in range(n):
 123         e=logs.get(A[j][l]*pow(A[j][k],-1,p)*base%p)
 124         if e is None:return None
 125         hist[e]+=1
 126     d=hist[0]
 127     if not 1<=d<n or n%d:return None
 128     return d if all(hist[e]==(d if e%d==0 else 0) for e in range(n)) else None

 131 def weyl_trial(A,p,g,seed,h=1,calibration=1):
 132     n=len(A);a=roots(n,p,g);rng=random.Random(seed);calls=0
 133     def apply(v):
 134         nonlocal calls
 135         calls+=1
 136         return vec(A,v,p)
 137     calibrated=apply([1]+[0]*(n-1))[0]==calibration
 138     count=[0,0]
 139     for _ in range(h):
 140         v=[rng.randrange(p) for _ in range(n)];y=apply(v)
 141         az=apply(v[-1:]+v[:-1]);ad=apply([x*z%p for x,z in zip(v,a)])
 142         count[0]+=sum(x!=y*z%p for x,y,z in zip(az,y,a))
 143         count[1]+=sum(x!=z for x,z in zip(ad,y[1:]+y[:1]))
 144     assert calls==1+3*h
 145     return {'calibrated':calibrated,'first_equation':count[0]==0,
 146             'second_equation':count[1]==0,'nonzero_first':count[0],
 147             'nonzero_second':count[1],'circuit_calls':calls,
 148             'passed':calibrated and not any(count)}

 229 def numerical_checks():
 230     out={'dtype':'complex128; numerical checks support but do not prove identities'}
 231     rng=np.random.default_rng(91837);records=[];sparse=[];row_sharp=[]
 232     for n in (3,4,5,6,8,9,12,16,24,32,64,128):
 233         x=np.arange(n);F=np.exp(-2j*np.pi*np.outer(x,x)/n)/np.sqrt(n)
 234         Z=np.roll(np.eye(n),1,axis=0);D=np.diag(np.exp(-2j*np.pi*x/n))
 235         K=rng.standard_normal((n,n))+1j*rng.standard_normal((n,n));M=F@K
 236         residual=np.linalg.norm(M@Z-D@M,'fro')**2+np.linalg.norm(M@D-Z.T@M,'fro')**2
 237         # Parseval identity in a freshly generated Weyl basis.
 238         weighted=0.
 239         for a in range(n):
 240             for b in range(n):
 241                 W=(np.exp(-2j*np.pi*a*x/n)[:,None]*np.roll(np.eye(n),b,axis=0))/np.sqrt(n)
 242                 coefficient=np.vdot(W,K)
 243                 weighted+=(abs(1-np.exp(-2j*np.pi*a/n))**2+abs(1-np.exp(-2j*np.pi*b/n))**2)*abs(coefficient)**2
 244         rel=abs(residual-weighted)/max(1,residual)
 245         fd=F@D;dist=np.linalg.norm(fd-F*np.vdot(F,fd)/n,'fro')
 246         rr=np.hypot(np.linalg.norm(fd@Z-D@fd,'fro'),np.linalg.norm(fd@D-Z.T@fd,'fro'))
 247         sharp=abs(rr/dist-2*np.sin(np.pi/n))
 248         comm=F@Z-Z@F;sv=np.linalg.svd(comm,compute_uv=False)
 249         formula=np.array([2*np.cos((2*(r//2)+1)*np.pi/(2*n)) for r in range(n)])
 250         spectrum_error=float(np.max(np.abs(sv-formula)))
 251         C=F.real;S=F.imag;J=np.eye(n)[(-x)%n]
 252         parity_error=max(np.linalg.norm(C@S),np.linalg.norm(C@C-(np.eye(n)+J)/2),np.linalg.norm(S@S-(np.eye(n)-J)/2))
 253         vals,O=np.linalg.eigh(C+np.sqrt(2)*S);q=np.diag(O.T@F@O)
 254         reconstruction=float(np.linalg.norm(O@np.diag(q)@O.T-F,2))
 255         c=2 if n%2==0 else 1
 256         real_count=int(sum(abs(z.imag)<1e-9 for z in q));imag_count=int(sum(abs(z.real)<1e-9 for z in q))
 257         assert rel<1e-10 and sharp<1e-10 and spectrum_error<1e-10 and parity_error<1e-9 and reconstruction<1e-10
 258         assert (real_count,imag_count)==((n+c)//2,(n-c)//2)
 259         records.append({'n':n,'weyl_quadratic_identity_relative_error':rel,'sharp_gap_absolute_error':float(sharp),
 260                         'commutator_spectrum_max_error':spectrum_error,'real_mask_reconstruction_error':reconstruction,
 261                         'real_mask_counts':[real_count,imag_count],
 262                         'stationary_r8_improved_floor':float(formula[8]/2) if n>8 else 0})
 263         for s in sorted(set([1,min(3,n-1),n-1])):
 264             k=1;b=np.zeros(n,dtype=complex);b[:s]=np.exp(2j*np.pi*k*x[:s]/n)*np.sin(np.pi*(x[:s]+1)/(s+1))
 265             ratio=np.linalg.norm(Z@b-np.exp(-2j*np.pi*k/n)*b)/np.linalg.norm(b)
 266             theoretical=2*np.sin(np.pi/(2*(s+1)))
 267             assert abs(ratio-theoretical)<1e-10
 268             sparse.append({'n':n,'s':s,'measured':float(ratio),'sharp_bound':float(theoretical)})
 269             # Sharpness when only one inverse row is prescribed sparse.
 270             u=F.conj().T[0,:];a=s/n;b=np.zeros(n,dtype=complex);b[:s]=u[:s]/a
 271             e=np.eye(n)[0,:];d=e-b@F

 278                               'inverse_row_residual':float(np.linalg.norm(B[0]-b))})
 279     out['weyl_spectrum_real_mask']=records;out['sparse_column_sharpness']=sparse
 280     out['one_sparse_inverse_row_sharpness']=row_sharp
 281     return out

 284 def sqrt_mod(a,p):
 285     """Independent Tonelli-Shanks implementation, to match normalized data."""
 286     a%=p
 287     if a==0:return 0
 288     if pow(a,(p-1)//2,p)!=1:raise ValueError('not square')
 289     q=p-1;s=0
 290     while q%2==0:q//=2;s+=1
 291     z=next(z for z in range(2,p) if pow(z,(p-1)//2,p)==p-1)
 292     c=pow(z,q,p);x=pow(a,(q+1)//2,p);t=pow(a,q,p);m=s
 293     while t!=1:
 294         i=1;v=t*t%p
 295         while v!=1:v=v*v%p;i+=1
 296         b=pow(c,2**(m-i-1),p);x=x*b%p;c=b*b%p;t=t*c%p;m=i
 297     return x

 300 def reconstruct_seeded_source_family(n,near=False):
 301     """Dense algebraic reconstruction from prose/source formula, no source import.
 302     Inverts whole matrices with elimination, independently of uploaded LU solves.
 303     """
 304     p=998244353;g=3;rng=random.Random(21 if near else 19);a=roots(n,p,g);M=eye(n)
 305     for stage in range(2):
 306         S=[[0]*n for _ in range(n)]
 307         for term in range(2):
 308             if near:
 309                 off=-(n-1-(stage+term)%3);diag=[2*n]*n;lo=[0]+[off]*(n-1);up=[off]*(n-1)+[0]
 310             else:
 311                 diag=[9+a[rng.randrange(n)] for _ in range(n)]
 312                 lo=[0]+[1+a[rng.randrange(n)] for _ in range(n-1)]
 313                 up=[1+a[rng.randrange(n)] for _ in range(n-1)]+[0]
 314             B=[[diag[i] if i==j else lo[i] if j==i-1 else up[i] if j==i+1 else 0 for j in range(n)] for i in range(n)]
 315             G=inv(B,p);perm=list(range(n));rng.shuffle(perm)
 316             l=[(2*n if near else 1)*a[rng.randrange(n)]%p for _ in range(n)]
 317             r=[a[rng.randrange(n)] for _ in range(n)]
 318             for i in range(n):
 319                 for j in range(n):S[perm[i]][perm[j]]=(S[perm[i]][perm[j]]+l[i]*G[i][j]*r[j])%p
 320         M=mul(S,M,p)
 321     return M
