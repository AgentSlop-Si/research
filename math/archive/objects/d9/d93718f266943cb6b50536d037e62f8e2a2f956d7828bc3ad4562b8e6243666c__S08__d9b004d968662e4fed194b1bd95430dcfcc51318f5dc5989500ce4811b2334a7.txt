   6 def prime(p):return p>=2 and all(p%d for d in range(2,math.isqrt(p)+1))
   7 def field(n):
   8     k=1
   9     while not prime(k*n+1):k+=1
  10     p=k*n+1;qs=[q for q in range(2,n+1) if n%q==0 and prime(q)]
  11     for a in range(2,p):
  12         w=pow(a,k,p)
  13         if all(pow(w,n//q,p)!=1 for q in qs):return p,w
  14     raise AssertionError('no root')
  15 def rank(M,p):
  16     a=[list(r) for r in M];nr=len(a);nc=len(a[0]) if a else 0;r=0
  17     for j in range(nc):
  18         pivot=next((i for i in range(r,nr) if a[i][j]%p),None)
  19         if pivot is None:continue
  20         a[pivot],a[r]=a[r],a[pivot];iv=pow(a[r][j],-1,p)
  21         for i in range(r+1,nr):
  22             q=a[i][j]*iv%p
  23             if q:a[i]=[(x-q*y)%p for x,y in zip(a[i],a[r])]
  24         r+=1
  25         if r==nr:break
  26     return r

  28 def inverse(M,p):
  29     n=len(M);A=[list(row)+[int(i==j) for j in range(n)] for i,row in enumerate(M)]
  30     for j in range(n):
  31         pivot=next((i for i in range(j,n) if A[i][j]%p),None)
  32         if pivot is None:return None
  33         A[pivot],A[j]=A[j],A[pivot];iv=pow(A[j][j],-1,p);A[j]=[v*iv%p for v in A[j]]
  34         for i in range(n):
  35             if i!=j:
  36                 q=A[i][j];A[i]=[(x-q*y)%p for x,y in zip(A[i],A[j])]
  37     return [row[n:] for row in A]
  38 def matmul(A,B,p):return [[sum(x*y for x,y in zip(row,col))%p for col in zip(*B)] for row in A]
  39 def delta(A,B,p):return [[(x-y)%p for x,y in zip(row,col)] for row,col in zip(A,B)]
  40 def poly(labels,n,w,p):
  41     P=[1]
  42     for j in labels:
  43         z=pow(w,-j,p);Q=[0]*(len(P)+1)
  44         for i,a in enumerate(P):Q[i]=(Q[i]-a*z)%p;Q[i+1]=(Q[i+1]+a)%p
  45         P=Q
  46     return P
  47 def transfer(P,p):
  48     # Generate companion-operator powers, not uploaded long division.
  49     m=len(P)-1;v=[1]+[0]*(m-1);columns=[]
  50     for e in range(2*m):
  51         if e>=m:columns.append(v[:])
  52         lead=v[-1];v=[(-lead*P[0])%p]+[(v[i-1]-lead*P[i])%p for i in range(1,m)]
  53     return [list(row) for row in zip(*columns)]
  54 def periodic(A,n):
  55     A=set(A)
  56     return any({(j+t)%n for j in A}==A for t in range(1,n))
  57 def exact_poly(labels,n):
  58     # Work in Z[y]/(y^N-1), then reduce all coefficients modulo y^(N/2)+1.
  59     P=[[1]+[0]*(n-1)]
  60     for j in labels:
  61         Q=[[0]*n for _ in range(len(P)+1)]
  62         for i,row in enumerate(P):
  63             for e,c in enumerate(row):
  64                 Q[i+1][e]+=c;Q[i][(e+j)%n]-=c
  65         P=Q
  66     return [tuple(row[e]-row[e+n//2] for e in range(n//2)) for row in P]

 148 # Standard independent radix-2 DIT; verify complete transforms and wire-form support.
 149 def fft(seq,p,w):
 150     n=len(seq);bits=n.bit_length()-1
 151     def rev(i):
 152         out=0
 153         for _ in range(bits):out=2*out+(i&1);i>>=1
 154         return out
 155     a=[seq[rev(i)] for i in range(n)];butterflies=0
 156     for level in range(1,bits+1):
 157         length=1<<level;half=length//2
 158         for start in range(0,n,length):
 159             for j in range(half):
 160                 u=a[start+j];v=a[start+j+half]*pow(w,-j*(n//length),p)%p
 161                 a[start+j]=(u+v)%p;a[start+j+half]=(u-v)%p;butterflies+=1
 162     return a,butterflies
