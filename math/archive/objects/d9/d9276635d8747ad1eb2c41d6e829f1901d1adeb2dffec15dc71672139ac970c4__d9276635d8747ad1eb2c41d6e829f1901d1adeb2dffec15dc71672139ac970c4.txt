"""Exact n=8 inverse-semiseparable Fourier feasibility equations.

For F = B P A, put C=A^{-1}. Then B=F C P^{-1}, avoiding
8x8 adjugates. An inverse of an invertible quasiseparable-order-one
stage is again QS-order-one. On a nonzero-corner chart it is in the
separated-generator stage model.

The 30 B membership equations are degree <=4 in the 26 parameters of
C (diagonal 1, normalized L[7]=1 and U[0]=1).
The coefficients live exactly in Z[z]/(z^4+1), z=exp(2 pi i /8).
This is a feasibility formulation, NOT an FFT exclusion certificate.
"""
from __future__ import annotations
from collections import Counter
from dataclasses import dataclass
import hashlib
import itertools
import json
from pathlib import Path
import numpy as np

N=8
SHUFFLE = (7,0,6,3,4,1,2,5)
ONE=(1,0,0,0)
ZERO=(0,0,0,0)

# Coefficient ring Z[zeta8], zeta8^4=-1

def cadd(a,b):return tuple(x+y for x,y in zip(a,b))
def cneg(a):return tuple(-x for x in a)
def cmul(a,b):
    c=[0]*4
    for i,x in enumerate(a):
        if not x:continue
        for j,y in enumerate(b):
            if not y:continue
            if i+j>=4:c[i+j-4]-=x*y
            else:c[i+j]+=x*y
    return tuple(c)

def cpowroot(k):
    k%=8
    a=[0]*4
    a[k if k<4 else k-4]=1 if k<4 else -1
    return tuple(a)

def poly_constant(c=ONE):
    return {():tuple(c)} if c!=ZERO else {}

def poly_var(j):return {(j,):ONE}

def padd(a,b,scale=1):
    out=dict(a)
    for mon,c in b.items():
        v=cadd(out.get(mon,ZERO),c if scale==1 else cneg(c))
        if v==ZERO:out.pop(mon,None)
        else:out[mon]=v
    return out

def pscale(a,c):
    return {mon:v for mon,x in a.items() if (v:=cmul(x,c))!=ZERO}

def pmul(a,b):
    out={}
    for am,ac in a.items():
        for bm,bc in b.items():
            mon=tuple(sorted(am+bm)); v=cadd(out.get(mon,ZERO),cmul(ac,bc))
            if v==ZERO:out.pop(mon,None)
            else:out[mon]=v
    return out

def stage_chart_polys():
    z=[poly_var(k) for k in range(26)]
    # d_i=1; L[1..6], L[7]=1; V[0..6]; U[0]=1,U[1..6]; W[1..7]
    L=[{}]+z[:6]+[poly_constant()]
    V=z[6:13]+[{}]
    U=[poly_constant()]+z[13:19]+[{}]
    W=[{}]+z[19:26]
    C=[]
    for i in range(N):
        row=[]
        for j in range(N):
            if i==j:row.append(poly_constant())
            elif i>j:row.append(pmul(L[i],V[j]))
            else:row.append(pmul(U[i],W[j]))
        C.append(row)
    assert all(C[i][i]==poly_constant() for i in range(N))
    return C

def fourier_image_polys(C):
    # B = F C P^(-1), where (C P^(-1))[:,j]=C[:,SHUFFLE[j]].
    return [[sum_polys([pscale(C[k][SHUFFLE[j]],cpowroot(i*k)) for k in range(N)])
             for j in range(N)] for i in range(N)]

def sum_polys(polys):
    x={}
    for p in polys:x=padd(x,p)
    return x

def eq_entries(B):
    out=[]
    for i in range(2,7):
        for j in range(1,i):
            out.append((f'lower_{i}_{j}',padd(pmul(B[i][j],B[7][0]),pmul(B[i][0],B[7][j]),-1)))
    for i in range(1,6):
        for j in range(i+1,7):
            out.append((f'upper_{i}_{j}',padd(pmul(B[i][j],B[0][7]),pmul(B[i][7],B[0][j]),-1)))
    assert len(out)==30
    return out

def poly_eval(a,values,prime,omega):
    powers=[1,omega%prime,pow(omega,2,prime),pow(omega,3,prime)]
    out=0
    for mon,c in a.items():
        term=sum(c[i]*powers[i] for i in range(4))%prime
        for v in mon:term=term*int(values[v])%prime
        out=(out+term)%prime
    return out

def root8(p,other_root=False):
    assert (p-1)%8==0
    for g in range(2,p):
        w=pow(g,(p-1)//8,p)
        if pow(w,4,p)==p-1:
            return pow(w,3,p) if other_root else w
    raise ValueError('no root')

def char_stage(z,p):
    z=list(map(lambda x:int(x)%p,z))
    assert len(z)==26
    L=[0]+z[:6]+[1]
    V=z[6:13]+[0]
    U=[1]+z[13:19]+[0]
    W=[0]+z[19:26]
    S=np.eye(N,dtype=np.int64)
    for i in range(N):
        for j in range(N):
            if i>j:S[i,j]=L[i]*V[j]%p
            elif i<j:S[i,j]=U[i]*W[j]%p
    return S

def generic_stage(rng,p):
    vals=rng.integers(1,p,size=34,dtype=np.int64)
    L=[0]+[int(v) for v in vals[8:14]]+[1]
    V=[int(v) for v in vals[14:21]]+[0]
    U=[1]+[int(v) for v in vals[21:27]]+[0]
    W=[0]+[int(v) for v in vals[27:34]]
    S=np.diag(vals[:8].astype(np.int64))
    for i in range(N):
        for j in range(N):
            if i>j:S[i,j]=L[i]*V[j]%p
            if i<j:S[i,j]=U[i]*W[j]%p
    return S

def invert_mod(a,p):
    a=np.asarray(a,dtype=np.int64)%p
    n=len(a)
    e=np.concatenate((a.copy(),np.eye(n,dtype=np.int64)),axis=1)
    for j in range(n):
        nonzero=np.flatnonzero(e[j:,j]);
        if not nonzero.size:raise ValueError('singular')
        k=j+int(nonzero[0]); e[[j,k]]=e[[k,j]]
        e[j,:]=e[j,:]*pow(int(e[j,j]),-1,p)%p
        for i in range(n):
            if i!=j and e[i,j]:e[i,:]=(e[i,:]-int(e[i,j])*e[j,:])%p
    assert np.array_equal(a@e[:,n:]%p,np.eye(n,dtype=np.int64))
    return e[:,n:]

def rankmod(a,p):
    a=np.array(a,dtype=np.int64,copy=True)%p
    m,n=a.shape;r=0
    for j in range(n):
        if r==m:break
        ix=np.flatnonzero(a[r:,j])
        if not ix.size:continue
        k=r+int(ix[0]);a[[r,k]]=a[[k,r]]
        a[r]=a[r]*pow(int(a[r,j]),-1,p)%p
        for i in range(r+1,m):
            if a[i,j]:a[i]=(a[i]-int(a[i,j])*a[r])%p
        r+=1
    return r

def identities(S,p):
    E=[]
    for i in range(2,7):
        for j in range(1,i):E.append((int(S[i,j])*int(S[7,0])-int(S[i,0])*int(S[7,j]))%p)
    for i in range(1,6):
        for j in range(i+1,7):E.append((int(S[i,j])*int(S[0,7])-int(S[i,7])*int(S[0,j]))%p)
    return E

def fftmod(p,omega):return np.array([[pow(omega,i*j,p) for j in range(N)] for i in range(N)],dtype=np.int64)

def modular_tests(nchecks=24):
    C=stage_chart_polys(); Bpoly=fourier_image_polys(C);eqs=eq_entries(Bpoly)
    assert len(eqs)==30
    stats={}
    for prime in (12289,65537):
        for variant in (0,1):
            w=root8(prime,variant)
            F=fftmod(prime,w)
            rng=np.random.default_rng(20261009+prime+variant)
            synthetic=0; rejected_random=0; actual_inv=0
            for iteration in range(nchecks):
                z=rng.integers(1,prime,size=26,dtype=np.int64)
                Q=char_stage(z,prime)
                A=invert_mod(Q,prime)
                # Verify Jacobi complementary-cut identities independently.
                for k in range(1,N):
                    assert rankmod(Q[k:,:k],prime)==rankmod(A[k:,:k],prime)
                    assert rankmod(Q[:k,k:],prime)==rankmod(A[:k,k:],prime)
                assert not any(identities(A,prime)), 'generic inverse left chart'
                assert int(A[7,0])!=0 and int(A[0,7])!=0
                actual_inv+=1
                B=generic_stage(rng,prime)
                M=B@A[np.array(SHUFFLE),:]%prime
                got=M@Q[:,np.array(SHUFFLE)]%prime
                assert np.array_equal(got,B)
                assert not any(identities(got,prime))
                # Evaluate F-based polynomials: exact cyclotomic coefficient
                # reduction must equal direct modular B identities.
                got=F@Q[:,np.array(SHUFFLE)]%prime
                ev=[poly_eval(eq,z,prime,w) for _,eq in eqs]
                assert ev==identities(got,prime),(ev,identities(got,prime))
                rejected_random+=bool(any(ev))
                synthetic+=1
            stats[str((prime,variant))]={'primitive_eighth_root':int(w),
               'positive_synthetic_factorizations':synthetic,
               'inverse_corner_generic_checks':actual_inv,
               'random_Fourier_candidates_nonzero':rejected_random,
               'exact_quotient_coefficient_evaluations':synthetic*30}
    return stats

def export(outpath):
    C=stage_chart_polys();B=fourier_image_polys(C);equations=eq_entries(B)
    stats={"number_equations":len(equations),"variables":26,
           "degree_max":max((len(mon) for _,e in equations for mon in e)),
           "degree_min_per_eq":[min(len(mon) for mon in e) for _,e in equations],
           "degree_max_per_eq":[max(len(mon) for mon in e) for _,e in equations],
           "monomial_counts":[len(e) for _,e in equations],
           "unique_monomials":len(set(mon for _,e in equations for mon in e)),
           "total_nonzero_terms":sum(len(e) for _,e in equations)}
    # Encode each monomial's variable indices and integer Z[z]/(z^4+1) coefficient.
    obj={'ring':'Z[z]/(z^4+1), z chosen primitive eighth root','shuffle':SHUFFLE,
      'variables':[f'x{i}' for i in range(26)],'summary':stats,
      'equations':[{'name':name,'terms':[[list(m),list(v)] for m,v in sorted(e.items())]} for name,e in equations]}
    raw=json.dumps(obj,separators=(',',':'),sort_keys=True).encode()
    Path(outpath).write_bytes(raw+b'\n')
    return stats,hashlib.sha256(raw+b'\n').hexdigest()

if __name__=='__main__':
    here=Path(__file__).resolve().parent
    stats,digest=export(here/'quartic_cyclotomic_equations.json')
    tests=modular_tests()
    result={'status':'exact reformulation, NOT feasibility resolution',
     'shuffle':SHUFFLE,'polynomial_stats':stats,'polynomial_sha256':digest,
     'modular_verification':tests}
    (here/'results.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result,indent=2))
