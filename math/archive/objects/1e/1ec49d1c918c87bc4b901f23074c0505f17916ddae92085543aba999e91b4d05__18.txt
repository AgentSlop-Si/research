   1 """Exact integer residue-count checks; no floating point or external code."""
   2 from itertools import combinations
   3 from math import comb
   4 import json
   5 from pathlib import Path
   6 
   7 def coefficients(p, support):
   8     # Each coefficient is an integer combination of 1,zeta,...,zeta^(p-1).
   9     out = [[1]+[0]*(p-1)]
  10     for j in support:
  11         nxt = [[0]*p for _ in range(len(out)+1)]
  12         for degree, c in enumerate(out):
  13             for r, value in enumerate(c):
  14                 nxt[degree][r] += value
  15                 nxt[degree+1][(r+j)%p] -= value
  16         out = nxt
  17     return out
  18 
  19 def is_zero(c):
  20     # For prime p, a degree < p integer polynomial vanishes at zeta_p
  21     # iff all p coefficients are equal (multiple of Phi_p).
  22     return len(set(c)) == 1
  23 
  24 counts=[]
  25 for p in [2,3,5,7,11,13]:
  26     sets=checks=0
  27     for k in range(1,p):
  28         assert all(comb(k,t)%p for t in range(k+1))
  29         for support in combinations(range(p),k):
  30             q=coefficients(p,support)
  31             assert not any(is_zero(c) for c in q)
  32             # The complement is independently among the subsets enumerated.
  33             sets += 1
  34             checks += len(q)
  35     counts.append(dict(prime=p,support_sets=sets,nonzero_coefficients=checks))
  36 result={'arithmetic':'exact Python integers and prime cyclotomic relation',
  37         'scope':'All nonempty proper root subsets at the listed prime lengths; not a runtime lower bound.',
  38         'counts':counts,'total_support_sets':sum(r['support_sets'] for r in counts),
  39         'total_coefficients':sum(r['nonzero_coefficients'] for r in counts)}
  40 Path(__file__).with_name('prime_annihilator_checks.json').write_text(json.dumps(result,indent=2))
  41 print(json.dumps(result,indent=2))
