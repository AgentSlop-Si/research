1: """Independent EXACT certificate for the original 256-point four-stage mixer.
2: 
3: No floating point is used. All numerical linear algebra in the verifier is
4: integer matrix multiplication with a proved conservative no-overflow bound.
5: Proof: for any symmetric positive definite diagonal-dominant integer H,
6: H=T^T(50 L-101 I+50 J)T and invertible integer T, the symmetric integer
7: M=50 L-101 I+50 J is positive definite, hence lambda_2(L)>101/50.
8: 
9: Dependencies: numpy. (The creation script additionally uses scipy.)
10: Run: python verify_exact.py
11: """
12: from pathlib import Path
13: import gzip,json
14: import numpy as np
15: 
16: ROOT=Path(__file__).resolve().parent
17: N=256
18: 
19: def affine(cols):
20:     n=1<<len(cols)
21:     p=np.zeros(n,dtype=np.int64)
22:     for i in range(n):
23:         v=0
24:         for j,c in enumerate(cols):
25:             if (i>>j)&1:v^=c
26:         p[i]=v
27:     assert len(set(map(int,p)))==n
28:     return p
29: 
30: def scan_path_edges(shuffles):
31:     n=len(shuffles[0]);idx=np.arange(n,dtype=np.int64)
32:     out=[]
33:     for j in range(len(shuffles)+1):
34:         out.extend((int(a),int(b)) for a,b in zip(idx[:-1],idx[1:]))
35:         if j<len(shuffles):idx=idx[shuffles[j]]
36:     return out
37: 
38: def laplacian(edges,n):
39:     L=np.zeros((n,n),dtype=np.int64)
40:     for i,j in edges:
41:         L[i,i]+=1; L[j,j]+=1; L[i,j]-=1;L[j,i]-=1
42:     return L
43: 
44: def run():
45:     layout=json.loads((ROOT/'AFFINE_LAYOUT.json').read_text())
46:     assert layout['k']==8 and len(layout['affine_shuffle_columns'])==3
47:     perms=[affine(c) for c in layout['affine_shuffle_columns']]
48:     edges=scan_path_edges(perms)
49:     assert len(edges)==4*(N-1)
50:     L=laplacian(edges,N)
51:     assert np.all(L==L.T) and np.all(L.sum(axis=1)==0)
52:     assert int(np.trace(L))==2*len(edges)
53: 
54:     with gzip.open(ROOT/'INTEGER_CONGRUENCE_CERTIFICATE.json.gz','rt',encoding='utf-8') as f:
55:         cert=json.load(f)
56:     assert (cert['N'],cert['shift_num'],cert['shift_den'])==(256,101,50)
57:     T=np.array(cert['T'],dtype=np.int64)
58:     assert T.shape==(N,N)
59:     assert np.all(T[np.tril_indices(N,-1)]==0), 'T must be upper triangular'
60:     assert np.all(np.diag(T)!=0), 'T must be invertible over Q'
61: 
62:     M=50*L-101*np.eye(N,dtype=np.int64)+50*np.ones((N,N),dtype=np.int64)
63:     # Bound EVERY partial dot product in BOTH multiplications below.
64:     maxrowM=int(np.max(np.sum(np.abs(M),axis=1)))
65:     maxT=int(np.max(np.abs(T)))
66:     maxcolT=int(np.max(np.sum(np.abs(T),axis=0)))
67:     overflow_bound=maxcolT*maxrowM*maxT
68:     assert overflow_bound < 2**62, 'Cannot safely multiply with int64'
69: 
70:     H=T.T@(M@T)
71:     assert np.array_equal(H,H.T)
72:     diagonal=H.diagonal()
73:     margins=diagonal-(np.abs(H).sum(axis=1)-np.abs(diagonal))
74:     assert np.all(diagonal>0) and np.all(margins>0)
75: 
76:     # An independent pure-Python big-integer contraction using only paths.
77:     # Formula follows M=50 Σ_e (e_i-e_j)(e_i-e_j)^T-101 I+50 J.
78:     checks=[(0,0),(1,10),(17,103),(201,221),(255,255)]
79:     checks.extend(((i*37+11)%N,(i*83+57)%N) for i in range(24))
80:     for a,b in checks:
81:         u=[int(z) for z in T[:,a]]
82:         v=[int(z) for z in T[:,b]]
83:         manual=(50*sum((u[i]-u[j])*(v[i]-v[j]) for i,j in edges)
84:                 -101*sum(ua*vb for ua,vb in zip(u,v))
85:                 +50*sum(u)*sum(v))
86:         assert manual==int(H[a,b]),(a,b,manual,int(H[a,b]))
87: 
88:     result={'N':N,'stages':4,'path_edges':len(edges),
89:             'threshold_numerator':101,'threshold_denominator':50,
90:             'exact_T_nonzeros':int(np.count_nonzero(T)),
91:             'exact_T_max_abs':maxT,'proof_int64_intermediate_absolute_bound':overflow_bound,
92:             'min_diagonal':int(diagonal.min()),
93:             'min_strict_diagonal_dominance_margin':int(margins.min()),
94:             'independent_big_integer_entries_checked':len(checks),
95:             'guaranteed_minimum_bisection_boundary':130,
96:             'theorem':'All moving-partition ranks <= available channel capacity for every nontrivial subset; no Fourier factorization claimed.'}
97:     print(json.dumps(result,indent=2))
98:     return result
99: if __name__=='__main__':run()
