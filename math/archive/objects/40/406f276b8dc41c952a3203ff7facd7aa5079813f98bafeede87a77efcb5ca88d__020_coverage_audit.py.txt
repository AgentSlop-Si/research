"""Exact finite audit of a broad A-k generalization: union of few low-rank
baseline families plus sparse residuals cannot cover every random input.
"""
import json
import random
from itertools import combinations
from dense_algebra import coverage_bound

def mask_subspace(basis):
    out={0}
    for v in basis:
        out |= {x^v for x in list(out)}
    return out

def run():
    n=12;r=2;k=2
    offsets=[0,0b101010101010,0b111111000000,0b111000111000]
    bases=[(0b111000111000,0b000111000111),
           (0b101010101010,0b010101010101),
           (0b100010001000,0b001000100010),
           (0b100000000001,0b000000011000)]
    residuals={0}
    for j in range(1,k+1):
        for tup in combinations(range(n),j):
            residuals.add(sum(1<<i for i in tup))
    representable=set()
    for shift,basis in zip(offsets,bases):
        representable |= {shift^v^e for v in mask_subspace(basis) for e in residuals}
    upper=coverage_bound(n,r,k,len(offsets))
    assert len(representable)<=upper['covered_upper']
    excluded=min(set(range(1<<n))-representable)
    assert excluded not in representable
    report={'bound':upper,'actually_representable':len(representable),
            'uncovered':(1<<n)-len(representable),
            'explicit_uncovered_binary':format(excluded,f'0{n}b'),
            'test_kind':'deterministic exhaustive enumeration'}
    print(json.dumps(report,indent=2))
    return report

if __name__=='__main__':run()
