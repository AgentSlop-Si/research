1: """Independent rank-one quasiseparable UNITARY positive control.
2: 
3: A single adjacent-Givens chain is unitary, qsep order one, and O(N) gates.
4: Its mixing entropy is too small to be a complete Fourier transform. This
5: checks a known Ailon-model argument, not a new lower bound on nonunitary FFTs.
6: """
7: import math,json
8: from exact_scan_macro import P,rank_mod
9: 
10: 
11: def entropy(matrix):
12:     return -sum((v*v)*math.log2(v*v) for row in matrix for v in row if abs(v)>1e-15)
13: 
14: 
15: def run(ns=(4,8,16,32)):
16:     report=[]
17:     p=P
18:     inv5=pow(5,-1,p)
19:     cm=3*inv5%p;sm=4*inv5%p
20:     for n in ns:
21:         exact=[[int(i==j) for j in range(n)] for i in range(n)]
22:         approx=[[float(i==j) for j in range(n)] for i in range(n)]
23:         entropies=[entropy(approx)]
24:         for idx in range(n-1):
25:             a,b=exact[idx][:],exact[idx+1][:]
26:             exact[idx]=[(cm*x+sm*y)%p for x,y in zip(a,b)]
27:             exact[idx+1]=[(-sm*x+cm*y)%p for x,y in zip(a,b)]
28:             a,b=approx[idx][:],approx[idx+1][:]
29:             approx[idx]=[(3*x+4*y)/5 for x,y in zip(a,b)]
30:             approx[idx+1]=[(-4*x+3*y)/5 for x,y in zip(a,b)]
31:             entropies.append(entropy(approx))
32:         max_cutrank=0
33:         for s in range(1,n):
34:             ru=rank_mod([row[s:] for row in exact[:s]])
35:             rd=rank_mod([row[:s] for row in exact[s:]])
36:             max_cutrank=max(max_cutrank,ru,rd)
37:         # Exact orthogonality over finite field: because (3/5)^2+(4/5)^2=1.
38:         for i in range(n):
39:             for j in range(n):
40:                 assert sum(exact[k][i]*exact[k][j] for k in range(n))%p==int(i==j)
41:         assert max_cutrank==1
42:         differences=[entropies[i+1]-entropies[i] for i in range(n-1)]
43:         assert max(differences)<=2+1e-10
44:         report.append({'n':n,'unitary_givens_gates':n-1,
45:                        'maximum_quasiseparable_offdiagonal_rank':max_cutrank,
46:                        'final_entropy_bits':entropies[-1],
47:                        'target_fourier_entropy_bits':n*math.log2(n),
48:                        'maximum_entropy_increase_per_gate':max(differences)})
49:     return report
50: 
51: if __name__=='__main__':print(json.dumps(run(),indent=2))
