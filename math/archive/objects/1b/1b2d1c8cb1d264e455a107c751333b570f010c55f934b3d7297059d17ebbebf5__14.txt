0055 def seq(nodes,w,n,p):
0056     b=len(nodes);q=[1]
0057     for z in nodes:q=conv(q,[1,-z%p],p)
0058     inv=[1]+[0]*(b-1)
0059     for i in range(1,b):inv[i]=-sum(q[j]*inv[i-j] for j in range(1,i+1))%p
0060     power=[1]*b;out=[]
0061     for k in range(min(b,n)):
0062         out.append(sum(a*t for a,t in zip(w,power))%p)
0063         power=[a*t%p for a,t in zip(nodes,power)]
0064     while len(out)<n:
0065         cursor=len(out);take=min(b,n-cursor)
0066         rhs=[-v%p for v in conv(out[-b:],q,p)[b:b+take]]
0067         out+=conv(rhs,inv,p)[:take]
0068     direct=[sum(w[j]*pow(nodes[j],k,p) for j in range(b))%p for k in range(n)]

0106 def cert(n,lens):
0107     mm=masks(lens)
0108     vals=[sum((F(mm[d][i],phi(n//math.gcd(n,d))) for d in mm),F(0)) for i in range(n)]

0110     old=sum((F(1,phi(n//math.gcd(n,d))) for d in mm),F(0))
0111     return max(vals,default=F(0)),grouped,old
