61: def error_ratio(out,reference,x):
62:     scale=max(float(np.max(np.abs(x.real))),float(np.max(np.abs(x.imag))))
63:     if scale==0:return float(np.max(np.abs(out)))
64:     scaled=x.real/scale+1j*(x.imag/scale)
65:     diff=(out.real-reference.real)/scale+1j*((out.imag-reference.imag)/scale)
66:     return float(np.max(np.abs(diff))/np.linalg.norm(scaled))

69: def usefulness(out,reference,x):
70:     scale=max(float(np.max(np.abs(x.real))),float(np.max(np.abs(x.imag))))
71:     if scale==0:return {'relative_l2_error':0.,'all_zero_output':True,'retained_energy_fraction':1.}
72:     scaled=x.real/scale+1j*(x.imag/scale)
73:     diff=(out.real-reference.real)/scale+1j*((out.imag-reference.imag)/scale)
74:     out_scaled=out.real/scale+1j*(out.imag/scale)
75:     denominator=float(np.vdot(scaled,scaled).real)
76:     return {'relative_l2_error':float(np.linalg.norm(diff)/np.sqrt(denominator)),
77:             'all_zero_output':bool(np.all(out==0)),
78:             'retained_energy_fraction':float(np.vdot(out_scaled,out_scaled).real/denominator)}
