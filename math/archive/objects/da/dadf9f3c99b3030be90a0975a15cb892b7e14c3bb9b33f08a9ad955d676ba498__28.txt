0008 class TestAdapter(unittest.TestCase):
0009     def test_sparse_delta_with_existing_engine(self):
0010         rng=np.random.default_rng(992)
0011         engine=HybridAdaptiveFFT(mode='plain')
0012         for n in [8,64,1024]:
0013             base=rng.standard_normal(n)
0014             f=engine.fft(base)
0015             updates=[(1,0.2),(n-1,0.4)]
0016             out,_=engine.fft_from_delta(f,updates)
0017             altered=base.copy();altered[1]+=0.2;altered[-1]+=0.4
0018             self.assertTrue(np.allclose(out,np.fft.fft(altered),atol=3e-10))
0019             self.assertEqual(engine.last_strategy,'metadata_sparse_factorized')
0020             compact=engine.rfft(base)
0021             out2,_=engine.rfft_from_delta(compact,n,updates)
0022             self.assertTrue(np.allclose(out2,np.fft.rfft(altered),atol=3e-10))
0023 
0024 if __name__=='__main__':unittest.main()

0026 class TestLazyAdapter(unittest.TestCase):
0027     def test_lazy_adapter_surfaces(self):
0028         rng=np.random.default_rng(993);n=128
0029         x=rng.standard_normal(n)+1j*rng.standard_normal(n)
0030         engine=HybridAdaptiveFFT(mode='plain');base=np.fft.fft(x)
0031         s=engine.lazy_spectrum(base);s.add_edit(3,1.5);x[3]+=1.5
0032         self.assertTrue(np.allclose(s.bins([0,4,77]),np.fft.fft(x)[[0,4,77]]))
0033         b=engine.buffered_spectrum(base,flush_threshold=2);b.add_edit(1,1);b.add_edit(2,2)
0034         self.assertEqual(b.flush_count,1)
0035         v=engine.prepared_bin_view(base,[1,2],[0,5]);v.add_edit(1,1)
0036         xx=x.copy(); # x already differs at 3; view is from original base, so use independent truth
0037         original=np.fft.ifft(base); original[1]+=1
0038         self.assertTrue(np.allclose(v.values(),np.fft.fft(original)[[0,5]],atol=2e-10))
