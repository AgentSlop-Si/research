   1 #!/usr/bin/env python3
   2 """Fresh finite controls for the aligned-resolvent corollary; no source imports."""
   3 import json, random
   4 from pathlib import Path
   5 
   6 P = 12289
   7 def eye(n): return [[int(i==j) for j in range(n)] for i in range(n)]
   8 def mm(a,b):
   9     return [[sum(x*y for x,y in zip(row,col))%P for col in zip(*b)] for row in a]
  10 def add(a,b): return [[(x+y)%P for x,y in zip(r,s)] for r,s in zip(a,b)]
  11 def rank(a):
  12     a=[r[:] for r in a]; row=0
  13     if not a: return 0
  14     for col in range(len(a[0])):
  15         pivot=next((i for i in range(row,len(a)) if a[i][col]%P),None)
  16         if pivot is None: continue
  17         a[row],a[pivot]=a[pivot],a[row]
  18         f=pow(a[row][col],-1,P);a[row]=[x*f%P for x in a[row]]
  19         for i in range(len(a)):
  20             if i!=row:
  21                 f=a[i][col];a[i]=[(x-f*y)%P for x,y in zip(a[i],a[row])]
  22         row+=1
  23         if row==len(a):break
  24     return row
  25 def inverse(a):
  26     n=len(a); b=[r[:]+e for r,e in zip(a,eye(n))]
  27     for col in range(n):
  28         pivot=next((i for i in range(col,n) if b[i][col]%P),None)
  29         if pivot is None: raise ValueError('singular')
  30         b[col],b[pivot]=b[pivot],b[col];f=pow(b[col][col],-1,P)
  31         b[col]=[x*f%P for x in b[col]]
  32         for i in range(n):
  33             if i!=col:
  34                 f=b[i][col];b[i]=[(x-f*y)%P for x,y in zip(b[i],b[col])]
  35     ans=[r[n:] for r in b];assert mm(a,ans)==eye(n);return ans
  36 def cuts(a,k): return ([r[:k] for r in a[k:]],[r[k:] for r in a[:k]])
  37 def mask(a,rng):
  38     l=[rng.randrange(1,P) for _ in a];r=[rng.randrange(1,P) for _ in a]
  39     return [[l[i]*a[i][j]*r[j]%P for j in range(len(a))] for i in range(len(a))]
  40 
  41 def run():
  42     rng=random.Random(1520);checks=[]
  43     # Singular principal blocks cannot be silently excluded.
  44     swap=[[0,1],[1,0]]; assert inverse(swap)==swap
  45     assert rank(cuts(swap,1)[0])==1
  46     for n in [4,8,16,32]:
  47         ts=[];xs=[]
  48         for case in range(4):
  49             while True:
  50                 t=[[0]*n for _ in range(n)]
  51                 for i in range(n):
  52                     t[i][i]=rng.randrange(P)
  53                     if i+1<n:
  54                         t[i][i+1]=rng.randrange(P);t[i+1][i]=rng.randrange(P)
  55                 try:x=inverse(t);break
  56                 except ValueError:pass
  57             ts.append(t);xs.append(mask(x,rng))
  58             for k in range(1,n):
  59                 for tx,xx in zip(cuts(t,k),cuts(x,k)):
  60                     assert rank(tx)==rank(xx)<=1
  61         assert mm(ts[0],ts[1])!=mm(ts[1],ts[0])
  62         s=[[0]*n for _ in range(n)];prod=eye(n)
  63         for count,x in enumerate(xs,1):
  64             s=add(s,x);prod=mm(prod,x)
  65             for k in range(1,n):
  66                 assert all(rank(a)<=count for a in cuts(s,k)+cuts(prod,k))
  67         omega=next(pow(g,(P-1)//n,P) for g in range(2,P)
  68                    if pow(pow(g,(P-1)//n,P),n//2,P)!=1)
  69         assert pow(omega,n,P)==1
  70         f=[[pow(omega,i*j,P) for j in range(n)] for i in range(n)]
  71         franks=[]
  72         for k in range(1,n):
  73             r=rank(cuts(f,k)[0]);assert r==min(k,n-k);franks.append(r)
  74         checks.append({'n':n,'tridiagonal_cases':4,'noncommuting':True,
  75                        'inverse_rank_identity_all_cuts':True,'sum_product_budgets_all_cuts':True,
  76                        'fourier_lower_cut_ranks':franks})
  77     result={'field':P,'singular_principal_control_passed':True,'cases':checks,
  78             'all_passed':True,'scope':'Exact finite-field controls. Complex all-N assertions follow from supplied proof, not extrapolation.'}
  79     out=Path(__file__).with_name('aligned_resolvent_results.json');out.write_text(json.dumps(result,indent=2)+'\n')
  80     print(json.dumps(result,indent=2))
  81 if __name__=='__main__':run()
