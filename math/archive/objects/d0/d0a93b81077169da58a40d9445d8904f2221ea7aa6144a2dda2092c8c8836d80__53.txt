   1 import unittest
   2 import random
   3 from wide_scan import *
   4 
   5 class RankAndScanTests(unittest.TestCase):
   6     def test_transform_field_roots(self):
   7         for n in (2,4,8,16,32,64):
   8             m=dft(n)
   9             self.assertEqual(rank(m),n)
  10             for i in range(n):
  11                 for j in range(n):
  12                     self.assertEqual(m[i][j],m[j][i])
  13     def test_vandermonde_balanced(self):
  14         for n in (4,8,16,32,64,128):
  15             self.assertEqual(offdiag_ranks(dft(n),n//2),(n//2,n//2))
  16     def test_scans_have_one_direction(self):
  17         for n in (4,8,16):
  18             a=[(3*i+2)%P for i in range(n)]
  19             b=[(5*i+1)%P for i in range(n)]
  20             for c in range(1,n):
  21                 self.assertEqual(offdiag_ranks(forward(a,b),c)[0],0)
  22                 self.assertLessEqual(offdiag_ranks(forward(a,b),c)[1],1)
  23                 self.assertEqual(offdiag_ranks(backward(a,b),c)[1],0)
  24                 self.assertLessEqual(offdiag_ranks(backward(a,b),c)[0],1)
  25     def test_matrix_product(self):
  26         rng=random.Random(172)
  27         for n in (2,4,7):
  28             x=[[rng.randrange(P) for _ in range(n)] for _ in range(n)]
  29             y=[[rng.randrange(P) for _ in range(n)] for _ in range(n)]
  30             prod=matmul(x,y)
  31             for i in range(n):
  32                 for j in range(n):
  33                     self.assertEqual(prod[i][j],sum(x[i][k]*y[k][j] for k in range(n))%P)
  34     def test_general_scan_bound(self):
  35         out=random_scan_compositions(trials=12)
  36         self.assertEqual(out['max_violation'],0)
  37     def test_parity_cut_rank(self):
  38         for n in (8,16,32,64,128):
  39             f=dft(n); o=parity_order(n)
  40             self.assertEqual(offdiag_ranks(permute(f,o,o),n//2),(n//4,n//4))
  41     def test_bit_reverse_cut_rank(self):
  42         for n in (8,16,32,64):
  43             f=dft(n); o=bit_reverse_order(n)
  44             self.assertEqual(offdiag_ranks(permute(f,o,o),n//2),(n//4,n//4))
  45     def test_8_exhaustive(self):
  46         report=exact_min_balanced_submatrix_rank(8)
  47         self.assertEqual(report['submatrix_count'],4900)
  48         self.assertEqual(report['minimum_rank'],2)
  49     def test_permutation_invalid(self):
  50         with self.assertRaises(ValueError):
  51             permute(dft(8),[0]*8,list(range(8)))
  52         with self.assertRaises(ValueError):
  53             crossing_count([0]*8,4)
  54     def test_primitive_backward_example(self):
  55         n=6
  56         F=forward([0]+[1]*(n-1),[1]*n)
  57         B=backward([1]*n,[1]*n)
  58         x=[0,2,1,3,0,7]
  59         self.assertEqual([sum(a*b for a,b in zip(row,x))%P for row in F],[0,2,3,6,6,13])
  60         self.assertEqual([sum(a*b for a,b in zip(row,x))%P for row in B],[13,13,11,10,7,7])
