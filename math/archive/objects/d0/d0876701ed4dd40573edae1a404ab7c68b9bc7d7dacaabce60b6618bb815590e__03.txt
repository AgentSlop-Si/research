   7	from check_four_character_identities import cover_and_splice,lift,build_walsh_route,LABELS,P
   8	I=tuple(range(4))
   9	def compose(a,b): return tuple(a[x] for x in b)
  10	def inverse(a): return tuple(a.index(x) for x in I)
  11	def chi(c,e): return (-1)**((c&e).bit_count()%2)
  12	def closure(generators):
  13	    group={I}; todo=[I]
  14	    while todo:
  15	        x=todo.pop()
  16	        for g in generators:
  17	            h=compose(g,x)
  18	            if h not in group: group.add(h); todo.append(h)
  19	    return group
  20	def audit(routes,a,b):
  21	    p,phis=routes[0]; n=a*b
  22	    gauges=[None]*n; gauges[0]=I; vertex=0; tree=[]
  23	    for _ in range(n-1):
  24	        tree.append(vertex)
  25	        gauges[p[vertex]]=compose(phis[vertex],gauges[vertex]);vertex=p[vertex]
  26	    assert all(g is not None for g in gauges)
  27	    generators=[]
  28	    for j,(p,phis) in enumerate(routes):
  29	        for v in range(n):
  30	            g=compose(inverse(gauges[p[v]]),compose(phis[v],gauges[v]))
  31	            generators.append((j,v,g))
  32	    assert all(g==I for j,v,g in generators if j==0 and v in set(tree))
  33	    group=closure(set(g for j,v,g in generators))
  34	    witness=next(((j,v,g,k,w,h,compose(g,h),compose(h,g))
  35	        for j,v,g in generators for k,w,h in generators if compose(g,h)!=compose(h,g)),None)
  36	    # All checked instances share this fixed Walsh character in the tree gauge.
  37	    fixed=[]
  38	    for c in range(1,4):
  39	        if all(all(chi(c,g[e])==chi(c,g[0])*chi(c,e) for e in I) for g in group): fixed.append(c)
  40	    assert len(group)==8 and witness is not None and fixed==[1]
  41	    rotations=[g for g in group if compose(compose(g,g),compose(g,g))==I and compose(g,g)!=I]
  42	    assert len(rotations)==2
  43	    r=rotations[0]
  44	    reflection=next(g for g in group if g!=I and compose(g,g)==I and compose(compose(g,r),g)==inverse(r))
  45	    assert closure([r,reflection])==group
  46	    return {'N':n,'joint_group_order':len(group),'group_elements':sorted(group),
  47	            'noncommuting_witness':{'edge_a':list(witness[:2]),'holonomy_a':witness[2],
  48	                'edge_b':list(witness[3:5]),'holonomy_b':witness[5],'ab':witness[6],'ba':witness[7]},
  49	            'dihedral_generators':[r,reflection],'fixed_nonzero_Walsh_characters':fixed,
  50	            'common_scalar_four_channel_gauge_exists':False,
  51	            'common_permutation_representation_blocks':[1,1,2]}

  53	def checked_lift(ts,ps,a,b,label):
  54	    tl,pl,rec=lift(ts,ps,a,b,label)
  55	    assert rec['premerge_cycle_lengths']==[a*b]*4
  56	    assert rec['merge_tails']==[0,b,2*a*b,2*a*b+b]
  57	    _,_,phis,_=build_walsh_route(ps,pl,a,b)
  58	    assert all(all(chi(1,g[e])==chi(1,g[0])*chi(1,e) for e in I) for g in phis)
  59	    high_x_swap=(2,3,0,1)
  60	    assert all(compose(g,high_x_swap)==compose(high_x_swap,g) for g in phis)
  61	    assert all(pl[(i+2*a*b)%(4*a*b)]==(pl[i]+2*a*b)%(4*a*b) for i in range(4*a*b))
  62	    # Sum the exact low-coordinate carries over the base grid.
  63	    axis,c,sign=label
  64	    if axis=='H':
  65	        total=sum((x+sign*(2*y+c))//a for x in range(a) for y in range(b))
  66	        assert total==sign*b*(b-1+c)
  67	    else:
  68	        total=sum((y+sign*(2*x+c))//b for x in range(a) for y in range(b))
  69	        assert total==sign*a*(a-1+c)
  70	    assert total%2==0
  71	    # The complete large-cycle monodromy starting above base vertex zero is
  72	    # exactly the same 4-cycle on literal high-bit sheet labels for every order.
  73	    monodromy=I; vertex=0
  74	    for _ in range(a*b):
  75	        monodromy=compose(phis[vertex],monodromy); vertex=ps[vertex]
  76	    assert vertex==0 and monodromy==(1,2,3,0)
  77	    return tl,pl,phis

  79	def mask_controls():
  80	    count=0
  81	    for a in itertools.product(range(2),repeat=4):
  82	        m=[[sum(chi(c,e)*a[e]*chi(d,e) for e in I) for d in I] for c in I]
  83	        fine=all(m[c][d]==0 for c in I for d in I if [0,1,2,2][c]!=[0,1,2,2][d])
  84	        coarse=all(m[c][d]==0 for c in I for d in I if (c//2)!=(d//2))
  85	        assert fine==(len(set(a))==1)
  86	        assert coarse==(a[0]==a[2] and a[1]==a[3])
  87	        count+=1
  88	    return {'binary_masks_exhaustively_checked':count,'fine_block_preserving':'fiber-constant masks only',
  89	            'coarse_2_plus_2_preserving':'a(0)=a(2) and a(1)=a(3)',
  90	            'nonsingular_two_value_example':[1,2,1,2],'generic_mixing_example':[1,2,3,4]}

  92	def bank_and_target_controls(a,b):
  93	    n=a*b; root=pow(3,(P-1)//(4*n),P); blockroot=pow(root,2*a,P)
  94	    def entry(i,j):
  95	        x,y=divmod(i,2*b);xx,z=divmod(j,2*b)
  96	        if x!=xx:return 0
  97	        return pow(blockroot,y*z,P) if x%a<a//2 else int(y==z)
  98	    for i in range(4*n):
  99	        for j in range(4*n):
 100	            assert entry(i,j)==entry((i+2*n)%(4*n),(j+2*n)%(4*n))
 101	    assert pow(root,2*n*(2*n+1),P)==P-1 and pow(root,0,P)==1
 102	    return {'N':n,'bank_simultaneous_shift_entries_checked':16*n*n,
 103	            'Fourier_target_shift_counterexample':[1,P-1]}
