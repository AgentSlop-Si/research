14: class Probe:
15:     def __init__(self,p):
16:         self.p,self.zero,self.one = p,0,1
17:         self.calls=Counter()
18:     def add(self,a,b):
19:         self.calls['add']+=1
20:         return (a+b)%self.p
21:     def sub(self,a,b):
22:         self.calls['sub']+=1
23:         return (a-b)%self.p
24:     def mul(self,a,b):
25:         self.calls['mul']+=1
26:         return a*b%self.p
27:     def inv(self,a):
28:         self.calls['inverse']+=1
29:         return pow(a,-1,self.p)

32: def direct(x,q,p):
33:     n=len(x)
34:     return [sum(value*pow(q,j*k,p) for j,value in enumerate(x))%p for k in range(n)]
35: 
36: 
37: def apply_filter(x,q,descriptor,pool,p):
38:     a,b,h,index=descriptor;n=len(x)
39:     return [(x[j]-pool[index]*pow(q,b*j%n,p)*x[(a*j+h)%n])%p for j in range(n)]
40: 
41: 
42: def matrix_of_filter(n,q,descriptor,pool,p):
43:     columns=[apply_filter([int(i==j) for i in range(n)],q,descriptor,pool,p) for j in range(n)]
44:     return [list(row) for row in zip(*columns)]
45: 
46: 
47: def mm(a,b,p):
48:     return [[sum(x*y for x,y in zip(row,col))%p for col in zip(*b)] for row in a]
49: 
50: 
51: def inverse(a,p):
52:     n=len(a)
53:     a=[row[:]+[int(i==j) for j in range(n)] for i,row in enumerate(a)]
54:     for j in range(n):
55:         pivot=next((i for i in range(j,n) if a[i][j]),None)
56:         if pivot is None:raise ValueError('singular')
57:         a[j],a[pivot]=a[pivot],a[j]
58:         factor=pow(a[j][j],-1,p)
59:         a[j]=[x*factor%p for x in a[j]]
60:         for i in range(n):
61:             if i!=j:
62:                 factor=a[i][j]
63:                 a[i]=[(x-factor*y)%p for x,y in zip(a[i],a[j])]
64:     return [row[n:] for row in a]

67: def generate(n,q,filters,pool,residual,p):
68:     matrix=[[int(i==j) for j in range(n)] for i in range(n)]
69:     for descriptor in filters:
70:         matrix=mm(matrix_of_filter(n,q,descriptor,pool,p),matrix,p)
71:     return [sum(a*b for a,b in zip(row,residual))%p for row in inverse(matrix,p)]

74: def checked(x,q,filters,pool,budget,p,fallback=False):
75:     probe=Probe(p)
76:     fn=composed_fft_or_fallback if fallback else try_composed_fourier
77:     y,meta=fn(probe,x,q,filters,pool,budget)
78:     assert meta['field_operations']==dict(probe.calls)
79:     assert sum((Counter(s) for s in meta['stages'].values()),Counter())==probe.calls
80:     if y is not None:
81:         assert y==direct(x,q,p)
82:         if meta['ordinary_fft_calls']==0:
83:             n=len(x);d=len(filters);r=len(pool);s=meta['actual_defects']
84:             bill=(1+r+6*d+2*s)*n+r*((n.bit_length()-1)+2)+d
85:             assert sum(probe.calls.values())==bill==meta['expected_total']
86:     return y,meta

89: def exhaustive():
90:     p,n,q=5,4,2
91:     filters=[(5,1,1,0),(9,3,1,0)];pool=[2]
92:     result={'cases':0,'accepted':0,'declined':0}
93:     for values in product(range(p),repeat=n):
94:         residual=list(values)
95:         for descriptor in filters:
96:             residual=apply_filter(residual,q,descriptor,pool,p)
97:         support=sum(bool(v) for v in residual)
98:         for budget in (0,1):
99:             y,meta=checked(list(values),q,filters,pool,budget,p)
100:             assert (y is not None)==(support<=budget)
101:             result['cases']+=1
102:             result['accepted' if y is not None else 'declined']+=1
103:     assert result['accepted']==18
104:     return result
