"""Independent exact test of central-bit Fourier rank obstruction.

This is a necessary condition for a product of order-one quasiseparable
stages interleaved with bit reversals. No FFT speedup is claimed.
Uses only Python's standard library.
"""
from __future__ import annotations
import math
import random


def bitrev(n):
    assert n > 0 and n & (n-1) == 0
    w=n.bit_length()-1
    return [int(f'{j:0{w}b}'[::-1],2) for j in range(n)]


def partition(n):
    k=n.bit_length()-1
    assert n>=4 and n&(n-1)==0
    if k%2:
        m=k//2
        A=[j for j in range(n) if (j>>m)&1==0]
        B=[j for j in range(n) if (j>>m)&1==1]
        return A,B,2*(1<<m)-1,('odd',1<<m)
    m=k//2
    A=[j for j in range(n) if ((j>>m)&1)==((j>>(m-1))&1)]
    B=[j for j in range(n) if ((j>>m)&1)!=((j>>(m-1))&1)]
    return A,B,1<<m,('even',1<<(m-1))


def primitive_root_of_unity(n,p):
    assert (p-1)%n==0
    for a in range(2,p):
        r=pow(a,(p-1)//n,p)
        if pow(r,n,p)==1 and pow(r,n//2,p)!=1:
            return r
    raise ValueError('no root')


def det_mod(a,p):
    n=len(a); A=[list(row) for row in a]; d=1
    for j in range(n):
        pivot=next((i for i in range(j,n) if A[i][j]%p),None)
        if pivot is None:return 0
        if pivot!=j:A[j],A[pivot]=A[pivot],A[j];d=-d
        d=d*A[j][j]%p
        inv=pow(A[j][j],-1,p)
        Aj=A[j]
        for i in range(j+1,n):
            f=A[i][j]*inv%p
            if not f:continue
            Ai=A[i]
            for h in range(j+1,n):Ai[h]=(Ai[h]-f*Aj[h])%p
            Ai[j]=0
    return d%p


def modular_fft_minor(n,p):
    A,B,cap,meta=partition(n)
    w=primitive_root_of_unity(n,p)
    powers=[pow(w,j,p) for j in range(n)]
    sub=[[powers[(r*c)%n] for c in B] for r in A]
    return det_mod(sub,p)


def stage(n,p,seed):
    rng=random.Random(seed)
    aa=[rng.randrange(p) for _ in range(n)]
    bb=[rng.randrange(p) for _ in range(n)]
    cc=[rng.randrange(p) for _ in range(n)]
    ee=[rng.randrange(p) for _ in range(n)]
    dd=[rng.randrange(p) for _ in range(n)]
    return [[(aa[i]*bb[j] if i>j else cc[i]*ee[j] if i<j else dd[i])%p for j in range(n)] for i in range(n)]


def matmul(A,B,p):
    T=list(zip(*B))
    return [[sum(x*y for x,y in zip(a,b))%p for b in T] for a in A]


def rank_mod(M,p):
    a=[list(row) for row in M]
    if not a:return 0
    nr,nc=len(a),len(a[0]);r=0
    for j in range(nc):
        i=next((i for i in range(r,nr) if a[i][j]),None)
        if i is None:continue
        a[i],a[r]=a[r],a[i]
        inv=pow(a[r][j],-1,p)
        for z in range(r+1,nr):
            if a[z][j]:
                coef=a[z][j]*inv%p
                a[z]=[(t-coef*s)%p for t,s in zip(a[z],a[r])]
        r+=1
        if r==nr:break
    return r


def product_rank(n,p,t):
    A,B,cap,_=partition(n)
    perm=bitrev(n)
    X=stage(n,p,100+n)
    for h in range(1,t):
        X=matmul(stage(n,p,100+n+h), [X[j] for j in perm], p)
    return rank_mod([[X[i][j] for j in B] for i in A],p)


def run():
    for k in range(2,10):
        n=1<<k
        A,B,cap,meta=partition(n)
        assert set(A).isdisjoint(B) and len(A)==len(B)==n//2
        perm=bitrev(n)
        assert {perm[j] for j in A}==set(A),n
        blocks_a=1+sum(A[j]>A[j-1]+1 for j in range(1,len(A)))
        blocks_b=1+sum(B[j]>B[j-1]+1 for j in range(1,len(B)))
        print(f'n={n} partition={meta} half={n//2} per_stage_bound={cap} depth_lower={math.ceil((n//2)/cap)} blocks_A={blocks_a} blocks_B={blocks_b}',flush=True)
        if n<=256:
            for p in (12289,65537):
                residue=modular_fft_minor(n,p)
                assert residue, (n,p)
                print(f'  det(F[A,B]) mod {p} = {residue}',flush=True)
        if n<=64:
            for t in (2,3,4):
                rr=product_rank(n,12289,t)
                assert rr<=t*cap
                print(f'  random {t}-stage rank={rr} <= {t*cap}',flush=True)

if __name__=='__main__':run()
