import unittest
from three_stage_control import witness

class NaiveThreeStageCounterexampleTest(unittest.TestCase):
    def test_four_full_rank_subblocks_despite_quasi_rank_one(self):
        for n in (8,16,32,64):
            proof=witness(n)
            self.assertEqual(proof['max_upper_cut_rank'],1)
            self.assertEqual(proof['max_lower_cut_rank'],1)
            self.assertTrue(all(x==n//2 for x in proof['four_arbitrary_half_ranks'].values()))
