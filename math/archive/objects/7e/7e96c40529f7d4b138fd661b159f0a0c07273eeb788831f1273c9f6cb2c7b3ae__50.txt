  32 class TestV21(unittest.TestCase):
  33     @classmethod
  34     def setUpClass(cls):
  35         cls.records={n:finite_field_case(n) for n in CASES}
  36 
  37     def test_all_fourier_blocks_invertible(self):
  38         for n,rec in self.records.items():
  39             self.assertEqual(rec['block_ranks'],[n-1]*n)
  40 
  41     def test_packet_offdiag_basis(self):
  42         for n,rec in self.records.items():
  43             self.assertEqual(rec['packet_offdiag_rank'],n*(n-1))
  44 
  45     def test_unique_top_degree_certificate(self):
  46         for n in CASES:
  47             for x in range(n):
  48                 c=leading_term_certificate(n,x)
  49                 self.assertEqual(c['top_degree'],(n-1)**2)
  50                 self.assertEqual(len(set(c['unique_top_columns'])),n-1)
  51                 self.assertNotIn(0,c['unique_top_columns'])
  52 
  53     def test_character_packets_have_degree_two_support(self):
  54         for n,rec in self.records.items():
  55             p,w,a=rec['field_prime'],rec['omega'],rec['alpha']
  56             for h in (1,2,3):
  57                 if h>=n:continue
  58                 for s in (0,1,2):
  59                     M=character_packet_matrix(n,p,w,a,h,s)
  60                     r,c,_=packet_support_degrees(M)
  61                     self.assertLessEqual(r,2)
  62                     self.assertLessEqual(c,2)
  63 
  64     def test_constant_diagonal_span_with_identity(self):
  65         for n,rec in self.records.items():
  66             p,w,a=rec['field_prime'],rec['omega'],rec['alpha']
  67             cols,_=packet_basis_columns(n,p,w,a)
  68             I=flatten(mat_identity(n))
  69             # deterministic arbitrary constant-diagonal target
  70             target=[]
  71             for i in range(n):
  72                 for j in range(n):
  73                     target.append(7%p if i==j else (3*i+5*j+11)%p)
  74             sol=solve_linear_columns(cols+[I],target,p)
  75             self.assertIsNotNone(sol)
  76 
  77     def test_exact_green_line_fourier_representation(self):
  78         for n,rec in self.records.items():
  79             self.assertTrue(rec['green_line_identity'])
  80 
  81     def test_common_weyl_line(self):
  82         for n,rec in self.records.items():
  83             p,w=rec['field_prime'],rec['omega']
  84             U=common_weyl_generator(n,p,w)
  85             inv2=pow(2,-1,n)
  86             for h in range(1,n):
  87                 left=weyl_matrix(n,p,w,h,-h)
  88                 right=matrix_power(U,h,p)
  89                 scalar=pow(w,(h*(h-1)*inv2)%n,p)
  90                 right=[[scalar*v%p for v in row] for row in right]
  91                 self.assertTrue(matrices_equal(left,right,p))
  92 
  93     def test_bluestein_chirp_polynomial_identity(self):
  94         for n,rec in self.records.items():
  95             self.assertTrue(rec['chirp_polynomial_identity'])
  96 
  97     def test_chirp_is_dense_in_both_domains(self):

  99             self.assertEqual(rec['chirp_spectrum_nonzero'],n)

 101     def test_selected_packet_count_gate(self):
 102         for n in CASES:
 103             # Each selected character packet has at most two entries per column.
 104             # A dense target therefore needs at least ceil(n/2) separately summed packets.
 105             self.assertGreaterEqual((n+1)//2,1)
 106             self.assertLess(2*((n+1)//2-1),n)

 111         self.assertEqual(saved['finite_field_cases'],expected)
