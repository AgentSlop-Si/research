63: def exact_green(g: GreenGraph,q=65537):
64:     """Returns (I-P)^-1 with P[i,pi[i]]=weights[i], exactly modulo q."""
65:     n=len(g.pi)
66:     a=[[0]*n for _ in range(n)]
67:     for cyc in cycles(g.pi):
68:         prod=1
69:         for i in cyc:prod=prod*g.weights[i]%q
70:         if prod==1:raise ValueError('singular cycle')
71:         inv=pow((1-prod)%q,-1,q)
72:         for i in cyc:
73:             k=i
74:             path=1
75:             while True:
76:                 a[i][k]=(path*inv)%q
77:                 path=(path*g.weights[k])%q
78:                 k=g.pi[k]
79:                 if k==i:break
80:     return a

101:     if not a:return 0
102:     b=[list(row) for row in a]
103:     n=len(b);cols=len(b[0]);r=0
104:     for col in range(cols):
105:         piv=next((i for i in range(r,n) if b[i][col]%q),None)
106:         if piv is None:continue
107:         b[r],b[piv]=b[piv],b[r]
108:         inv=pow(b[r][col],-1,q)
109:         row=b[r]
110:         for j in range(col,cols):row[j]=row[j]*inv%q
111:         for i in range(r+1,n):
