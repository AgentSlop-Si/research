import unittest
from condition_reset_experiment import run_case

class ConditionResetTests(unittest.TestCase):
    def test_no_multiplicative_condition_growth(self):
        for K in (1e2,1e4):
            r=run_case(N=16,B=4,macro_levels=10,K=K,seed=7)
            self.assertLess(r['max_global_condition'], K*1.0001)
            self.assertLess(r['final_condition'], 1.0000001)

if __name__=='__main__':
    unittest.main()
