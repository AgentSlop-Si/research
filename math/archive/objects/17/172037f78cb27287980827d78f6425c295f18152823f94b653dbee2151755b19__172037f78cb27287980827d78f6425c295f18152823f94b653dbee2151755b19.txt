#!/usr/bin/env python3
"""Verify every sealed payload byte; no uploaded source is imported."""
import hashlib,json,pathlib
root=pathlib.Path(__file__).resolve().parents[1]
seal=json.loads((root/'SEALED_MANIFEST.json').read_text())
for item in seal['files']:
 path=root/item['path'];data=path.read_bytes()
 assert len(data)==item['size_bytes'],item['path']
 assert hashlib.sha256(data).hexdigest()==item['sha256'],item['path']
actual={str(p.relative_to(root)) for p in root.rglob('*') if p.is_file()}-{'SEALED_MANIFEST.json','SEALED_MANIFEST.sha256'}
assert actual=={x['path'] for x in seal['files']},'unsealed or missing payload file'
expected=(root/'SEALED_MANIFEST.sha256').read_text().split()[0]
assert hashlib.sha256((root/'SEALED_MANIFEST.json').read_bytes()).hexdigest()==expected
print(f"PASS: {len(seal['files'])} files, every byte verified, no unsealed payload files")
