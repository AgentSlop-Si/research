   8 def add(a,b,c=1):return [[(x+c*y)%P for x,y in zip(r,s)] for r,s in zip(a,b)]
   9 def scale(a,v):return [[x*v%P for x in row] for row in a]
  10 def diagonal(v):return [[x if i==j else 0 for j in range(len(v))] for i,x in enumerate(v)]
  11 def successor(order):
  12     p=[0]*len(order)
  13     for i,j in enumerate(order):p[j]=order[(i+1)%len(order)]
  14     return p
  15 
  16 def deficit(p,q):
  17     inv=[0]*len(p)
  18     for i,j in enumerate(p):inv[j]=i
  19     seen=set();count=0
  20     for v in range(len(p)):
  21         if v in seen:continue
  22         count+=1
  23         while v not in seen:seen.add(v);v=inv[q[v]]
  24     return len(p)-count

  26 def cycle_orders(n,number,seed):
  27     rng=random.Random(seed+17*n);out=[];perms=[]
  28     while len(out)<number:
  29         order=list(range(n));rng.shuffle(order);p=successor(order)
  30         slope=(p[1]-p[0])%n
  31         if all(p[i]==(slope*i+p[0])%n for i in range(n)):continue
  32         if any(all(p[q[i]]==q[p[i]] for i in range(n)) for q in perms):continue
  33         out.append(order);perms.append(p)
  34     return out

  36 def green(order,alpha):
  37     n=len(order);pos=[0]*n
  38     for i,j in enumerate(order):pos[j]=i
  39     den=F.inv(1-pow(alpha,n,P));powers=[pow(alpha,j,P)*den%P for j in range(n)]
  40     return [[powers[(pos[i]-pos[j])%n] for j in range(n)] for i in range(n)]
  41 
  42 def denominator(order,alpha):
  43     n=len(order);out=F.eye(n)
  44     for j,i in enumerate(successor(order)):out[i][j]=(out[i][j]-alpha)%P
  45     return out

  47 def stage(order,alpha,index,patched=False):
  48     n=len(order);a,b=1+index,n//2+1+index
  49     modified=[b if x==a else a if x==b else x for x in order]
  50     g0=green(order,alpha);g1=green(modified,alpha);delta=deficit(successor(order),successor(modified));assert delta==2
  51     assert F.mul(denominator(order,alpha),g0)==F.eye(n)
  52     if patched:
  53         l=[1]*n;r=[1]*n;l[(3+index)%n]=7+index;r[(7+index)%n]=11+index
  54         g1=[[x*l[i]*r[j]%P for j,x in enumerate(row)] for i,row in enumerate(g1)]
  55     s=add(g0,g1);si=F.inverse(s);core=scale(denominator(order,alpha),F.inv(2));rank=F.rank(add(si,core,-1))
  56     assert rank<=delta+2*patched
  57     if not patched:
  58         assert rank==delta
  59         assert F.rank(add(g1,g0,-1))==delta<n
  60     return si,core,rank

  62 def nnz(a):return max(sum(bool(x) for x in row) for row in a)
  63 def witness(core,k):
  64     n=len(core);m=n//(k+1);bad={j for row in core[:m] for j,x in enumerate(row) if x};columns=[j for j in range(n) if j not in bad][:m]
  65     assert len(columns)==m;z=root(n,P,3)
  66     if m:assert F.det([[pow(z,i*j,P) for j in columns] for i in range(m)])
  67     return columns

  99 def general_sparse():
