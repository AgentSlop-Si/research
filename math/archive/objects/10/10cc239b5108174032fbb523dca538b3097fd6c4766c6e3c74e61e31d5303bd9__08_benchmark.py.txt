#!/usr/bin/env python3
from __future__ import annotations
import csv, statistics, time
from pathlib import Path
import numpy as np
import prototype as p
import unit_step_core as core


def median_time(fn, repeats: int = 5) -> float:
    fn()
    vals=[]
    for _ in range(repeats):
        t=time.perf_counter(); fn(); vals.append(time.perf_counter()-t)
    return statistics.median(vals)


def main() -> None:
    rng=np.random.default_rng(20261009)
    rows=[]
    for n in [511,1023,2047,4095,8191]:
        q=p.next_power_of_two(n); m=p.choose_dyadic_block(q); rank=10
        x=rng.normal(size=n)+1j*rng.normal(size=n)
        state=core.draw_state(q,rng)
        ref=lambda:p.reference_bessel_chirp_row(x,rank,m,state)
        streamed=lambda:p.streamed_bessel_chirp_row(x,rank,m,state)
        fused0=lambda:p.fused_addition_bessel_chirp_row(x,rank,m,state,0)
        fused1=lambda:p.fused_addition_bessel_chirp_row(x,rank,m,state,1)
        t_ref=median_time(ref)
        t_stream=median_time(streamed)
        t_f0=median_time(fused0)
        t_f1=median_time(fused1)
        exact=streamed()
        rows.append({
            'N':n,'Q':q,'M':m,'rank':rank,
            'reference_seconds':t_ref,
            'streamed_exact_seconds':t_stream,
            'fused_H0_seconds':t_f0,
            'fused_H1_seconds':t_f1,
            'streamed_speedup_vs_reference':t_ref/t_stream,
            'fused_H0_speedup_vs_reference':t_ref/t_f0,
            'fused_H1_speedup_vs_reference':t_ref/t_f1,
            'fused_H0_max_row_difference':float(np.max(np.abs(fused0()-exact))),
            'fused_H1_max_row_difference':float(np.max(np.abs(fused1()-exact))),
        })
    out=Path('benchmark_results.csv')
    with out.open('w',newline='',encoding='utf-8') as f:
        w=csv.DictWriter(f,fieldnames=list(rows[0]));w.writeheader();w.writerows(rows)
    for row in rows: print(row)

if __name__=='__main__': main()
