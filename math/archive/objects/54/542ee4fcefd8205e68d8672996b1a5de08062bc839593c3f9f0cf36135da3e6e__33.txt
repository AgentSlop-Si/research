   1 """Independent finite-field auditing of the exact complex theorems.
   2 The rigorous characteristic-zero proofs are in PROOF_CERTIFICATE.md.
   3 """
   4 import random,unittest
   5 from fractions import Fraction
   6 from math import ceil
   7 from itertools import combinations
   8 import numpy as np
   9 from bitrev_rank import exact_cut_rank,bitrev_perm,predicted_qs_rank
  10 from coset_rank import (prim_root,rank_mod,fourier_submatrix,dyadic_union_columns,
  11                         rank_coset_formula,single_consecutive_coset_formula)
  12 
  13 PRIMES=(12289,65537)
  14 
  15 
  16 def formula_sum(n):
  17     m=n.bit_length()-1
  18     if m%2==0: return (4*n*n+3*n*m-4*n)//24
  19     return (4*n*n+3*n*m+n)//24
  20 
  21 
  22 def reversal_rank_sums(n):
  23     return sum(exact_cut_rank(n,k) for k in range(1,n))
  24 
  25 class ExactFourierBitrevTest(unittest.TestCase):
  26     def test_all_finite_field_cuts(self):
  27         for n in (4,8,16,32,64):
  28             b=bitrev_perm(n)
  29             for p in PRIMES:
  30                 M=fourier_submatrix(n,b,b,p)
  31                 for k in range(1,n):
  32                     r=rank_mod(M[:k,k:],p)
  33                     self.assertEqual(r,exact_cut_rank(n,k),(n,p,k))
  34                     # Symmetric Fourier matrix means lower left is transposed upper right.
  35                     self.assertEqual(r,rank_mod(M[k:,:k],p))
  36 
  37     def test_N128_selected_cuts(self):
  38         n=128;b=bitrev_perm(n)
  39         selected=(1,2,3,7,8,15,16,31,32,47,61,62,63,64,65,66,67,81,95,96,111,112,120,125,126,127)
  40         for p in PRIMES:
  41             M=fourier_submatrix(n,b,b,p)
  42             for k in selected:
  43                 self.assertEqual(rank_mod(M[:k,k:],p),exact_cut_rank(n,k),(n,p,k))
  44 
  45     def test_n8_exhaustive_critical_minors(self):
  46         n=8; sub=list(combinations(range(n),3));cols=list(combinations(range(n),5))
  47         for p in PRIMES:
  48             M=fourier_submatrix(n,list(range(n)),list(range(n)),p)
  49             for rows in sub:
  50                 for C in cols:
  51                     self.assertEqual(rank_mod(M[np.ix_(rows,C)],p),3,(p,rows,C))
  52 
  53     def test_random_independent_endpoint_orders(self):
  54         rng=random.Random(17081)
  55         for n in (16,32,64):
  56             L=n//4;cut=n//2-1
  57             for p in PRIMES:
  58                 F=fourier_submatrix(n,list(range(n)),list(range(n)),p)
  59                 for _ in range(24):
  60                     row=list(range(n));col=list(range(n))
  61                     rng.shuffle(row);rng.shuffle(col)
  62                     M=F[np.ix_(row,col)]
  63                     rank=rank_mod(M[:cut,cut:],p)
  64                     self.assertGreaterEqual(rank,L+1,(n,p,row,col,rank))
  65 
  66     def test_trace_square_exact_inequality(self):
  67         for n in (8,16,32,64,128,256,512):
  68             L=n//4
  69             val=Fraction(L,1)+Fraction(3,4)+Fraction(1,4*L)+Fraction(1,16*L*L)
  70             self.assertLess(val,L+1)
  71             self.assertEqual(ceil(val),L+1)
  72 
  73     def test_optimal_peak_and_sum_formulas(self):
  74         for m in range(1,13):
  75             n=1<<m
  76             self.assertEqual(predicted_qs_rank(n),1 if n==2 else (1 if n==4 else n//4+1))
  77             s=reversal_rank_sums(n)
  78             self.assertEqual(s,formula_sum(n),(n,s,formula_sum(n)))
  79             self.assertGreaterEqual(6*s,n*n-1)
  80 
  81     def test_coset_formula(self):
  82         rng=random.Random(260910)
  83         for n in (8,16,32,64):
  84             for p in PRIMES:
  85                 for q in (2,4,8):
  86                     if q>n:continue
  87                     for _ in range(12):
  88                         rows=sorted(rng.sample(range(n),rng.randint(1,n)))
  89                         bases=sorted(rng.sample(range(q),rng.randint(1,q)))
  90                         cols=dyadic_union_columns(n,q,bases)
  91                         a=rank_mod(fourier_submatrix(n,rows,cols,p),p)
  92                         b=rank_coset_formula(n,q,bases,rows,p)
  93                         self.assertEqual(a,b,(n,p,q,rows,bases))
  94 
  95     def test_consecutive_single_coset_formula(self):
  96         rng=random.Random(20261009)
  97         for n in (8,16,32,64):
  98             for p in PRIMES:
  99                 for q in (2,4,8):
 100                     if q>n:continue
 101                     for _ in range(10):
 102                         rows=sorted(rng.sample(range(n),rng.randint(1,n)))
 103                         b=rng.randrange(q);k=rng.randint(1,n//q)
 104                         cols=[b+q*j for j in range(k)]
 105                         rk=rank_mod(fourier_submatrix(n,rows,cols,p),p)
 106                         expect=single_consecutive_coset_formula(n,q,b,rows,k)
 107                         self.assertEqual(rk,expect,(n,p,q,b,k))

 109     def test_prior_endpoint_positive_control(self):
 110         n=16
 111         order=[0,8,4,12,1,9,5,13,2,10,6,14,3,11,7,15]
 112         for p in PRIMES:
 113             M=fourier_submatrix(n,order,order,p)
 114             self.assertEqual(rank_mod(M[3:, :9],p),8)
 115             # Applying a naive Vandermonde rule would incorrectly yield 9.
 116             self.assertEqual(min(13,9),9)
