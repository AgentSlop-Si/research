  17 def isprime(n):
  18     if n<2: return False
  19     for d in range(2,int(n**0.5)+1):
  20         if n%d==0:return False
  21     return True

  24 def factor(n):
  25     out=[];k=2
  26     while k*k<=n:
  27         if n%k==0:
  28             out.append(k)
  29             while n%k==0:n//=k
  30         k+=1
  31     if n>1:out.append(n)
  32     return out

  35 def field_root(n):
  36     for c in range(1,4000):
  37         p=c*n+1
  38         if not isprime(p):continue
  39         for g in range(2,p):
  40             w=pow(g,(p-1)//n,p)
  41             if all(pow(w,n//d,p)!=1 for d in factor(n)):
  42                 return p,w
  43     raise RuntimeError(n)

  46 def modular_rank(table,p):
  47     if not table:return 0
  48     a=[list(row) for row in table];n=len(a);m=len(a[0]);r=0
  49     for c in range(m):
  50         pr=None
  51         for i in range(r,n):
  52             if a[i][c]%p:
  53                 pr=i;break
  54         if pr is None:continue
  55         a[r],a[pr]=a[pr],a[r]
  56         z=pow(a[r][c],p-2,p)
  57         for j in range(c,m):a[r][j]=(a[r][j]*z)%p
  58         for i in range(r+1,n):
  59             v=a[i][c]
  60             if v:
  61                 for j in range(c,m): a[i][j]=(a[i][j]-v*a[r][j])%p
  62         r+=1
  63         if r==n:break
  64     return r

  67 def direct_field_dft(signal,p,omega,frequency):
  68     n=len(signal)
  69     ratio=pow(omega,(-frequency)%n,p)
  70     z=1;result=0
  71     for value in signal:
  72         result=(result+value*z)%p
  73         z=z*ratio%p
  74     return result

  77 def audit():
  78     rng=random.Random(172663)
  79     results={'status':'PASS','model':'exact Fp where p ≡ 1 mod N, plus float64 SVD checks',
  80              'raw_tensor_coefficients_checked':0,

  83              'sparse_core_random_gauge_cases':0,

  85              'spot_checked_large_output_coefficients':0,

  87              'matrix_cut_profiles':[],
  88              'unitary_singular_value_max_discrepancy':0.0,

  95                  for s in range(Q) for h in range(M)]
  96         crossed=[[f(s,h,t,u) for h in range(M) for u in range(M)]
  97                   for s in range(Q) for t in range(Q)]

  99                     for s in range(Q) for u in range(M)]

 103         results['independent_modular_rank_checks']+=3
 104         results['matrix_cut_profiles'].append({'Q':Q,'M':M,'N':N,'ranks':ranks,'field_prime':p})
 105         A,B,core=endpoint_sparse_core(Q,M,p,w)
 106         for a in range(Q):
 107             for h in range(M):
 108                 for t in range(Q):
 109                     for b in range(M):
 110                         # Direct summation over both *external* basis coordinates.

 133                           for s in range(Q) for u in range(M))%p
 134                 assert core[a][h][t][b]==brute
 135             results['sparse_core_random_gauge_cases']+=1
