   1 #!/usr/bin/env python3
   2 """Phase9B dual sparse innovations: independent exact NTT & adversarial tests."""
   3 from __future__ import annotations
   4 import random,unittest
   5 from phase9_dense_cycle import P,Costs,exact_reference_dft,root_of_unity
   6 from phase9_spectral_innovations import (
   7     make_spectral_cycle, exact_spectral_innovation_cycle,
   8     recover_spectral_innovations, verify_spectral_innovations)

  12 class SpectralTests(unittest.TestCase):
  13     def check(self,k,m,events,seed=17,lam_override=None):
  14         x,meta=make_spectral_cycle(k,m,events,seed,lam_override)
  15         answer,cert,c=exact_spectral_innovation_cycle(
  16             x,m,max((len(u) for u in events.values()),default=0),3,
  17             Costs(),random.Random(seed+990),True)
  18         self.assertIsNotNone(answer,cert)
  19         self.assertEqual(answer,exact_reference_dft(x))
  20         if k*m<=128:self.assertEqual(answer,quadratic_reference(x))
  21         self.assertEqual(cert['total_innovation_modes'],sum(map(len,events.values())))
  22         return x,meta,cert,c
  23 
  24     def test_a_dense_error_every_transition(self):
  25         for k,m in [(16,8),(32,8),(64,8),(64,16),(128,16),(256,16),(512,16)]:
  26             for s in (0,1,2,3):
  27                 rng=random.Random(k*317+m*23+s)
  28                 events={t:[(e,rng.randrange(1,P)) for e in rng.sample(range(k),s)]
  29                         for t in range(1,m-1)} if s else {}
  30                 self.check(k,m,events,seed=11+s)
  31 
  32     def test_b_exact_singular_denominator(self):
  33         for k,m in [(16,8),(32,8),(64,16)]:
  34             n=k*m;w=root_of_unity(n);rng=random.Random(k)
  35             lam=[rng.randrange(1,P) for _ in range(k)]
  36             lam[0]=1
  37             lam[1]=pow(w,1+k,P)
  38             ev={t:[((2*t+3)%k,100+t)] for t in range(1,m-1)}
  39             self.check(k,m,ev,seed=40,lam_override=lam)
  40 
  41     def test_c_one_fourier_mode_is_dense_position_innovation(self):
  42         k,m=32,8
  43         events={t:[(t+1,t+33)] for t in range(1,m-1)}
  44         x,meta,cert,c=self.check(k,m,events)
  45         w=root_of_unity(k)
  46         for t in range(1,m-1):
  47             ell,val=events[t][0]
  48             coeff=val*pow(k,P-2,P)%P
  49             err=[coeff*pow(w,ell*b,P)%P for b in range(k)]
  50             self.assertEqual(sum(z!=0 for z in err),k)
  51         self.assertTrue(all(meta['kernel']))
  52         self.assertEqual(len(set(meta['lambda'])),k)
  53

  71     def test_f_invisible_probes_detected_by_freivalds(self):
  72         k,m=32,8;n=k*m
  73         x,meta=make_spectral_cycle(k,m,{},10)
  74         w=root_of_unity(k)
  75         nodes=[pow(w,z,P) for z in range(3)]
  76         amps=[(nodes[1]-nodes[2])%P,(nodes[2]-nodes[0])%P,
  77               (nodes[0]-nodes[1])%P]
  78         self.assertEqual(sum(amps)%P,0)
  79         self.assertEqual(sum(a*r for a,r in zip(amps,nodes))%P,0)
  80         # Inject 3-mode error at final column only, so no later residual.
  81         mutated=x[:]
  82         for b in range(k):
  83             e=sum(a*pow(node,b,P) for a,node in zip(amps,nodes))%P
  84             mutated[b*m+m-1]=(mutated[b*m+m-1]+e)%P
  85         cols=[[mutated[b*m+t] for b in range(k)] for t in range(m)]
  86         f0,lam,tab=learn_eigenvalues(cols,n,Costs())
  87         ev=recover_spectral_innovations(cols,lam,tab,1,Costs())
  88         self.assertEqual(sum(map(len,ev)),0) # invisible to two spatial probes
  89         self.assertFalse(verify_spectral_innovations(
  90             cols,lam,ev,tab,m,Costs(),3,random.Random(448)))
  91         y,cert,_=exact_spectral_innovation_cycle(
  92             mutated,m,1,3,Costs(),random.Random(448),True)
  93         self.assertIsNone(y)
  94         self.assertEqual(cert['reason'],'independent recurrence fingerprint failed')
  95 
  96     def test_g_finite_size_full_rank_witness(self):
  97         k,m=16,8
  98         rng=random.Random(17)
  99         events={t:[((t+1)%k,rng.randrange(1,P))] for t in range(1,m-1)}
 100         x,meta,cert,c=self.check(k,m,events,123)
 101         columns=[[x[b*m+t] for t in range(m)] for b in range(k)]
 102         self.assertEqual(matrix_rank_mod(columns),m)
 103         self.assertEqual(matrix_rank_mod([[x[i+j] for j in range(k*m//2)]
 104                                          for i in range(k*m//2)]),k*m//2)
 105         self.assertTrue(all(x))
