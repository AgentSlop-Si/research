   7 def roots(n,p):
   8     if (p-1)%n:raise ValueError('unsupported length')
   9     primes=[q for q in [2,3] if n%q==0]
  10     for g in range(2,p):
  11         w=pow(g,(p-1)//n,p)
  12         if pow(w,n,p)==1 and all(pow(w,n//q,p)!=1 for q in primes):break
  13     else:raise ValueError('root')
  14     a=[1]
  15     for _ in range(1,n):a.append(a[-1]*w%p)
  16     return a

  18 def binary(x,R,p,lift=False,skip_identity=True):

  25                 if lift:
  26                     coeff=-2%p;bottom=(top+coeff*v)%p;mul+=int(not skip_identity or coeff!=1)
  27                 else:bottom=(u-v)%p
  28                 add+=2;w=R[j*step]
  29                 if w!=1:bottom=bottom*w%p;mul+=1

  32     bits=n.bit_length()-1;res=[0]*n

  34         j=0;s=i
  35         for _ in range(bits):j=2*j+(s&1);s>>=1
  36         res[j]=v
  37     return res,(add,mul)

  39 def split(x,R,p):
  40     n=len(x);counts=[0,0]
  41     def rec(v,stride):
  42         m=len(v)
  43         if m==1:return v[:]
  44         if m==2:counts[0]+=2;return [(v[0]+v[1])%p,(v[0]-v[1])%p]
  45         h=m//2;q=m//4;s=[(v[j]+v[j+h])%p for j in range(h)];d=[(v[j]-v[j+h])%p for j in range(h)];counts[0]+=m
  46         c=[];e=[];imag=R[q*stride]
  47         for j in range(q):

  49             for which in [1,3]:
  50                 tw=R[which*j*stride]
  51                 if tw!=1:
  52                     if which==1:u=u*tw%p
  53                     else:v2=v2*tw%p

  57         out[::2]=ev;out[1::4]=od1;out[3::4]=od3;return out
  58     return rec(x,1),tuple(counts)

  60 def gt(x,R,p,method='split',identity_aware=False):
  61     n=len(x);g=n//3;rg=R[::3];eg=3*pow(3,-1,g);e3=g*pow(g,-1,3);ys=[];adds=muls=0
  62     for a in range(3):
  63         row=[x[(j*eg+a*e3)%n] for j in range(g)]
  64         y,c=split(row,rg,p) if method=='split' else binary(row,rg,p,lift=method=='lift')
  65         ys.append(y);adds+=c[0];muls+=c[1]
  66     inv2=pow(2,-1,p);v=-inv2%p;w=(R[g]-R[2*g])*inv2%p;out=[0]*n
  67     for j in range(g):
  68         a,b,c=(row[j] for row in ys);t=(b+c)%p;d=(b-c)%p;mid=(a+v*t)%p;z=w*d%p
  69         out[3*j%n]=(a+t)%p;out[(3*j+g)%n]=(mid+z)%p;out[(3*j+2*g)%n]=(mid-z)%p
  70         adds+=6;muls+=int(v!=1)+int(w!=1) if identity_aware else 2
  71     return out,(adds,muls)

  73 def entropy(A):return -sum((abs(z)**2)*math.log2(abs(z)**2) for row in A for z in row if abs(z)>1e-14)
  74 def fourier(n):return [[cmath.exp(2j*math.pi*i*j/n)/math.sqrt(n) for j in range(n)] for i in range(n)]
  75 def apply_gate(A,ids,U):
  76     B=[r[:] for r in A]
  77     for i,a in enumerate(ids):B[a]=[sum(U[i][j]*A[ids[j]][k] for j in range(len(ids))) for k in range(len(A))]
  78     return B
