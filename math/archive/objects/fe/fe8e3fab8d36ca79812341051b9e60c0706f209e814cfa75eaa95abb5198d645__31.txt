   1 #!/usr/bin/env python3
   2 """Independent exhaustive 504 single-Toffoli layout search, all parity inputs."""
   3 from independent_checks import affine,mcnot,compose,BASE
   4 from itertools import combinations
   5 import json
   6 
   7 def boundary_table(order):
   8  n=len(order);a=[0]*n
   9  for i in range(n-1):a[order[i]^order[i+1]]+=1
  10  step=1
  11  while step<n:
  12   for i in range(0,n,2*step):
  13    for j in range(i,i+step):x,y=a[j],a[j+step];a[j],a[j+step]=x+y,x-y
  14   step*=2
  15  return [(n-1-v)//2 for v in a]
  16 def best(perms):
  17  n=len(perms[0]);order=list(range(n));caps=boundary_table(order)
  18  for p in perms:
  19   order=compose(order,p);ds=boundary_table(order);caps=[a+b for a,b in zip(caps,ds)]
  20  bestgap=-n;bestmask=None
  21  for mask in range(1,n):
  22   h=1<<mask.bit_length();d=n//h
  23   if d==1:rank=n//2
  24   else:
  25    counts=[0]*d
  26    for i,v in enumerate(order):
  27     if (v&mask).bit_count()%2:counts[i%d]+=1
  28    rank=sum(min(c,h//2) for c in counts)
  29   gap=rank-caps[mask]
  30   if gap>bestgap:bestgap,bestmask=gap,mask
  31  return bestgap,bestmask
  32 
  33 if __name__=='__main__':
  34  source=json.loads((BASE/'sources/EXACT_CERTIFICATES.json').read_text())['nonaffine_256'];ps=[affine(c) for c in source['affine_shuffle_columns']]
  35  base=best(ps);assert base[0]==-39
  36  records=[]
  37  for j in range(3):
  38   for t in range(8):
  39    for controls in combinations([v for v in range(8) if v!=t],2):
  40     new=[p[:] for p in ps];new[j]=compose(new[j],mcnot(8,controls,t));gap,mask=best(new)
  41     records.append({'shuffle':j,'controls':controls,'target':t,'best_gap':gap,'witness_mask':mask})
  42  assert len(records)==504 and max(r['best_gap'] for r in records)==-10 and all(r['best_gap']<0 for r in records)
  43  out={'status':'PASS','baseline_best_gap':base[0],'number_of_single_Toffoli_layouts':504,'input_masks_each':255,'best_gap':-10,'records':records}
  44  (BASE/'SINGLE_EDIT_SEARCH.json').write_text(json.dumps(out,indent=2,sort_keys=True)+'\n');print('PASS all 504 layouts and 128520 parity probes; best gap -10')
