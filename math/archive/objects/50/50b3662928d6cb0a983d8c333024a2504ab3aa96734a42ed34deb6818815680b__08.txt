0034 def build_load():
0035     cmd = ['gcc',*FLAGS,str(HERE/'leaf_fft.c'),'-lm','-o',str(HERE/'libleaf_fft.so')]
0036     subprocess.run(cmd, check=True)

0040     lib.fft_plan_destroy.argtypes=[P]

0048 def ptr(a): return ct.c_void_p(a.ctypes.data)
0049 def stats(a):
0050     a=np.asarray(a,dtype=float)
0051     med=float(np.median(a))
0052     return {'median_us':med*1e6,'p10_us':float(np.percentile(a,10))*1e6,
0053             'p90_us':float(np.percentile(a,90))*1e6,
0054             'relative_mad':float(np.median(np.abs(a-med))/med),
0055             'samples_us':(a*1e6).tolist()}

0074 def correctness(lib,tables,sizes):
0075     worst={m:0. for m in METHODS};cases=0
0076     # Complete 4-, 8- and 16-input table validation, not just sampled patterns.
0077     table_errors={}
0078     for leaf,t in tables.items():
0079         rows=1<<leaf;bits=int(math.log2(leaf))
0080         rev=np.array([int(f'{i:0{bits}b}'[::-1],2) for i in range(leaf)])
0081         v=((np.arange(rows,dtype=np.uint32)[:,None]>>np.arange(leaf))&1).astype(np.float64)
0082         expected=np.fft.fft(v[:,rev],axis=1)
0083         actual=np.ctypeslib.as_array(ct.cast(t,ct.POINTER(ct.c_double)),shape=(rows*leaf*2,)).view(np.complex128).reshape(rows,leaf)
0084         err=float(np.max(np.abs(expected-actual)))
0085         np.testing.assert_allclose(actual,expected,atol=1e-12,rtol=1e-12)
0086         table_errors[str(leaf)]=err
0087     for n in sorted(set([16,32,64,128,512,*sizes])):
0088         p=lib.fft_plan_create(n); assert p
0089         y=np.empty(n,dtype=np.complex128)
0090         inputs=[np.zeros(n,dtype=np.uint8),np.ones(n,dtype=np.uint8),np.arange(n,dtype=np.uint8)%2]
0091         impulse=np.zeros(n,dtype=np.uint8);impulse[n//3]=1;inputs.append(impulse)
0092         for seed in [7,19,123,2026,58127]:
0093             inputs.append(np.random.default_rng(seed).integers(0,2,n,dtype=np.uint8))
0094         for x in inputs:
0095             ref=np.fft.fft(x)
0096             for method,m in enumerate(METHODS):
0097                 t=tables.get(LEAF.get(m,0)) if m.startswith('lookup') else None
0098                 lib.fft_execute(method,p,ptr(x),ptr(y),t)
0099                 err=float(np.max(np.abs(y-ref)));worst[m]=max(worst[m],err)
0100                 np.testing.assert_allclose(y,ref,atol=2e-9,rtol=2e-11)
0101                 cases+=1
0102         lib.fft_plan_destroy(p)
0103     return dict(passed=True,full_fft_cases=cases,random_seeds=[7,19,123,2026,58127],
0104                 max_absolute_error=worst,exhaustive_leaf_table_max_absolute_error=table_errors,
0105                 asserted_atol=2e-9,asserted_rtol=2e-11)

0108 def measure_tables(lib):
0109     tables={};summary={}
0110     for leaf in (4,8,16):
0111         times=[]
0112         # 11 independent allocate-and-populate calls. OS/cache state is uncontrolled;
0113         # these are construction costs, not claims of first-page-fault cold latency.
0114         for _ in range(11):
0115             start=time.perf_counter();t=lib.fft_table_create(leaf);elapsed=time.perf_counter()-start
0116             assert t;times.append(elapsed);lib.fft_table_destroy(t)
0117         tables[leaf]=lib.fft_table_create(leaf)
0118         summary[str(leaf)]={'entries':1<<leaf,'bytes':(1<<leaf)*leaf*16,
0119                             'construction':stats(times),'first_construction_us':times[0]*1e6}
0120     return tables,summary

0123 def bench_size(lib,tables,n,rounds,target,pool_count,seed):
0124     rng=np.random.default_rng(seed)
0125     pool=rng.integers(0,2,(pool_count,n),dtype=np.uint8)
0126     output=np.empty(n,dtype=np.complex128)
0127     start=time.perf_counter();p=lib.fft_plan_create(n);plan_s=time.perf_counter()-start;assert p
0128     records={m:[] for m in METHODS}; repeats={}
0129     for method,m in enumerate(METHODS):
0130         t=tables.get(LEAF.get(m,0)) if m.startswith('lookup') else None
0131         estimate=lib.fft_bench(method,p,ptr(pool),pool_count,ptr(output),t,pool_count*2)
0132         repeats[m]=max(pool_count,int(math.ceil(target/estimate/pool_count)*pool_count))
0133     order_rng=random.Random(seed)
0134     for _ in range(rounds):
0135         order=list(range(len(METHODS)));order_rng.shuffle(order)
0136         for method in order:
0137             m=METHODS[method]; t=tables.get(LEAF.get(m,0)) if m.startswith('lookup') else None
0138             records[m].append(lib.fft_bench(method,p,ptr(pool),pool_count,ptr(output),t,repeats[m]))
0139     lib.fft_plan_destroy(p)
0140     summary={m:dict(stats(v),transforms_per_sample=repeats[m]) for m,v in records.items()}
0141     base=summary['radix2_fused2']['median_us']
0142     bestcode=min((summary[m]['median_us'],m) for m in METHODS if not m.startswith('lookup'))
0143     for m,s in summary.items():
0144         s['speedup_vs_radix2']=base/s['median_us']
0145         s['speedup_vs_best_arithmetic']=bestcode[0]/s['median_us']
0146     return dict(n=n,input_seed=seed,input_pool_count=pool_count,plan_create_us=plan_s*1e6,
0147                 best_arithmetic=bestcode[1],methods=summary),pool

0150 def bench_python_baselines(pool,rounds,target,seed):
0151     count,n=pool.shape
0152     floats=pool.astype(np.float64)
0153     methods={
0154       'scipy_fft_uint8': lambda j: spfft.fft(pool[j],workers=1),
0155       'scipy_fft_float64': lambda j: spfft.fft(floats[j],workers=1),
0156       'numpy_fft_uint8': lambda j: np.fft.fft(pool[j]),
0157       'scipy_rfft_float64_half_spectrum': lambda j: spfft.rfft(floats[j],workers=1),
0158     }
0159     repeats={};records={m:[] for m in methods}
0160     for m,f in methods.items():
0161         start=time.perf_counter()
0162         for r in range(count): f(r)
0163         estimate=(time.perf_counter()-start)/count
0164         repeats[m]=max(count,int(math.ceil(target/estimate/count)*count))
0165     order_rng=random.Random(seed)
0166     for _ in range(rounds):
0167         order=list(methods);order_rng.shuffle(order)
0168         for m in order:
0169             start=time.perf_counter()
0170             for r in range(repeats[m]): methods[m](r%count)
0171             records[m].append((time.perf_counter()-start)/repeats[m])
0172     return {m:dict(stats(v),transforms_per_sample=repeats[m]) for m,v in records.items()}

0175 def main():
0176     ap=argparse.ArgumentParser()
0177     ap.add_argument('--sizes',nargs='+',type=int,default=[4096,16384,65536,262144,1048576])
0178     ap.add_argument('--rounds',type=int,default=11)

0183     ap.add_argument('--skip-practical',action='store_true',help='Skip Python library baselines, e.g. for a focused C rerun')
0184     args=ap.parse_args()
0185     if args.quick:
0186         args.sizes=[4096,16384];args.rounds=3;args.target_ms=10
0187     affinity=sorted(os.sched_getaffinity(0));selected=None
0188     try: os.sched_setaffinity(0,{affinity[0]});selected=affinity[0]
0189     except OSError: pass
0190     lib,cmd=build_load();tables,table_summary=measure_tables(lib)
0191     result={'metadata':get_metadata(cmd,affinity,selected),'methodology':{
0192         'description':'Single-threaded full complex FFT of binary uint8 inputs, preallocated C output and plans, identical merge engine.',
0193         'timed_operations':'Input bit-reversal gathers, key packing, complete row copies or arithmetic leaves, every upper merge stage; C calls inside a timed C batch.',
0194         'excluded_operations':'Plan/table creation, random input generation, validation, C output allocation.',
0195         'baseline':'Iterative radix-2 with fused bit reversal and length-2 arithmetic, specialized length-4 stage; no recursion or hashes.',
0196         'codelet_controls':'Fused bit reversal and real-input arithmetic length-4/8/16 leaves with the same upper merges.',
0197         'key_order':'Key bit j is input[global_bit_reverse(leaf_start+j)]. Table spectra are computed from this bit-reversed leaf layout.',
0198         'input_pool':'16 distinct seeded random arrays cycled per sample by default; each sample is a whole number of cycles.',
0199         'warmup':'At least two input-pool cycles per method before samples. No artificial cache flushing.',
0200         'samples':'Randomized method order every round; median of batch per-transform averages.',
0201         'python_baselines':'SciPy and NumPy end-to-end calls include output allocation/Python dispatch; not allocation-matched to C. float64 conversion is outside timing for float64 variants.',
0202         'half_spectrum_note':'rfft returns N/2+1 values, so its performance is context only, not an equivalent full-output comparison.',
0203         'table_build_note':'Allocation and populate measured separately; first and 11-call median shown. No claim of a controlled cache-cold first use.',
0204         'benchmark_rounds':args.rounds,'target_sample_ms':args.target_ms},'tables':table_summary}
0205     result['correctness']=correctness(lib,tables,args.sizes)
0206     print('Correctness:',json.dumps(result['correctness']),flush=True)
0207     result['sizes']=[]
0208     for n in args.sizes:
0209         rec,pool=bench_size(lib,tables,n,args.rounds,args.target_ms/1000,args.pool_count,18000+n)
0210         rec['practical_baselines']=({} if args.skip_practical else bench_python_baselines(pool,args.rounds,args.target_ms/1000,38000+n))
0211         # Construction amortization only when lookup beats best arithmetic leaf.
0212         for leaf in (4,8,16):
0213             delta=rec['methods'][rec['best_arithmetic']]['median_us']-rec['methods'][f'lookup{leaf}']['median_us']
0214             rec['methods'][f'lookup{leaf}']['table_build_break_even_transforms_vs_best_arithmetic']=(table_summary[str(leaf)]['construction']['median_us']/delta if delta>0 else None)
0215         result['sizes'].append(rec)
0216         args.output.write_text(json.dumps(result,indent=2)+'\n')
0217         print(f'N={n}: '+', '.join(f'{m}={s["median_us"]:.2f} us' for m,s in rec['methods'].items()),flush=True)
0218         print('  practical: '+', '.join(f'{m}={s["median_us"]:.2f} us' for m,s in rec['practical_baselines'].items()),flush=True)
0219     for t in tables.values():lib.fft_table_destroy(t)
0220     print('Saved',args.output,flush=True)
