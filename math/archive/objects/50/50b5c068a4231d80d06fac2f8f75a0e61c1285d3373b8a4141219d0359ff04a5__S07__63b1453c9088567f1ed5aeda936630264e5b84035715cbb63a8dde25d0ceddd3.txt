  30     def test_dense_star_invertible_but_not_dft(self):

  34             F=dft_matrix(n,p,w)

  40     def test_shared_binomial_evaluator_full_random(self):
  41         rng=random.Random(8037)
  42         for n in (2,4,8,16,32,64,128):
  43             p,w=fixture(n)
  44             for _ in range(5):
  45                 x=[rng.randrange(p) for _ in range(n)]
  46                 y,log=eval_radix2_remainder(x,p,w)
  47                 self.assertEqual(y,dft_ref(x,p,w))
  48                 self.assertEqual(log.butterflies,n*(n.bit_length()-1)//2)
  49                 self.assertEqual(log.adds,2*log.butterflies)
  50                 self.assertEqual(log.muls,log.butterflies)
  51                 self.assertEqual(log.root_prep_muls,n-1)
  52                 self.assertEqual(log.writes,n)

  54     def test_generic_root_division_evaluator(self):
  55         rng=random.Random(991)
  56         for n in (4,8,16,32):
  57             p,w=fixture(n)
  58             for policy in ('subgroup','contiguous','noncoset16'):
  59                 for _ in range(2):
  60                     x=[rng.randrange(p) for _ in range(n)]
  61                     y,log=eval_generic_remainder(x,p,w,policy)
  62                     self.assertEqual(y,dft_ref(x,p,w))
  63                     self.assertGreater(log.plan_muls+log.plan_adds,0)
  64                     self.assertEqual(log.writes,n)

  66     def test_noncoset_partition_has_more_online_work(self):

 107 class CRTBridgeTests(unittest.TestCase):
