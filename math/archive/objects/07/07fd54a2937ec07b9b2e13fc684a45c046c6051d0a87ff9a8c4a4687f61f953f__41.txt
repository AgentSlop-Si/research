0064 def mersenne(exponent):
0065     require(exponent >= 3 and all(exponent % d for d in range(2, isqrt(exponent)+1)),
0066             'Composite Lucas-Lehmer exponent')
0067     p = 2**exponent - 1
0068     s = 4
0069     for _ in range(exponent - 2):
0070         s = (s*s-2) % p
0071     require(s == 0, 'Lucas-Lehmer failed')
0072     return p

0075 def exactness(h):
0076     require(h in (23, 25), 'Uncertified dimension')
0077     primes = [mersenne(e) for e in (61, 31, 19)]
0078     Z = 3*h-7
0079     B0 = (4*h-2)*Z + 27*(h+1)
0080     D = 3*(h+1)*(h-1)**2
0081     B = 2*(h-1)*B0
0082     bounds = {r: sum(comb(h,j)*factorial(j)*B**j*D**(r-j)
0083                      for j in range(r+1)) for r in (2,3,4)}
0084     require(bounds[2] < primes[0], 'Rank-two minor modulus insufficient')
0085     require(max(bounds[3],bounds[4]) < prod(primes), 'CRT minor modulus insufficient')
0086     require(3*(h+1)*(h-1) < min(primes), 'Denominator may vanish')
0087     return dict(h=h, primes=primes, prime_product=prod(primes),
0088                 single_numerator_bound=B0, common_denominator_bound=D,
0089                 correction_numerator_bound=B, minor_bounds=bounds,
0090                 modulus_gaps={2: primes[0]-bounds[2],
0091                               3: prod(primes)-bounds[3],4: prod(primes)-bounds[4]})

0094 def tree(edges, n):
0095     require(len(edges) == n-1 and len(set(edges)) == n-1, 'Wrong tree edge count')
0096     seen = {0}
0097     while True:
0098         new = seen | {v for u,v in edges if u in seen} | {u for u,v in edges if v in seen}
0099         if new == seen:
0100             break
0101         seen = new
0102     require(len(seen) == n, 'Disconnected restriction graph')

0163 @lru_cache(None)
0164 def logs(x):
0165     x = Q(x)
0166     require(x >= 1, 'Log domain')
0167     k = 0
0168     while x > 2:
0169         x /= 2
0170         k += 1
0171     def small(y):
0172         z = (y-1)/(y+1)
0173         lo = 2*sum((z**(2*j+1)/Q(2*j+1) for j in range(32)), Q())
0174         return lo, lo+2*z**65/(65*(1-z*z))
0175     lo,hi = small(x)
0176     lo2,hi2 = small(Q(2))
0177     lo,hi = lo+k*lo2,hi+k*hi2
0178     scale = 10**30
0179     return Q((lo*scale).numerator//(lo*scale).denominator,scale), \
0180         Q(-(-(hi*scale).numerator//(hi*scale).denominator),scale)

0183 def moment(m,W,rows,saving):
0184     require(type(saving) is Q and 0 < saving < 1, 'Exact positive saving required')
0185     lower,upper = Q(),Q()
0186     enclosure = {}
0187     for t,n in sorted(rows.items()):
0188         require(type(t) is int and type(n) is int and 0<t<m and n>0, 'Invalid child')
0189         lo,hi = logs(Q(m,t))
0190         u,v = saving*lo,saving*hi
0191         require(0 <= u <= v < 1, 'Exponential enclosure domain')
0192         weight = Q(t*n,m*W)
0193         lower += weight*(1+u+u*u/2+u*u*u/6)
0194         upper += weight*(1+v+v*v/(2*(1-v/3)))
0195         enclosure[t] = dict(lower=lo,upper=hi)
0196     return dict(saving=saving,exponent=1-saving,lower=lower,upper=upper,
0197                 strict_gap=1-upper,logarithms=enclosure)

0226                 'Invalid fixed blocks')

0230         blocks[h] -= h
