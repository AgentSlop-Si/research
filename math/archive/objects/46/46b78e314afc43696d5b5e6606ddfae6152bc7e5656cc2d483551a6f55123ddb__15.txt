0010 def main():
0011     assert is_prime(P)

0020     equations=[]

0039         for i in (0,1):

0048         else:

0050             assert len(independent)==7

0052                 'determinant_mod_p':determinant([[plus[r][c] for c in pp] for r in independent])}
0053             rec['forced_kernel']='span((1,-1,0,0,0,0,0,0))'
0054         counts[expected]+=1;records.append(rec)

0056     # and 32 fixed points in P0^{-1}P1 give rank 64-(16+32)=16 over Q.
