  20 def ident(n):
  21     return [[int(i == j) for j in range(n)] for i in range(n)]
  22 
  23 def mm(a, b, p=None):
  24     v = [[sum(a[i][k] * b[k][j] for k in range(len(b)))
  25           for j in range(len(b[0]))] for i in range(len(a))]
  26     return [[x % p for x in row] for row in v] if p else v
  27 
  28 def inv(a, p=None):
  29     n = len(a)
  30     z = [[(v % p if p else Q(v)) for v in row] + ident(n)[i]
  31          for i, row in enumerate(a)]
  32     for j in range(n):
  33         pivot = next(i for i in range(j, n) if z[i][j])
  34         z[j], z[pivot] = z[pivot], z[j]
  35         f = pow(z[j][j], -1, p) if p else 1 / z[j][j]
  36         z[j] = [(v * f % p if p else v * f) for v in z[j]]
  37         for i in range(n):
  38             if i == j:
  39                 continue
  40             f = z[i][j]
  41             z[i] = [((v - f * w) % p if p else v - f * w)
  42                     for v, w in zip(z[i], z[j])]
  43     return [row[n:] for row in z]
  44 
  45 def rk(a, p):
  46     z = [[v % p for v in row] for row in a]
  47     row = 0
  48     for col in range(len(z[0])):
  49         ix = next((i for i in range(row, len(z)) if z[i][col]), None)
  50         if ix is None:
  51             continue
  52         z[row], z[ix] = z[ix], z[row]
  53         s = pow(z[row][col], -1, p)
  54         z[row] = [v * s % p for v in z[row]]
  55         for i in range(row + 1, len(z)):
  56             s = z[i][col]
  57             z[i] = [(v - s * w) % p for v, w in zip(z[i], z[row])]
  58         row += 1
  59         if row == len(z):
  60             break
  61     return row

  63 def tri(n, d, b):
  64     return [[d if i == j else b if abs(i-j) == 1 else 0
  65              for j in range(n)] for i in range(n)]
  66 
  67 def dets(n, d, b):
  68     v = [1, d]
  69     for k in range(2, n + 1):
  70         v.append(d * v[-1] - b * b * v[-2])
  71     return v
  72 
  73 def green_closed(n, d, b):
  74     v = dets(n, d, b)
  75     return [[Q((-b)**abs(i-j) * v[min(i,j)] * v[n-1-max(i,j)], v[n])
  76              for j in range(n)] for i in range(n)]
  77 
  78 def add(a, b):
  79     return [[v+w for v, w in zip(x,y)] for x,y in zip(a,b)]
  80 
  81 def norm1(a):
  82     return max(sum(abs(a[i][j]) for i in range(len(a)))
  83                for j in range(len(a[0])))
  84 
  85 def modq(a):
  86     return [[v.numerator * pow(v.denominator, -1, P) % P for v in row]
  87             for row in a]
  88 
  89 def nz(a):
  90     return sum(bool(v) for row in a for v in row)
  91 
  92 def cuts(a):
  93     n = len(a)
  94     return max(max(rk([row[s:] for row in a[:s]], P),
  95                    rk([row[:s] for row in a[s:]], P)) for s in range(1,n))
  96 
  97 def bitrev(i, bits):
  98     return sum(((i >> k) & 1) << (bits-1-k) for k in range(bits))
  99 
 100 def butterfly(n, span, w):
 101     a = [[0] * n for _ in range(n)]
 102     half = span // 2
 103     for base in range(0,n,span):
 104         for j in range(half):
 105             x,y = base+j,base+j+half
 106             t = pow(w, j*(n//span), P)
 107             a[x][x] = a[y][x] = 1
 108             a[x][y],a[y][y] = t,(-t)%P
 109     return a

 111 def do_check():
 112     report = {'method':'No uploaded Python imported, compiled, or executed.',
 113               'sha256':[], 'rational_green':[], 'modular_green':[],
 114               'gauge':[], 'stage_bounds':[]}
 115     for line in ((SRC/'SHA256SUMS.txt').read_text().splitlines() if SRC else []):
 116         h, name = line.split('  ')
 117         good = hashlib.sha256((SRC/name).read_bytes()).hexdigest() == h
 118         assert good, name
 119         report['sha256'].append({'file':name,'matches':good})
 120     expected = json.loads((SRC/'norm_results.json').read_text()) if SRC else None
 121     for n in (2,4,8,16,32):
 122         t = tri(n,3,-1)
 123         a,b,v = green_closed(n,3,-1),green_closed(n,4,-1),green_closed(n,7,-2)
 124         s = add(a,b)
 125         si = [[Q(t[i][j],2) + Q(i==j,4) - v[i][j]/4
 126                for j in range(n)] for i in range(n)]
 127         assert mm(t,a) == ident(n)
 128         assert mm(s,si) == ident(n)
 129         assert all(x>0 for row in s for x in row)
 130         assert all(si[i][j] < 0 for i in range(n) for j in range(n) if i!=j)
 131         assert all(si[i][i] > 0 for i in range(n))
 132         ns,ni = norm1(s),norm1(si)
 133         assert ns < Q(3,2) and ni < Q(17,6) and ns*ni < Q(17,4)
 134         if expected:
 135             supplied = next(x for x in expected['exact_real_column_norm'] if x['n']==n)
 136             assert ns == Q(supplied['norm1_num'],supplied['norm1_den'])
 137             supplied = next(x for x in expected['exact_condition1'] if x['n']==n)
 138             assert ns*ni == Q(supplied['kappa_1_numerator'],supplied['kappa_1_denominator'])
 139         det_s = Q(dets(n,7,-2)[n],dets(n,3,-1)[n]*dets(n,4,-1)[n])
 140         assert 0 < det_s < Q(11,20)**(n//2)*Q(7,12)**(n%2) < 1
 141         report['rational_green'].append({'n':n,'norm1':str(ns),'inverse_norm1':str(ni),
 142             'kappa1':str(ns*ni),'kappa1_float':float(ns*ni),'det_S':str(det_s),
 143             'identities_and_signs':True})
 144         if n>=4:
 145             sm,sim = modq(s),modq(si)
 146             assert inv(sm,P)==sim
 147             assert nz(sm)==n*n and nz(sim)==n*n
 148             assert cuts(sm)<=2 and cuts(sim)<=2
 149             report['modular_green'].append({'n':n,'S_nonzeros':nz(sm),
 150                 'inverse_nonzeros':nz(sim),'S_cut_order':cuts(sm),'inverse_cut_order':cuts(sim)})
 151     for n in (4,8,16,32):
 152         L = n.bit_length()-1
 153         w = pow(3,(P-1)//n,P)
 154         u,v = list(range(1,n+1)),[3*i+1 for i in range(n)]
 155         beta = 7 * pow(1+7*sum(x*y for x,y in zip(u,v)),-1,P) % P
 156         h = [[(int(i==j)+7*u[i]*v[j])%P for j in range(n)] for i in range(n)]
 157         hi = [[(int(i==j)-beta*u[i]*v[j])%P for j in range(n)] for i in range(n)]
 158         assert mm(h,hi,P)==ident(n)
 159         base = [butterfly(n,1<<(i+1),w) for i in range(L)]
 160         dressed = []
 161         for i,B in enumerate(base):
 162             G = mm(h,B,P) if i==0 else mm(B,hi,P) if i==L-1 else mm(mm(h,B,P),hi,P)
 163             assert nz(G)==n*n
 164             dressed.append(G)
 165         recovered = mm(dressed[0],inv(base[0],P),P)
 166         assert recovered==h
 167         assert rk([[(h[i][j]-int(i==j))%P for j in range(n)] for i in range(n)],P)==1
 168         m = ident(n)
 169         for G in dressed:
 170             m = mm(G,m,P)
 171         m = [[m[i][bitrev(j,L)] for j in range(n)] for i in range(n)]
 172         f = [[pow(w,i*j,P) for j in range(n)] for i in range(n)]
 173         assert m==f
 174         rand = random.Random(20261009+n)
 175         for rep in range(12):
 176             x = [rand.randrange(P) for j in range(n)]
 177             via_m = mm(m,[[v] for v in x],P)
 178             direct = [[sum(x[j]*pow(w,k*j,P) for j in range(n))%P] for k in range(n)]
 179             assert via_m==direct
 180         report['gauge'].append({'n':n,'dense_stages':L,'full_fourier_equality':True,
 181             'rank_one_recovered':True,'fresh_vectors':12})
 182     for n in (16,64,256,1024,4096,65536,1048576):
