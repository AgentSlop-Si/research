1: """Rank certificate for nonlinear feature banks + linear readout.
2: 
3: If a bank has M=N scalar features and its linear readout computes all N
4: independent Fourier outputs on {0,1}^N, then each feature must itself be a
5: Boolean-linear function. A nonlinear degree of freedom needs an EXTRA feature.
6: This is a restricted feature-bank model, NOT a general FFT lower bound.
7: """
8: from __future__ import annotations
9: from itertools import product
10: import json
11: from pathlib import Path
12: from degree_boundary import P, direct_dft, matrix_rank_mod
13: 
14: 
15: def verify(n):
16:     assert n<=10 and n>=2 and not n & (n-1)
17:     E=[];F=[];Sq=[];linear=[]
18:     for bits in product((0,1),repeat=n):
19:         s=sum(bits)%P
20:         z=s*s%P
21:         E.append([(t+z)%P for t in bits])
22:         F.append(direct_dft(bits))
23:         Sq.append([z])
24:         linear.append(list(bits))
25:     rA=matrix_rank_mod(E)
26:     with_outputs=[e+f for e,f in zip(E,F)]
27:     with_extra=[e+sq for e,sq in zip(E,Sq)]
28:     with_extra_outputs=[e+sq+f for e,sq,f in zip(E,Sq,F)]
29:     rAF=matrix_rank_mod(with_outputs)
30:     rAE=matrix_rank_mod(with_extra)
31:     rAEF=matrix_rank_mod(with_extra_outputs)
32:     assert (rA,rAF,rAE,rAEF)==(n,n+1,n+1,n+1), (n,rA,rAF,rAE,rAEF)
33:     return {'N':n,'rank_N_nonlinear_features':rA,
34:             'rank_after_adding_all_FFT_outputs':rAF,
35:             'rank_after_adding_one_square_of_sum_feature':rAE,
36:             'rank_after_adding_square_and_FFT_outputs':rAEF,
37:             'conclusion':'N mixed features cannot linearly read out all FFT outputs; N+1 features can'}
38: 
39: if __name__=='__main__':
40:     out=[verify(n) for n in (2,4,8)]
41:     Path(__file__).with_name('feature_rank_results.json').write_text(json.dumps(out,indent=2)+'\n')
42:     print(json.dumps(out,indent=2))
