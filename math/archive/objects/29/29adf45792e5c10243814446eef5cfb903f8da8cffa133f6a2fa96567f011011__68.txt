   1 """Exact linear-arithmetic all-pass application.
   2 
   3 Input scalars must implement exact field arithmetic if exactness is required.
   4 Fractions and SymPy exact scalars work. Ordinary Python float/complex inputs
   5 produce floating-point results, not an exact-arithmetic certificate.
   6 """
   7 from fractions import Fraction
   8 
   9 class AllPassCycle:
  10     """Prepare U(M)=(I-2M)(2I-M)^-1 and its inverse V(M).
  11 
  12     order[j] is the jth coordinate around the directed cycle.
  13     weights[j] is M[order[j], order[j-1]], including the wrap at j=0.
  14     The supplied weights must have modulus one and total product -1.
  15     The caller supplies those exact weights; arbitrary phase synthesis is
  16     not part of this evaluator's promise.
  17     """
  18     def __init__(self, n, order=None, weights=None):
  19         if n < 3:
  20             raise ValueError('This evaluator is presented for N >= 3.')
  21         self.n=n
  22         self.order=list(range(n)) if order is None else list(order)
  23         if len(self.order)!=n or set(self.order)!=set(range(n)):
  24             raise ValueError('order must be a permutation of range(n)')
  25         weights=[-1]+[1]*(n-1) if weights is None else list(weights)
  26         if len(weights)!=n:
  27             raise ValueError('weights must have length N')
  28         total=1
  29         for w in weights: total*=w
  30         if total != -1:
  31             raise ValueError('cycle weight product must equal -1 exactly')
  32         self.gauge=[1]
  33         for j in range(1,n): self.gauge.append(self.gauge[-1]*weights[j])
  34         self.gauge_inverse=[1/g if not isinstance(g,int) else Fraction(1,g)
  35                             for g in self.gauge]
  36         self.half=Fraction(1,2)
  37         a=Fraction(1)
  38         for _ in range(n-1): a*=self.half
  39         self.h=1/(2+a)
  40 
  41     def apply(self, x, inverse=False):
  42         """Apply U(M), or V(M)=U(M)^-1 when inverse=True."""
  43         if len(x)!=self.n:
  44             raise ValueError('input must have length N')
  45         xx=[self.gauge_inverse[j]*x[self.order[j]] for j in range(self.n)]
  46         if inverse: xx.reverse()
  47         b=0
  48         for j in range(1,self.n): b=(xx[j]+b)*self.half
  49         z=[(xx[0]-b)*self.h]
  50         for j in range(1,self.n): z.append((xx[j]+z[-1])*self.half)
  51         yy=[2*xx[j]-3*z[j] for j in range(self.n)]
  52         if inverse: yy.reverse()
  53         out=[None]*self.n
  54         for j in range(self.n): out[self.order[j]]=self.gauge[j]*yy[j]
  55         return out
