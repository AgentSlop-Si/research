"""One fully transparent example: 1234×5678 via seven exact evaluations.
The same bilinear multiplication count is valid for arbitrary length-4 inputs.
"""
from dense_algebra import schoolbook, karatsuba, interpolation_product
from universal_dispatch import convolve
from signed_digit import shift_multiply

a=[4,3,2,1];b=[8,7,6,5]
for x in range(len(a)+len(b)-1):
    ax=sum(v*x**i for i,v in enumerate(a))
    bx=sum(v*x**i for i,v in enumerate(b))
    print(f'x={x}  A(x)={ax:>6}  B(x)={bx:>6}  product={ax*bx:>10}')
print('Schoolbook :', schoolbook(a,b))
print('Karatsuba  :', karatsuba(a,b))
print('Interpolation:',interpolation_product(a,b))
print('Adaptive   :', convolve(a,b))
print('Exact integer:',shift_multiply(1234,5678).product)
