  14 def matrix_rank_mod(matrix):
  15     if not matrix:return 0
  16     a=[row[:] for row in matrix];rows=len(a);cols=len(a[0]);r=0
  17     for j in range(cols):
  18         p=next((i for i in range(r,rows) if a[i][j]),None)
  19         if p is None:continue
  20         a[r],a[p]=a[p],a[r]
  21         factor=pow(a[r][j],P-2,P)
  22         for k in range(j,cols):a[r][k]=a[r][k]*factor%P
  23         for i in range(r+1,rows):
  24             z=a[i][j]
  25             if z:
  26                 for k in range(j,cols):a[i][k]=(a[i][k]-z*a[r][k])%P
  27         r+=1
  28         if r==rows:break
  29     return r

  32 def quadratic_reference(x):
  33     n=len(x);wn=root_of_unity(n)
  34     table=[pow(wn,-a% n,P) for a in range(n)]
  35     return [sum(x[b]*table[a*b %n] for b in range(n))%P for a in range(n)]

  38 class Phase9Tests(unittest.TestCase):
  39     def run_case(self,k,m,innov,s,seed,expect_mode=None,lam_override=None):
  40         x,metadata=make_dense_cycle(k,m,innov,seed,lam_override)
  41         answer,cert,counts=exact_dense_cycle_with_innovations(
  42             x,m,s,3,3,Costs(),random.Random(991+seed),True)
  43         self.assertIsNotNone(answer,f'{k},{m},{s},{cert}')
  44         self.assertEqual(answer,exact_reference_dft(x))
  45         if k*m<=128:self.assertEqual(answer,quadratic_reference(x))
  46         if expect_mode:self.assertEqual(cert['mode'],expect_mode)
  47         self.assertEqual(cert['innovation_total'],sum(map(len,innov.values())))
  48         self.assertEqual(cert['verification_rounds'],3)
  49         return x,metadata,cert,counts

 116     def test_h_syndrome_alias_requires_fingerprint(self):
 117         k,m=32,8
 118         x,meta=make_dense_cycle(k,m,{},seed=9)
 119         wn=root_of_unity(k);theta=[pow(wn,-j%k,P) for j in range(3)]
 120         amps=[(theta[1]-theta[2])%P,(theta[2]-theta[0])%P,
 121               (theta[0]-theta[1])%P]
 122         self.assertEqual(sum(amps)%P,0)
 123         self.assertEqual(sum(a*t for a,t in zip(amps,theta))%P,0)
 124         corrupted=x[:]
 125         for b,a in enumerate(amps):corrupted[b*m+3]=(corrupted[b*m+3]+a)%P
 126         cols=[[corrupted[b*m+t] for b in range(k)] for t in range(m)]
 127         f0,lam,tab=learn_eigenvalues(cols,k*m,Costs())
 128         ev=find_innovations(cols,lam,tab,1,Costs())
 129         self.assertEqual(sum(map(len,ev)),0)  # deliberately invisible to 2 probes!
 130         self.assertFalse(probabilistic_verify(cols,lam,ev,tab,m,Costs(),3,random.Random(281)))
 131         z,cert,_=exact_dense_cycle_with_innovations(
 132             corrupted,m,1,None,3,Costs(),random.Random(281),True)
 133         self.assertIsNone(z)
 134         self.assertEqual(cert['reason'],'independent recurrence fingerprint failed')
