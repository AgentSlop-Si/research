"""Exact structured full DFT via a sparse filtered polynomial.

Independent audit/demo, 2026-10-08. Standard-library only; no FFT or cached
spectrum. Classical generating-function algebra, not a novelty claim.
The caller supplies a prime p and an exact order-N root omega in F_p.
Run: python structured_dft.py
"""
from __future__ import annotations
import json


def _check_root(omega: int, n: int, p: int) -> None:
    """Verify order n assuming the caller's p is prime (not a primality test)."""
    if p < 2 or not 0 < n <= p - 1 or (p - 1) % n:
        raise ValueError("Need a prime p and n dividing p-1")
    if pow(omega, n, p) != 1:
        raise ValueError("omega does not have order n")
    left = n
    r = 2
    factors = []
    while r * r <= left:
        if left % r == 0:
            factors.append(r)
            while left % r == 0:
                left //= r
        r += 1
    if left > 1:
        factors.append(left)
    if any(pow(omega, n // r, p) == 1 for r in factors):
        raise ValueError("omega does not have exact order n")


def _evaluate(coefficients: list[int], z: int, p: int) -> int:
    out = 0
    for value in reversed(coefficients):
        out = (out * z + value) % p
    return out


def _batch_inverse(values: list[int], p: int) -> list[int]:
    """O(len(values)) field multiplications and one nonzero inversion."""
    if not values:
        return []
    prefixes = [1]
    for value in values:
        if value % p == 0:
            raise ValueError("Batch inversion requires nonzero values")
        prefixes.append(prefixes[-1] * value % p)
    suffix_inverse = pow(prefixes[-1], -1, p)
    inverse = [0] * len(values)
    for i in range(len(values) - 1, -1, -1):
        inverse[i] = prefixes[i] * suffix_inverse % p
        suffix_inverse = suffix_inverse * values[i] % p
    return inverse


def structured_dft(a: list[int], q: list[int], p: int, omega: int):
    """Return [A(omega**k)] exactly, with diagnostic facts.

    Correct for every a and every q with q[0]=1, including dense residuals.
    Fast only if deg(q) and nnz(q*a) are small. All right-boundary terms are
    retained. Root validation assumes primality of p, explicitly supplied.
    """
    if not a or not q or q[0] % p != 1:
        raise ValueError("Require nonempty a, q and q[0]=1 in F_p")
    n = len(a)
    _check_root(omega, n, p)
    a = [v % p for v in a]
    q = [v % p for v in q]
    while len(q) > 1 and q[-1] == 0:
        q.pop()
    degree = len(q) - 1
    # Full ordinary polynomial product, including coefficients n through n+d-1.
    filtered = [0] * (n + degree)
    for j, c in enumerate(q):
        if c:
            for i, v in enumerate(a):
                filtered[i + j] = (filtered[i + j] + c * v) % p
    # At n-th roots, reduce the numerator modulo z^n-1, not modulo z^n.
    folded = [0] * n
    for i, v in enumerate(filtered):
        folded[i % n] = (folded[i % n] + v) % p
    terms = [(i, v) for i, v in enumerate(folded) if v]
    numerators = [0] * n
    for exponent, coefficient in terms:
        step = pow(omega, exponent, p)
        value = coefficient
        for k in range(n):
            numerators[k] = (numerators[k] + value) % p
            value = value * step % p
    roots = [1]
    for _ in range(n - 1):
        roots.append(roots[-1] * omega % p)
    denominators = [_evaluate(q, z, p) for z in roots]
    regular = [k for k, denominator in enumerate(denominators) if denominator]
    singular = [k for k, denominator in enumerate(denominators) if not denominator]
    reciprocals = _batch_inverse([denominators[k] for k in regular], p)
    out = [0] * n
    for k, reciprocal in zip(regular, reciprocals):
        out[k] = numerators[k] * reciprocal % p
    # At most degree exceptional bins; no derivatives or characteristic caveat.
    for k in singular:
        out[k] = _evaluate(a, roots[k], p)
    return out, {
        "length": n,
        "filter_degree": degree,
        "filtered_support": sum(v != 0 for v in filtered),
        "folded_support": len(terms),
        "recurrence_holds_interior": all(v == 0 for v in filtered[degree:n]),
        "singular_bins": singular,
    }


def _direct_dft(a, p, omega):
    # Separate reference: explicit coefficient/root-power sum.
    return [sum(v * pow(omega, k * j, p) for j, v in enumerate(a)) % p
            for k in range(len(a))]


def self_test():
    p, n, omega = 97, 8, 64
    cases = [
        ("constant: singular DC", [3] * n, [1, -1]),
        ("arithmetic: singular DC", [2 + 3 * i for i in range(n)], [1, -2, 1]),
        ("geometric: singular non-DC", [pow(omega, i, p) for i in range(n)], [1, -omega]),
        ("Fibonacci", [1, 1, 2, 3, 5, 8, 13, 21], [1, -1, -1]),
        ("piecewise constant", [3, 3, 3, 5, 5, 5, 5, 2], [1, -1]),
        ("dense residual: correctness only", [7, 0, 21, 4, 8, 19, 91, 3], [1, -2, 3]),
    ]
    results = []
    for name, a, q in cases:
        got, facts = structured_dft(a, q, p, omega)
        assert got == _direct_dft(a, p, omega), name
        results.append({"case": name, "spectrum": got, **facts})
    assert results[0]["spectrum"] == [24, 0, 0, 0, 0, 0, 0, 0]
    assert results[0]["singular_bins"] == [0]
    assert results[2]["singular_bins"] == [7]
    assert results[2]["spectrum"] == [0, 0, 0, 0, 0, 0, 0, 8]
    # Boundary regression: A=3(1+...+z^7), Q=1-z, D=3-3z^8.
    # Omitting -3z^8 gives false nonzero bins when Q(z) != 0.
    wrongly_omitted_tail_bin_one = 3 * pow((1 - omega) % p, -1, p) % p
    assert wrongly_omitted_tail_bin_one != results[0]["spectrum"][1]
    for a, q in [([7], [1]), ([0], [1, -1]), ([3, 4], [1, 2])]:
        small_n = len(a)
        small_root = 1 if small_n == 1 else 96
        got, _ = structured_dft(a, q, p, small_root)
        assert got == _direct_dft(a, p, small_root)
    try:
        structured_dft([1] * 8, [1], 97, 1)
    except ValueError:
        pass
    else:
        raise AssertionError("Nonprimitive root was accepted")
    return {"status": "passed", "prime": p, "omega": omega,
            "cases": results,
            "boundary_regression_false_bin_1": wrongly_omitted_tail_bin_one,
            "small_edge_cases": 3, "invalid_root_rejected": True}


if __name__ == "__main__":
    print(json.dumps(self_test(), indent=2))
