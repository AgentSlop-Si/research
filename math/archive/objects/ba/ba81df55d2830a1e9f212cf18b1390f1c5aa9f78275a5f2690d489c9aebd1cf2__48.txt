   6 def matzero(n,m=None):return [[0]*(n if m is None else m) for _ in range(n)]
   7 def mm(A,B,p):
   8     C=matzero(len(A),len(B[0]))
   9     for i,row in enumerate(A):
  10         for k,x in enumerate(row):
  11             if x:
  12                 for j,y in enumerate(B[k]):
  13                     if y:C[i][j]=(C[i][j]+x*y)%p
  14     return C
  15 def mv(A,x,p):return [sum(a*b for a,b in zip(row,x))%p for row in A]
  16 def rank(A,p):
  17     A=[row[:] for row in A];r=0
  18     for j in range(len(A[0])):
  19         k=next((k for k in range(r,len(A)) if A[k][j]),None)
  20         if k is None:continue
  21         A[k],A[r]=A[r],A[k];z=pow(A[r][j],-1,p);A[r]=[x*z%p for x in A[r]]
  22         for k in range(r+1,len(A)):
  23             z=A[k][j];A[k]=[(x-z*y)%p for x,y in zip(A[k],A[r])]
  24         r+=1
  25         if r==len(A):break
  26     return r
  27 def plus(x,y,p,k=1):return [(a+k*b)%p for a,b in zip(x,y)]
  28 def qmatrix(N,w,p):
  29     Q=matzero(N)
  30     for j in range(N):Q[2*j%N][j]=(Q[2*j%N][j]+1)%p;Q[(2*j+1)%N][j]=(Q[(2*j+1)%N][j]+pow(w,j,p))%p
  31     return Q
  32 def oracle_feedback(N,b,w,p,H):
  33     r=N>>b;Q=qmatrix(N,w,p);inds=list(range(0,N,1<<b));E=[[int(i==s) for s in inds] for i in range(N)];state=mm(Q,E,p)
  34     A=[state[s][:] for s in inds]
  35     for s in inds:state[s]=[0]*r
  36     out=[matzero(r) for _ in range(H+1)]
  37     for L in range(2,H+1):
  38         state=mm(Q,state,p);out[L]=[state[s][:] for s in inds]
  39         for s in inds:state[s]=[0]*r
  40     return Q,A,out

  42 class Plan:
  43     def __init__(self,N,b,w,p):
  44         self.N=N;self.n=N.bit_length()-1;self.b=b;self.r=N>>b;self.d=self.n-b;self.p=p
  45         eta=w;self.setup_mul=0
  46         for _ in range(b):eta=eta*eta%p;self.setup_mul+=1
  47         zeta=w*pow(eta,-1,p)%p;self.setup_mul+=1;self.setup_inv=1
  48         self.ph=[1]*self.r;zp=[1]*self.r
  49         for k in range(1,self.r):self.ph[k]=self.ph[k-1]*eta%p;zp[k]=zp[k-1]*zeta%p;self.setup_mul+=2
  50         self.fix=[1]*self.r;tail=[0]*self.r;run=[0]*self.r;self.valid=[False]*self.r
  51         for k in range(1,self.r):
  52             a=k>>1;self.fix[k]=self.fix[a]
  53             if k&1:self.fix[k]=self.fix[k]*zp[a]%p;self.setup_mul+=1
  54             else:tail[k]=tail[a]+1
  55             run[k]=max(run[a],tail[k]);self.valid[k]=bool(k&1) and run[k]<b
  56         assert self.setup_mul==b+1+2*(self.r-1)+self.r//2
  57         self.count={'state_visits':0,'route_checks':0,'route_updates':0,'mul':0,'add':0,'tree_updates':0,'output_scalars':0}
  58     def step(self,v):
  59         out=[0]*self.r;p=self.p
  60         for j,x in enumerate(v):out[2*j%self.r]=(out[2*j%self.r]+x)%p;out[(2*j+1)%self.r]=(out[(2*j+1)%self.r]+self.ph[j]*x)%p
  61         self.count['state_visits']+=self.r;self.count['mul']+=self.r;self.count['add']+=2*self.r
  62         return out
  63     def route(self,v,t,target):
  64         for a,x in enumerate(v):
  65             k=a% (1<<t);self.count['route_checks']+=1
  66             if k>=1<<(t-1) and self.valid[k]:
  67                 y=((a-k)*(1<<self.b)+k)%self.r;target[y]=(target[y]+self.fix[k]*x)%self.p
  68                 self.count['route_updates']+=1;self.count['mul']+=1;self.count['add']+=1
  69     def const_feedback(self,v,H):
  70         res=[[0]*self.r for _ in range(H+1)];cur=v[:]
  71         for t in range(1,max(H-self.b,0)+1):cur=self.step(cur);self.route(cur,t,res[self.b+t])
  72         return res
  73     def solve(self,f,H):
  74         assert H<=self.n and len(f)==H+1 and all(len(x)==self.r for x in f)
  75         rho=[x[:] for x in f];sol=[]
  76         for j in range(H+1):
  77             v=rho[j][:];sol.append(v);self.count['output_scalars']+=self.r
  78             if j<H:
  79                 for a,x in enumerate(v):k=2*a%self.r;rho[j+1][k]=(rho[j+1][k]+x)%self.p
  80                 self.count['tree_updates']+=self.r;self.count['add']+=self.r
  81             cur=v
  82             for t in range(1,max(H-j-self.b,0)+1):cur=self.step(cur);self.route(cur,t,rho[j+self.b+t])
  83         return sol

  85 def source_jet(x,N,b,Q,p,H):
  86     inds=list(range(0,N,1<<b));state=x[:]
  87     for i in inds:state[i]=0
  88     out=[[x[i] for i in inds]]
  89     for L in range(1,H+1):
  90         state=mv(Q,state,p);out.append([state[i] for i in inds])
  91         for i in inds:state[i]=0
  92     return out
  93 
  94 def apply_oracle(v,A,K,p,H):
  95     out=[x[:] for x in v]
  96     for j in range(H+1):
  97         if j<H:out[j+1]=plus(out[j+1],mv(A,v[j],p),p,-1)
  98         for L in range(2,H-j+1):out[j+L]=plus(out[j+L],mv(K[L],v[j],p),p,-1)
  99     return out
