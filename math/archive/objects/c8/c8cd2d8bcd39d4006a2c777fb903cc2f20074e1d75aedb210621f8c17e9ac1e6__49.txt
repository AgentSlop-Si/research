0008 def route(m):
0009     n=m*m;counts=[0]*(n+1);area=[(i//m+1)*(i%m+1) for i in range(n)]
0010     for a in area:counts[a]+=1
0011     offsets=[];v=0
0012     for c in counts:offsets.append(v);v+=c
0013     order=[0]*n
0014     for i,a in enumerate(area):order[offsets[a]]=i;offsets[a]+=1
0015     sigma=[0]*n
0016     for k,i in enumerate(order):sigma[i]=order[n-1-k]
0017     return sigma

0019 def solve(x,m,q,transpose=False):
0020     n=m*m;y=[0]*n
0021     if not transpose:
0022         for i in range(n):
0023             t,u=divmod(i,m)
0024             y[i]=(x[i]+(pow(q,u,p)*y[i-m] if t else 0)+(y[i-1] if u else 0))%p
0025     else:
0026         for i in range(n-1,-1,-1):
0027             t,u=divmod(i,m)
0028             y[i]=(x[i]+(pow(q,u,p)*y[i+m] if t+1<m else 0)+(y[i+1] if u+1<m else 0))%p
0029     return y

0031 def stage(x,m,q,alpha,beta,transpose=False):
0032     sigma=route(m);qi=pow(q,-1,p)
0033     a=solve(x,m,q,transpose)
0034     if not transpose:
0035         tmp=solve(x[::-1],m,qi);b=[tmp[i] for i in sigma]
0036     else:
0037         tmp=solve([x[i] for i in sigma],m,qi,True);b=tmp[::-1]
0038     return [(alpha*u+beta*v)%p for u,v in zip(a,b)]

0040 def normalize(T,F,r,c):
0041     n=len(T)
0042     if any(T[r][j]==0 for j in range(n)) or any(T[i][c]==0 for i in range(n)):
0043         return {'status':'reference_zero_impossible','valid':False}
0044     E=[F[r][j]*pow(T[r][j],-1,p)%p for j in range(n)]
0045     D=[F[i][c]*pow(T[i][c]*E[c]%p,-1,p)%p for i in range(n)]
0046     U=[[D[i]*T[i][j]*E[j]%p for j in range(n)] for i in range(n)]
0047     assert D[r]==1 and U[r]==F[r] and all(U[i][c]==F[i][c] for i in range(n))
0048     mismatch=sum(U[i][j]!=F[i][j] for i in range(n) for j in range(n))
0049     cross_mismatch=0
0050     for i in range(n):
0051         for j in range(n):
0052             a=T[i][j]*T[r][c]*pow(T[i][c]*T[r][j]%p,-1,p)%p
0053             b=F[i][j]*F[r][c]*pow(F[i][c]*F[r][j]%p,-1,p)%p
0054             assert (U[i][j]==F[i][j])==(a==b)
0055             cross_mismatch+=a!=b
0056     assert mismatch==cross_mismatch
0057     return {'status':'normalized','valid':mismatch==0,'mismatches':mismatch,'cross_ratio_mismatches':cross_mismatch}

0064         if not tr:return [diag[i]*x[perm[i]]%p for i in range(n)]
