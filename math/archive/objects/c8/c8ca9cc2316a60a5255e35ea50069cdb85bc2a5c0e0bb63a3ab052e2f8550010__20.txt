   1 import json,time
   2 import numpy as np
   3 try:
   4     from scipy.fft import fft

   6     from numpy.fft import fft
   7 from lazy_spectrum import LazySpectrum, BufferedSpectrum, suggest_flush_threshold
   8 
   9 rng=np.random.default_rng(202610092)
  10 rows=[]
  11 
  12 def run_once(fn):
  13     t0=time.perf_counter_ns();out=fn();return (time.perf_counter_ns()-t0)/1e6,out
  14 
  15 for n in [65536,262144]:
  16   updates=512;q=64
  17   for H in [4,16,64]:
  18     x0=rng.standard_normal(n)+1j*rng.standard_normal(n)
  19     locs=rng.choice(n,size=updates,replace=False)
  20     vals=rng.normal(scale=.05,size=updates)+1j*rng.normal(scale=.05,size=updates)
  21     bins=np.sort(rng.choice(n,size=q,replace=False))
  22 
  23     def fresh():
  24       x=x0.copy();_=fft(x);out=None
  25       for i,(j,d) in enumerate(zip(locs,vals),1):
  26         x[j]+=d
  27         if i%H==0:out=fft(x)[bins]
  28       return out
  29 
  30     def never_flush():
  31       x=x0.copy();s=LazySpectrum.from_spectrum(fft(x));out=None
  32       for i,(j,d) in enumerate(zip(locs,vals),1):
  33         x[j]+=d;s.add_edit(int(j),complex(d))
  34         if i%H==0:out=s.bins(bins)
  35       return out
  36 
  37     tf,of=run_once(fresh)
  38     tn,on=run_once(never_flush)
  39     best=(1e99,None,None)
  40     candidates=sorted(set([16,32,64,128,256,512,suggest_flush_threshold(n,q,H,backend='native')]))
  41     candrows=[]
  42     for B in candidates:
  43       if B>updates: continue
  44       def buffered(B=B):
  45         x=x0.copy();s=BufferedSpectrum.from_spectrum(fft(x),flush_threshold=B,flush_method='native');out=None
  46         for i,(j,d) in enumerate(zip(locs,vals),1):
  47           x[j]+=d;s.add_edit(int(j),complex(d))
  48           if i%H==0:out=s.bins(bins)
  49         return out,s.flush_count,s.pending_count
  50       tb,ob=run_once(buffered)
  51       candrows.append((B,tb,ob[1],ob[2]))
  52       if tb<best[0]:best=(tb,B,ob)
  53     x=x0.copy();x[locs]+=vals;truth=fft(x)[bins]
  54     assert np.allclose(of,truth,rtol=1e-9,atol=1e-7)
  55     assert np.allclose(on,truth,rtol=1e-9,atol=1e-7)
  56     assert np.allclose(best[2][0],truth,rtol=1e-9,atol=1e-7)
  57     row=dict(n=n,updates=updates,bins=q,updates_per_query=H,fresh_ms=tf,never_flush_ms=tn,
  58              best_buffered_ms=best[0],best_threshold=best[1],best_flush_count=best[2][1],
  59              buffered_vs_fresh=tf/best[0],buffered_vs_never=tn/best[0],proxy_threshold=suggest_flush_threshold(n,q,H,backend='native'),
  60              candidates=[{'threshold':b,'ms':t,'flushes':f,'pending':p} for b,t,f,p in candrows])
  61     rows.append(row);print(row)
  62 with open('benchmark_buffered_results.json','w') as f:json.dump(rows,f,indent=2)
