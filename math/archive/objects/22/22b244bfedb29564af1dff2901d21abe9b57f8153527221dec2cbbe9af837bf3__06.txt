  13 def packed_sums(rows, U, word_bits=64):
  14     G, B = len(rows), len(rows[0])
  15     q = (2*U*G).bit_length()
  16     lanes = word_bits//q
  17     assert lanes >= 1
  18     accum = [0]*((B+lanes-1)//lanes)
  19     for row in rows:
  20         for base in range(0, B, lanes):
  21             word = sum((row[t]+U) << (q*(t-base))
  22                        for t in range(base, min(base+lanes, B)))
  23             assert 0 <= word < 1 << word_bits
  24             accum[base//lanes] += word
  25             assert accum[base//lanes] < 1 << word_bits
  26     got = [((accum[t//lanes] >> (q*(t%lanes))) & ((1<<q)-1))-G*U
  27            for t in range(B)]
  28     want = [sum(row[t] for row in rows) for t in range(B)]
  29     assert got == want
  30     return got

  44 phase_cases = 0

  51             assert 2*distance <= N

  54 decoder_cases = 0
  55 max_error = max_ratio = 0.0
  56 phases = [1,-1,1j,-1j]
  57 for N in [1,2,3,5,8,17,31,64,127]:
  58     for B in [1,3,8,17,64,129]:

  60             h1,h2 = 1/32,1/64
  61             Q0 = 256
  62             Q1 = 1 << math.ceil(math.log2(256*B))
  63             U = 33
  64             atoms = [(rng.randrange(N),rng.randrange(4)) for _ in range(g*G)]
  65             bound = math.pi/Q0 + math.pi*(B-1)/Q1 + (h1+h2)/math.sqrt(2)
  66             for m0 in range(0,N,B):
  67                 count=min(B,N-m0)
  68                 real_rows,imag_rows=[],[]
  69                 for group in range(G):
  70                     vals=[]
  71                     for t in range(count):
  72                         val=0j
  73                         for j,s in atoms[group*g:(group+1)*g]:
  74                             qa=((2*Q0*((j*m0)%N)+N)//(2*N))%Q0
  75                             qb=((2*Q1*j+N)//(2*N))%Q1
  76                             val += phases[s]*cmath.exp(-2j*math.pi*(qa/Q0+t*qb/Q1))
  77                         vals.append(val/g)
  78                     real_rows.append([round(v.real/h1) for v in vals])
  79                     imag_rows.append([round(v.imag/h1) for v in vals])
  80                 sr=packed_sums(real_rows,U)
  81                 si=packed_sums(imag_rows,U)
  82                 for t in range(count):

  84                                    round((h1*si[t]/G)/h2)*h2)

  86                               for j,s in atoms)/(g*G)
  87                     error=abs(approx-exact)

  89                     max_error=max(max_error,error)
  90                     max_ratio=max(max_ratio,error/bound)
  91             decoder_cases += 1
