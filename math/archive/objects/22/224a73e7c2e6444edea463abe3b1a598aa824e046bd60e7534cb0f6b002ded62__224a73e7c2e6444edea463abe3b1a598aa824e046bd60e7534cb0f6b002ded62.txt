"""Two-sided Fourier support elimination for globally separated rank-one stages.

F_n = B P A, C=A^-1, B=F_n C P^-1.
Permutation p has p[0]=n-1,p[1]=0,p[2:]=1..n-2 in any order.
Work on nonzero chart: diagonal(C)=1,L[n-1]=1,U[0]=1; t=V0!=0,
s=h0!=0, and S_k!=0 for all k=1..n-2. Also require det C,
regular inverse corners, and B corners !=0 for a genuine factorization.

Raw variables: t, V1..V(n-2), h0..h(n-2), beta, gamma  => 2n.
Construct L and eta analytically so the n-3 lower j=1 and n-3 upper
row=1 equations of B vanish identically.
Remaining equations (n-3)(n-4); exact matrix identities, not FFT speedup.
"""
from __future__ import annotations
import json
import numpy as np
from pathlib import Path

P8=[7,0,6,3,4,1,2,5]

def primitive_root(p,n):
    assert (p-1)%n==0
    primes=[q for q in range(2,n+1) if n%q==0 and all(q%d for d in range(2,int(q**.5)+1))]
    for a in range(2,p):
        w=pow(a,(p-1)//n,p)
        if pow(w,n,p)==1 and all(pow(w,n//q,p)!=1 for q in primes):return w
    raise RuntimeError('no primitive root')

def factor_from_dual(n,p,omega,perm,y):
    assert len(y)==2*n
    assert perm[0]==n-1 and perm[1]==0 and set(perm[2:])==set(range(1,n-1))
    t=y[0]%p
    V=[t]+[x%p for x in y[1:n-1]]+[0]
    H=[x%p for x in y[n-1:2*n-2]]
    assert len(H)==n-1
    beta,gamma=y[-2]%p,y[-1]%p
    s=H[0]
    assert t!=0 and s!=0
    z=omega%p; xi=pow(z,-1,p)
    D=t*(z-1)%p
    q=[(pow(z,k,p)-gamma)%p for k in range(n)]
    bb=(t-1+beta*(s-1))*pow(z-1,-1,p)%p
    aa=(1-beta*s-bb)%p
    L=[0]+[(aa+bb*pow(xi,k,p)+beta*H[k])*pow(t,-1,p)%p for k in range(1,n-1)]+[1]
    S=[sum(q[i]*H[i] for i in range(k))%p for k in range(1,n-1)]
    if any(x==0 for x in S): raise ZeroDivisionError('S_k anchor zero')
    R=[sum(q[i]*L[i] for i in range(k+1,n))%p for k in range(1,n-1)]
    ETA=[0]+[-(q[k]+V[k]*R[k-1])*pow(S[k-1],-1,p)%p for k in range(1,n-1)]+[1]
    U=[1]+[H[k]*pow(s,-1,p)%p for k in range(1,n-1)]+[0]
    W=[0]+[s*ETA[k]%p for k in range(1,n)]
    C=np.eye(n,dtype=np.int64)
    for i in range(n):
        for j in range(n):
            if i>j: C[i,j]=L[i]*V[j]%p
            elif i<j:C[i,j]=U[i]*W[j]%p
    F=np.array([[pow(z,i*j,p) for j in range(n)] for i in range(n)],dtype=np.int64)
    B=F@C[:,np.array(perm)]%p
    assert all((B[i,1]-beta*B[i,0])%p==0 for i in range(2,n))
    assert all((B[1,j]-gamma*B[0,j])%p==0 for j in range(2,n))
    return C,B,dict(t=t,s=s,beta=beta,gamma=gamma,L=L,V=V,U=U,W=W,S=S,ETA=ETA,perm=perm,D=D)

def constraints(B,p):
    n=len(B); lower=[];upper=[]
    for i in range(2,n-1):
        for j in range(1,i):
            lower.append((f'lower_{i}_{j}',(int(B[i,j])*int(B[n-1,0])-int(B[i,0])*int(B[n-1,j]))%p))
    for i in range(1,n-2):
        for j in range(i+1,n-1):
            upper.append((f'upper_{i}_{j}',(int(B[i,j])*int(B[0,n-1])-int(B[i,n-1])*int(B[0,j]))%p))
    return lower,upper

def test_mod(n=8,prime=12289,repeats=100,seed=24001):
    w=primitive_root(prime,n)
    rng=np.random.default_rng(seed)
    perm=P8 if n==8 else [n-1,0]+list(map(int,rng.permutation(np.arange(1,n-1))))
    vals=[]
    positive=0
    for m in range(repeats):
        # Integer sampling avoids numpy matmul overflow modulo >~30k.
        y=list(map(int,rng.integers(1,prime,size=2*n,dtype=np.int64)))
        try:C,B,data=factor_from_dual(n,prime,w,perm,y)
        except ZeroDivisionError:continue
        lower,upper=constraints(B,prime)
        assert len(lower)==(n-2)*(n-3)//2
        assert len(upper)==(n-2)*(n-3)//2
        assert all(v==0 for name,v in lower if name.endswith('_1'))
        assert all(v==0 for name,v in upper if name.startswith('upper_1_'))
        assert sum(name.endswith('_1') for name,_ in lower)==n-3
        assert sum(name.startswith('upper_1_') for name,_ in upper)==n-3
        remaining=[v for name,v in lower+upper if not (name.endswith('_1') or name.startswith('upper_1_'))]
        assert len(remaining)==(n-3)*(n-4)
        assert any(remaining) # almost sure; reject accidental random root
        vals.append((name,float(len(remaining))) for name in (f'n={n}',))
        positive+=1
    assert positive>0
    return dict(n=n,prime=prime,primitive_root=w,perm=perm,checked=positive,variables=2*n,
                eliminated=2*(n-3),remaining_equations=(n-3)*(n-4),status='verified elimination only')

if __name__=='__main__':
    out=[]
    for n in (8,16):
        for prime in (12289,65537):
            out.append(test_mod(n,prime,40,20261009+n+prime))
    print(json.dumps(out,indent=2))
    Path(__file__).with_name('dual_support_results.json').write_text(json.dumps(out,indent=2)+'\n')
