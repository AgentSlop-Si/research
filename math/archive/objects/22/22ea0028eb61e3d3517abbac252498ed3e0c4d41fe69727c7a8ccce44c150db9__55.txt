87:     def test_sub_fft_cost_regimes(self) -> None:
88:         # For powers of two N, choose power-of-two M near log N or log log N.
89:         # The normalized cost must trend to zero along both regimes.
90:         ratios_fast = []
91:         ratios_trade = []
92:         for log2_n in [32, 64, 128, 256, 512, 1024]:
93:             m_fast = 1 << int(math.floor(math.log2(log2_n)))
94:             m_trade = 1 << max(1, int(math.floor(math.log2(math.log2(log2_n)))))
95:             # Divide the proxy by N before evaluating, avoiding gigantic integers.
96:             fast_per_n = 1.0 + math.log2(m_fast) + (log2_n - math.log2(m_fast)) / m_fast
97:             trade_per_n = 1.0 + math.log2(m_trade) + (log2_n - math.log2(m_trade)) / m_trade
98:             ratios_fast.append(fast_per_n / log2_n)
99:             ratios_trade.append(trade_per_n / log2_n)
100:         self.assertLess(ratios_fast[-1], ratios_fast[0])
101:         self.assertLess(ratios_trade[-1], ratios_trade[0])
102:         self.assertLess(ratios_fast[-1], 0.03)
103:         self.assertLess(ratios_trade[-1], 0.2)
