   1 """Apply the all-forward t=N/2 boundary product in O(N) scalar arithmetic.
   2 
   3 The formula is scalar-unitary and two-sparse, not a Fourier algorithm.
   4 Supply exact field scalars for exact results; floating-point input remains
   5 floating-point. Coefficients have Theta(N) bits, so no linear bit-cost claim.
   6 """
   7 from fractions import Fraction
   8 
   9 
  10 def power_of_two_by_squaring(exponent):
  11     """O(log exponent) scalar multiplications; integers are not bounded-word."""
  12     answer, base = 1, 2
  13     while exponent:
  14         if exponent & 1:
  15             answer *= base
  16         exponent >>= 1
  17         if exponent:
  18             base *= base
  19     return answer
  20

  22 class FusedHalfCycle:
  23     """Prepare product_{lambda^(N/2)=1} K(lambda M) for even N.
  24 
  25     order[j] is the jth coordinate around the directed cycle.
  26     weights[j] is M[order[j],order[j-1]], with the wrap at j=0.
  27     Supplied weights must be unit modulus with total product -1.
  28     Preparation uses O(N) scalar operations for the weighted-cycle gauge,
  29     plus O(log N) scalar multiplications to construct 2^(N/2).
  30     """
  31     def __init__(self, n, order=None, weights=None):
  32         if n < 2 or n % 2:
  33             raise ValueError('N must be positive and even, with N >= 2')
  34         self.n = n
  35         self.m = n//2
  36         self.order = list(range(n)) if order is None else list(order)
  37         if len(self.order) != n or set(self.order) != set(range(n)):
  38             raise ValueError('order must be a permutation of range(N)')
  39         weights = [-1]+[1]*(n-1) if weights is None else list(weights)
  40         if len(weights) != n:
  41             raise ValueError('weights must have length N')
  42         total = 1
  43         for weight in weights:
  44             total *= weight
  45         if total != -1:
  46             raise ValueError('cycle weight product must equal -1 exactly')
  47         self.gauge = [1]
  48         for j in range(1, n):
  49             self.gauge.append(self.gauge[-1]*weights[j])
  50         self.gauge_inverse = [Fraction(1,g) if isinstance(g,int) else 1/g
  51                               for g in self.gauge]
  52         a = power_of_two_by_squaring(self.m)
  53         self.diagonal = Fraction(a+1, 2)
  54         self.halfcycle = Fraction(a-1, 2)
  55 
  56     def apply(self, x):
  57         if len(x) != self.n:
  58             raise ValueError('input must have length N')
  59         xx = [self.gauge_inverse[j]*x[self.order[j]] for j in range(self.n)]
  60         output = [None]*self.n
  61         for j in range(self.n):
  62             shifted = -xx[j+self.m] if j < self.m else xx[j-self.m]
  63             value = self.diagonal*xx[j]+self.halfcycle*shifted
  64             output[self.order[j]] = self.gauge[j]*value
  65         return output
