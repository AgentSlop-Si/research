  10 def rank(A,p):
  11     A=[list(row) for row in A]
  12     if not A:return 0
  13     m,n=len(A),len(A[0]);q=0
  14     for j in range(n):
  15         pivot=next((i for i in range(q,m) if A[i][j]%p),None)
  16         if pivot is None:continue
  17         A[q],A[pivot]=A[pivot],A[q]
  18         inv=pow(A[q][j],-1,p)
  19         A[q]=[x*inv%p for x in A[q]]
  20         for i in range(q+1,m):
  21             c=A[i][j]%p
  22             A[i]=[(x-c*y)%p for x,y in zip(A[i],A[q])]
  23         q+=1
  24         if q==m:break
  25     return q

  27 def inv(A,p):
  28     n=len(A);T=[list(A[i])+[int(i==j) for j in range(n)] for i in range(n)]
  29     for j in range(n):
  30         k=next(i for i in range(j,n) if T[i][j]%p)
  31         T[j],T[k]=T[k],T[j]; c=pow(T[j][j],-1,p)
  32         T[j]=[c*x%p for x in T[j]]
  33         for i in range(n):
  34             if i!=j:
  35                 c=T[i][j];T[i]=[(x-c*y)%p for x,y in zip(T[i],T[j])]
  36     return [r[n:] for r in T]

  38 def mul(A,B,p):return [[sum(x*y for x,y in zip(row,col))%p for col in zip(*B)] for row in A]
  39 def sub(A,R,H):return [[A[i][j] for j in H] for i in R]
  40 def qs(A,p):

  45 def root(n,p):
  46     prime_divisors=[d for d in range(2,n+1) if n%d==0 and all(d%e for e in range(2,math.isqrt(d)+1))]
  47     for b in range(2,p):
  48         z=pow(b,(p-1)//n,p)
  49         if pow(z,n,p)==1 and all(pow(z,n//d,p)!=1 for d in prime_divisors):return z
  50     raise ValueError('No primitive root')

  52 def gate_values(pi,u,v,ranks):
  53     cu,cd,bu,bd=ranks;n=len(pi);a=min(pi[u:v+1]);b=max(pi[u:v+1]);t=v-u+1
  54     budget=(bu if u else 0)+(bd if v+1<n else 0)
  55     return ((min(t,a),budget+(cu if a else 0)),
  56             (min(t,n-1-b),budget+(cd if b+1<n else 0)),
  57             (min(t,n-1-b+a),budget+(cu if a else 0)+(cd if b+1<n else 0)))

  62 def rolling_extrema(a,width):
  63     lo,hi=deque(),deque()
  64     for j,x in enumerate(a):
  65         while lo and a[lo[-1]]>=x:lo.pop()
  66         while hi and a[hi[-1]]<=x:hi.pop()
  67         lo.append(j);hi.append(j)
  68         while lo and lo[0]<=j-width:lo.popleft()
  69         while hi and hi[0]<=j-width:hi.popleft()
  70         if j+1>=width:yield j-width+1,a[lo[0]],a[hi[0]]

  72 def fast_screen(pi,ranks=(1,1,1,1)):
  73     """Exact all-window reduction: at most six endpoint and three sliding tests."""
  74     n=len(pi);cu,cd,bu,bd=ranks
  75     for typ,cr in enumerate((cu,cd,cu+cd)):
  76         for side,br in ((0,bd),(1,bu)):
  77             w=br+cr+1
  78             if w<n:
  79                 u,v=(0,w-1) if side==0 else (n-w,n-1)
  80                 x,y=gate_values(pi,u,v,ranks)[typ]
  81                 if x>y:return False
  82         w=bu+bd+cr+1
  83         if w<=n-2:
  84             for u,a,b in rolling_extrema(pi,w):
  85                 if 1<=u and u+w<=n-1:
  86                     if typ==0 and a>=w:return False
  87                     if typ==1 and n-1-b>=w:return False
  88                     if typ==2 and a>0 and b<n-1 and n-1-b+a>=w:return False
  89     return True

  95 def general_checks(M,C,B,pi,p):
  96     n=len(pi);cu,cd=qs(C,p);bu,bd=qs(B,p)

 102             R=list(range(u))+list(range(v+1,n));beta=(bu if u else 0)+(bd if v+1<n else 0)

 108 def ct_control(a,b,p):
 109     """Derive two stages from j=s+a*t, k=q+b*p, with dyadic small orders."""
 110     n=a*b;z=root(n,p)

 114     after_first=[(s,q) for s in range(a) for q in ob]
 115     after_shuffle=[(s,q) for q in range(b) for s in oa]
 116     output_pairs=[(ell,q) for q in range(b) for ell in oa]
 117     A=[[pow(z,a*q*t,p) if s==ss else 0 for ss,t in input_pairs] for s,q in after_first]
 118     B=[[pow(z,(q+b*ell)*s,p) if q==qq else 0 for s,qq in after_shuffle] for ell,q in output_pairs]
