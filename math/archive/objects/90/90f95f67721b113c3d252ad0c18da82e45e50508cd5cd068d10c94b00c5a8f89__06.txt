0024 def factors(n):
0025     out = {}
0026     p = 2
0027     while p*p <= n:
0028         while n % p == 0:
0029             out[p] = out.get(p, 0) + 1
0030             n //= p
0031         p += 1
0032     if n > 1:
0033         out[n] = out.get(n, 0) + 1
0034     return out

0037 def phi(n):
0038     r = n
0039     for p in factors(n):
0040         r -= r//p
0041     return r

0044 def mu(n):
0045     f = factors(n)
0046     return 0 if max(f.values(), default=0) > 1 else (-1)**len(f)

0049 def divisors(n):
0050     return [d for d in range(1, math.isqrt(n)+1) if n % d == 0] + [n//d for d in range(1, math.isqrt(n)+1) if n%d == 0 and d*d != n]

0053 def ramanujan(n, t):
0054     return sum(d*mu(n//d) for d in divisors(math.gcd(n, t)))

0057 def exact_bound_data(n, m):
0058     ph = phi(n)
0059     positive = Counter(math.gcd(n, d) for d in range(1, m))
0060     prefixes = {g: [0]*m for g in positive}
0061     for g in positive:
0062         for u in range(1, m):
0063             prefixes[g][u] = prefixes[g][u-1] + (math.gcd(n, u) == g)
0064     maxima = {g: max(prefixes[g][u] + prefixes[g][m-1-u] for u in range(m)) for g in positive}
0065     V = sum((F(2*c, phi(n//g)) for g, c in positive.items()), F(0))
0066     C = sum((F(maxima[g], phi(n//g)) for g in positive), F(0))
0067     for g, c in maxima.items():
0068         assert c <= (m-1)//g
0069         assert F(ph, phi(n//g)) <= g
0070         assert c <= 2*positive[g]
0071     assert C <= V <= F(m*(m-1), ph)
0072     assert C <= F((m-1)*len(positive), ph)
0073     return C, V, maxima

0076 def exact_q_numerator(n, m):
0077     """Returns A with Q=A/(N phi(N)) from integer Ramanujan sums."""
0078     ds = [set(range(u-m+1, u+1)) - {0} for u in range(m)]
0079     table = [ramanujan(n, j) for j in range(n)]
0080     return np.array([[sum(table[(d*(j-i)) % n] for d in ds[i%m] & ds[j%m])
0081                       for j in range(n)] for i in range(n)], dtype=object)

0084 def exact_ldl_positive(a):
0085     """No floating point: ordinary rational LDL for positive definite input."""
0086     a = [[F(v) for v in row] for row in a]
0087     n = len(a)
0088     for k in range(n):
0089         pivot = a[k][k]
0090         assert pivot > 0, (n, k, pivot)
0091         for i in range(k+1, n):
0092             for j in range(i, n):
0093                 a[j][i] -= a[i][k]*a[j][k]/pivot
0094                 a[i][j] = a[j][i]

0097 def grid_constant_output(value_over_tau):
0098     """Independent exact bin-center calculation for identical input rows."""
0099     centers = []

0101         bin_id = (value_over_tau - shift) // 7
0102         centers.append(7*(F(bin_id)+F(1, 2))+shift)
0103     return sorted(centers)[1]
