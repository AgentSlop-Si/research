6. The original formula as actual Python code
Before showing the fast FFT algorithm, let's implement the formula exactly as written.

import math
import cmath

def dft(a):
    N = len(a)
    result = []

    for k in range(N):
        total = 0

        for j in range(N):
            angle = -2 * math.pi * j * k / N

            multiplier = cmath.exp(1j * angle)

            contribution = a[j] * multiplier

            total += contribution

        result.append(total)

    return result


A = [4, 3, 2, 1, 0, 0, 0, 0]

12. But where does the FFT algorithm hold these values in memory?
Let's finally turn the complete algorithm into code.

Here is a recursive FFT that uses exactly the mathematics we've been calculating.

import math

def transform(a, inverse=False):
    n = len(a)

    # Only one number left
    if n == 1:
        return [complex(a[0])]

    # Step 1: Split into even and odd indices
    even = transform(a[0::2], inverse)
    odd = transform(a[1::2], inverse)

    # Step 2: Allocate space for n outputs
    result = [0j] * n

    # Step 3: Choose rotation direction
    if inverse:
        sign = 1
    else:
        sign = -1

    # Step 4: Calculate butterflies
    for k in range(n // 2):

        angle = sign * 2 * math.pi * k / n

        w = complex(
            math.cos(angle),
            math.sin(angle)
        )

        t = w * odd[k]

        result[k] = even[k] + t

        result[k + n // 2] = even[k] - t

    return result


def fft(a):
    return transform(a, inverse=False)


def ifft(a):
    result = transform(a, inverse=True)
    n = len(a)

    return [value / n for value in result]
