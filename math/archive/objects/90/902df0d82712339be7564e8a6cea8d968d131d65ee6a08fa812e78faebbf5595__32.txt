  40 def inverse(a):
  41     n = len(a)
  42     rows = [list(row) + ident for row, ident in zip(a, eye(n))]
  43     for col in range(n):
  44         pivot = next((r for r in range(col, n) if rows[r][col] % P), None)
  45         if pivot is None:
  46             raise ValueError('singular')
  47         rows[col], rows[pivot] = rows[pivot], rows[col]
  48         scale = pow(rows[col][col], -1, P)
  49         rows[col] = [v*scale % P for v in rows[col]]
  50         for r in range(n):
  51             if r != col:
  52                 scale = rows[r][col]
  53                 if scale:
  54                     rows[r] = [(x-scale*y) % P for x, y in zip(rows[r], rows[col])]
  55     return [row[n:] for row in rows]

  58 def rank(a):
  59     a = [[x % P for x in row] for row in a]
  60     count = 0
  61     for col in range(len(a[0])):
  62         pivot = next((r for r in range(count, len(a)) if a[r][col]), None)
  63         if pivot is None:
  64             continue
  65         a[count], a[pivot] = a[pivot], a[count]
  66         scale = pow(a[count][col], -1, P)
  67         a[count] = [v*scale % P for v in a[count]]
  68         for r in range(count+1, len(a)):
  69             scale = a[r][col]
  70             a[r] = [(x-scale*y) % P for x, y in zip(a[r], a[count])]
  71         count += 1
  72         if count == len(a):
  73             break
  74     return count

  77 def null_vector(rows, n):
  78     """One nonzero kernel vector, or None. Independent elimination."""
  79     a = [[v % P for v in row] for row in rows]
  80     pivots = []
  81     k = 0
  82     for col in range(n):
  83         r = next((r for r in range(k, len(a)) if a[r][col]), None)
  84         if r is None:
  85             continue
  86         a[k], a[r] = a[r], a[k]
  87         scale = pow(a[k][col], -1, P)
  88         a[k] = [v*scale % P for v in a[k]]
  89         for r in range(len(a)):
  90             if r != k:
  91                 factor = a[r][col]
  92                 a[r] = [(x-factor*y) % P for x, y in zip(a[r], a[k])]
  93         pivots.append(col)
  94         k += 1
  95         if k == len(a):
  96             break
  97     free = next((j for j in range(n) if j not in pivots), None)
  98     if free is None:
  99         return None
 100     x = [0]*n
 101     x[free] = 1
 102     for r, col in enumerate(pivots):
 103         x[col] = -a[r][free] % P
 104     assert all(sum(u*v for u, v in zip(row, x)) % P == 0 for row in rows)
 105     return x

 108 def fourier(n):
 109     w = pow(3, (P-1)-(P-1)//n, P)
 110     assert pow(w, n, P) == 1 and pow(w, n//2, P) != 1
 111     scale = pow(int(sqrt_mod(n, P)), -1, P)
 112     return [[scale*pow(w, i*j, P) % P for j in range(n)] for i in range(n)], w

 115 def sampled_green(n, seed):
 116     """Same published parameter stream; different dense path-product evaluator.
 117 
 118     Each term is formed from the finite geometric inverse formula, not the
 119     source's linear recurrence. This is intentionally a dense diagnostic.
 120     """
 121     rng = random.Random(seed)
 122     w = pow(3, (P-1)-(P-1)//n, P)
 123     roots = [pow(w, k, P) for k in range(n)]
 124     total = [[0]*n for _ in range(n)]
 125     term_checks = []
 126     for _ in range(2):
 127         order = list(range(n))
 128         rng.shuffle(order)
 129         while True:
 130             weights = [rng.choice(roots) for _ in range(n)]
 131             product = math.prod(weights) % P
 132             if product != 1:
 133                 break
 134         left = [rng.choice(roots) for _ in range(n)]
 135         right = [rng.choice(roots) for _ in range(n)]
 136         denominator = pow((1-product) % P, -1, P)
 137         unscaled = [[0]*n for _ in range(n)]
 138         for source in range(n):
 139             path_weight = 1
 140             for distance in range(n):
 141                 destination = (source+distance) % n
 142                 if distance:
 143                     path_weight = path_weight*weights[destination] % P
 144                 unscaled[destination][source] = path_weight*denominator % P
 145                 i, j = order[destination], order[source]
 146                 total[i][j] = (total[i][j] + left[destination]*right[source]
 147                                *unscaled[destination][source]) % P
 148         # Validate every entry of (I-S)G=I using the sparse denominator.
 149         assert all((unscaled[i][j]-weights[i]*unscaled[(i-1)%n][j]) % P
 150                    == int(i == j) for i in range(n) for j in range(n))
 151         term_checks.append(True)
 152     return total

 155 def gram_pair(a, b):
 156     n = len(a)
 157     bi = inverse(b)
 158     jbi = [bi[(-i) % n] for i in range(n)]
 159     return mul(tr(a), a), mul(tr(bi), jbi)

 162 def signatures(a):
 163     invdiag = [pow(a[i][i], -1, P) for i in range(len(a))]
 164     h = [[v*v*invdiag[i]*invdiag[j] % P for j, v in enumerate(row)]
 165          for i, row in enumerate(a)]
 166     return h, [tuple(sorted(row)) for row in h]

 169 def match_grams(k, c):
 170     n = len(k)
 171     zk = sum(k[i][i] == 0 for i in range(n))
 172     zc = sum(c[i][i] == 0 for i in range(n))
 173     if zk != zc:
 174         return 'excluded', None, 'diagonal-zero-count'
 175     if zk:
 176         return 'inconclusive', None, 'zero-diagonal'
 177     h, hk = signatures(k)
 178     g, hc = signatures(c)
 179     if Counter(hk) != Counter(hc):
 180         return 'excluded', None, 'row-signature-multiset'
 181     if len(set(hk)) != n:
 182         return 'ambiguous', None, 'repeated-signatures'
 183     lookup = {sig: i for i, sig in enumerate(hk)}
 184     perm = [lookup[sig] for sig in hc]
 185     if any(g[i][j] != h[perm[i]][perm[j]] for i in range(n) for j in range(n)):
 186         return 'excluded', None, 'full-invariant-matrix'
 187     return 'candidate', perm, 'matched'

 190 def reconstruct_mask(k, c, perm):
 191     n = len(k)
 192     invperm = [perm.index(i) for i in range(n)]
 193     target = [[c[invperm[i]][invperm[j]] for j in range(n)] for i in range(n)]
 194     if any(v == 0 for row in k for v in row):
 195         return None
 196     ratio = [[target[i][j]*pow(k[i][j], -1, P) % P for j in range(n)] for i in range(n)]
 197     roots = sqrt_mod(ratio[0][0], P, all_roots=True)
 198     if not roots or roots[0] == 0:
 199         return None
 200     q0 = int(roots[0])
 201     q = [ratio[i][0]*pow(q0, -1, P) % P for i in range(n)]
 202     if any(q[i] == 0 for i in range(n)):
 203         return None
 204     return q if all(q[i]*k[i][j]*q[j] % P == target[i][j]
 205                     for i in range(n) for j in range(n)) else None

 208 def sandwich(a, q, perm, b):
 209     middle = [[0]*len(a) for _ in a]
 210     for j, i in enumerate(perm):
 211         middle[i] = [q[i]*x % P for x in b[j]]
 212     return mul(a, middle)

 215 def make_b(a, q, perm, target):
 216     core = mul(inverse(a), target)
 217     scaled = [[x*pow(q[i], -1, P) % P for x in row] for i, row in enumerate(core)]
 218     return [scaled[i] for i in perm]

 221 def parity_objects(n):
 222     f, w = fourier(n)
 223     h = n//2
 224     sq2 = int(sqrt_mod(2, P))
 225     invsq2 = pow(sq2, -1, P)
 226     scale = f[0][0]
 227     imag = pow(w, n//4, P)
 228     inv2 = pow(2, -1, P)
 229     # (-i)^-1 = i, and (omega^-j - omega^j)/(2i) = sin(theta_j).
 230     iunit = pow(imag, -1, P)
 231     inv2i = pow(2*iunit, -1, P)
 232     hmat = [[0]*n for _ in range(n)]
 233     hmat[0][0] = hmat[h][1] = 1
 234     for j in range(1, h):
 235         hmat[j][j+1] = hmat[n-j][j+1] = invsq2
 236         hmat[j][h+j] = invsq2
 237         hmat[n-j][h+j] = -invsq2 % P
 238     q = [1]*(h+1) + [imag]*(h-1)
 239     # Build C directly from algebraic cos/sin entries, not H^T F.
 240     c = [[scale]*n, [scale*pow(P-1, k, P) % P for k in range(n)]]
 241     cosine, sine = [], []
 242     for j in range(1, h):
 243         cosine.append([sq2*scale*(pow(w,j*k,P)+pow(w,-j*k,P))*inv2 % P for k in range(n)])
 244         sine.append([sq2*scale*(pow(w,-j*k,P)-pow(w,j*k,P))*inv2i % P for k in range(n)])
 245     c += cosine+sine
 246     # O(N) coefficient preparation by an advancing root and inverse root.
 247     r = [[0]*n for _ in range(n)]
 248     r[0][0], r[1][1] = 1, P-1
 249     wp, wm, wi = 1, 1, pow(w, -1, P)
 250     for j in range(1, h):
 251         wp, wm = wp*w % P, wm*wi % P
 252         co, si = (wp+wm)*inv2 % P, (wm-wp)*inv2i % P
 253         a, b = j+1, h+j
 254         r[a][a] = r[b][b] = co
 255         r[a][b], r[b][a] = -si % P, si
 256     return f, hmat, q, c, r

 259 def covariance_matrix(c, r):
 260     rc = mul(r, c)
 261     n = len(c)
 262     return [[(c[i][(j+1)%n]-rc[i][j]) % P for j in range(n)] for i in range(n)]
