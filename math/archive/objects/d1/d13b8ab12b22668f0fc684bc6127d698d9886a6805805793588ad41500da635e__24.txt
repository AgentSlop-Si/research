0023 def canonical(M):
0024     return M.applyfunc(s.expand)
0025 
0026 
0027 def exact_equal(A, B):
0028     if A.shape != B.shape or any(s.expand(x) != 0 for x in A - B):
0029         raise AssertionError(f"Exact matrices differ: {A.shape}, {B.shape}")

0032 def is_unitary(M):
0033     exact_equal(canonical(M.conjugate().T * M), s.eye(M.cols))
0034 
0035 
0036 def shear_left(M, target, source, coefficient):
0037     """One actual scalar shear, not a simultaneous block operation."""
0038     if target == source:
0039         raise ValueError("A shear needs distinct coordinates")
0040     N = M.copy()
0041     for k in range(M.cols):
0042         N[target, k] = s.expand(M[target, k] + coefficient * M[source, k])
0043     return N
0044 
0045 
0046 def shear_inverse_right(M, target, source, coefficient):
0047     N = M.copy()
0048     for k in range(M.rows):
0049         N[k, source] = s.expand(M[k, source] - coefficient * M[k, target])
0050     return N
0051 
0052 
0053 def embed(M, coordinates, dimension):
0054     N = s.eye(dimension)
0055     for i, a in enumerate(coordinates):
0056         for j, b in enumerate(coordinates):
0057             N[a, b] = M[i, j]
0058     return N

0061 def borrowed_add_word(U, source, target, scratch):
0062     """Dense O(B^2) ADD_U via four-shear commutators; arbitrary scratch.
0063 
0064     z += c*x; y += z; z -= c*x; y -= z gives y += c*x.
0065     This is an independent toy compiler, not the manuscript's replay compiler.
0066     """
0067     gates = []
0068     for j in range(U.rows):
0069         for i in range(U.cols):
0070             c = U[j, i]
0071             if c != 0:
0072                 gates += [(scratch, source[i], c), (target[j], scratch, s.Integer(1)),
0073                           (scratch, source[i], -c), (target[j], scratch, s.Integer(-1))]
0074     return gates
0075 
0076 
0077 def exchange_word(U):
0078     n = U.rows
0079     a, b, z = list(range(n)), list(range(n, 2*n)), 2*n
0080     return (borrowed_add_word(U, a, b, z) +
0081             borrowed_add_word(-U.conjugate().T, b, a, z) +
0082             borrowed_add_word(U, a, b, z))
0083 
0084 
0085 def expected_exchange(U):
0086     n = U.rows
0087     zero = s.zeros(n)
0088     W = zero.row_join(-U.conjugate().T).col_join(U.row_join(zero))
0089     return s.diag(W, s.Integer(1))

0092 def to_numpy(M):
0093     return np.array(M.evalf(17).tolist(), dtype=np.complex128)

0096 def local_prefixes(U):
0097     d = 2*U.rows + 1
0098     P = s.eye(d)
0099     Pinv = s.eye(d)
0100     prefixes = [(P, Pinv)]
0101     for j, i, c in exchange_word(U):
0102         P = shear_left(P, j, i, c)
0103         Pinv = shear_inverse_right(Pinv, j, i, c)
0104         exact_equal(P * Pinv, s.eye(d))
0105         # Determinants are also implied by the elementary form of each gate.
0106         if s.expand(P.det()) != 1:
0107             raise AssertionError("Non-unimodular exact prefix")
0108         prefixes.append((P, Pinv))
0109     exact_equal(P, expected_exchange(U))
0110     return prefixes

0113 def check_serial_fiber(Q, coordinates, U, prefixes, numeric=False):
0114     """Check every scalar prefix against embed(P_t) Q, with unitary Q."""
0115     is_unitary(Q)
0116     M = Q.copy()
0117     worst = 1.0
0118     for t, (j, i, c) in enumerate(exchange_word(U), 1):
0119         M = shear_left(M, coordinates[j], coordinates[i], c)
0120         P, _ = prefixes[t]
0121         exact_equal(M, canonical(embed(P, coordinates, Q.rows) * Q))
0122         if numeric:
0123             worst = max(worst, float(np.linalg.cond(to_numpy(M))))
0124     is_unitary(M)
0125     return M, worst

0128 def permutation(p):
0129     M = s.zeros(len(p))
0130     for j, i in enumerate(p):
0131         M[j, i] = 1
0132     return M

0142     def test_01_scalar_anchor_counterexample(self):
0143         S = 8 * s.eye(2)
0144         whole = s.diag(S, s.eye(2))
0145         exact_equal(S.conjugate().T * S, 64 * s.eye(2))

0147         # From these exact Gram spectra: kappa(S)=1, kappa(S direct-sum I)=8.

0324     def test_12_all_length_bluestein_numerical_identities(self):
0325         # Bounded numerical sanity checks only. No spectral-norm SVD sweep.
0326         # Fourier convention: exp(+2*pi*i*j*k/N); normalized output 1/sqrt(N).
0327         rng = np.random.default_rng(20261009)
0328         rows = []
0329         for n in (1, 2, 3, 5, 6, 7, 9, 10, 15, 17):
0330             m = 1 << ((2*n-2).bit_length())
0331             j = np.arange(n)
0332             positive_chirp = np.exp(1j*np.pi*j*j/n)
0333             b = np.zeros(m, dtype=np.complex128)
0334             b[0] = 1
0335             for k in range(1, n):
0336                 b[k] = np.exp(-1j*np.pi*k*k/n)
0337                 b[m-k] = b[k]
0338             self.assertEqual(int(np.count_nonzero(b)), 2*n-1)
0339             # The signed support is -(n-1),...,(n-1); neither endpoint +/-n
0340             # is inserted. For m=2n this is exactly one missing periodic term.
0341             if m == 2*n:
0342                 self.assertEqual(b[n], 0)
0343             D = m*np.fft.ifft(b)/math.sqrt(n)
0344             bound = 9+4*math.log(2*n)
0345             self.assertLessEqual(float(np.max(np.abs(D))), bound)
0346 
0347             # Full small matrix identity: Y V* D V X = normalized F_n.
0348             X = np.zeros((m, n), dtype=np.complex128)
0349             X[:n] = np.diag(positive_chirp)
0350             actual = np.fft.fft(D[:, None]*np.fft.ifft(X, axis=0, norm='ortho'),
0351                                  axis=0, norm='ortho')[:n]
0352             actual = positive_chirp[:, None]*actual
0353             reference = np.exp(2j*np.pi*j[:, None]*j[None, :]/n)/math.sqrt(n)
0354             matrix_error = float(np.max(np.abs(actual-reference)))
0355             self.assertLess(matrix_error, 2e-12)
0356 
0357             # Dirty buffers are arbitrary AND correlated with the source.
0358             x = rng.normal(size=n)+1j*rng.normal(size=n)
0359             y0 = rng.normal(size=n)+1j*rng.normal(size=n)
0360             z0 = rng.normal(size=m)+1j*rng.normal(size=m)
0361             z0[:n] += x
0362             w0 = rng.normal(size=m)+1j*rng.normal(size=m)+0.3*z0
0363             y, z, w = y0.copy(), z0.copy(), w0.copy()
0364 
0365             def pass_once(with_source, sign):
0366                 nonlocal y, z, w
0367                 if with_source:
0368                     z[:n] += positive_chirp*x
0369                 z = np.fft.ifft(z, norm='ortho')
0370                 w += D*z
0371                 w = np.fft.fft(w, norm='ortho')
0372                 y += sign*positive_chirp*w[:n]
0373                 w = np.fft.ifft(w, norm='ortho')
0374                 w -= D*z
0375                 z = np.fft.fft(z, norm='ortho')
0376                 if with_source:
0377                     z[:n] -= positive_chirp*x
0378 
0379             pass_once(True, 1)
0380             first_restore = max(np.max(np.abs(z-z0)), np.max(np.abs(w-w0)))
0381             pass_once(False, -1)
0382             dirty_error = float(np.max(np.abs(y-y0-reference@x)))
0383             scratch_error = float(max(np.max(np.abs(z-z0)), np.max(np.abs(w-w0))))
0384             self.assertLess(float(first_restore), 2e-12)
0385             self.assertLess(dirty_error, 3e-12)
0386             self.assertLess(scratch_error, 3e-12)
0387 
0388             qj = np.arange(2*n)
0389             gauss_spectrum = (2*n)*np.fft.ifft(np.exp(-1j*np.pi*qj*qj/n))
0390             magnitudes = np.abs(gauss_spectrum)
0391             gauss_error = float(np.max(np.minimum(magnitudes, np.abs(magnitudes-2*math.sqrt(n)))))
0392             self.assertLess(gauss_error, 2e-11)
0393             self.assertEqual(int(np.count_nonzero(magnitudes > math.sqrt(n))), n)
0394             gj = np.arange(4*n)
0395             G4n = np.sum(np.exp(2j*np.pi*gj*gj/(4*n)))
0396             sqrt_from_gauss = G4n/(2*(1+1j))
0397             normalization_error = float(abs(sqrt_from_gauss-math.sqrt(n)))
0398             self.assertLess(normalization_error, 2e-11)
0399             rows.append({'Fourier_sign': '+', 'N': n, 'M': m, 'max_abs_diagonal_D': float(np.max(np.abs(D))),
0400                          'proposed_diagonal_bound': bound,
0401                          'matrix_identity_max_abs_error': matrix_error,
0402                          'dirty_add_max_abs_error': dirty_error,
0403                          'scratch_restore_max_abs_error': scratch_error,
0404                          'complete_gauss_magnitude_error': gauss_error,
0405                          'gauss_sqrt_normalization_error': normalization_error})
0406         RESULTS['bluestein_numerical_only'] = rows
