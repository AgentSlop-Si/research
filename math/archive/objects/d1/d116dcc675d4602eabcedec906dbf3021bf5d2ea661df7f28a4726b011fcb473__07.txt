   8 def mask_set(n,m):return {j for j in range(n) if (j&m).bit_count()%2==0}

  10 def network_bound(n,perms,cols,rows):
  11     stages=len(perms)+1;inf=n+1;V=8*n*stages+2;src=V-2;sink=V-1;graph=[[] for _ in range(V)]
  12     def node(s,kind,j,half):return 8*n*s+2*n*kind+2*j+half
  13     def edge(u,v,cap):
  14         graph[u].append([v,cap,len(graph[v])]);graph[v].append([u,0,len(graph[u])-1])
  15     for s in range(stages):
  16         for j in range(n):
  17             edge(node(s,0,j,0),node(s,0,j,1),1);edge(node(s,1,j,0),node(s,1,j,1),1);edge(node(s,0,j,1),node(s,1,j,0),inf)
  18         for j in range(1,n):
  19             edge(node(s,2,j,0),node(s,2,j,1),1);edge(node(s,0,j-1,1),node(s,2,j,0),inf);edge(node(s,2,j,1),node(s,1,j,0),inf)
  20             if j<n-1:edge(node(s,2,j,1),node(s,2,j+1,0),inf)
  21         for j in range(n-1):
  22             edge(node(s,3,j,0),node(s,3,j,1),1);edge(node(s,0,j+1,1),node(s,3,j,0),inf);edge(node(s,3,j,1),node(s,1,j,0),inf)
  23             if j:edge(node(s,3,j,1),node(s,3,j-1,0),inf)
  24     for s,pi in enumerate(perms):
  25         for j,old in enumerate(pi):edge(node(s,1,old,1),node(s+1,0,j,0),inf)
  26     for j in cols:edge(src,node(0,0,j,0),inf)
  27     for i in rows:edge(node(stages-1,1,i,1),sink,inf)
  28     flow=0
  29     while True:
  30         pred=[None]*V;pred[src]=(-1,-1);queue=deque([src])
  31         while queue and pred[sink] is None:
  32             u=queue.popleft()
  33             for j,(v,cap,_) in enumerate(graph[u]):
  34                 if cap and pred[v] is None:pred[v]=(u,j);queue.append(v)
  35         if pred[sink] is None:return flow
  36         v=sink;delta=inf
  37         while v!=src:u,j=pred[v];delta=min(delta,graph[u][j][1]);v=u
  38         v=sink
  39         while v!=src:
  40             u,j=pred[v];back=graph[u][j][2];graph[u][j][1]-=delta;graph[v][back][1]+=delta;v=u
  41         flow+=delta

  43 def shuffles(k):
  44     rng=random.Random(20261016+17*k);perms=[]
  45     for s in range(3):
  46         rows=[1<<j for j in range(k)]
  47         for _ in range(2*k*k):
  48             i,j=rng.sample(range(k),2)
  49             if rng.randrange(4)==0:rows[i],rows[j]=rows[j],rows[i]
  50             else:rows[i]^=rows[j]
  51         perms.append([sum(((x&row).bit_count()%2)<<i for i,row in enumerate(rows)) for x in range(1<<k)])
  52     return perms
