  10 class Poly:
  11     def __init__(self, terms=None): self.d={k:int(v) for k,v in (terms or {}).items() if v}
  12     @staticmethod
  13     def c(n): return Poly({(0,()):n})
  14     @staticmethod
  15     def x(i): return Poly({(0,(i,)):1})
  16     @staticmethod
  17     def z(e):
  18         e%=8
  19         return Poly({(e%4,()):1 if e<4 else -1})
  20     def __add__(self,o):
  21         if isinstance(o,int): o=Poly.c(o)
  22         d=self.d.copy()
  23         for k,v in o.d.items(): d[k]=d.get(k,0)+v
  24         return Poly(d)
  25     __radd__=__add__
  26     def __neg__(self): return Poly({k:-v for k,v in self.d.items()})
  27     def __sub__(self,o): return self+(-o if isinstance(o,Poly) else -int(o))
  28     def __rsub__(self,o): return -self+o
  29     def __mul__(self,o):
  30         if isinstance(o,int): o=Poly.c(o)
  31         d=collections.defaultdict(int)
  32         for (a,m),v in self.d.items():
  33             for (b,n),w in o.d.items():
  34                 e=a+b
  35                 d[(e%4,tuple(sorted(m+n)))]+=v*w*(1 if e<4 else -1)
  36         return Poly(d)
  37     __rmul__=__mul__
  38     def __eq__(self,o): return isinstance(o,Poly) and self.d==o.d
  39     def evaluate(self,x,p,w):
  40         return sum(c*pow(w,e,p)*math.prod(x[i] for i in m) for (e,m),c in self.d.items())%p
  41     def grouped(self):
  42         d=collections.defaultdict(lambda:[0]*4)
  43         for (e,m),v in self.d.items(): d[m][e]=v
  44         return dict(d)

  48 def det(A):
  49     """Subset expansion, independently usable over Fraction or Poly."""
  50     n=len(A); dp={0:1}
  51     for row in range(n):
  52         nq={}
  53         for mask,val in dp.items():
  54             for col in range(n):
  55                 if mask>>col&1: continue
  56                 sign=-1 if (mask>>(col+1)).bit_count()%2 else 1
  57                 nm=mask|(1<<col)
  58                 term=val*A[row][col]*sign
  59                 nq[nm]=nq.get(nm,0)+term
  60         dp=nq
  61     return dp[(1<<n)-1]

  63 def stage(L,V,U,W,D):

 101 def inverse_mod(A,p):
 102     n=len(A);a=[[int(v)%p for v in r]+[int(i==j) for j in range(n)] for i,r in enumerate(A)]
 103     for c in range(n):
 104         row=next((i for i in range(c,n) if a[i][c]),None)
 105         if row is None: return None
 106         a[c],a[row]=a[row],a[c];q=pow(a[c][c],-1,p);a[c]=[v*q%p for v in a[c]]
 107         for i in range(n):
 108             if i!=c:
 109                 q=a[i][c];a[i]=[(u-q*v)%p for u,v in zip(a[i],a[c])]
 110     return [r[n:] for r in a]
