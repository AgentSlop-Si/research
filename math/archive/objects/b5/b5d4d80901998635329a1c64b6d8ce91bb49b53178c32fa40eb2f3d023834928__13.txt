  10 def zero(n, m=None):
  11     return [[0] * (n if m is None else m) for _ in range(n)]
  12 
  13 
  14 def ident(n):
  15     a = zero(n)
  16     for i in range(n):
  17         a[i][i] = 1
  18     return a
  19

  21 def mul(a, b, p=P):
  22     c = zero(len(a), len(b[0]))
  23     brows = [[(j, v) for j, v in enumerate(row) if v] for row in b]
  24     for i, row in enumerate(a):
  25         for k, v in enumerate(row):
  26             if v:
  27                 for j, w in brows[k]:
  28                     c[i][j] += v * w
  29         c[i] = [v % p for v in c[i]]
  30     return c
  31 
  32 
  33 def add(a, b, sign=1, p=P):
  34     return [[(v + sign * w) % p for v, w in zip(r, s)] for r, s in zip(a, b)]
  35 
  36 
  37 def scale(a, c, p=P):
  38     return [[v * c % p for v in row] for row in a]
  39 
  40 
  41 def transpose(a):
  42     return [list(r) for r in zip(*a)]
  43

  45 def inv(a, p=P):
  46     n = len(a)
  47     z = [row[:] + ident(n)[i] for i, row in enumerate(a)]
  48     for j in range(n):
  49         pivot = next(i for i in range(j, n) if z[i][j])
  50         z[j], z[pivot] = z[pivot], z[j]
  51         z[j] = [v * pow(z[j][j], -1, p) % p for v in z[j]]
  52         for i in range(n):
  53             if i != j and z[i][j]:
  54                 c = z[i][j]
  55                 z[i] = [(v - c * w) % p for v, w in zip(z[i], z[j])]
  56     return [row[n:] for row in z]
  57

  59 def rank(a, p=P):
  60     z = [row[:] for row in a]
  61     r = 0
  62     for j in range(len(z[0])):
  63         pivot = next((i for i in range(r, len(z)) if z[i][j]), None)
  64         if pivot is None:
  65             continue
  66         z[r], z[pivot] = z[pivot], z[r]
  67         v = pow(z[r][j], -1, p)
  68         z[r] = [x * v % p for x in z[r]]
  69         for i in range(r + 1, len(z)):
  70             c = z[i][j]
  71             if c:
  72                 z[i] = [(x - c * y) % p for x, y in zip(z[i], z[r])]
  73         r += 1
  74     return r

  77 def root(n, p=P, primitive=3):
  78     w = pow(primitive, (p - 1) // n, p)
  79     assert pow(w, n, p) == 1
  80     assert n == 1 or pow(w, n // 2, p) != 1
  81     return w
  82 
  83 
  84 def fourier(n, w, p=P):
  85     return [[pow(w, i * j, p) for j in range(n)] for i in range(n)]
  86 
  87 
  88 def weyl(n, eta, a, b, p=P):
  89     z = zero(n)
  90     for j in range(n):
  91         i = (j + b) % n
  92         z[i][j] = pow(eta, a * i, p)
  93     return z

  96 def degree(a):
  97     return {
  98         'nnz': sum(bool(x) for row in a for x in row),
  99         'row': max(sum(bool(x) for x in row) for row in a),
 100         'column': max(sum(bool(x) for x in row) for row in zip(*a)),
 101     }

 104 def put(a, b, i, j):
 105     for r, row in enumerate(b):
 106         for s, v in enumerate(row):
 107             a[i + r][j + s] = v

 110 def block_bank(m, q, active, inverse=False):
 111     f = fourier(m, root(m))
 112     if inverse:
 113         f = scale(fourier(m, pow(root(m), -1, P)), pow(m, -1, P))
 114     b = zero(m * q)
 115     for r in range(q):
 116         put(b, f if r in active else ident(m), r * m, r * m)
 117     return b

 120 def controls_weyl():
 121     count = 0
 122     for m in (2, 4, 8, 16, 32):
 123         eta = root(m)
 124         f = fourier(m, eta)
 125         fi = scale(fourier(m, pow(eta, -1, P)), pow(m, -1, P))
 126         pairs = list(itertools.product(range(m), repeat=2)) if m <= 16 else [(0, 0), (1, 1), (3, 7), (31, 17)]
 127         for a, b in pairs:
 128             lhs = mul(mul(f, weyl(m, eta, a, b)), fi)
 129             rhs = scale(weyl(m, eta, b, -a), pow(eta, a * b, P))
 130             assert lhs == rhs
 131             count += 1
 132     return count

 135 def controls_uncertainty():
 136     summary = []
 137     for p, generator, m, alphabet, exhaustive in ((5, 2, 4, range(5), True), (17, 3, 2, range(17), True), (17, 3, 8, (0, 1), False)):
 138         f = fourier(m, root(m, p, generator), p)
 139         count = 0
 140         minimum = m * m
 141         for x in itertools.product(alphabet, repeat=m):
 142             s = sum(v != 0 for v in x)
 143             if not s:
 144                 continue
 145             fx = [sum(a * b for a, b in zip(row, x)) % p for row in f]
 146             t = sum(v != 0 for v in fx)
 147             c = [[x[(i - j) % m] for j in range(m)] for i in range(m)]
 148             assert rank(c, p) == t
 149             assert s * t >= m
 150             minimum = min(minimum, s * t)
 151             count += 1
 152         summary.append({'field': p, 'm': m, 'nonzero_vectors': count, 'exhaustive_over_field': exhaustive, 'minimum_support_product': minimum})
 153     return summary

 156 def active_sets(q):
 157     if q <= 4:
 158         return [{j for j in range(q) if (mask >> j) & 1} for mask in range(1 << q)]
 159     choices = [set(), set(range(q)), {0}, set(range(q // 2)), set(range(0, q, 2)), set(range(1, q))]
 160     return choices

 163 def controls_ct():
 164     summary = []
 165     count = 0
 166     for m, q in ((2, 2), (2, 4), (4, 2), (4, 4), (4, 8), (8, 4), (8, 8), (16, 4)):
 167         n = m * q
 168         omega = root(n)
 169         eta = pow(omega, q, P)
 170         omega_q = pow(omega, m, P)
 171         f = fourier(n, omega)
 172         fm = fourier(m, eta)
 173         s = weyl(n, omega, 0, 1)
 174         d = weyl(n, omega, 1, 0)
 175         si = weyl(n, omega, 0, -1)
 176         qmat = zero(n)
 177         for r in range(q):
 178             for t in range(m):
 179                 qmat[r * m + t][r + q * t] = 1
 180         qinv = transpose(qmat)
 181         t1, t2 = mul(mul(qmat, s), qinv), mul(mul(qmat, d), qinv)
 182         for active in active_sets(q):
 183             bank = block_bank(m, q, active)
 184             bi = block_bank(m, q, active, inverse=True)
 185             pmat = zero(n)
 186             for u in range(q):
 187                 for t in range(m):
 188                     for r in range(q):
 189                         coefficient = pow(omega_q, u * r, P) * pow(omega, r * t, P) % P
 190                         for h in range(m):
 191                             entry = int(t == h) if r in active else fm[t][h]
 192                             pmat[u * m + t][r * m + h] = coefficient * entry % P
 193             assert mul(mul(pmat, bank), qmat) == f
 194             a1, a2 = mul(mul(bank, t1), bi), mul(mul(bank, t2), bi)
 195             assert mul(a1, bank) == mul(bank, t1)
 196             assert mul(a2, bank) == mul(bank, t2)
 197             assert mul(pmat, a1) == mul(d, pmat)
 198             assert mul(pmat, a2) == mul(si, pmat)
 199             assert mul(t1, qmat) == mul(qmat, s)
 200             assert mul(t2, qmat) == mul(qmat, d)
 201             k = sum((r in active) != (((r + 1) % q) in active) for r in range(q))
 202             deg = degree(a1)
 203             assert deg['nnz'] == n + k * m * (m - 1)
 204             assert deg['row'] == deg['column'] == (m if k else 1)
 205             assert degree(a2)['nnz'] == degree(t1)['nnz'] == degree(t2)['nnz'] == n
 206             count += 1
 207             if q == m and active == set(range(q // 2)):
 208                 summary.append({'N': n, 'm': m, 'q': q, 'active': sorted(active), 'boundary_edges': k, 'A_shift': deg, 'other_three_nnz_each': n, 'charged_transform_count': {'size_m_in_bank': len(active), 'size_m_hidden_in_P': q - len(active), 'size_q_in_P': m}})
 209     return {'cases': count, 'square_contiguous_controls': summary}

 212 def controls_directionality():
 213     out = []
 214     for m in (2, 4, 8, 16):
 215         f = fourier(m, root(m))
 216         t = zero(m)
 217         for i in range(m):
 218             t[i][0] = 1
 219         a = mul(f, t)
 220         assert degree(a)['row'] == degree(t)['row'] == 1
 221         assert degree(t)['column'] == m
 222         assert degree(a)['nnz'] == 1
 223         ar, tr = transpose(t), transpose(a)
 224         assert mul(ar, f) == tr
 225         assert degree(ar)['column'] == degree(tr)['column'] == 1
 226         out.append({'m': m, 'left_mixed_row_degrees': [degree(t)['row'], degree(a)['row']], 'left_input_column_degree': degree(t)['column'], 'right_mixed_column_degrees': [degree(ar)['column'], degree(tr)['column']]})
 227     return out

 230 def mono(n, offset, seed):
 231     a = zero(n)
 232     for j in range(n):
 233         a[(j + offset) % n][j] = 1 + ((seed + j) % 13)
 234     return a

 237 def controls_state():
 238     out = []
 239     for n in (4, 8):
 240         m, q = 2, n // 2
 241         bank = block_bank(m, q, {0})
 242         size = 14 * n
 243         ks = ident(size)
 244         layers = []
 245         realization = {0: ident(n)}
 246         specs = ((0, 1, 2, 3), (3, 4, 5, 6), (7, 8, 9, 10), (10, 11, 12, 13))
 247         for ell, (inp, left, right, output) in enumerate(specs):
 248             hs = zero(n)
 249             for r, wire in enumerate((left, right)):
 250                 u = ident(n)
 251                 weights = [2 + ((j + ell + r) % 5) for j in range(n)]
 252                 product = 1
 253                 for j, weight in enumerate(weights):
 254                     u[(j + 1) % n][j] = -weight % P
 255                     product = product * weight % P
 256                 assert product != 1
 257                 l = mono(n, ell + r + 1, ell + 2 * r)
 258                 rr = mono(n, 2 * ell + r, 3 * ell + r)
 259                 put(ks, u, wire * n, wire * n)
 260                 put(ks, scale(rr, -1), wire * n, inp * n)
 261                 put(ks, scale(l, -1), output * n, wire * n)
 262                 g = mul(inv(u), rr)
 263                 hs = add(hs, mul(l, g))
 264                 realization[wire] = mul(g, realization[inp])
 265             layers.append(hs)
 266             realization[output] = mul(hs, realization[inp])
 267             if output == 6:
 268                 realization[7] = mul(bank, realization[6])
 269         assert degree(ks)['row'] <= 3 and degree(ks)['column'] <= 3
 270         k = [row[:] for row in ks]
 271         put(k, scale(bank, -1), 7 * n, 6 * n)
 272         v = []
 273         for wire in range(14):
 274             v.extend(realization[wire])
 275         b0 = zero(size, n)
 276         put(b0, ident(n), 0, 0)
 277         assert mul(k, v) == b0
 278         target = mul(mul(mul(mul(layers[3], layers[2]), bank), layers[1]), layers[0])
 279         assert realization[13] == target
 280         # Arbitrary degree-two witnesses check the exact residual decomposition.
 281         sr = add(ident(size), mono(size, 5, 2))
 282         rr = add(scale(ident(size), 3), mono(size, 11, 4))
 283         residual = add(mul(k, sr), mul(rr, k), -1)
 284         sparse_residual = add(mul(ks, sr), mul(rr, ks), -1)
 285         bank_edge = zero(size)
 286         put(bank_edge, bank, 7 * n, 6 * n)
 287         rhs = add(add(sparse_residual, mul(bank_edge, sr), -1), mul(rr, bank_edge))
 288         assert residual == rhs
 289         center = [[sparse_residual[7 * n + i][6 * n + j] for j in range(n)] for i in range(n)]
 290         expected_center = [[(sr[7 * n + i][6 * n + j] - rr[7 * n + i][6 * n + j]) % P for j in range(n)] for i in range(n)]
 291         assert center == expected_center
 292         out.append({'N': n, 'state_dimension': size, 'K_s_degree': degree(ks), 'transfer_and_residual_identities': True})
 293     return out
