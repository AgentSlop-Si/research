"""Independent Phase 11 audit. Execute from this directory.

Reference paths: explicit DFT sums on small lengths, Phase10 decimation-in-
frequency iterative/recursive FFT on large lengths, direct selected-bin sums.
The rational-Green gate is also checked by explicit Gaussian elimination.
"""
from __future__ import annotations
import json
import random
import sys
from pathlib import Path
from phase11 import (
    LazySpectralUnion, direct_dft, prime_root_for_power_two,
    sharp_rational_rank_witness, rational_mask_difference_rank,
    direct_difference_matrix, mat_rank, symbols_for_levels,
    apply_one_pole_green,
)

HERE=Path(__file__).resolve().parent
# Prefer the frozen standalone reference; fall back to the installed Phase10
# sibling checkpoint for convenience in the live project tree.
LOCAL_REFERENCE=HERE/'controls'/'phase10_reference.py'
if LOCAL_REFERENCE.exists():
    sys.path.insert(0,str(HERE/'controls'))
    from phase10_reference import classical_fft
else:
    sys.path.insert(0,str(HERE.parent/'hero17_phase10_projector_union'))
    from phase10 import classical_fft

rng=random.Random(171711)
audit={
    'phase':'11 — lazy spectral refinement and rational Green correction rank',
    'scope':'Exact finite-field arithmetic. No general sub-n-log-n algorithm.',
    'predecessor':'Phase 10 14 tests and Phase 9 15 tests rerun separately.',
}

# Full direct DFTs. Multiple data types deliberately challenge the projected
# state identity: random dense, impulse, constant, alternating, ramp, chirp.
cases=0; checked=0
for n in [2,4,8,16,32,64,128]:
    p,w=prime_root_for_power_two(n)
    data=[
        [rng.randrange(p) for _ in range(n)],
        [0]*(n-1)+[1],
        [1]*n,
        [(-1)**j%p for j in range(n)],
        [(11+3*j+5*j*j)%p for j in range(n)],
        [pow(w,(j*j)%n,p) for j in range(n)],
    ]
    for x in data:
        engine=LazySpectralUnion(n,p,w)
        got,ledger=engine.transform(x)
        expected=direct_dft(x,p,w)
        assert got==expected,(n,'direct disagreement')
        assert ledger['online_adds']==n*(n.bit_length()-1)
        cases+=1;checked+=n
# Nontrivial rational sharp witnesses and full independent matrix ranks.
rank_cases=0;explicit=0
for n in [8,16,32,64,128]:
    p,w=prime_root_for_power_two(n)
    for _ in range(10):
        s=rng.randrange(max(2,n//4), min(n-1,3*n//4)+1)
        K=set(rng.sample(range(n),s))
        for d in [1,2,3,4]:
            if d>min(s,n-s):continue
            pcoef,qcoef,rank=sharp_rational_rank_witness(n,p,w,K,d)
            assert rank==n-2*d
            rank_cases+=1
            if n<=16:
                _,values=rational_mask_difference_rank(n,p,w,K,pcoef,qcoef)
                assert mat_rank(direct_difference_matrix(n,p,w,K,values),p)==rank
                explicit+=1

# The dyadic projected state graph has no identical formal linear forms
# (including across levels), and each complete level contains n independent
# forms. This is verified separately from the numerical executor.
seen=0;symbolic_nodes=0;rank_state_levels=0
for n in [2,4,8,16,32,64]:
    p,w=prime_root_for_power_two(n)
    forms=[]
    for d,m,level in symbols_for_levels(n,p,w):
        forms.extend(level)
        symbolic_nodes+=len(level)
        if n<=32:
            A=[[0]*n for _ in range(n)]
            for i,row in enumerate(level):
                for j,v in row:A[i][j]=v
            assert mat_rank(A,p)==n
            rank_state_levels+=1
    assert len(set(forms))==len(forms)
    seen+=1

# Large complete arbitrary-input NTT. Distinct reference algorithm imported
# from Phase10; additional bins are direct n-term sums, not FFT-derived.
large_n=65536;p,w=prime_root_for_power_two(large_n)
x=[rng.randrange(p) for _ in range(large_n)]
engine=LazySpectralUnion(large_n,p,w)
out,L=engine.transform(x)
ref,R=classical_fft(x,p,w)
assert out==ref
spotbins=[0,1,2,3,large_n//2,large_n-1]+rng.sample(range(large_n),35)
W=[1]*large_n
for j in range(1,large_n):W[j]=W[j-1]*w%p
for k in spotbins:
    f=sum(x[j]*W[(-j*k)%large_n] for j in range(large_n))%p
    assert out[k]==f,(k,'large spot mismatch')

# A fully dense one-pole cyclic Green matrix applies in linear arithmetic,
# even though its Fourier-eigenvalue distance from every 0/1 projector
# has rank n. Independently verify all n recurrence equations and 41 bins.
alpha=2
assert pow(alpha,large_n,p)!=1
green,green_cost=apply_one_pole_green(x,p,alpha)
for j in range(large_n):
    assert (green[j]-alpha*green[(j+1)%large_n])%p==x[j]
for k in spotbins:
    gbin=sum(green[j]*W[(-j*k)%large_n] for j in range(large_n))%p
    xbin=sum(x[j]*W[(-j*k)%large_n] for j in range(large_n))%p
    assert gbin*(1-alpha*W[k])%p==xbin

# Compare independent arithmetic ledgers; base classical executor is DIF,
# the projected state recurrence is DIT in a frequency-residue view.
counts=[]
for n in [256,1024,4096,65536]:
    pp,ww=prime_root_for_power_two(n)
    sample=x if n==65536 else [rng.randrange(pp) for _ in range(n)]
    eng=LazySpectralUnion(n,pp,ww)
    y,cost=eng.transform(sample)
    yy,base=classical_fft(sample,pp,ww)
    assert y==yy
    counts.append({
        'n':n,
        'butterflies':cost['butterflies'],
        'lazy_online_field_ops':cost['online_adds']+cost['online_muls'],
        'classical_online_field_ops':base.online_adds+base.online_muls,
        'lazy_total_field_ops_charged':cost['total_field_ops']+eng.prep['total_field_ops'],
        'classical_total_field_ops_charged':base.total_field(),
        'lazy_state_writes':cost['state_writes'],
        'lazy_max_retained_field_words':3*n,
    })

# One scalar e0 measurement from each of ceil(log2 n) masks cannot recover
# n arbitrary amplitudes, even if those masks have unique bit signatures.
N=16
binary_masks=[[int(bool(k&(1<<b))) for k in range(N)] for b in range(4)]
assert len(set(zip(*binary_masks)))==N
assert mat_rank(binary_masks,p)==4

audit.update({
    'full_small_direct_dfts':cases,
    'full_small_direct_coefficients':checked,
    'large_full_dft_length':large_n,
    'large_full_dft_coefficients_exactly_compared':large_n,
    'large_independent_direct_bin_checks':len(spotbins),
    'sharp_rank_witness_cases':rank_cases,
    'explicit_matrix_rank_cross_checks':explicit,
    'symbolic_no_duplicate_sizes':seen,
    'symbolic_nodes_checked':symbolic_nodes,
    'full_state_basis_rank_checks':rank_state_levels,
    'field_operation_comparison':counts,
    'large_green_full_recurrence_residual_checks':large_n,
    'large_green_independent_spectral_bin_checks':len(spotbins),
    'large_green_online_field_ops':green_cost['online_adds']+green_cost['online_muls'],
    'large_green_total_field_ops':green_cost['total_field_ops'],
    'large_green_rank_distance_from_any_projector':large_n,
    'binary_mask_one_coordinate_rank':4,
    'binary_mask_distinct_frequency_signatures':N,
    'conclusion':'No full-DFT arithmetic speedup: lazy-state online counts equal radix2. Green rational correction rank gate sharp but restricted.',
})
(HERE/'AUDIT_RESULTS.json').write_text(json.dumps(audit,indent=2)+'\n')
print(json.dumps(audit,indent=2))
