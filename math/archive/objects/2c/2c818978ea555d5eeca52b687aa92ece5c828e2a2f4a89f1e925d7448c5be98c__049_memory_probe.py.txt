"""Isolated-process peak RSS; includes interpreter/libraries and one batch.

This is an observed process peak, not an exact allocator-by-allocator bound.
"""
import argparse
import json
import os
from pathlib import Path
import platform
import resource
import subprocess
import sys


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--worker',choices=['scipy_fresh','phase5_auto','graph_sparse','graph_complete'])
    args=parser.parse_args()
    if args.worker:
        from benchmark_graph import make_case,fresh,phase5
        from graph_reuse import run_graph
        base,patches,hints,_=make_case('chain',262144,64)
        unit=1 if platform.system()=='Darwin' else 1024
        before=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*unit
        if args.worker=='scipy_fresh': output,stats=fresh(base,patches)
        elif args.worker=='phase5_auto': output,stats=phase5(base,patches)
        else: output,_,stats=run_graph(base,patches,hints if args.worker=='graph_sparse' else None)
        print(json.dumps({'method':args.worker,'n':len(base),'versions':len(patches),
            'peak_rss_bytes':resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*unit,
            'pre_execution_peak_rss_bytes':before,'explicit_output_bytes':output.nbytes}))
        return
    records=[]
    for method in ['scipy_fresh','phase5_auto','graph_sparse','graph_complete']:
        env=dict(os.environ,OPENBLAS_NUM_THREADS='1',OMP_NUM_THREADS='1')
        p=subprocess.run([sys.executable,__file__,'--worker',method],capture_output=True,text=True,check=True,env=env)
        record=json.loads(p.stdout)
        records.append(record); print(json.dumps(record),flush=True)
    (Path(__file__).parent/'results'/'memory_results.json').write_text(json.dumps({'notes':__doc__,'rows':records},indent=2))


if __name__=='__main__': main()
