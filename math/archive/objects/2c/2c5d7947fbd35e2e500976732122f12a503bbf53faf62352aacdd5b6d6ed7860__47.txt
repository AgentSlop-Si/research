   1 """Fresh small controls for a new assessment-level generalization; no source imports."""
   2 import json, random
   3 from pathlib import Path
   4 import sympy as S
   5 w,a=S.symbols('w a')
   6 rng=random.Random(20261010)
   7 records=[]
   8 for m in range(1,4):
   9  for trial in range(2):
  10   A=S.Matrix(m,m,[rng.randrange(-3,5) for _ in range(m*m)])
  11   B=S.Matrix(m,m,[rng.randrange(-3,5) for _ in range(m*m)])
  12   def pencil(M,e):
  13    return S.Poly(S.expand(S.Matrix(m,m,lambda i,j:M[i,j]*(1-a*w**(e+i+j))).det()),a).as_expr()
  14   P,Q=pencil(A,0),pencil(B,1)
  15   # Fixed-degree Sylvester determinant also covers singular coefficient matrices.
  16   pc=[S.Poly(P,a).nth(k) for k in range(m,-1,-1)]
  17   qc=[S.Poly(Q,a).nth(k) for k in range(m,-1,-1)]
  18   syl=S.Matrix([[0]*j+c+[0]*(m-1-j) for c in (pc,qc) for j in range(m)])
  19   R=S.Poly(S.expand(syl.det()),w)
  20   v=m*(m-1)*(4*m+1)//6; D=m*m*(2*m-1); b=(m-1)*m*(2*m-1)//6
  21   U,rem=S.div(R,S.Poly(w**v*(w-1)**(m*m),w))
  22   assert rem.is_zero
  23   coeff=U.as_expr().expand()
  24   assert S.expand(w**(2*b)*coeff.subs(w,1/w)-coeff)==0
  25   assert S.expand(w**D*R.as_expr().subs(w,1/w)-(-1)**(m*m)*R.as_expr())==0
  26   valuation=min(k[0] for k,c in R.terms()) if not R.is_zero else None
  27   records.append({'m':m,'trial':trial,'A':list(A),'B':list(B),'predicted_v':v,'actual_v':valuation,'degree':R.degree(),'quotient_degree':U.degree(),'pass':True})
  28 for m in range(1,9):
  29  v=sum(min(2*i,2*j+1) for i in range(m) for j in range(m))
  30  assert v==m*(m-1)*(4*m+1)//6
  31  # At A=B=I, each difference has exactly one zero at w=1 in characteristic zero.
  32  assert sum(2*m-2*j-1 for j in range(m))==m*m
  33 records.append({'diagonal_specialization':'A=B=I, m=1..8','valuation_formula':True,'w1_multiplicity':True})
  34 path=Path('publication_assessment_20261010/GENERAL_PENCIL_CONTROLS.json')
  35 path.write_text(json.dumps(records,indent=2,default=str)+'\n')
  36 print('Fresh controls passed:',len(records)-1,'arbitrary matrix pairs; diagonal formulas m=1..8')
