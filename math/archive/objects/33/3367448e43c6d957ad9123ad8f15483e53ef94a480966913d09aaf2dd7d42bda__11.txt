  23 def eye(n: int) -> Matrix:
  24     return [[int(i == j) for j in range(n)] for i in range(n)]
  25 
  26 
  27 def zeros(n: int) -> Matrix:
  28     return [[0] * n for _ in range(n)]

  31 def add(a: Matrix, b: Matrix, p: int, sign: int = 1) -> Matrix:
  32     return [[(x + sign * y) % p for x, y in zip(ar, br)] for ar, br in zip(a, b)]
  33 
  34 
  35 def scale(a: Matrix, c: int, p: int) -> Matrix:
  36     return [[c * x % p for x in row] for row in a]
  37 
  38 
  39 def mul(a: Matrix, b: Matrix, p: int) -> Matrix:
  40     bt = list(zip(*b))
  41     return [[sum(x * y for x, y in zip(row, col)) % p for col in bt] for row in a]
  42

  44 def mv(a: Matrix, x: Sequence[int], p: int) -> Vector:
  45     return [sum(u * v for u, v in zip(row, x)) % p for row in a]
  46

  48 def inv(a: Matrix, p: int) -> Matrix:
  49     n = len(a)
  50     aug = [list(row) + ident for row, ident in zip(a, eye(n))]
  51     for col in range(n):
  52         pivot = next((r for r in range(col, n) if aug[r][col] % p), None)
  53         if pivot is None:
  54             raise ValueError("singular test-oracle matrix")
  55         aug[col], aug[pivot] = aug[pivot], aug[col]
  56         z = pow(aug[col][col] % p, -1, p)
  57         aug[col] = [v * z % p for v in aug[col]]
  58         for r in range(n):
  59             if r != col and aug[r][col]:
  60                 z = aug[r][col]
  61                 aug[r] = [(x - z * y) % p for x, y in zip(aug[r], aug[col])]
  62     return [row[n:] for row in aug]
  63

  65 def rank(a: Matrix, p: int) -> int:
  66     if not a:
  67         return 0
  68     b = [row[:] for row in a]
  69     r = 0
  70     for col in range(len(b[0])):
  71         pivot = next((i for i in range(r, len(b)) if b[i][col] % p), None)
  72         if pivot is None:
  73             continue
  74         b[r], b[pivot] = b[pivot], b[r]
  75         z = pow(b[r][col] % p, -1, p)
  76         b[r] = [x * z % p for x in b[r]]
  77         for i in range(r + 1, len(b)):
  78             z = b[i][col]
  79             if z:
  80                 b[i] = [(x - z * y) % p for x, y in zip(b[i], b[r])]
  81         r += 1
  82         if r == len(b):
  83             break
  84     return r

  87 def root_of_unity(n: int, p: int) -> int:
  88     if n < 2 or n & (n - 1) or (p - 1) % n:
  89         raise ValueError("test roots require dyadic n >= 2 dividing p-1")
  90     w = pow(3, (p - 1) // n, p)
  91     if pow(w, n, p) != 1 or pow(w, n // 2, p) == 1:
  92         raise ValueError("3 did not yield a primitive root at this test length")
  93     return w
  94 
  95 
  96 def fourier(n: int, p: int) -> Matrix:
  97     w = root_of_unity(n, p)
  98     return [[pow(w, i * j, p) for j in range(n)] for i in range(n)]

 101 def shift_diagonal(n: int, p: int) -> tuple[Matrix, Matrix]:
 102     w = root_of_unity(n, p)
 103     s, d = zeros(n), zeros(n)
 104     for j in range(n):
 105         s[(j + 1) % n][j] = 1
 106         d[j][j] = pow(w, j, p)
 107     return s, d

 110 def poly_matrix(coeff: Sequence[int], k: Matrix, p: int) -> Matrix:
 111     n = len(k)
 112     out = zeros(n)
 113     for c in reversed(coeff):
 114         out = add(mul(out, k, p), scale(eye(n), c, p), p)
 115     return out

 118 def inverse_coefficients(mu: Sequence[int], p: int) -> Vector:
 119     """Coefficients of (mu(1)-mu(z))/((1-z)*mu(1)), ascending order."""
 120     if len(mu) < 2 or mu[-1] % p == 0:
 121         raise ValueError("annihilator must have positive stated degree")
 122     m = sum(mu) % p
 123     if m == 0:
 124         raise ValueError("resonance: mu(1)=0 does not certify invertibility")
 125     mi = pow(m, -1, p)
 126     suffix = 0
 127     q = [0] * (len(mu) - 1)
 128     for j in range(len(mu) - 1, 0, -1):
 129         suffix = (suffix + mu[j]) % p
 130         q[j - 1] = suffix * mi % p
 131     return q

 134 def certificate(a: Matrix, b: Matrix, mu: Sequence[int], p: int) -> dict[str, bool]:
 135     n = len(a)
 136     s, d = shift_diagonal(n, p)
 137     k = mul(b, a, p)
 138     h = add(eye(n), k, p, -1)
 139     j = add(eye(n), mul(a, b, p), p, -1)
 140     cubic = add(mul(mul(h, d, p), b, p), mul(mul(b, s, p), j, p), p, -1)
 141     seed = mv(h, [1] * n, p) == [row[0] for row in b]
 142     mu_valid = len(mu) >= 2 and mu[-1] % p != 0
 143     closure = mu_valid and poly_matrix(mu, k, p) == zeros(n)
 144     nonresonant = mu_valid and sum(mu) % p != 0
 145     return {"cubic_phase": cubic == zeros(n), "seed": seed,
 146             "annihilator": closure, "nonresonant": nonresonant}

 149 def dense_kernel(a: Matrix, b: Matrix, mu: Sequence[int], x: Sequence[int], p: int) -> Vector:
 150     q = inverse_coefficients(mu, p)
 151     u = mv(b, x, p)
 152     v = [q[-1] * z % p for z in u]
 153     for c in reversed(q[:-1]):
 154         v = mv(b, mv(a, v, p), p)
 155         v = [(z + c * t) % p for z, t in zip(v, u)]
 156     return v

 159 @dataclass
 160 class Counts:
 161     mul: int = 0
 162     add: int = 0
 163     inverse: int = 0
 164 
 165     def multiply(self, a: int, b: int, p: int) -> int:
 166         self.mul += 1
 167         return a * b % p

 169     def plus(self, a: int, b: int, p: int) -> int:
 170         self.add += 1

 173     def invert(self, a: int, p: int) -> int:

 178 class GreenPlan:
 179     """L (I-diag(weights) P_cycle)^-1 R with fully charged scalar scan.
 180 
 181     The cycle list c means (P z)[c[t]] = z[c[t+1]]. Individual zero
 182     weights are valid. Only product(weights)==1 is singular.
 183     Numeric parameter generation is NOT included in the scan setup counts.
 184     """
 185     def __init__(self, cycle: Sequence[int], weights: Sequence[int],
 186                  left: Sequence[int], right: Sequence[int], p: int):
 187         self.n = len(cycle)
 188         if self.n == 0 or any(len(x) != self.n for x in (weights, left, right)):
 189             raise ValueError("nonempty equal-length arrays required")
 190         seen = [False] * self.n
 191         for c in cycle:
 192             if not isinstance(c, int) or c < 0 or c >= self.n or seen[c]:
 193                 raise ValueError("cycle must list every coordinate exactly once")
 194             seen[c] = True
 195         self.cycle, self.p = list(cycle), p
 196         self.a = [x % p for x in weights]
 197         self.left = [x % p for x in left]
 198         self.right = [x % p for x in right]
 199         self.prefix: Vector = []
 200         self.setup = Counts()
 201         total = 1
 202         for c in self.cycle:
 203             self.prefix.append(total)
 204             total = self.setup.multiply(total, self.a[c], p)
 205         denom = self.setup.plus(1, -total, p)
 206         if denom == 0:
 207             raise ValueError("singular Green cycle: product(weights)=1")
 208         self.denom_inv = self.setup.invert(denom, p)
 209 
 210     def apply(self, x: Sequence[int], count: Counts | None = None) -> Vector:
 211         if len(x) != self.n:
 212             raise ValueError("wrong vector length")
 213         c = count if count is not None else Counts()
 214         p = self.p
 215         v = [c.multiply(r, t, p) for r, t in zip(self.right, x)]
 216         total = 0
 217         for pref, index in zip(self.prefix, self.cycle):
 218             total = c.plus(total, c.multiply(pref, v[index], p), p)
 219         z = [0] * self.n
 220         z[self.cycle[0]] = c.multiply(total, self.denom_inv, p)
 221         nxt = z[self.cycle[0]]
 222         for t in range(self.n - 1, 0, -1):
 223             index = self.cycle[t]
 224             z[index] = c.plus(v[index], c.multiply(self.a[index], nxt, p), p)
 225             nxt = z[index]
 226         return [c.multiply(l, t, p) for l, t in zip(self.left, z)]
 227 
 228     def denominator(self) -> Matrix:
 229         out = eye(self.n)
 230         for t, index in enumerate(self.cycle):
 231             nxt = self.cycle[(t + 1) % self.n]
 232             out[index][nxt] = (out[index][nxt] - self.a[index]) % self.p
 233         return out
 234 
 235     def oracle_matrix(self) -> Matrix:
 236         out = inv(self.denominator(), self.p)
 237         return [[self.left[i] * out[i][j] * self.right[j] % self.p
 238                  for j in range(self.n)] for i in range(self.n)]

 241 class PairPlan:
 242     def __init__(self, g0: GreenPlan, g1: GreenPlan):
 243         if (g0.n, g0.p) != (g1.n, g1.p):
 244             raise ValueError("incompatible Green plans")
 245         self.g0, self.g1, self.n, self.p = g0, g1, g0.n, g0.p
 246 
 247     def apply(self, x: Sequence[int], count: Counts | None = None) -> Vector:
 248         c = count if count is not None else Counts()
 249         v, w = self.g0.apply(x, c), self.g1.apply(x, c)
 250         return [c.plus(a, b, self.p) for a, b in zip(v, w)]
 251 
 252     def oracle_matrix(self) -> Matrix:
 253         return add(self.g0.oracle_matrix(), self.g1.oracle_matrix(), self.p)

 256 def fast_polynomial_kernel(a: PairPlan, b: PairPlan, q: Sequence[int],
 257                            x: Sequence[int]) -> tuple[Vector, Counts]:
 258     """Evaluate p(BA)B x. This is an inverse ONLY with a valid annihilator."""
 259     if len(q) == 0 or (a.n, a.p) != (b.n, b.p):
 260         raise ValueError("nonempty polynomial and compatible operators required")
 261     c = Counts()
 262     u = b.apply(x, c)
 263     v = [c.multiply(q[-1], t, a.p) for t in u]
 264     for coeff in reversed(q[:-1]):
 265         v = b.apply(a.apply(v, c), c)
 266         v = [c.plus(z, c.multiply(coeff, t, a.p), a.p) for z, t in zip(v, u)]
 267     return v, c

 270 def random_green(n: int, p: int, rng: random.Random, zero_edge: bool = False) -> GreenPlan:
 271     cycle = list(range(n))
 272     rng.shuffle(cycle)
 273     while True:
 274         weights = [rng.randrange(1, p) for _ in range(n)]
 275         if zero_edge:
 276             weights[rng.randrange(n)] = 0
 277         left = [rng.randrange(1, p) for _ in range(n)]
 278         right = [rng.randrange(1, p) for _ in range(n)]
 279         try:
 280             return GreenPlan(cycle, weights, left, right, p)
 281         except ValueError:
 282             continue

 285 def random_pair(n: int, p: int, rng: random.Random) -> PairPlan:
 286     return PairPlan(random_green(n, p, rng), random_green(n, p, rng))

 293 def nilpotent(n: int, degree: int, p: int) -> Matrix:
 294     """Dense-looking similarity of block Jordan nilpotents. Exact degree=degree."""
 295     if not 1 <= degree <= n:
 296         raise ValueError("invalid nilpotency degree")
 297     j = zeros(n)
 298     for start in range(0, n, degree):
 299         for i in range(start, min(n, start + degree) - 1):
 300             j[i][i + 1] = 1
 301     g = [[int(i >= k) for k in range(n)] for i in range(n)]
 302     return mul(mul(g, j, p), inv(g, p), p)

 305 def oracle_feedback(target: Matrix, k: Matrix, p: int) -> tuple[Matrix, Matrix]:
 306     """Uses a dense target oracle: algebra control, NOT candidate synthesis."""
 307     b = mul(add(eye(len(k)), k, p, -1), target, p)
 308     a = mul(inv(b, p), k, p)
 309     assert mul(b, a, p) == k
 310     return a, b

 528                          GreenPlan(orders[3], [0]*n, [v*half % p for v in ds], [1]*n, p))
 529             k = mul(b.oracle_matrix(), a.oracle_matrix(), p)
 530             assert k == [[ds[i] if i == j else 0 for j in range(n)] for i in range(n)]
 531             columns = []
 532             v = [1] * n
 533             for _ in range(n):
 534                 columns.append(v)
 535                 v = mv(k, v, p)
 536             assert rank([list(row) for row in zip(*columns)], p) == n
 537             cases += 1
 538     return {"status": "pass", "cases": cases,
 539             "scope": "Explicit Vandermonde witness for the generic-degree-N theorem for any fixed four cycle orders."}

 542 def test_pure_feedback_obstruction() -> dict:
 543     rng = random.Random(631)
 544     details = []
 545     for p in FIELDS:
 546         for n in (16, 32, 64, 128):
 547             finv = inv(fourier(n, p), p)
 548             for repetition in range(3):
 549                 pg = random_green(n, p, rng)
 550                 qg = random_green(n, p, rng)
 551                 # The no-go proof uses the denominators, not endpoint-scaled maps.
 552                 pm, qm = pg.denominator(), qg.denominator()
 553                 lhs = mul(pm, finv, p)
 554                 rhs = add(mul(pm, qm, p), eye(n), p, -1)
 555                 lhs_min = min(sum(z != 0 for z in row) for row in lhs)
 556                 rhs_max = max(sum(z != 0 for z in row) for row in rhs)
 557                 assert lhs_min >= (n + 1) // 2 and rhs_max <= 4 and lhs != rhs
 558                 details.append({"p": p, "N": n, "repetition": repetition,
 559                                 "min_support_P_Finverse": lhs_min,
 560                                 "max_support_PQ_minus_I": rhs_max})
 561     return {"status": "pass", "cases": len(details), "details": details,
 562             "scope": "Independent arithmetic checks of a support obstruction proved in THEORY.md."}
