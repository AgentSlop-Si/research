   1 """FFT symmetry-cache research prototype.
   2 
   3 Illustrates how circular shifts, reversals, and binary complements can share
   4 an FFT result. It is NOT a faster general FFT; canonicalization, lookup,
   5 complex arithmetic, output materialization, and numerical precision cost time.
   6 
   7 Python >= 3.10, standard library only.
   8 """
   9 from __future__ import annotations
  10 import cmath
  11 import math
  12 import random
  13 import statistics
  14 from time import perf_counter

  17 def ordinary_fft(a):
  18     n = len(a)
  19     if n == 1: return (complex(a[0]),)
  20     E=ordinary_fft(a[0::2]); O=ordinary_fft(a[1::2]); out=[0j]*n
  21     for k in range(n//2):
  22         v=cmath.exp(-2j*math.pi*k/n)*O[k]
  23         out[k]=E[k]+v
  24         out[k+n//2]=E[k]-v
  25     return tuple(out)
  26 
  27 
  28 def least_rotation(v):
  29     """Booth algorithm: starting position of minimal cyclic rotation, O(m)."""
  30     n=len(v)
  31     if n==1:return 0
  32     doubled=v+v
  33     i,j,k=0,1,0
  34     while i<n and j<n and k<n:
  35         a=doubled[i+k]; b=doubled[j+k]
  36         if a==b:
  37             k+=1
  38         elif a>b:
  39             i=i+k+1
  40             if i<=j:i=j+1
  41             k=0
  42         else:
  43             j=j+k+1
  44             if j<=i:j=i+1
  45             k=0
  46     return min(i,j)
  47

  49 def canonical_binary(v):
  50     """Return (lexically minimal orbit representative, sign, shift, complement).
  51 
  52     Representative c[j] = (1 - v[(shift+sign*j)%m]) if complement else
  53                        v[(shift+sign*j)%m].
  54     sign is +1 or -1; use inverse DFT identity to get FFT(v) from FFT(c).
  55     Symmetry group size <= 4m (shift, reversal, optional complement).
  56     """
  57     n=len(v)
  58     v=tuple(v)
  59     assert all(x in (0,1) for x in v)
  60     best=None
  61     info=None
  62     for sign in (1,-1):
  63         base=tuple(v[(sign*j)%n] for j in range(n))
  64         for complement in (False,True):
  65             arr=tuple(1-x for x in base) if complement else base
  66             start=least_rotation(arr)
  67             candidate=arr[start:]+arr[:start]
  68             shift=(sign*start)%n
  69             order=(bool(complement),sign==-1,shift)
  70             if best is None or candidate<best or (candidate==best and order<info[3]):
  71                 best=candidate
  72                 info=(sign,shift,complement,order)
  73     sign,shift,complement,_=info
  74     return best,sign,shift,complement
  75

  77 def recover_fft(canonical_output, sign, shift, complement):
  78     """Reconstruct FFT(v) from FFT(c), where c is an orbit transform of v."""
  79     n=len(canonical_output)
  80     if sign==1 and shift==0 and not complement:
  81         return canonical_output
  82     out=[]
  83     z=1+0j
  84     step=cmath.exp(-2j*math.pi*shift/n)
  85     for k in range(n):
  86         value=canonical_output[(sign*k)%n]
  87         if complement:
  88             value=(n if k==0 else 0)-value
  89         out.append(z*value)
  90         z*=step
  91     return tuple(out)
  92 
  93 
  94 class SymmetryFFT:
  95     def __init__(self):
  96         self.cache={}
  97         self.hits=0
  98         self.misses=0
  99         self.butterflies=0
 100         self.reconstructed=0
 101         self.canonical_scanned=0
 102 
 103     def fft(self, v):
 104         n=len(v)
 105         if n==1:
 106             return (complex(v[0]),)
 107         key,sign,shift,comp=canonical_binary(v)
 108         self.canonical_scanned+=n
 109         cache_key=(n,key)
 110         F=self.cache.get(cache_key)
 111         if F is None:
 112             self.misses+=1
 113             even=self.fft(key[0::2]); odd=self.fft(key[1::2])
 114             out=[0j]*n
 115             for k in range(n//2):
 116                 w=cmath.exp(-2j*math.pi*k/n)*odd[k]
 117                 out[k]=even[k]+w
 118                 out[k+n//2]=even[k]-w
 119                 self.butterflies+=1
 120             F=tuple(out)
 121             self.cache[cache_key]=F
 122         else:
 123             self.hits+=1
 124         if sign!=1 or shift!=0 or comp:
 125             self.reconstructed+=n
 126         return recover_fft(F,sign,shift,comp)

 129 def interleaved_orbit_inputs(n, block_len=128, seed=7):
 130     """At radix-2 recursion depth, length-block_len children are orbit variants."""
 131     assert n%block_len==0
 132     rand=random.Random(seed)
 133     template=[rand.randrange(2) for _ in range(block_len)]
 134     groups=n//block_len
 135     vec=[0]*n
 136     for offset in range(groups):
 137         orientation=rand.choice((-1,1))
 138         shift=rand.randrange(block_len)
 139         comp=rand.randrange(2)
 140         for j in range(block_len):
 141             vec[offset+groups*j]=template[(shift+orientation*j)%block_len]^comp
 142     return vec

 207 def affine_fft_from_base(base_fft, a, b, complement=False):
 208     """FFT of B[j]=A[(a*j+b)%m] xor complement (for binary A).
 209 
 210     Requires gcd(a,m)=1. Reconstruction is O(m) without recomputing an FFT.
 211     """
 212     n=len(base_fft)
 213     ainv=pow(a,-1,n)
 214     out=[]
 215     phase=1+0j
 216     step=cmath.exp(2j*math.pi*ainv*b/n)
 217     for k in range(n):
 218         value=phase*base_fft[(ainv*k)%n]
 219         if complement:
 220             value=(n if k==0 else 0)-value
 221         out.append(value)
 222         phase*=step
 223     return tuple(out)

 226 def sparse_edit_fft(base_fft, changes):
 227     """FFT of modified input using a list of (position, additive_delta).
 228 
 229     O(n * number_of_changes) operations plus O(n) output writes.
 230     """
 231     n=len(base_fft)
 232     out=list(base_fft)
 233     for pos,delta in changes:
 234         w=cmath.exp(-2j*math.pi*pos/n)
 235         z=1+0j
 236         for k in range(n):
 237             out[k]+=delta*z
 238             z*=w
 239     return tuple(out)
