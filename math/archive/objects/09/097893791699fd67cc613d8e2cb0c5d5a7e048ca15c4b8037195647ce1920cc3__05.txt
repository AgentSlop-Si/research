1: #!/usr/bin/env python3
2: """Predecode-median Fourier sketch (v6 research prototype).
3: 
4: The algorithm continues the v5 frequency-shift sketch, but moves the robust
5: aggregation before the K families of short M-point decoders.  Each randomized
6: row stores only one length-K mixed spectrum plus its shifts/phases.  For every
7: local Fourier coordinate A_t[p], we gather the row estimates, take separate
8: real/imaginary medians, and run the deterministic M-point decoder once.
9: 
10: This is a research prototype in the algebraic complex-arithmetic model.  It is
11: not a machine-precision FFT replacement.
12: """

30: def unitary_fft(x: ComplexArray, axis: int = -1) -> ComplexArray:
31:     x = np.asarray(x, dtype=np.complex128)
32:     n = x.shape[axis]
33:     if n <= 0:
34:         raise ValueError("FFT axis must be non-empty")
35:     return np.fft.fft(x, axis=axis) / math.sqrt(n)

42:         raise ValueError("FFT axis must be non-empty")
43:     return np.fft.ifft(x, axis=axis) * math.sqrt(n)

53: @dataclass(frozen=True)
54: class SketchRandomness:
55:     shifts: IntArray
56:     phases: ComplexArray
57: 
58:     def validate(self, k: int, m: int) -> None:
59:         if self.shifts.shape != (m,) or self.phases.shape != (m,):
60:             raise ValueError("invalid randomness shape")
61:         if np.any(self.shifts < 0) or np.any(self.shifts >= k):
62:             raise ValueError("shift outside Z_K")
63:         if not np.allclose(np.abs(self.phases), 1.0):
64:             raise ValueError("phases must have unit magnitude")

67: @dataclass(frozen=True)
68: class CompressedRows:
69:     """Compressed randomized rows needed for deferred recovery."""
70: 
71:     spectra: ComplexArray  # shape (R,K)
72:     shifts: IntArray       # shape (R,M)
73:     phases: ComplexArray   # shape (R,M)
74: 
75:     @property
76:     def rows(self) -> int:
77:         return int(self.spectra.shape[0])
78: 
79:     @property
80:     def k(self) -> int:
81:         return int(self.spectra.shape[1])
82: 
83:     @property
84:     def m(self) -> int:
85:         return int(self.shifts.shape[1])
86: 
87:     def validate(self) -> None:
88:         r, k = self.spectra.shape
89:         if self.shifts.shape != (r, self.m) or self.phases.shape != (r, self.m):
90:             raise ValueError("compressed row dimensions disagree")
91:         if np.any(self.shifts < 0) or np.any(self.shifts >= k):
92:             raise ValueError("invalid compressed-row shift")

95: def draw_randomness(k: int, m: int, rng: np.random.Generator) -> SketchRandomness:
96:     if k <= 0 or m <= 0:
97:         raise ValueError("k,m must be positive")
98:     return SketchRandomness(
99:         shifts=rng.integers(0, k, size=m, dtype=np.int64),
100:         phases=FOURTH_ROOTS[rng.integers(0, 4, size=m)],
101:     )

104: def compress_one_row(x: ComplexArray, m: int, state: SketchRandomness) -> ComplexArray:
105:     """Return Z[s] = sum_u conj(xi_u) A_u[s+q_u]."""
106:     x = np.asarray(x, dtype=np.complex128)
107:     if x.ndim != 1 or x.size == 0 or m <= 0 or x.size % m:
108:         raise ValueError("m must divide a non-empty one-dimensional x")
109:     n = int(x.size)
110:     k = n // m
111:     state.validate(k, m)
112:     x2 = x.reshape(k, m)
113:     r = np.arange(k, dtype=np.float64)[:, None]
114:     modulation = np.conj(state.phases)[None, :] * np.exp(
115:         -2j * np.pi * r * state.shifts[None, :] / k
116:     )
117:     z = np.sum(x2 * modulation, axis=1)
118:     return unitary_fft(z)

121: def build_compressed_rows(
122:     x: ComplexArray,
123:     m: int,
124:     rows: int,
125:     rng: np.random.Generator,
126: ) -> CompressedRows:
127:     """Build R independent compressed rows without materializing R full outputs."""
128:     x = np.asarray(x, dtype=np.complex128)
129:     if rows <= 0 or rows % 2 != 1:
130:         raise ValueError("rows must be a positive odd integer")
131:     if x.ndim != 1 or x.size == 0 or m <= 0 or x.size % m:
132:         raise ValueError("m must divide a non-empty one-dimensional x")
133:     k = x.size // m
134:     spectra = np.empty((rows, k), dtype=np.complex128)
135:     shifts = np.empty((rows, m), dtype=np.int64)
136:     phases = np.empty((rows, m), dtype=np.complex128)
137:     for j in range(rows):
138:         state = draw_randomness(k, m, rng)
139:         shifts[j] = state.shifts
140:         phases[j] = state.phases
141:         spectra[j] = compress_one_row(x, m, state)
142:     result = CompressedRows(spectra=spectra, shifts=shifts, phases=phases)
143:     result.validate()
144:     return result

147: def local_estimate_from_row(
148:     spectrum: ComplexArray,
149:     state: SketchRandomness,
150: ) -> ComplexArray:
151:     """Recover all local estimates Ahat[p,t] from one compressed row."""
152:     spectrum = np.asarray(spectrum, dtype=np.complex128)
153:     if spectrum.ndim != 1 or spectrum.size == 0:
154:         raise ValueError("spectrum must be a non-empty vector")
155:     k = spectrum.size
156:     m = state.shifts.size
157:     state.validate(k, m)
158:     p = np.arange(k, dtype=np.int64)[:, None]
159:     return state.phases[None, :] * spectrum[(p - state.shifts[None, :]) % k]

162: def decode_local_spectra(local_spectra: ComplexArray) -> ComplexArray:
163:     """Decode exact/estimated A_t[p] into standard-order length-N DFT values."""
164:     local_spectra = np.asarray(local_spectra, dtype=np.complex128)
165:     if local_spectra.ndim != 2 or 0 in local_spectra.shape:
166:         raise ValueError("local_spectra must have shape (K,M)")
167:     k, m = local_spectra.shape
168:     n = k * m
169:     p = np.arange(k, dtype=np.float64)[:, None]
170:     t = np.arange(m, dtype=np.float64)[None, :]
171:     pre = local_spectra * np.exp(-2j * np.pi * p * t / n)
172:     values = unitary_fft(pre, axis=1)  # [p,v]
173:     return np.ascontiguousarray(values.T.reshape(n))

186: def coordinatewise_complex_median(values: ComplexArray, axis: int = 0) -> ComplexArray:
187:     values = np.asarray(values, dtype=np.complex128)
188:     if values.shape[axis] % 2 != 1:
189:         raise ValueError("median axis must have odd length")
190:     return np.median(values.real, axis=axis) + 1j * np.median(values.imag, axis=axis)

193: def predecode_median_from_compressed(rows: CompressedRows) -> ComplexArray:
194:     """Take local medians and execute the short decoder only once.
195: 
196:     The function processes one p-slice at a time.  It stores R*K compressed
197:     spectral values, not R*N materialized local estimates or outputs.
198:     """
199:     rows.validate()
200:     rcount, k = rows.spectra.shape
201:     m = rows.m
202:     if rcount % 2 != 1:
203:         raise ValueError("an odd number of rows is required")
204:     n = k * m
205:     out = np.empty(n, dtype=np.complex128)
206:     row_ids = np.arange(rcount, dtype=np.int64)[:, None]
207:     t = np.arange(m, dtype=np.float64)
208:     v_indices = np.arange(m, dtype=np.int64)
209: 
210:     for p in range(k):
211:         gather_idx = (p - rows.shifts) % k
212:         estimates = rows.phases * rows.spectra[row_ids, gather_idx]
213:         local_median = coordinatewise_complex_median(estimates, axis=0)
214:         pre = local_median * np.exp(-2j * np.pi * p * t / n)
215:         decoded = unitary_fft(pre)
216:         out[p + k * v_indices] = decoded
217:     return out

254: def local_variance_vector(x: ComplexArray, m: int) -> npt.NDArray[np.float64]:
255:     """Exact variance of each Ahat[p,t]-A[p,t], independent of p."""
256:     x = np.asarray(x, dtype=np.complex128)
257:     if x.ndim != 1 or x.size == 0 or m <= 0 or x.size % m:
258:         raise ValueError("m must divide x.size")
259:     x2 = x.reshape(x.size // m, m)
260:     energies = np.sum(np.abs(x2) ** 2, axis=0)
261:     total = float(np.sum(energies))
262:     k = x.size // m
263:     return np.maximum(0.0, total - energies) / k

266: def predecode_median_uniform_bound(
267:     x: ComplexArray,
268:     m: int,
269:     rows: int,
270:     delta: float,
271:     *,
272:     refined: bool = True,
273: ) -> float:
274:     """Proved simultaneous absolute output-error bound.
275: 
276:     With rows=2*t-1 and C_t=binom(2*t-1,t), the local coordinate medians
277:     obey a 2*t-moment bound.  The final M-point decoder is then charged once.
278:     """
279:     x = np.asarray(x, dtype=np.complex128)
280:     if rows <= 0 or rows % 2 != 1:
281:         raise ValueError("rows must be positive and odd")
282:     if not (0.0 < delta < 1.0):
283:         raise ValueError("delta must lie in (0,1)")
284:     if x.ndim != 1 or x.size == 0 or m <= 0 or x.size % m:
285:         raise ValueError("m must divide x.size")
286:     tmed = (rows + 1) // 2
287:     c_t = math.comb(2 * tmed - 1, tmed)
288:     amp = math.sqrt(2.0) * (2.0 * x.size * c_t / delta) ** (1.0 / (2.0 * tmed))
289:     if refined:
290:         variances = local_variance_vector(x, m)
291:         return amp * float(np.sum(np.sqrt(variances))) / math.sqrt(m)
292:     return amp * math.sqrt(m * (m - 1) / x.size) * float(np.linalg.norm(x))

295: def predecode_cost_proxy(n: int, m: int, rows: int) -> float:
296:     """Arithmetic proxy for R local rows plus one final decoder.
297: 
298:     R*N*(1 + log2(K)/M) + N*log2(M), K=N/M.
299:     """
300:     if n <= 1 or m <= 0 or n % m or rows <= 0:
301:         raise ValueError("invalid dimensions")
302:     k = n // m
303:     return rows * n * (2.0 + math.log2(k) / m) + n * math.log2(m)

322: def local_sparse_failure_bound(n: int, m: int, rows: int, h: int) -> float:
323:     """Union-bound failure probability for h-sparse local spectra.
324: 
325:     Each source column has at most h nonzero length-K Fourier coefficients.
326:     A row is collision-free for a target local coordinate with probability at
327:     least 1-rho, rho <= (M-1)h/K.  A median fails only if at least t rows are
328:     bad, where rows=2*t-1.
329:     """
330:     if n <= 0 or m <= 0 or n % m or rows <= 0 or rows % 2 != 1 or h < 0:
331:         raise ValueError("invalid arguments")
332:     k = n // m
333:     tmed = (rows + 1) // 2
334:     rho = min(1.0, (m - 1) * h / k)
335:     return min(1.0, n * math.comb(rows, tmed) * rho**tmed)

338: def make_local_sparse_signal(
339:     k: int,
340:     m: int,
341:     h: int,
342:     rng: np.random.Generator,
343: ) -> tuple[ComplexArray, ComplexArray]:
344:     """Create x whose K-point transform in each of M columns is h-sparse."""
345:     if k <= 0 or m <= 0 or not (0 <= h <= k):
346:         raise ValueError("invalid k,m,h")

352:     x = unitary_ifft(a, axis=0).reshape(k * m)

356: def tone_moment_theory(k: int, m: int, amplitude: float = 1.0) -> dict[str, float]:
357:     """Exact one-output moments for a single local Fourier tone.
358: 
359:     At p equal to the tone frequency, C~Bin(M-1,1/K) colliders contribute
360:     independent fourth-root phases.  Conditional on C, E|sum_{j<=C} zeta_j|^4
361:     = 2*C^2-C.
362:     """
363:     if k <= 0 or m <= 1:
364:         raise ValueError("require k>0,m>1")
365:     c = m - 1
366:     variance = amplitude**2 * c / (m * k)
367:     fourth = amplitude**4 * (c / k + 2.0 * c * (c - 1) / (k * k)) / (m * m)
368:     kurtosis = fourth / (variance * variance)
369:     return {
370:         "variance": variance,
371:         "fourth_moment": fourth,
372:         "kurtosis": kurtosis,
373:         "closed_form_kurtosis": k / c + 2.0 - 2.0 / c,
374:     }

377: def simulate_tone_moments(
378:     k: int,
379:     m: int,
380:     trials: int,
381:     rng: np.random.Generator,
382: ) -> dict[str, float]:
383:     """Fast direct simulation of the single-local-tone output error."""
384:     q = rng.integers(0, k, size=(trials, m), dtype=np.int64)
385:     xi = FOURTH_ROOTS[rng.integers(0, 4, size=(trials, m))]
386:     t0 = 0
387:     alpha = np.exp(-2j * np.pi * np.arange(m) / (k * m))
388:     hit = q[:, 1:] == q[:, [t0]]
389:     error = (
390:         np.conj(xi[:, t0])
391:         * np.sum(alpha[None, 1:] * xi[:, 1:] * hit, axis=1)
392:         / math.sqrt(m)
393:     )
394:     abs2 = np.abs(error) ** 2
395:     return {
396:         "empirical_variance": float(np.mean(abs2)),
397:         "empirical_fourth_moment": float(np.mean(abs2**2)),
398:         "empirical_kurtosis": float(np.mean(abs2**2) / np.mean(abs2) ** 2),
399:         "zero_fraction": float(np.mean(abs2 == 0.0)),
400:     }

403: def _case_signals(k: int, m: int, rng: np.random.Generator) -> dict[str, ComplexArray]:
404:     n = k * m
405:     r = np.arange(k, dtype=np.float64)
406:     tone = max(1, k // 5)
407:     out: dict[str, ComplexArray] = {
408:         "flat": np.ones(n, dtype=np.complex128),
409:         "random_phase": np.exp(2j * np.pi * rng.random(n)),
410:         "gaussian": rng.normal(size=n) + 1j * rng.normal(size=n),
411:     }
412:     local = np.zeros((k, m), dtype=np.complex128)
413:     local[:, m // 3] = np.exp(2j * np.pi * tone * r / k)
414:     out["local_spike_tone"] = local.reshape(n)
415:     pattern = np.exp(2j * np.pi * rng.random(m))
416:     out["repeated_pattern_tone"] = (
417:         np.exp(2j * np.pi * tone * r[:, None] / k) * pattern[None, :]
418:     ).reshape(n)
419:     spike = np.zeros(n, dtype=np.complex128)
420:     spike[n // 3] = 1.0
421:     out["single_spike"] = spike
422:     return out
