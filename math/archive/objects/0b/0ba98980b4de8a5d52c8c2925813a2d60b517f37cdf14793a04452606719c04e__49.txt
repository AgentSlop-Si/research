9: def matrix_of(field, n, action):
10:     columns = [action([field.one if i==j else field.zero for i in range(n)]) for j in range(n)]
11:     return [[columns[j][i] for j in range(n)] for i in range(n)]

14: def scaled(field, s, a):
15:     return [[field.mul(s, v) for v in row] for row in a]

18: def power(field, a, k):
19:     out = identity(field, len(a))
20:     while k:
21:         if k&1: out = mm(field, out, a)
22:         k >>= 1
23:         if k: a = mm(field, a, a)
24:     return out

27: def one_case(field, n, label):
28:     q = field.root(n); plan = prepare_grading(field, n, q)
29:     base = prepare(field, n, q); sparse = prepare_sparse(field, n, q)
30:     C = build_operators(field, sparse, base.kernel)['C']
31:     R = matrix_of(field, n, lambda x: apply_r(field, plan, x))
32:     D = matrix_of(field, n, lambda x: apply_d(field, plan, x))
33:     H = matrix_of(field, n, lambda x: horner_c(field, sparse, base.kernel, x)[0])
34:     ident = identity(field, n)
35:     assert mm(field, R, C) == scaled(field, field.power(q,n-1), mm(field,C,R))
36:     assert mm(field, H, R) == mm(field, D, H)
37:     assert power(field,R,n) == ident == power(field,D,n)
38:     K = power(field,R,n//2)
39:     assert mm(field,K,K) == ident
40:     assert mm(field,K,C) == scaled(field,field.neg(field.one),mm(field,C,K))
41:     S = matrix_of(field,n,lambda x:split_pairs(field,plan,x))
42:     Si = matrix_of(field,n,lambda x:join_pairs(field,plan,x))
43:     assert mm(field,S,Si) == ident == mm(field,Si,S)
44:     split = mm(field,mm(field,S,C),Si)
45:     m=n//2
46:     assert all(split[i][j]==field.zero for i in range(n) for j in range(n) if (i<m)==(j<m))
47:     outputs=0
48:     for j in range(n):
49:         x=[field.one if i==j else field.zero for i in range(n)]
50:         assert apply_r(field,plan,apply_r(field,plan,x),inverse=True)==x
51:         assert join_pairs(field,plan,split_pairs(field,plan,x))==x
52:         y=mv(field,H,x)
53:         for _ in range(4):
54:             x=apply_r(field,plan,x); y=apply_d(field,plan,y)
55:             assert y==mv(field,H,x)
56:             assert sum(v!=field.zero for v in x)==1
57:             outputs+=1
58:     mixed=[field.integer((3*i+2)%11-5) for i in range(n)]
59:     counted=Counted(field)
60:     apply_r(counted,plan,mixed); apply_d(counted,plan,mixed)
61:     assert dict(counted.count)=={'mul':3*n-1,'add':n-1}
62:     c=Counted(field); split_pairs(c,plan,mixed)
63:     assert dict(c.count)=={'mul':3*m,'add':m,'sub':m}
64:     c=Counted(field); join_pairs(c,plan,mixed)
65:     assert dict(c.count)=={'add':m,'sub':m,'mul':m}
66:     assert all(field.mul(plan['r'][i],plan['inverse_r'][i])==field.one for i in range(n))
67:     assert all(field.mul(plan['h'][i],plan['h'][(i+m)%n])==field.one for i in range(n))
68:     # This version duplicates a root table deliberately; count the complete setup.
69:     assert plan['preparation']=={'mul':21*n//2+(n.bit_length()-1)-3,
70:         'sub':2*n-1,'constant_inverse':2,'neg':n+1}
71:     return dict(n=n,field=label,basis_vectors=n,related_outputs=outputs,
72:                 preparation=plan['preparation'],input_output_update=dict(counted.count))
