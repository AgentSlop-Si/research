   7 # Independent fixed-degree field implementation. Inversion uses elimination.
   8 class K:
   9     def __init__(self, value=0):
  10         if isinstance(value,K): self.c=value.c; return
  11         c=list(value) if isinstance(value,(list,tuple)) else [value]
  12         c=[Q(x) for x in c]+[Q(0)]*max(0,4-len(c))
  13         for j in range(len(c)-1,3,-1):
  14             for k in range(1,5): c[j-k]-=c[j]
  15         self.c=tuple(c[:4])
  16     def __add__(self,v):
  17         if isinstance(v,E):return NotImplemented
  18         v=K(v); return K([a+b for a,b in zip(self.c,v.c)])
  19     __radd__=__add__
  20     def __neg__(self):return K([-x for x in self.c])
  21     def __sub__(self,v):
  22         if isinstance(v,E):return NotImplemented
  23         return self+-K(v)
  24     def __rsub__(self,v):return K(v)+-self
  25     def __mul__(self,v):
  26         if isinstance(v,E):return NotImplemented
  27         v=K(v); r=[Q(0)]*7
  28         for i in range(4):
  29             for j in range(4):r[i+j]+=self.c[i]*v.c[j]
  30         return K(r)
  31     __rmul__=__mul__
  32     def inv(self):
  33         cols=[(self*K([0]*j+[1])).c for j in range(4)]
  34         a=[[cols[j][i] for j in range(4)]+[Q(i==0)] for i in range(4)]
  35         for j in range(4):
  36             p=next((i for i in range(j,4) if a[i][j]),None)
  37             if p is None:raise ZeroDivisionError('nonunit cyclotomic element')
  38             a[j],a[p]=a[p],a[j]
  39             v=a[j][j];a[j]=[x/v for x in a[j]]
  40             for i in range(4):
  41                 if i!=j:
  42                     v=a[i][j];a[i]=[x-v*y for x,y in zip(a[i],a[j])]
  43         return K([a[i][4] for i in range(4)])
  44     def __truediv__(self,v):
  45         if isinstance(v,E):return NotImplemented
  46         return self*K(v).inv()
  47     def __rtruediv__(self,v):return K(v)*self.inv()
  48     def __eq__(self,v):
  49         if isinstance(v,E):return NotImplemented
  50         return self.c==K(v).c
  51     def __bool__(self):return any(self.c)
  52     def __pow__(self,n):
  53         if n<0:return self.inv()**(-n)
  54         out=K(1);base=self
  55         while n:
  56             if n&1:out=out*base
  57             base=base*base;n//=2
  58         return out
  59     def residue(self):
  60         return sum((x.numerator*pow(x.denominator,-1,241)%241)*pow(91,i,241) for i,x in enumerate(self.c))%241
  61     def data(self):return [str(x) for x in self.c]

  63 class E:
  64     p0=K(0);p1=K(0)
  65     def __init__(self,a=0,b=0):
  66         if isinstance(a,E):self.a,self.b=a.a,a.b
  67         else:self.a,self.b=K(a),K(b)
  68     def __add__(self,v):
  69         v=E(v);return E(self.a+v.a,self.b+v.b)
  70     __radd__=__add__
  71     def __neg__(self):return E(-self.a,-self.b)
  72     def __sub__(self,v):return self+-E(v)
  73     def __rsub__(self,v):return E(v)+-self
  74     def __mul__(self,v):
  75         v=E(v)
  76         return E(self.a*v.a-self.p0*self.b*v.b,self.a*v.b+self.b*v.a-self.p1*self.b*v.b)
  77     __rmul__=__mul__
  78     def conj(self):return E(self.a-self.p1*self.b,-self.b)
  79     def norm(self):return self.a*self.a-self.p1*self.a*self.b+self.p0*self.b*self.b
  80     def inv(self):
  81         c=self.conj();n=self.norm();return E(c.a/n,c.b/n)
  82     def __truediv__(self,v):return self*E(v).inv()
  83     def __rtruediv__(self,v):return E(v)*self.inv()
  84     def __eq__(self,v):
  85         v=E(v);return self.a==v.a and self.b==v.b
  86     def __bool__(self):return bool(self.a) or bool(self.b)
  87     def __pow__(self,n):
  88         if n<0:return self.inv()**(-n)
  89         out=E(1);base=self
  90         while n:
  91             if n&1:out=out*base
  92             base=base*base;n//=2
  93         return out
  94     def data(self):return {'constant':self.a.data(),'root':self.b.data()}

  96 # Sparse integer multivariate polynomials: alpha,beta,gamma,epsilon,d.
  97 class P:
  98     dim=5
  99     def __init__(self,v=0):self.t=dict(v) if isinstance(v,dict) else ({(0,)*self.dim:int(v)} if v else {})
 100     def __add__(self,v):
 101         v=v if isinstance(v,P) else P(v);r=dict(self.t)
 102         for m,c in v.t.items():r[m]=r.get(m,0)+c
 103         return P({m:c for m,c in r.items() if c})
 104     __radd__=__add__
 105     def __neg__(self):return P({m:-c for m,c in self.t.items()})
 106     def __sub__(self,v):return self+-(v if isinstance(v,P) else P(v))
 107     def __rsub__(self,v):return P(v)+-self
 108     def __mul__(self,v):
 109         v=v if isinstance(v,P) else P(v);r={}
 110         for m,c in self.t.items():
 111             for n,b in v.t.items():
 112                 k=tuple(x+y for x,y in zip(m,n));r[k]=r.get(k,0)+c*b
 113         return P({m:c for m,c in r.items() if c})
 114     __rmul__=__mul__
 115     def __pow__(self,n):
 116         r=P(1)
 117         for _ in range(n):r=r*self
 118         return r
 119     def __eq__(self,v):return self.t==(v.t if isinstance(v,P) else P(v).t)

 121 def symbolic_checks():
 122     vs=[P({tuple(int(j==i) for j in range(5)):1}) for i in range(5)]
 123     a,b,g,e,d=vs
 124     L=b*b-g*e;M=-2*a*b+a*g+a*e+b*g+b*e-2*g*e;C=(b-g)**2
 125     poly=L*d*d+M*d+C
 126     residual=(b-g)*(b*d*d+(b-a)*d+b-g)-(e-b)*d*(g*d+2*g-a-b)
 127     assert residual==poly
 128     # Norm of t's numerator and denominator agree over the quadratic.
 129     s=a+b
 130     numerator=C*(g*g*L-g*s*M+s*s*C)
 131     denominator=g*g*C*L-g*(2*g-s)*M*L+(2*g-s)**2*L*L
 132     assert numerator==denominator
 133     # Cubic Fourier recovery: (1-z) Q(z) = 1 - [z^3-c(1-z)^3].
 134     z,c=vs[:2]
 135     assert (1-z)*((1+c)*z*z+(1-2*c)*z+1+c)==1-(z**3-c*(1-z)**3)
 136     return {'synthesis_remaining_equation_identity':True,'reciprocal_pole_norm_identity':True,'fourier_cubic_recovery_identity':True}

 138 def matrix_multiply(A,B):
 139     return [[sum(A[i][k]*B[k][j] for k in range(len(B))) for j in range(len(B[0]))] for i in range(len(A))]

 141 def synthesize(q,save_full=False):
 142     eta=K([0,1]);g=1+2*(eta**2+eta**3)
 143     assert eta**5==1 and eta!=1 and g*g==5
 144     den=1+5*q**3*g
 145     assert den*(1-5*q**3*g)==1-125*q**6
 146     B=[[(eta**(i*j)-q*g*eta**(-3*(i+j)**2)+(5*q**3*g*eta**(-3*i*i) if i==j else 0))/den for j in range(5)] for i in range(5)]
 147     for i in range(5):
 148         for j in range(5):
 149             assert B[i][j]==B[j][i]
 150             assert B[i][j]==B[(-i)%5][(-j)%5]
 151     a,b,c,e,f,h=B[0][1],B[0][2],B[1][2],B[1][3],B[1][4],B[2][3]
 152     al,be,ga,ep=c/(a*b),e/(a*b),f/(a*a),h/(b*b)
 153     p2=be*be-ga*ep
 154     E.p1=(-2*al*be+al*ga+al*ep+be*ga+be*ep-2*ga*ep)/p2
 155     E.p0=(be-ga)**2/p2
 156     disc=E.p1*E.p1-4*E.p0
 157     d=E(0,1);assert d*d+E.p1*d+E.p0==0
 158     tden=E(ga)*(d+2)-al-be
 159     t=(E(ga)*d+E(al+be)*d*d)/tden
 160     kap=(t+d*d)/ga
 161     x=(t-d)/kap
 162     r=[None,x,(al-x)/d,(be-x)/d,(ga-x)/d]
 163     z=kap/(1+t)
 164     u=[E(1)]+[B[0][i]*r[i] for i in range(1,5)]
 165     v=[z/t]+[B[0][i]-z*u[i] for i in range(1,5)]
 166     diagonal=[B[i][i]-(1+t)*u[i]*v[i] for i in range(5)]
 167     assert t.norm()==1 and t.conj()==1/t
 168     guards={'a':a,'b':b,'gamma':ga,'beta_minus_gamma':be-ga,'polynomial_leading':p2,'discriminant':disc,'d_norm':d.norm(),'t_denominator_norm':tden.norm(),'kappa_norm':kap.norm(),'t_norm':t.norm(),'one_plus_t_norm':(1+t).norm(),'one_minus_t_norm':(1-t).norm()}
 169     for name,arr in [('u',u),('v',v),('diagonal',diagonal)]:
 170         for i,value in enumerate(arr):guards[name+str(i)+'_norm']=value.norm()
 171     assert len(guards)==27 and all(guards.values())
 172     residues={name:x.residue() for name,x in guards.items()}
 173     assert all(residues.values())
 174     target_residues=[[x.residue() for x in row] for row in B]
 175     expected=[[161,20,102,102,20],[20,182,62,26,92],[102,62,236,28,26],[102,26,28,236,62],[20,92,26,62,182]]
 176     assert target_residues==expected
 177     assert den.residue()==34 and E.p1.residue()==30 and E.p0.residue()==124
 178     assert disc.residue()==163 and pow(163,120,241)==240
 179     for i in range(5):
 180         for j in range(i+1,5):assert u[i]*v[j]+t*v[i]*u[j]==B[i][j]
 181     # Independently form both Green operators from powers of their weighted cycles.
 182     identity=[[E(int(i==j)) for j in range(5)] for i in range(5)]
 183     greens=[]
 184     for pole in [t,1/t]:
 185         W=[[E(0) for _ in range(5)] for _ in range(5)]
 186         for i in range(5):W[i][(i+1)%5]=pole if i==4 else E(1)
 187         power=identity;total=[[E(0) for _ in range(5)] for _ in range(5)]
 188         for k in range(5):
 189             total=[[total[i][j]+power[i][j] for j in range(5)] for i in range(5)]
 190             power=matrix_multiply(power,W)
 191         assert all(power[i][j]==pole*identity[i][j] for i in range(5) for j in range(5))
 192         G=[[total[i][j]/(1-pole) for j in range(5)] for i in range(5)]
 193         inverse=[[identity[i][j]-W[i][j] for j in range(5)] for i in range(5)]
 194         assert matrix_multiply(inverse,G)==identity and matrix_multiply(G,inverse)==identity
 195         greens.append(G)
 196     for i in range(5):
 197         for j in range(5):
 198             result=(1-t)*u[i]*greens[0][i][j]*v[j]+(t-1)*v[i]*greens[1][i][j]*u[j]+(diagonal[i] if i==j else 0)
 199             assert result==B[i][j] and result.conj()==B[i][j]
 200     # Apply actual cyclic scans to two dense vectors and all five basis vectors.
 201     def solve(pole,inp):
 202         out=[E(0)]*5;out[0]=sum(inp)/(1-pole)
 203         out[4]=inp[4]+pole*out[0]
 204         for i in range(3,0,-1):out[i]=inp[i]+out[i+1]
 205         for i in range(5):assert out[i]-(pole if i==4 else 1)*out[(i+1)%5]==inp[i]
 206         return out
 207     inputs=[[int(i==j) for i in range(5)] for j in range(5)]+[[1,2,3,4,5],[2,-1,0,3,-2]]
 208     for inp in inputs:
 209         y=solve(t,[v[i]*inp[i] for i in range(5)])
 210         z=solve(1/t,[u[i]*inp[i] for i in range(5)])
 211         output=[(1-t)*u[i]*y[i]+(t-1)*v[i]*z[i]+diagonal[i]*inp[i] for i in range(5)]
 212         assert output==[sum(B[i][j]*inp[j] for j in range(5)) for i in range(5)]
 213     data={'q':q,'physical_N':5*q*q,'denominator_residue':den.residue(),'quadratic_residues':[E.p0.residue(),E.p1.residue(),1],'guard_residues':residues,'all_27_guards_nonzero':True,'all_25_factor_entries_checked':True,'both_roots_checked':True,'all_100_inverse_entries_checked':True,'scan_entries_checked':35,'pole_norm':t.norm().data()}
 214     if save_full:
 215         data.update({'target':[[x.data() for x in row] for row in B],'quadratic_constant':E.p0.data(),'quadratic_linear':E.p1.data(),'t':t.data(),'u':[x.data() for x in u],'v':[x.data() for x in v],'diagonal_correction':[x.data() for x in diagonal],'guard_norms':{name:x.data() for name,x in guards.items()}})
 216     return data

 218 def congruence_check():
 219     # Polynomial arithmetic over F241 verifies q=1+241k universally, not by sampling k.
 220     def add(a,b):
 221         r=[((a[i] if i<len(a) else 0)+(b[i] if i<len(b) else 0))%241 for i in range(max(len(a),len(b)))];
 222         while len(r)>1 and r[-1]==0:r.pop()
 223         return r
 224     def mul(a,b):
 225         r=[0]*(len(a)+len(b)-1)
 226         for i,x in enumerate(a):
 227             for j,y in enumerate(b):r[i+j]=(r[i+j]+x*y)%241
 228         return add(r,[0])
 229     q=add([1,241],[0]);q3=mul(mul(q,q),q)
 230     assert q==[1] and q3==[1]
 231     g=(1+2*(pow(91,2,241)+pow(91,3,241)))%241
 232     assert g==103 and add([1],mul([5*g],q3))==[34]
 233     for i in range(5):
 234         for j in range(5):
 235             numerator=add([pow(91,i*j,241)],mul([-g*pow(91,(-3*(i+j)**2)%5,241)],q))
 236             if i==j:numerator=add(numerator,mul([5*g*pow(91,(-3*i*i)%5,241)],q3))
 237             assert len(numerator)==1
 238     return {'polynomial_substitution_q_equals_1_plus_241k':True,'target_denominator_constant':34,'all_25_target_entries_constant':True}

 240 def bluestein_checks():
 241     def prime(p):return p>=2 and all(p%d for d in range(2,math.isqrt(p)+1))
 242     def factors(n):
 243         out=[];d=2
 244         while d*d<=n:
 245             if n%d==0:
 246                 out.append(d)
 247                 while n%d==0:n//=d
 248             d+=1
 249         if n>1:out.append(n)
 250         return out
 251     def dft(x,root,p):
 252         return [sum(x[j]*pow(root,i*j,p) for j in range(len(x)))%p for i in range(len(x))]
 253     cases=[]
 254     for n in range(2,13):
 255         m=2*n-1;period=math.lcm(2*n,m);p=period+1
 256         while not prime(p):p+=period
 257         fac=factors(p-1)
 258         gen=next(g for g in range(2,p) if all(pow(g,(p-1)//r,p)!=1 for r in fac))
 259         theta=pow(gen,(p-1)//(2*n),p);omega=theta*theta%p;root=pow(gen,(p-1)//m,p)
 260         x=[(j*j+3*j+2)%p for j in range(n)]
 261         a=[x[j]*pow(theta,j*j,p)%p if j<n else 0 for j in range(m)]
 262         b=[0]*m
 263         for ell in range(-(n-1),n):b[ell%m]=pow(theta,(-ell*ell)%(2*n),p)
 264         fa,fb=dft(a,root,p),dft(b,root,p)
 265         product=[y*z%p for y,z in zip(fa,fb)]
 266         forward=dft(product,root,p)
 267         conv=[forward[(-i)%m]*pow(m,-1,p)%p for i in range(m)]
 268         output=[pow(theta,i*i,p)*conv[i]%p for i in range(n)]
 269         assert output==dft(x,omega,p)
 270         cases.append({'n':n,'padded_m':m,'prime':p,'verified_outputs':n})
 271     grid=[]
 272     for n in [1,2,3,4,10,100,10**6,10**9,10**12,10**30]:
 273         target=2*n-1;r=math.isqrt((target+4)//5)
 274         while 5*r*r<target:r+=1
 275         k=max(0,(r-1+2409)//2410);q=1+2410*k;m=5*q*q
 276         assert m>=target and (k==0 or 5*(q-2410)**2<target)
 277         assert m<target+4820*(math.isqrt(5*target)+1)+5*2410**2
 278         grid.append({'n':n,'q':q,'m':m,'minimal_grid_choice':True})
 279     return {'exact_convolution_controls':cases,'supported_length_padding_controls':grid}
