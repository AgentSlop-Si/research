#!/usr/bin/env python3
"""Independent numerical controls for the v10 Bessel/Chebyshev lift.

No uploaded module is imported or executed. The formulas are reconstructed here.
These floating-point checks support, but do not replace, the accompanying proof
audit. Run from any directory: python -I independent_lift_audit.py --output out.json
Dependencies: numpy, mpmath. No network or original archive is needed.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import math
import platform
from fractions import Fraction
from pathlib import Path

import mpmath as mp
import numpy as np


def dyadic(n):
    return 1 << (n - 1).bit_length()


def block_size(q):
    m = 1 << max(0, (q.bit_length() - 1).bit_length() - 1)
    while m * m > q:
        m //= 2
    return m


def exact_map(n):
    q = dyadic(n)
    j = [(2 * k * q + n) // (2 * n) for k in range(n)]
    rem = [k * q - j[k] * n for k in range(n)]
    assert len(set(j)) == n
    assert min(j) >= 0 and max(j) < q
    assert all(2 * abs(v) <= n for v in rem)
    return q, np.array(j), np.array([v / n for v in rem])


def bessel_series(r, z):
    """Direct alternating defining series, independently summed at 90 digits."""
    z = mp.mpf(z)
    t = (z / 2) ** r / mp.factorial(r)
    result = t
    for ell in range(1, 1000):
        t *= -(z / 2) ** 2 / (ell * (r + ell))
        result += t
        if abs(t) < mp.mpf('1e-85') * max(abs(result), mp.mpf('1e-300')):
            return result
    raise RuntimeError('Series did not converge')


def tail(rank):
    r = mp.mpf(rank)
    b = mp.pi / 2
    return 2 * mp.exp(mp.pi**2 / (4 * (r + 1))) * b**r / mp.factorial(rank) / (1 - b / (r + 1))


def cheb_rows(y, rank):
    rows = [np.ones_like(y)]
    if rank > 1:
        rows.append(y.copy())
    for _ in range(2, rank):
        rows.append(2 * y * rows[-1] - rows[-2])
    return np.asarray(rows[:rank])


def lift_factors(n, rank):
    q, j, theta = exact_map(n)
    weights = np.array([[float(bessel_series(r, mp.pi * t / q)) for t in range(n)] for r in range(rank)])
    coeff = cheb_rows(2 * theta, rank).astype(complex)
    for r in range(1, rank):
        coeff[r] *= 2 * (-1j)**r
    p = coeff.T @ weights
    return q, j, weights, coeff, p


def dft_matrix(q):
    return np.exp(-2j * np.pi * np.outer(np.arange(q), np.arange(q)) / q) / math.sqrt(q)


def projection(q, m, a, beta):
    """Explicit scaled block projector P: P^2=m P and E[P]=I."""
    out = np.zeros((q, q), dtype=complex)
    for b in range(q // m):
        phase = np.array([np.exp(2j * np.pi * ((a * t * (t-1) // 2 + (a*m*b + beta)*t) % q) / q) for t in range(m)])
        out[b*m:(b+1)*m, b*m:(b+1)*m] = np.outer(phase.conj(), phase)
    return out


def test_map():
    maxoffset = 0
    for n in range(1, 1001):
        _, _, theta = exact_map(n)
        maxoffset = max(maxoffset, float(np.max(np.abs(theta))))
    examples = []
    for exponent, which in [(28, 'below'), (53, 'above'), (60, 'below'), (60, 'above')]:
        q = 1 << exponent
        n = q - 1
        k = (n-1)//2 if which == 'below' else (n+1)//2
        je = (2*k*q+n)//(2*n)
        te = Fraction(k*q-je*n, n)
        sf = float(k) * (q/n)
        jf = math.floor(sf+0.5)
        tf = sf-jf
        defect = Fraction(jf) + Fraction(tf) - Fraction(k*q,n)
        examples.append(dict(N=n,Q=q,k=k,exact_j=je,exact_theta=str(te),float_j=jf,float_theta=tf,combined_frequency_defect=str(defect),combined_frequency_defect_float=float(defect)))
    assert examples[0]['exact_j'] != examples[0]['float_j']
    assert abs(examples[1]['float_theta']) > 0.5
    assert examples[2]['float_j'] == examples[3]['float_j']
    return dict(exact_lengths_checked=1000,max_absolute_offset=maxoffset,float_counterexamples=examples)


def test_kernel():
    records=[]
    for rank in [2,4,6,10,14,20,30]:
        largest=mp.mpf(0)
        for iz in range(21):
            z=mp.pi*iz/20
            js=[bessel_series(r,z) for r in range(rank)]
            for iy in range(25):
                y=mp.mpf(iy)/12-1
                t0,t1=mp.mpf(1),y
                p=js[0]
                for r in range(1,rank):
                    tr=t1
                    p+=2*(-mp.j)**r*js[r]*tr
                    t0,t1=t1,2*y*t1-t0
                largest=max(largest,abs(mp.exp(-mp.j*z*y)-p))
        bnd=tail(rank)
        assert largest <= bnd
        records.append(dict(rank=rank,max_grid_error=mp.nstr(largest,30),tail=mp.nstr(bnd,30),ratio=float(largest/bnd)))
    return records


def test_lift(rng):
    records=[]
    for n in [1,2,3,7,10,15,17,31,45,61]:
        rank=10
        q,j,b,c,p=lift_factors(n,rank)
        x=rng.normal(size=n)+1j*rng.normal(size=n)
        fft_out=np.zeros(n,dtype=complex)
        for r in range(rank):
            v=np.zeros(q,dtype=complex); v[:n]=x*b[r]
            fft_out+=c[r]*np.fft.fft(v)[j]/math.sqrt(n)
        center=np.exp(-2j*np.pi*np.outer(j,np.arange(n))/q)*p/math.sqrt(n)
        direct_out=center@x
        target=dft_matrix(n)@x
        discrepancy=float(np.max(np.abs(fft_out-direct_out)))
        ratio=float(np.max(np.abs(fft_out-target))/(float(tail(rank))*np.linalg.norm(x)))
        assert discrepancy < 1e-11
        assert ratio < 1
        records.append(dict(N=n,Q=q,rank=rank,lift_vs_direct_error=discrepancy,truncation_bound_ratio=ratio))
    return records


def test_covariance(rng):
    records=[]
    for n in [3,7,10,15,17,31]:
        rank=8
        q,j,b,c,p=lift_factors(n,rank)
        m=block_size(q); K=q//m
        F=dft_matrix(q)
        center=math.sqrt(q/n)*F[j,:n]*p
        mean=np.zeros_like(center)
        cov=np.zeros((n,n,n),dtype=complex)
        combine_defect=0.0
        states=0
        for a in range(1,q,2):
            for beta in range(q):
                P=projection(q,m,a,beta)
                assert np.max(np.abs(P@P-m*P))<1e-12
                fp=(F@P)[j,:n]
                combined=math.sqrt(q/n)*fp*p
                separate=np.zeros_like(combined)
                for r in range(rank):
                    separate+=math.sqrt(q/n)*c[r,:,None]*fp*b[r,None,:]
                combine_defect=max(combine_defect,float(np.max(np.abs(combined-separate))))
                err=combined-center
                cov+=np.einsum('ki,kj->kij',err.conj(),err)
                mean+=combined
                states+=1
        mean/=states;cov/=states
        Q0=(m*np.eye(q)-(np.arange(q)[:,None]%K==np.arange(q)[None,:]%K))/q
        defect=0.0
        maxnorm=0.0
        for k in range(n):
            modulation=np.exp(2j*np.pi*j[k]*(np.arange(q)[:,None]-np.arange(q)[None,:])/q)
            expected=(q/n)*(p[k].conj()[:,None]*(Q0*modulation)[:n,:n]*p[k][None,:])
            defect=max(defect,float(np.max(np.abs(cov[k]-expected))))
            maxnorm=max(maxnorm,float(np.linalg.eigvalsh(cov[k])[-1]))
        bound=(m/n)*(1+float(tail(rank)))**2
        # M=1 gives exactly zero covariance, not covariance norm M/Q.
        assert defect<1e-11
        assert maxnorm <= bound*(1+1e-10)
        assert np.max(np.abs(mean-center))<1e-11
        records.append(dict(N=n,Q=q,M=m,rank=rank,states=states,mean_operator_defect=float(np.max(np.abs(mean-center))),covariance_operator_defect=defect,shared_state_identity_defect=combine_defect,max_covariance_eigenvalue=maxnorm,theorem_bound=bound,ratio=maxnorm/bound))
    return records


def test_forward_recurrence():
    records=[]
    for z in [math.pi,0.1,math.pi/2**10,math.pi/2**20,math.pi/2**30]:
        values=[float(bessel_series(0,z)),float(bessel_series(1,z))]
        for r in range(1,12):
            values.append((2*r/z)*values[-1]-values[-2])
        truth=bessel_series(12,z)
        records.append(dict(z=z,rank=12,rounded_seed_forward_value=values[12],reference=mp.nstr(truth,30),absolute_error=mp.nstr(abs(values[12]-truth),30)))
    assert abs(records[3]['rounded_seed_forward_value'])>1e40
    return records


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--output',type=Path,default=Path(__file__).with_name('independent_results.json'))
    args=parser.parse_args()
    mp.mp.dps=90
    rng=np.random.default_rng(9171710)
    data=dict(status='PASS',scope='Independently authored numerical controls; uploaded code never imported or executed',environment=dict(python=platform.python_version(),numpy=np.__version__,mpmath=mp.__version__),mapping=test_map(),kernel_tail=test_kernel(),lift=test_lift(rng),covariance=test_covariance(rng),forward_recurrence_instability=test_forward_recurrence())
    data['script_sha256']=hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    args.output.write_text(json.dumps(data,indent=2)+'\n')
    print(json.dumps(dict(status=data['status'],max_kernel_tail_ratio=max(v['ratio'] for v in data['kernel_tail']),max_lift_identity_error=max(v['lift_vs_direct_error'] for v in data['lift']),max_covariance_identity_defect=max(v['covariance_operator_defect'] for v in data['covariance']),max_covariance_bound_ratio=max(v['ratio'] for v in data['covariance']),output=str(args.output)),indent=2))


if __name__=='__main__':
    main()
