import unittest
from real_mask_audit import audit_real_mask

class RealMaskTests(unittest.TestCase):
    def _test_size(self,n):
        r=audit_real_mask(n)
        self.assertEqual(r['rank_real'],n//2+1)
        self.assertEqual(r['rank_imag'],n//2-1)
        self.assertEqual(r['measured_real_mask_entries'],n//2+1)
        self.assertEqual(r['measured_imaginary_mask_entries'],n//2-1)
        self.assertEqual(r['measured_mixed_mask_entries'],0)
        self.assertLess(r['offdiagonal_eigenbasis_error'],1e-11)
        self.assertLess(r['Fourier_reconstruction_error'],1e-11)
    def test_4(self):self._test_size(4)
    def test_8(self):self._test_size(8)
    def test_16(self):self._test_size(16)
    def test_32(self):self._test_size(32)
    def test_64(self):self._test_size(64)
    def test_128(self):self._test_size(128)

if __name__=='__main__':unittest.main()
