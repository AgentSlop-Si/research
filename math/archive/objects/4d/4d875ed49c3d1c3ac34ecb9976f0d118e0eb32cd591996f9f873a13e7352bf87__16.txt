  13 def negacyclic(a,b):
  14     # Exact product in Z[z]/(z^4+1), representing Q(zeta_8)
  15     c=[0]*4
  16     for i,u in enumerate(a):
  17         for j,v in enumerate(b):
  18             t=i+j
  19             c[t%4]+=(1 if t<4 else -1)*u*v
  20     return tuple(c)

  22 def ring_add(*terms):
  23     return tuple(sum(a[j] for a in terms) for j in range(4))
  24 
  25 def ring_neg(a):return tuple(-v for v in a)

  27 def ring_scale(a,k):return tuple(v*k for v in a)

  29 def ring_ff8(x):
  30     f=[]
  31     for k in range(8):
  32         c=[0]*4
  33         for j,v in enumerate(x):
  34             r=(k*j)%8
  35             c[r%4]+=v*(1 if r<4 else -1)
  36         f.append(tuple(c))
  37     return f

  39 def cubic_paired_single_missing_exact_characteristic_zero():
  40     """Verify a degree-3 witness for one *single* omitted Fourier bin.
  41     This is noncompetitive with ordinary complex conjugation on binary inputs.
  42     Identity: 4X7 = -(X0-4)Q - R - 2 X1 X2 X4, with
  43     Q=X1X6+X2X5+X3X4 and R=X1²X5+X1X3²+X2²X3.
  44     """
  45     c0=(4,0,0,0)
  46     for x in all_binary(8):
  47         f=ring_ff8(x)
  48         mul=lambda *arr: _mul_many(arr)
  49         q=ring_add(mul(f[1],f[6]),mul(f[2],f[5]),mul(f[3],f[4]))
  50         r=ring_add(mul(f[1],f[1],f[5]),mul(f[1],f[3],f[3]),mul(f[2],f[2],f[3]))
  51         centered=ring_add(f[0],ring_neg(c0))
  52         rhs=ring_add(ring_neg(mul(centered,q)),ring_neg(r),ring_scale(mul(f[1],f[2],f[4]),-2))
  53         assert rhs==ring_scale(f[7],4),(x,rhs,ring_scale(f[7],4))
  54     return True
  55 
  56 def _mul_many(arr):
  57     z=(1,0,0,0)
  58     for a in arr:z=negacyclic(z,a)
  59     return z
