20: def make_input(field,n,q,parameter,a,b,shift,defects):
21:     """Independent forward-affine closure; no calls to production cycle solver."""
22:     powers=[field.power(q,j) for j in range(n)]
23:     permutation=[(a*j+shift)%n for j in range(n)]
24:     order=cycle_order(permutation)
25:     coefficients=[field.mul(parameter,powers[(b*j)%n]) for j in range(n)]
26:     rhs=[defects.get(j,field.zero) for j in range(n)]
27:     alpha,beta=field.one,field.zero
28:     for j in order:
29:         reciprocal=field.inv(coefficients[j])

31:         beta=field.mul(field.sub(beta,rhs[j]),reciprocal)
32:     first=field.mul(beta,field.inv(field.sub(field.one,alpha)))
33:     out=[field.zero]*n; value=first
