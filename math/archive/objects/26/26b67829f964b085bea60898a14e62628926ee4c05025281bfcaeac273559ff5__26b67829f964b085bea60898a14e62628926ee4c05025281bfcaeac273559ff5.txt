def trim(a):
 while len(a)>1 and a[-1]==0:a.pop()
 return a

def div(a,b):
 a=a[:];q=[0]*max(1,len(a)-len(b)+1)
 while len(a)>=len(b) and any(a):
  d=len(a)-len(b);v=a[-1]//b[-1];assert v*b[-1]==a[-1];q[d]=v
  for j in range(len(b)):a[d+j]-=v*b[j]
  trim(a)
 return trim(q),trim(a)

@functools.lru_cache(None)
def cyclo(n):
 a=[-1]+[0]*(n-1)+[1]
 for d in range(1,n):
  if n%d==0:a,rem=div(a,list(cyclo(d)));assert not any(rem)
 return tuple(a)

def residues(n):
 f=list(cyclo(n));d=len(f)-1;out=[]
 for e in range(n):
  a=[0]*e+[1];_,a=div(a,f);out.append(a+[0]*(d-len(a)))
 return out

def cycle(n,rng):
 order=list(range(n));rng.shuffle(order);p=[0]*n
 for i,j in zip(order,order[1:]+order[:1]):p[i]=j
 return p

def support_case(n,alpha,beta,rp,cp,left,right,a,b):
 rs=residues(n);d=len(rs[0]);inverse=[beta.index(j) for j in range(n)];supports=[]
 for i in range(n):
  row=set();ia=alpha[i]
  for j in range(n):
   k=inverse[j]
   terms=[(left[i]*right[j],rp[i]*cp[j]),(-a[i]*left[ia]*right[j],rp[ia]*cp[j]),(-b[k]*left[i]*right[k],rp[i]*cp[k]),(a[i]*b[k]*left[ia]*right[k],rp[ia]*cp[k])]
   value=[sum(c*rs[e%n][z] for c,e in terms) for z in range(d)]
   if any(value):row.add(j)
  supports.append(row)
 bound=(n+1)//2
 assert all(len(supports[i]|supports[alpha[i]])>=bound for i in range(n))
 total=sum(map(len,supports));target=(n*bound+1)//2;assert total>=target
 return {'N':n,'nnz':total,'nnz_lower_bound':target,'min_adjacent_union':min(len(supports[i]|supports[alpha[i]]) for i in range(n))}
