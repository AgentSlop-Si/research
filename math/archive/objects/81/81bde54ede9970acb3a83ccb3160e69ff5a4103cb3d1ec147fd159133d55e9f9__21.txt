  18 def eye(n):
  19     return [[int(i == j) for j in range(n)] for i in range(n)]

  22 def add(a, b, p=P):
  23     return [[(u + v) % p for u, v in zip(x, y)] for x, y in zip(a, b)]

  26 def scale(a, c, p=P):
  27     return [[c * v % p for v in row] for row in a]

  30 def mul(a, b, p=P):
  31     out = [[0] * len(b[0]) for _ in a]
  32     for i, row in enumerate(a):
  33         for k, u in enumerate(row):
  34             if u:
  35                 for j, v in enumerate(b[k]):
  36                     out[i][j] = (out[i][j] + u * v) % p
  37     return out

  40 def sub(a, rr, cc):
  41     return [[a[i][j] for j in cc] for i in rr]

  44 def rank(a, p=P):
  45     if not a or not a[0]:
  46         return 0
  47     b = [row[:] for row in a]
  48     r = 0
  49     for c in range(len(b[0])):
  50         pivot = next((i for i in range(r, len(b)) if b[i][c] % p), None)
  51         if pivot is None:
  52             continue
  53         b[r], b[pivot] = b[pivot], b[r]
  54         z = pow(b[r][c], -1, p)
  55         b[r] = [v * z % p for v in b[r]]
  56         for i in range(r + 1, len(b)):
  57             z = b[i][c]
  58             if z:
  59                 b[i] = [(u - z * v) % p for u, v in zip(b[i], b[r])]
  60         r += 1
  61         if r == len(b):
  62             break
  63     return r

  66 def inv(a, p=P):
  67     n = len(a)
  68     e = eye(n)
  69     b = [row[:] + e[i] for i, row in enumerate(a)]
  70     for c in range(n):
  71         pivot = next((i for i in range(c, n) if b[i][c] % p), None)
  72         if pivot is None:
  73             raise ValueError("singular")
  74         b[c], b[pivot] = b[pivot], b[c]
  75         z = pow(b[c][c], -1, p)
  76         b[c] = [v * z % p for v in b[c]]
  77         for i in range(n):
  78             if i != c:
  79                 z = b[i][c]
  80                 if z:
  81                     b[i] = [(u - z * v) % p for u, v in zip(b[i], b[c])]
  82     return [row[n:] for row in b]

  85 def determinant(a, p):
  86     b = [row[:] for row in a]
  87     ans = 1
  88     for c in range(len(b)):
  89         pivot = next((i for i in range(c, len(b)) if b[i][c] % p), None)
  90         if pivot is None:
  91             return 0
  92         if pivot != c:
  93             b[c], b[pivot] = b[pivot], b[c]
  94             ans = -ans
  95         ans = ans * b[c][c] % p
  96         z = pow(b[c][c], -1, p)
  97         for i in range(c + 1, len(b)):
  98             t = b[i][c] * z % p
  99             b[i] = [(u - t * v) % p for u, v in zip(b[i], b[c])]
 100     return ans % p

 103 def cut_ranks(a):
 104     n = len(a)
 105     return [(rank(sub(a, range(k), range(k, n))),
 106              rank(sub(a, range(k, n), range(k)))) for k in range(1, n)]

 109 def qsep(a):
 110     return max(max(pair) for pair in cut_ranks(a)) if len(a) > 1 else 0

 113 def poly(a, coefficients):
 114     n = len(a)
 115     e = eye(n)
 116     out = [[0] * n for _ in range(n)]
 117     for coefficient in reversed(coefficients):
 118         out = add(mul(a, out), scale(e, coefficient))
 119     return out

 122 def signed_cycle(n):
 123     a = [[0] * n for _ in range(n)]
 124     for j in range(n - 1):
 125         a[j + 1][j] = 1
 126     a[0][n - 1] = P - 1
 127     return a

 130 def base_matrix(n, kind, rng):
 131     if kind == "cycle":
 132         return signed_cycle(n)
 133     a = [[0] * n for _ in range(n)]
 134     for i in range(n):
 135         a[i][i] = 7 + i
 136         if i:
 137             a[i][i - 1] = -2 % P
 138         if i + 1 < n:
 139             a[i][i + 1] = -1 % P
 140     if kind == "tridiagonal_plus_rank1":
 141         u = [rng.randrange(1, 10) for _ in range(n)]
 142         v = [rng.randrange(1, 10) for _ in range(n)]
 143         a = add(a, [[x * y % P for y in v] for x in u])
 144     return a

 147 def check_inverse_and_rational(rng):
 148     records = []
 149     special = []
 150     for n in (4, 8, 16):
 151         m = n // 2
 152         a = [[int(j == (i + m) % n) for j in range(n)] for i in range(n)]
 153         assert rank(sub(a, range(m), range(m))) == 0
 154         assert cut_ranks(a) == cut_ranks(inv(a))
 155         special.append({"n": n, "principal_rank": 0,
 156                         "middle_offdiagonal_rank": m})
 157     for n in (12, 20, 28):
 158         for kind in ("cycle", "tridiagonal", "tridiagonal_plus_rank1"):
 159             a = base_matrix(n, kind, rng)
 160             assert cut_ranks(a) == cut_ranks(inv(a))
 161             h = qsep(a)
 162             for dp, dq in ((0, 1), (1, 0), (1, 1), (2, 2),
 163                            (3, 1), (1, 3), (4, 3), (3, 4)):
 164                 cp = [rng.randrange(1, 100) for _ in range(dp + 1)]
 165                 cq = [rng.randrange(1, 100) for _ in range(dq + 1)]
 166                 # Nonsingularity is checked, never assumed from random sampling.
 167                 denominator = poly(a, cq)
 168                 inverse = inv(denominator)
 169                 assert cut_ranks(denominator) == cut_ranks(inverse)
 170                 value = mul(poly(a, cp), inverse)
 171                 actual = qsep(value)
 172                 assert actual <= h * max(dp, dq)
 173                 records.append({"n": n, "base": kind, "h": h,
 174                                 "degrees": [dp, dq], "qsep": actual,
 175                                 "bound": h * max(dp, dq)})
 176             # Repeated pole and a numerator/denominator cancellation.
 177             # q(z)=(z-100)^4, p(z)=(z-100)^2*(1+3z).
 178             linear = add(a, scale(eye(n), -100))
 179             den = mul(mul(linear, linear), mul(linear, linear))
 180             numerator = mul(mul(linear, linear), add(eye(n), scale(a, 3)))
 181             value = mul(numerator, inv(den))
 182             reduced = mul(add(eye(n), scale(a, 3)), inv(mul(linear, linear)))
 183             assert value == reduced
 184             assert qsep(value) <= 2 * h
 185     return {"random_rational_cases": records,
 186             "singular_principal_block_controls": special,
 187             "repeated_pole_cancellation_cases": 9}

 190 def cycle_action(x, shift):
 191     n = len(x)
 192     out = [0] * n
 193     for j, v in enumerate(x):
 194         k = j + shift
 195         out[k % n] = -v if k >= n else v
 196     return out

 199 def apply_k_mod(x, lam, p=P):
 200     # Solve (I-lambda*M) z=x by a cyclic scalar recurrence.
 201     tail = 0
 202     for v in x[1:]:
 203         tail = (v + lam * tail) % p
 204     z0 = (x[0] - lam * tail) * pow(1 + pow(lam, len(x), p), -1, p) % p
 205     z = [z0]
 206     for v in x[1:]:
 207         z.append((v + lam * z[-1]) % p)
 208     return [(v + w) % p for v, w in zip(x, z)]

 211 def apply_fused_mod(x, p=P):
 212     m = len(x) // 2
 213     z = pow(2, m, p)
 214     alpha = (z + 1) * pow(2, -1, p) % p
 215     beta = (z - 1) * pow(2, -1, p) % p
 216     return [(alpha * v + beta * w) % p
 217             for v, w in zip(x, cycle_action(x, m))]

 220 def apply_k_fraction(x, lam):
 221     tail = Fraction(0)
 222     for v in x[1:]:
 223         tail = v + lam * tail
 224     z = [(x[0] - lam * tail) / (1 + lam ** len(x))]
 225     for v in x[1:]:
 226         z.append(v + lam * z[-1])
 227     return [v + w for v, w in zip(x, z)]

 230 def check_halfcycle(rng):
 231     modular = []
 232     for n in (2, 4, 8, 16, 32, 64, 128):
 233         m = n // 2
 234         root = pow(3, (P - 1) // m, P)
 235         assert pow(root, m, P) == 1
 236         if m > 1:
 237             assert pow(root, m // 2, P) != 1
 238         for _ in range(8):
 239             x = [rng.randrange(P) for _ in range(n)]
 240             y = x[:]
 241             phase = 1
 242             for j in range(m):
 243                 y = apply_k_mod(y, phase)
 244                 phase = phase * root % P
 245             assert y == apply_fused_mod(x)
 246         # Verify all columns and support count of the fused evaluator.
 247         columns = [apply_fused_mod([int(i == j) for i in range(n)])
 248                    for j in range(n)]
 249         assert all(sum(v != 0 for v in c) == 2 for c in columns)
 250         matrix = [list(row) for row in zip(*columns)]
 251         half_cut_rank = rank(sub(matrix, range(m), range(m, n)))
 252         assert half_cut_rank == m
 253         modular.append({"n": n, "word_length": m, "fresh_inputs": 8,
 254                         "basis_columns": n, "nonzeros_per_column": 2,
 255                         "natural_half_cut_rank": half_cut_rank})
 256     rational = []
 257     for n, phases in ((2, (1,)), (4, (1, -1))):
 258         for trial in range(5):
 259             x = [Fraction(rng.randrange(-30, 31), rng.randrange(1, 12))
 260                  for _ in range(n)]
 261             y = x[:]
 262             for phase in phases:
 263                 y = apply_k_fraction(y, Fraction(phase))
 264             m = n // 2
 265             a = Fraction(2 ** m + 1, 2)
 266             b = Fraction(2 ** m - 1, 2)
 267             expected = [a * v + b * w for v, w in zip(x, cycle_action(x, m))]
 268             assert y == expected
 269         rational.append({"n": n, "fresh_exact_rational_inputs": 5})
 270     return {"modular": modular, "rational": rational}

 273 def check_ports_and_fourier(rng):
 274     ports = []
 275     for n in (8, 12, 16, 32):
 276         size = n + 7
 277         a = base_matrix(size, "tridiagonal_plus_rank1", rng)
 278         ai = inv(a)
 279         h = qsep(a)
 280         for trial in range(10):
 281             inputs = rng.sample(range(size), n)
 282             outputs = rng.sample(range(size), n)
 283             split = sorted(inputs)[n // 2 - 1] + 1
 284             il = [i for i, pos in enumerate(inputs) if pos < split]
 285             ir = [i for i, pos in enumerate(inputs) if pos >= split]
 286             ol = [i for i, pos in enumerate(outputs) if pos < split]
 287             ore = [i for i, pos in enumerate(outputs) if pos >= split]
 288             value = sub(ai, outputs, inputs)
 289             ranks = [rank(sub(value, ol, ir)), rank(sub(value, ore, il))]
 290             assert max(ranks) <= h
 291             assert max(len(ol) * len(ir), len(ore) * len(il)) >= (n * n) // 4
 292             ports.append({"n": n, "system_size": size, "h": h,
 293                           "ports_left": [len(il), len(ol)], "ranks": ranks})
 294     fourier = []
 295     for n in (4, 8, 16, 32, 64):
 296         root = pow(3, (P - 1) // n, P)
 297         powers = [pow(root, j, P) for j in range(n)]
 298         for trial in range(25):
 299             rows = rng.sample(range(n), n // 2)
 300             cols = rng.sample(range(n), n - n // 2)
 301             matrix = [[powers[(r * c) % n] for c in cols] for r in rows]
 302             r = rank(matrix)
 303             lower = (len(rows) * len(cols) + n - 1) // n
 304             assert r >= lower
 305             gains_r = [rng.randrange(1, P) for _ in rows]
 306             gains_c = [rng.randrange(1, P) for _ in cols]
 307             scaled = [[gains_r[i] * v * gains_c[j] % P for j, v in enumerate(row)]
 308                       for i, row in enumerate(matrix)]
 309             assert rank(scaled) == r
 310             fourier.append({"n": n, "rank_mod_p": r, "lower_bound": lower})
 311     return {"arbitrary_ports": ports, "scaled_permuted_fourier_cuts": fourier}

 314 def signal_edges(n, ranks, shuffles):
 315     # A separate graph construction from the documented recurrences.
 316     infinity = 10 ** 9
 317     edges = {}
 318     def connect(u, v, cap=infinity):
 319         assert (u, v) not in edges
 320         edges[u, v] = cap
 321     def port(s, kind, j, cap):
 322         connect((s, kind, j, "i"), (s, kind, j, "o"), cap)
 323     for s, width in enumerate(ranks):
 324         for j in range(n):
 325             port(s, "x", j, 1)
 326             port(s, "y", j, 1)
 327             connect((s, "x", j, "o"), (s, "y", j, "i"))
 328         for j in range(1, n):
 329             port(s, "f", j, width)
 330             connect((s, "x", j - 1, "o"), (s, "f", j, "i"))
 331             connect((s, "f", j, "o"), (s, "y", j, "i"))
 332             if j > 1:
 333                 connect((s, "f", j - 1, "o"), (s, "f", j, "i"))
 334         for j in range(n - 1):
 335             port(s, "b", j, width)
 336             connect((s, "x", j + 1, "o"), (s, "b", j, "i"))
 337             connect((s, "b", j, "o"), (s, "y", j, "i"))
 338             if j < n - 2:
 339                 connect((s, "b", j + 1, "o"), (s, "b", j, "i"))
 340     for s, permutation in enumerate(shuffles):
 341         assert sorted(permutation) == list(range(n))
 342         for destination, origin in enumerate(permutation):
 343             connect((s, "y", origin, "o"), (s + 1, "x", destination, "i"))
 344     return edges

 347 def maxflow(edges, source, sink):
 348     # Independent integer Edmonds-Karp, used only on this small certificate.
 349     residual = {}
 350     neighbours = {}
 351     for (u, v), cap in edges.items():
 352         residual[u, v] = residual.get((u, v), 0) + cap
 353         residual.setdefault((v, u), 0)
 354         neighbours.setdefault(u, []).append(v)
 355         neighbours.setdefault(v, []).append(u)
 356     total = 0
 357     while True:
 358         previous = {source: None}
 359         queue = deque([source])
 360         while queue and sink not in previous:
 361             u = queue.popleft()
 362             for v in neighbours[u]:
 363                 if v not in previous and residual[u, v]:
 364                     previous[v] = u
 365                     queue.append(v)
 366         if sink not in previous:
 367             return total
 368         amount = 10 ** 30
 369         v = sink
 370         while previous[v] is not None:
 371             u = previous[v]
 372             amount = min(amount, residual[u, v])
 373             v = u
 374         v = sink
 375         while previous[v] is not None:
 376             u = previous[v]
 377             residual[u, v] -= amount
 378             residual[v, u] += amount
 379             v = u
 380         total += amount

 383 def check_scan(certificate):
 384     obj = json.loads(certificate.read_text())
 385     n = obj["n"]
 386     assert n == 64 and obj["stage_ranks"] == [1, 1]
 387     permutation = obj["shuffle_stage_2_to_1"]
 388     rr, cc = obj["R_stage2"], obj["C_stage1"]
 389     edges = signal_edges(n, [1, 1], [permutation])
 390     source, sink = ("SRC",), ("SINK",)
 391     for c in cc:
 392         edges[source, (0, "x", c, "i")] = 10 ** 9
 393     for r in rr:
 394         edges[(1, "y", r, "o"), sink] = 10 ** 9
 395     nodes = {u for edge in edges for u in edge}
 396     left = {tuple(x) for x in obj["network_source_partition"]}
 397     assert len(left) == len(obj["network_source_partition"])
 398     assert left <= nodes and source in left and sink not in left
 399     crossing = {(u, v): cap for (u, v), cap in edges.items() if u in left and v not in left}
 400     claimed = {(tuple(e["tail"]), tuple(e["head"])): e["capacity"]
 401                for e in obj["network_mincut_crossing_edges"]}
 402     assert crossing == claimed
 403     capacity = sum(crossing.values())
 404     assert capacity == 16
 405     flow = maxflow(edges, source, sink)
 406     assert flow == capacity
 407     bitrev = lambda k: int(format(k, "06b")[::-1], 2)
 408     physical_rows = [bitrev(k) for k in rr]
 409     physical_cols = [bitrev(k) for k in cc]
 410     minor_rows = obj["nonzero_17_minor_physical_rows"]
 411     minor_cols = obj["nonzero_17_minor_physical_cols"]
 412     assert len(set(minor_rows)) == len(set(minor_cols)) == 17
 413     assert set(minor_rows) <= set(physical_rows)
 414     assert set(minor_cols) <= set(physical_cols)
 415     fields = []
 416     for record in obj["finite_fields"]:
 417         p, z = record["p"], record["root"]
 418         assert p > 2 and all(p % d for d in range(2, int(p ** 0.5) + 1))
 419         assert pow(z, 64, p) == 1 and pow(z, 32, p) == p - 1
 420         matrix = [[pow(z, i * j, p) for j in minor_cols] for i in minor_rows]
 421         det = determinant(matrix, p)
 422         full_rank = rank([[pow(z, i * j, p) for j in physical_cols]
 423                           for i in physical_rows], p)
 424         assert det == record["shared_minor_det"] != 0
 425         assert full_rank == record["rank"] == 17
 426         fields.append({"prime": p, "determinant": det,
 427                        "full_rectangle_rank_mod_p": full_rank})
 428     rset, cset = set(rr), set(cc)
 429     br = sum((j in rset) != (j + 1 in rset) for j in range(n - 1))
 430     bc = sum((j in cset) != (j + 1 in cset) for j in range(n - 1))
 431     overlap = len({permutation[j] for j in rr} & cset)
 432     assert (br, bc, overlap) == (4, 4, 9)
 433     return {"n": n, "graph_nodes": len(nodes), "graph_edges": len(edges),
 434             "cut_capacity": capacity, "independent_maxflow": flow,
 435             "old_interval_bound": br + bc + overlap,
 436             "fourier_minor_size": 17, "fields": fields,
 437             "uploaded_code_executed": False}

 440 def check_actual_networks(rng):
 441     records = []
 442     for n in (8, 12):
 443         for stages in (2, 3):
 444             for width in (1, 2):
 445                 matrices = []
 446                 for _ in range(stages):
 447                     a = scale(eye(n), rng.randrange(1, 20))
 448                     for component in range(width):
 449                         for upper in (False, True):
 450                             u = [rng.randrange(1, 20) for _ in range(n)]
 451                             v = [rng.randrange(1, 20) for _ in range(n)]
 452                             transition = [rng.randrange(1, 20) for _ in range(n)]
 453                             for j in range(n):
 454                                 running = v[j]
 455                                 for i in range(j + 1, n):
 456                                     row, col = (j, i) if upper else (i, j)
 457                                     a[row][col] = (a[row][col] + u[i] * running) % P
 458                                     running = running * transition[i] % P
 459                     assert qsep(a) <= width
 460                     matrices.append(a)
 461                 shuffles = [rng.sample(range(n), n) for _ in range(stages - 1)]
 462                 value = matrices[0]
 463                 for index in range(1, stages):
 464                     shuffled = [value[j] for j in shuffles[index - 1]]
 465                     value = mul(matrices[index], shuffled)
 466                 base_edges = signal_edges(n, [width] * stages, shuffles)
 467                 for trial in range(5):
 468                     rr = rng.sample(range(n), rng.randrange(1, n + 1))
 469                     cc = rng.sample(range(n), rng.randrange(1, n + 1))
 470                     edges = dict(base_edges)
 471                     for c in cc:
 472                         edges[("SRC",), (0, "x", c, "i")] = 10 ** 9
 473                     for r in rr:
 474                         edges[(stages - 1, "y", r, "o"), ("SINK",)] = 10 ** 9
 475                     flow = maxflow(edges, ("SRC",), ("SINK",))
 476                     actual = rank(sub(value, rr, cc))
 477                     assert actual <= flow
 478                     records.append({"n": n, "stages": stages, "width": width,
 479                                     "rectangle": [len(rr), len(cc)],
 480                                     "actual_rank": actual, "maxflow": flow})
 481     return records

 484 def check_chirp_polynomial(rng):
 485     p = 2013265921
 486     records = []
 487     for n in (2, 3, 4, 5, 6, 8, 10, 15, 16, 30, 32, 64):
 488         assert (p - 1) % (2 * n) == 0
 489         zeta = pow(31, (p - 1) // (2 * n), p)
 490         assert pow(zeta, n, p) == p - 1
 491         s = 1 - n % 2
 492         coefficients = [pow(zeta, (-j * j + s * j) % (2 * n), p)
 493                         for j in range(n)]
 494         left = [pow(zeta, (k * k - s * k) % (2 * n), p) for k in range(n)]
 495         right = [pow(zeta, (j * j + s * j) % (2 * n), p) for j in range(n)]
 496         matrix = [[0] * n for _ in range(n)]
 497         # Assemble the polynomial using the actual signed-cycle power action,
 498         # rather than filling the claimed c_(k-j) formula.
 499         for degree, coefficient in enumerate(coefficients):
 500             for j in range(n):
 501                 k = j + degree
 502                 matrix[k % n][j] = (coefficient if k < n else -coefficient) % p
 503         dressed = [[left[k] * matrix[k][j] * right[j] % p for j in range(n)]
 504                    for k in range(n)]
 505         expected = [[pow(zeta, 2 * k * j, p) for j in range(n)] for k in range(n)]
 506         assert dressed == expected
 507         for trial in range(3):
 508             x = [rng.randrange(p) for _ in range(n)]
 509             assert [sum(v * z for v, z in zip(row, x)) % p for row in dressed] == [
 510                 sum(pow(zeta, 2 * k * j, p) * x[j] for j in range(n)) % p
 511                 for k in range(n)]
 512         records.append({"n": n, "parity_shift": s, "degree": n - 1,
 513                         "all_entries_verified": n * n, "fresh_inputs": 3})
 514     return records
