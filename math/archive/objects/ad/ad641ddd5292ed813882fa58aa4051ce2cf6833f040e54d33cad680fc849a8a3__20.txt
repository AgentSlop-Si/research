  15 def eye(n):
  16     return [[int(i == j) for j in range(n)] for i in range(n)]

  18 def product(a,b,p):
  19     bt=list(zip(*b))
  20     return [[sum(x*y for x,y in zip(row,col))%p for col in bt] for row in a]

  22 def add(a,b,p):
  23     return [[(x+y)%p for x,y in zip(u,v)] for u,v in zip(a,b)]

  25 def inverse(a,p):
  26     n=len(a); x=[[(v%p) for v in row]+ident for row,ident in zip(a,eye(n))]
  27     for c in range(n):
  28         r=next((r for r in range(c,n) if x[r][c]),None)
  29         if r is None: raise ValueError('Singular modular matrix')
  30         x[c],x[r]=x[r],x[c]
  31         inv=pow(x[c][c],-1,p)
  32         x[c]=[t*inv%p for t in x[c]]
  33         for r in range(n):
  34             if r!=c and x[r][c]:
  35                 f=x[r][c]
  36                 x[r]=[(u-f*v)%p for u,v in zip(x[r],x[c])]
  37     return [r[n:] for r in x]

  39 def rank(a,p):
  40     if not a:return 0
  41     # Sparse online row elimination: useful for the commutant equations too.
  42     pivots={}
  43     for row in a:
  44         row={j:int(x)%p for j,x in enumerate(row) if int(x)%p}
  45         while row:
  46             lead=min(row)
  47             if lead not in pivots:
  48                 scale=pow(row[lead],-1,p)
  49                 pivots[lead]={j:x*scale%p for j,x in row.items()}
  50                 break
  51             scale=row[lead]
  52             for j,v in pivots[lead].items():
  53                 t=(row.get(j,0)-scale*v)%p
  54                 if t:row[j]=t
  55                 elif j in row:del row[j]
  56     return len(pivots)

  58 def field(n):
  59     if (P-1)%n==0:return P,pow(3,-((P-1)//n),P)

  70     return [sum(x*v[j] for j,x in row)%p for row in h]

  72 def parity_audit(n):
  73     p,w=field(n);half=n//2;W=[[pow(w,i*j,p) for j in range(n)] for i in range(n)]
  74     potentials=[(pow(w,i,p)+pow(w,-i,p))%p for i in range(n)]
  75     H=[[((int((i-j)%n==1)+int((i-j)%n==n-1))+(potentials[i] if i==j else 0))%p for j in range(n)] for i in range(n)]
  76     assert product(W,H,p)==product(H,W,p)
  77     assert rank(H,p)==n-2
  78     sectors=[]
  79     for sign in (1,-1):
  80         positions=list(range(half+1)) if sign==1 else list(range(1,half))
  81         # Non-orthonormal real parity columns avoid all modular square roots.
  82         cols=[]
  83         for j in positions:
  84             c=[0]*n;c[j]=1
  85             if j not in (0,half):c[n-j]=sign%p
  86             cols.append(c)
  87         B=[list(row) for row in zip(*cols)]
  88         HB=product(H,B,p);WB=product(W,B,p)
  89         h=[HB[j] for j in positions];f=[WB[j] for j in positions]
  90         assert product(B,h,p)==HB and product(B,f,p)==WB
  91         assert rank(h,p)==len(h)-1
  92         d=len(h);hs=[[(j,v) for j,v in enumerate(row) if v] for row in h]
  93         # Fit from a cyclic column, then evaluate independently on every column.
  94         e=[1]+[0]*(d-1);v=e[:];powers=[]
  95         for k in range(d):powers.append(v);v=sparse_step(hs,v,p)
  96         coeff=[0]*d
  97         for row in range(d-1,-1,-1):
  98             rhs=(f[row][0]-sum(coeff[k]*powers[k][row] for k in range(row+1,d)))%p
  99             coeff[row]=rhs*pow(powers[row][row],-1,p)%p
 100         for j in range(d):
 101             v=[0]*d
 102             for c in coeff[::-1]:
 103                 v=sparse_step(hs,v,p);v[j]=(v[j]+c)%p
 104             assert v==[f[i][j] for i in range(d)]
 105         target=pow(2,-1,p) if sign==1 else (pow(w,half-1,p)-pow(w,-(half-1),p))%p
 106         assert coeff[-1]==target!=0
 107         # Jacobi null vector, then verify Fourier phase on that line.
 108         z=[1]+[0]*(d-1)
 109         for i in range(d-1):
 110             z[i+1]=(-sum(h[i][j]*z[j] for j in range(i+1))*pow(h[i][i+1],-1,p))%p
 111         assert matvec(h,z,p)==[0]*d
 112         wz=matvec(f,z,p);lam=wz[0]
 113         assert wz==[lam*x%p for x in z] and lam*lam%p==sign*n%p
 114         sectors.append({'parity':sign,'dimension':d,'degree':len(coeff)-1,'top_coefficient_unnormalized':coeff[-1], 'zero_line_fourier_eigenvalue_unnormalized':lam,'all_entries_equal':True})
 115     assert sectors[0]['zero_line_fourier_eigenvalue_unnormalized']!=sectors[1]['zero_line_fourier_eigenvalue_unnormalized']
 116     return {'n':n,'prime':p,'root':w,'harper_nullity':2,'sectors':sectors}

 118 def intertwiner_audit(n):
 119     p,w=field(n);phase=[(pow(w,i,p)+pow(w,-i,p))%p for i in range(n)]
 120     allowed=[(i,j) for i in range(n) for j in range(n) if phase[i]==phase[j]]
 121     assert set(allowed)=={(i,j) for i in range(n) for j in (i,(-i)%n)}
 122     equations=[[0]*len(allowed) for _ in range(n*n)]
 123     for col,(a,b) in enumerate(allowed):
 124         for k in ((b+1)%n,(b-1)%n):equations[a*n+k][col]+=1
 125         for k in ((a+1)%n,(a-1)%n):equations[k*n+b][col]-=1
 126     dim=len(allowed)-rank(equations,p);assert dim==2
 127     W=[[pow(w,i*j,p) for j in range(n)] for i in range(n)]
 128     def defects(M):
 129         return [sum((M[i][(j+1)%n]+M[i][(j-1)%n]-phase[i]*M[i][j])%p!=0 for i in range(n) for j in range(n)),sum((phase[j]*M[i][j]-M[(i+1)%n][j]-M[(i-1)%n][j])%p!=0 for i in range(n) for j in range(n))]
 130     FJ=[[W[i][(-j)%n] for j in range(n)] for i in range(n)]
 131     candidate=[[(3*W[i][j]+5*FJ[i][j])%p for j in range(n)] for i in range(n)]
 132     assert defects(W)==defects(FJ)==defects(candidate)==[0,0]
 133     assert rank([[W[i][1],FJ[i][1]] for i in range(n)],p)==2
 134     assert rank([[W[i][0],FJ[i][0]] for i in range(n)],p)==1
 135     return {'n':n,'prime':p,'commutant_dimension':dim,'e1_calibration_rank':2,'e0_calibration_rank':1}

 140 def diagonal_similarity(u,v,p):
 141     n=len(u);g=[[] for _ in u]
 142     for i in range(n):
 143         for j in range(n):
 144             a,b=u[i][j]%p,v[i][j]%p
 145             if (a==0)!=(b==0):return None
 146             if a:
 147                 r=a*pow(b,-1,p)%p
 148                 g[j].append((i,r));g[i].append((j,pow(r,-1,p)))
 149     q=[None]*n
 150     for s in range(n):
 151         if q[s] is not None:continue
 152         q[s]=1;queue=collections.deque([s])
 153         while queue:
 154             i=queue.popleft()
 155             for j,r in g[i]:
 156                 x=q[i]*r%p
 157                 if q[j] is None:q[j]=x;queue.append(j)
 158                 elif q[j]!=x:return None
 159     assert all(u[i][j]*q[j]%p==q[i]*v[i][j]%p for i in range(n) for j in range(n))
 160     return q

 162 def source_protocol_factor(n,rng,w):
 163     # Reconstruct only the published coefficient-generation protocol.
 164     # Solver below is independently written full pivoted elimination.
 165     total=[[0]*n for _ in range(n)]
 166     for _ in range(2):
 167         order=list(range(n));rng.shuffle(order)
 168         phase=[pow(w,(i*i+3*i+rng.randrange(n))%n,P) for i in range(n)]
 169         lo=[(1+pow(w,i+rng.randrange(n),P))%P for i in range(n-1)]
 170         hi=[(2+pow(w,2*i+rng.randrange(n),P))%P for i in range(n-1)]
 171         b=[[0]*n for _ in range(n)]
 172         for i in range(n):
 173             b[i][i]=(7+phase[i])%P
 174             if i+1<n:b[i+1][i]=lo[i];b[i][i+1]=hi[i]
 175         inv=inverse(b,P)
 176         left=[pow(w,rng.randrange(n),P) for _ in range(n)]
 177         right=[pow(w,rng.randrange(n),P) for _ in range(n)]
 178         for i in range(n):
 179             for j in range(n):total[order[i]][order[j]]=(total[order[i]][order[j]]+left[i]*inv[i][j]*right[j])%P
 180     return total

 223 def separator(orders):
 224     n=len(orders[0]);r=list(orders[0][:n//2]);c=list(orders[0][n-n//2:])
 225     for order in orders[1:]:
 226         rr=set(r);cc=set(c);selected=[x for x in order if x in rr or x in cc]
 227         left=set(selected[:len(r)]);right=set(selected[len(r):])
 228         options=[([x for x in r if x in left],[x for x in c if x in right]),([x for x in r if x in right],[x for x in c if x in left])]
 229         r,c=max(options,key=lambda pair:len(pair[0]))
 230     assert len(r)==len(c)
 231     for order in orders:
 232         pos={x:i for i,x in enumerate(order)}
 233         assert max(pos[x] for x in r)<min(pos[x] for x in c) or max(pos[x] for x in c)<min(pos[x] for x in r)
 234     return r,c

 340 def read_source_tree(path):

 382 def independent_separator(row_orders,column_orders):
 383     n=len(row_orders[0]);I=list(range(n));J=list(range(n))
 384     for ro,co in zip(row_orders,column_orders):
 385         rp={label:i for i,label in enumerate(ro)};cp={label:i for i,label in enumerate(co)}
 386         rr=sorted(I,key=rp.__getitem__);cc=sorted(J,key=cp.__getitem__);k=len(I)//2
 387         if not k:return [],[]
 388         if rp[rr[k-1]]<cp[cc[-k]]:I,J=rr[:k],cc[-k:]
 389         else:
 390             assert cp[cc[k-1]]<rp[rr[-k]]
 391             I,J=rr[-k:],cc[:k]

 394         assert max(rp[x] for x in I)<min(cp[x] for x in J) or max(cp[x] for x in J)<min(rp[x] for x in I)
 395     return I,J
