  27 def prepare_orbit(field,n,q,coefficients):
  28     if not coefficients or len(coefficients)>n:
  29         raise ValueError('1..N coefficients required')
  30     kernel=prepare(field,n,q)
  31     c=Counted(field)
  32     roots=[field.one]
  33     for _ in range(1,n):

  36     last_g=c.mul(kernel.factorial[-1],roots[-1])

  50 def partial_forward(c,plan,x):
  51     """First R Fourier coefficients, R the dyadic rounded requested count."""
  52     n,r=plan.n,plan.rounded
  53     def visit(start,stride,size):
  54         if size==r:
  55             values=[x[start+stride*k] for k in range(r)]
  56             return values if r==1 else fft(c,plan.leaf,values)
  57         even=visit(start,2*stride,size//2)
  58         odd=visit(start+stride,2*stride,size//2)
  59         return [c.add(a,c.mul(plan.roots[k*(n//size)],b))
  60                 for k,(a,b) in enumerate(zip(even,odd))]
  61     return visit(0,1,n)

  64 def partial_transpose(c,plan,coefficients):
  65     """Transpose (not conjugate transpose) of partial_forward, all N outputs."""
  66     n,r=plan.n,plan.rounded
  67     if len(coefficients)!=r:
  68         raise ValueError('R coefficients required')
  69     out=[c.zero]*n
  70     def visit(start,stride,size,values):
  71         if size==r:
  72             result=values if r==1 else fft(c,plan.leaf,values)
  73             for k,a in enumerate(result):
  74                 out[start+stride*k]=a
  75             return
  76         right=[c.mul(plan.roots[k*(n//size)],a) for k,a in enumerate(values)]
  77         visit(start,2*stride,size//2,values)
  78         visit(start+stride,2*stride,size//2,right)
  79     visit(0,1,n,coefficients)
