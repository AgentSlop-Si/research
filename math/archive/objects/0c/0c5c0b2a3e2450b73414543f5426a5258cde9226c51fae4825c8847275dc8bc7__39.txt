  16 def inv(a, p):
  17     if a % p == 0:
  18         raise ZeroDivisionError
  19     return pow(a, -1, p)
  20 
  21 
  22 def sqrt_cipolla(a, p):
  23     """Independent finite-field square root; setup cost is not counted as free."""
  24     a %= p
  25     if a == 0:
  26         return 0
  27     if pow(a, (p-1)//2, p) != 1:
  28         raise ValueError('no square root')
  29     t = 0
  30     while pow((t*t-a) % p, (p-1)//2, p) != p-1:
  31         t += 1
  32     d = (t*t-a) % p
  33     def mul(u, v):
  34         return ((u[0]*v[0] + d*u[1]*v[1]) % p,
  35                 (u[0]*v[1] + u[1]*v[0]) % p)
  36     r, b, e = (1, 0), (t, 1), (p+1)//2
  37     while e:
  38         if e & 1:
  39             r = mul(r, b)
  40         b = mul(b, b)
  41         e //= 2
  42     assert r[1] == 0 and r[0]*r[0] % p == a
  43     return min(r[0], p-r[0])

  46 def prime_factors(n):
  47     out, d = [], 2
  48     while d*d <= n:
  49         if n % d == 0:
  50             out.append(d)
  51             while n % d == 0:
  52                 n //= d
  53         d += 1
  54     if n > 1:
  55         out.append(n)
  56     return out
  57 
  58 
  59 def generator(p):
  60     factors = prime_factors(p-1)
  61     return next(g for g in range(2, p)
  62                 if all(pow(g, (p-1)//q, p) != 1 for q in factors))

  65 def successor_linear(order):
  66     """O(N) validation and construction, with no sorting or lookup table oracle."""
  67     n = len(order)
  68     if n == 0:
  69         raise ValueError('empty order')
  70     seen = [False]*n
  71     for j in order:
  72         if not isinstance(j, int) or j < 0 or j >= n or seen[j]:
  73             raise ValueError('not a permutation')
  74         seen[j] = True
  75     successor = [0]*n
  76     for a, b in zip(order, order[1:]):
  77         successor[a] = b
  78     successor[order[-1]] = order[0]
  79     return successor
  80 
  81 
  82 def dot(a, b, p):
  83     return sum(x*y for x, y in zip(a, b)) % p
  84 
  85 
  86 def mv(A, x, p):
  87     return [dot(row, x, p) for row in A]
  88 
  89 
  90 def transpose(A):
  91     return [list(row) for row in zip(*A)]
  92 
  93 
  94 def mm(A, B, p):
  95     return [[dot(row, col, p) for col in transpose(B)] for row in A]
  96

  98 def solve_and_det(A, b, p):
  99     n = len(A)
 100     if not n:
 101         return [], 1
 102     M = [[x % p for x in row] + [v % p] for row, v in zip(A, b)]
 103     det = 1
 104     for j in range(n):
 105         pivot = next((i for i in range(j, n) if M[i][j]), None)
 106         if pivot is None:
 107             raise ArithmeticError('singular matrix')
 108         if pivot != j:
 109             M[pivot], M[j] = M[j], M[pivot]
 110             det = -det
 111         pivot_value = M[j][j]
 112         det = det*pivot_value % p
 113         q = inv(pivot_value, p)
 114         M[j] = [v*q % p for v in M[j]]
 115         for i in range(n):
 116             if i != j:
 117                 t = M[i][j]
 118                 M[i] = [(a-t*b) % p for a, b in zip(M[i], M[j])]
 119     return [row[-1] for row in M], det % p

 122 class AuditTransform:
 123     def __init__(self, n, p):
 124         assert n >= 4 and n & (n-1) == 0 and (p-1) % n == 0
 125         self.n, self.p = n, p
 126         self.w = pow(generator(p), -(p-1)//n, p)
 127         assert pow(self.w, n, p) == 1 and pow(self.w, n//2, p) != 1
 128         self.sn, self.s2 = sqrt_cipolla(n, p), sqrt_cipolla(2, p)
 129         self.isn, self.is2 = inv(self.sn, p), inv(self.s2, p)
 130         self.i = inv(pow(self.w, n//4, p), p)
 131         assert self.i*self.i % p == p-1
 132         self.C = transpose([self.column(j) for j in range(n)])
 133         self.F = [[pow(self.w, k*j, p) for j in range(n)] for k in range(n)]
 134         I = [[int(a == b) for b in range(n)] for a in range(n)]
 135         assert mm(self.C, transpose(self.C), p) == I
 136         self.angles = []
 137         self.R = [[0]*n for _ in range(n)]
 138         self.R[0][0], self.R[1][1] = 1, p-1
 139         for k in range(1, n//2):
 140             a, b = pow(self.w, k, p), pow(self.w, -k, p)
 141             c = (a+b)*inv(2, p) % p
 142             s = (b-a)*inv(2*self.i, p) % p
 143             self.angles.append((c, s))
 144             u, v = k+1, n//2+k
 145             self.R[u][u], self.R[u][v] = c, -s % p
 146             self.R[v][u], self.R[v][v] = s, c
 147         # CZ shifts column j to C[:,j+1]. This directly checks every sign.
 148         CZ = [[row[(j+1) % n] for j in range(n)] for row in self.C]
 149         assert CZ == mm(self.R, self.C, p)
 150 
 151     def column(self, j):
 152         n, p, w = self.n, self.p, self.w
 153         out = [0]*n
 154         out[0] = self.isn
 155         out[1] = self.isn*pow(w, n*j//2, p) % p
 156         a, b, sa, sb = 1, 1, pow(w, j, p), pow(w, -j, p)
 157         for k in range(1, n//2):
 158             a, b = a*sa % p, b*sb % p
 159             out[k+1] = (a+b)*self.isn*self.is2 % p
 160             out[n//2+k] = (a-b)*self.i*self.isn*self.is2 % p
 161         return out
 162 
 163     def convert(self, f):
 164         n, p = self.n, self.p
 165         out = [0]*n
 166         out[0], out[1] = f[0]*self.isn % p, f[n//2]*self.isn % p
 167         for k in range(1, n//2):
 168             out[k+1] = (f[k]+f[-k])*self.is2*self.isn % p
 169             out[n//2+k] = (f[k]-f[-k])*self.i*self.is2*self.isn % p
 170         return out
 171 
 172     def block_inverse(self, z, b):
 173         n, p = self.n, self.p
 174         assert pow(z, n, p) != 1
 175         y = [0]*n
 176         y[0], y[1] = b[0]*inv(1-z, p) % p, b[1]*inv(1+z, p) % p
 177         for k, (c, s) in enumerate(self.angles, 1):
 178             q, t = (1-z*c) % p, z*s % p
 179             d = inv(q*q+t*t, p)
 180             u, v = k+1, n//2+k
 181             y[u] = (q*b[u]-t*b[v])*d % p
 182             y[v] = (t*b[u]+q*b[v])*d % p
 183         return y
 184 
 185     def input(self, order, z):
 186         x, power = [0]*self.n, 1
 187         for j in order:
 188             x[j], power = power, power*z % self.p
 189         return x
 190 
 191     def woodbury(self, order, z):
 192         n, p = self.n, self.p
 193         z %= p
 194         factor = (1-pow(z, n, p)) % p
 195         if factor == 0:
 196             raise ValueError('singular base resolvent')
 197         w = successor_linear(order)
 198         J = [j for j in range(n) if w[j] != (j+1) % n]
 199         b = self.column(order[0])
 200         y = self.block_inverse(z, b)
 201         AU, V = [], []
 202         for j in J:
 203             u = [(a-b) % p for a, b in zip(self.column(w[j]), self.column((j+1) % n))]
 204             AU.append(self.block_inverse(z, u))
 205             V.append(self.column(j))
 206         K = [[(int(a == b)-z*dot(V[a], AU[b], p)) % p for b in range(len(J))]
 207              for a in range(len(J))]
 208         h = [dot(v, y, p) for v in V]
 209         c, determinant = solve_and_det(K, h, p)
 210         assert determinant == 1
 211         # Independent closed form in original coordinate basis.
 212         denominator_inverse = inv(factor, p)
 213         for a, j in enumerate(J):
 214             assert h[a] == pow(z, (j-order[0]) % n, p)*denominator_inverse % p
 215             for b, ell in enumerate(J):
 216                 q = (pow(z, (j-w[ell]) % n, p)-pow(z, (j-ell-1) % n, p))*denominator_inverse % p
 217                 assert K[a][b] == (int(a == b)-z*q) % p
 218         return [(factor*(y[k]+z*sum(cj*u[k] for cj, u in zip(c, AU)))) % p
 219                 for k in range(n)], len(J)
 220 
 221     def sparse_twisted(self, order, z):
 222         """All-z O(N(s+1)) field arithmetic, given order and transform constants.
 223 
 224         One inversion after batch preparation. A root singularity is evaluated
 225         by an explicit N-term sum; no Fourier-character shortcut is assumed.
 226         """
 227         n, p = self.n, self.p
 228         z %= p
 229         w = successor_linear(order)
 230         x = self.input(order, z)
 231         J = [j for j in range(n) if w[j] != (j+1) % n]
 232         d = [0]*n
 233         d[order[0]] = (1-pow(z, n, p)) % p
 234         for j in J:
 235             a = z*x[j] % p
 236             d[w[j]] = (d[w[j]]+a) % p
 237             d[(j+1) % n] = (d[(j+1) % n]-a) % p
 238         assert d == [(x[j]-z*x[j-1]) % p for j in range(n)]
 239         support = [(j, v) for j, v in enumerate(d) if v]
 240         assert len(support) <= 2*len(J)+1
 241         numer = [0]*n
 242         for j, v in support:
 243             power, step = 1, pow(self.w, j, p)
 244             for k in range(n):
 245                 numer[k] = (numer[k]+v*power) % p
 246                 power = power*step % p
 247         den, power = [], 1
 248         for k in range(n):
 249             den.append((1-z*power) % p)
 250             power = power*self.w % p
 251         regular = [k for k in range(n) if den[k]]
 252         exceptional = [k for k in range(n) if not den[k]]
 253         assert len(exceptional) <= 1
 254         pref = [1]
 255         for k in regular:
 256             pref.append(pref[-1]*den[k] % p)
 257         back, f = inv(pref[-1], p), [0]*n
 258         for t in range(len(regular)-1, -1, -1):
 259             k = regular[t]
 260             f[k] = numer[k]*back*pref[t] % p
 261             back = back*den[k] % p
 262         for k in exceptional:
 263             step, power, value = pow(self.w, k, p), 1, 0
 264             for a in x:
 265                 value = (value+a*power) % p
 266                 power = power*step % p
 267             f[k] = value
 268         return self.convert(f), len(J), len(support), len(exceptional)

 271 def sample_orders(n, rng):
 272     base = list(range(n))
 273     out = [base, base[1:]+base[:1], base[n//2:]+base[:n//2], list(reversed(base))]
 274     for a, b in [(0, 1), (1, n//2), (0, n-1)]:
 275         q = base[:]
 276         q[a], q[b] = q[b], q[a]
 277         out.append(q)
 278     q = base[:]
 279     q[0], q[n//2] = q[n//2], q[0]
 280     q[1], q[n-1] = q[n-1], q[1]
 281     out.append(q)
 282     for _ in range(4):
 283         q = base[:]
 284         rng.shuffle(q)
 285         out.append(q)
 286     return list(dict.fromkeys(tuple(q) for q in out))

 289 def main():
 290     rng = random.Random(240023)
 291     groups, nonsingular, singular, covariance = [], 0, 0, 0
 292     for p, n in [(17,4), (17,8), (17,16), (97,4), (97,8), (97,16), (97,32),
 293                  (193,64), (998244353,4), (998244353,8), (998244353,16),
 294                  (998244353,32), (998244353,64)]:
 295         T = AuditTransform(n, p)
 296         covariance += 1
 297         orders = list(itertools.permutations(range(n))) if n == 4 else sample_orders(n, rng)
 298         roots = [pow(T.w, k, p) for k in range(n)]
 299         if p <= 97:
 300             nodes = list(range(p))
 301         else:
 302             nodes = list(dict.fromkeys([0,1,2,p-1]+roots+[rng.randrange(p) for _ in range(6)]))
 303         before = nonsingular, singular
 304         for order in orders:
 305             for z in nodes:
 306                 x = T.input(order, z)
 307                 expected = mv(T.C, x, p)
 308                 sparse, s, support, exceptional = T.sparse_twisted(order, z)
 309                 assert sparse == expected
 310                 lean = permuted_geometric_dft(order, z, p, T.w)
 311                 assert lean == mv(T.F, x, p)
 312                 assert T.convert(lean) == expected
 313                 if pow(z, n, p) != 1:
 314                     got, same_s = T.woodbury(order, z)
 315                     assert got == expected and same_s == s and exceptional == 0
 316                     nonsingular += 1
 317                 else:
 318                     try:
 319                         T.woodbury(order, z)
 320                     except ValueError:
 321                         pass
 322                     else:
 323                         raise AssertionError('root should be rejected by nonsingular executor')
 324                     assert exceptional == 1
 325                     singular += 1
 326         groups.append({'p':p, 'n':n, 'orders':len(orders), 'nodes':len(nodes),
 327                        'woodbury_nonsingular':nonsingular-before[0],
 328                        'all_z_sparse_singular':singular-before[1]})
 329     # Counterexample to reusing the unpermuted character branch.
 330     T = AuditTransform(8, 97)
 331     order = list(range(8))
 332     order[1], order[4] = order[4], order[1]
 333     z = pow(T.w, -1, 97)
 334     ordinary = mv(T.C, T.input(list(range(8)), z), 97)
 335     permuted = mv(T.C, T.input(order, z), 97)
 336     assert ordinary != permuted
 337     witness = {'p':97, 'n':8, 'omega':T.w, 'sqrt_n':T.sn, 'sqrt_2':T.s2,
 338                'i':T.i, 'order':order, 'z':z, 'z_to_n':pow(z,8,97),
 339                'unpermuted_Cg':ordinary, 'required_CPinverse_g':permuted}
 340     invalid = [[], [0,0], [1], [0,2], [-1,0], [0,'1'], [0,1,3]]
 341     for order in invalid:
 342         try:
 343             successor_linear(order)
 344         except ValueError:
 345             pass
 346         else:
 347             raise AssertionError('invalid permutation was accepted')
 348     result = {'status':'PASS', 'submitted_code_executed_or_imported':False,
 349               'matrix_orthogonality_and_covariance_configurations':covariance,
 350               'woodbury_nonsingular_exact_comparisons':nonsingular,
 351               'woodbury_small_determinant_equals_one_checks':nonsingular,
 352               'closed_coordinate_resolvent_comparisons':'Every small K and rhs entry in every nonsingular case',
 353               'all_z_sparse_exact_comparisons':nonsingular+singular,
 354               'standalone_linear_evaluator_exact_comparisons':nonsingular+singular,
 355               'singular_root_exact_comparisons':singular,
 356               'invalid_permutation_rejections':len(invalid), 'groups':groups,
 357               'singular_character_counterexample':witness}
 358     path = Path(__file__).with_name('CHECK_RESULTS.json')
 359     path.write_text(json.dumps(result, indent=2)+'\n')
 360     print(json.dumps({k:v for k,v in result.items() if k not in ('groups','singular_character_counterexample')}, indent=2))
 361     print('Wrote', path)
