   1 from math import comb, log

   4 def constants(h: int):
   5     v = comb(h, 3)
   6     n0 = v**3
   7     i0 = 3*v*v
   8     d_nbr = comb(h-3, 3) + 3*(h-3)
   9     m = h**3
  10     W = 2*n0 + i0*(v*d_nbr + (h+1))
  11     Delta = 2*v*v*(v - 3*h*(h+1))
  12     p = (W-1).bit_length()
  13     Wstar = 1 << p
  14     S = Wstar*m - Delta
  15     lam = S / Wstar
  16     theta = log(lam) / log(m)
  17     return locals()
