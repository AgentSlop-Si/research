82: def inv_mod(a, p: int = MOD):
83:     n = len(a)
84:     aug = [list(map(lambda z: z % p, a[i])) + eye(n)[i] for i in range(n)]
85:     r = 0
86:     for c in range(n):
87:         pivot = next((i for i in range(r, n) if aug[i][c] % p), None)
88:         if pivot is None:
89:             raise ValueError("singular")
90:         aug[r], aug[pivot] = aug[pivot], aug[r]
91:         z = inv(aug[r][c], p)
92:         aug[r] = [(v * z) % p for v in aug[r]]
93:         for i in range(n):
94:             if i != r and aug[i][c] % p:
95:                 f = aug[i][c]
96:                 aug[i] = [(x - f * y) % p for x, y in zip(aug[i], aug[r])]
97:         r += 1
98:     return [row[n:] for row in aug]
