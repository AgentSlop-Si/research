 209     c = Ledger()
 210     n = validate(pairs, c)
 211     products = cached_prefix(cache, pairs, n, len(pairs), c)
 212     schedule = ['cached']*len(products)
 213     for pair in pairs[len(products):]:
 214         products.append(seed_product(pair, n, c))
 215         schedule.append('fresh')
 216     return finish(products, c, schedule)

 219 def secant_batch(pairs, max_patch=2, cache=None):
 220     """Fixed verified 2-parent schedule; dense residuals fall back to paid seeds."""
 221     c = Ledger()
 222     n = validate(pairs, c)
 223     if not isinstance(max_patch, int) or isinstance(max_patch, bool) or max_patch < 0:
 224         raise ValueError('max_patch must be a nonnegative integer')
 225     for pair in pairs[1:]:
 226         for u, v in zip(pairs[0][1], pair[1]):
 227             c['same_b_digit_comparisons'] += 1
 228             if u != v:
 229                 raise ValueError('secant mode requires the same B in every query')
 230     products = cached_prefix(cache, pairs, n, min(2, len(pairs)), c)
 231     schedule = ['cached']*len(products)
 232     residual_supports = []
 233     for i in range(len(products), len(pairs)):
 234         if i < 2:
 235             products.append(seed_product(pairs[i], n, c))
 236             schedule.append('fresh')
 237             continue
 238         c['secant_relations_tested'] += 1
 239         raw = combine([pairs[i][0], pairs[i-1][0], pairs[i-2][0]], [1,-2,1], n+1, c)
 240         numeric = signed_patch(raw, n+1, c)
 241         coordinate = []
 242         for j, value in enumerate(raw):
 243             c['coordinate_patch_zero_tests'] += 1
 244             if value:
 245                 coordinate.append((j, 1 if value > 0 else -1, abs(value)))
 246                 c['coordinate_patch_records'] += 1
 247         patch = numeric if len(numeric) < len(coordinate) else coordinate
 248         residual_supports.append(len(patch))
 249         if len(patch) > max_patch:
 250             products.append(seed_product(pairs[i], n, c))
 251             schedule.append('fresh_dense_residual')
 252             continue
 253         coefficients = combine([products[i-1], products[i-2]], [2,-1], 2*n+1, c)
 254         for j, sign, digit in patch:
 255             for k, value in enumerate(pairs[i][1]):
 256                 term = digit*value
 257                 c['digit_mul'] += 1
 258                 if sign > 0:
 259                     coefficients[j+k] += term
 260                     c['word_add'] += 1
 261                 else:
 262                     coefficients[j+k] -= term
 263                     c['word_sub'] += 1
 264                 c['residual_convolution_updates'] += 1
 265         products.append(product_digits(coefficients, n, c))
 266         schedule.append('secant')
 267     return finish(products, c, schedule, residual_supports=residual_supports)

 270 def polynomial_batch(pairs, max_degree=MAX_DEGREE, cache=None):
 271     """Verify each operand trajectory, then use their product-degree recurrence."""
 272     c = Ledger()
 273     n = validate(pairs, c)
 274     if (not isinstance(max_degree, int) or isinstance(max_degree, bool)
 275             or not 0 <= max_degree <= MAX_DEGREE):
 276         raise ValueError('supported discovery degree is 0 through 3')
 277     degrees = [discover_degree(pairs, j, max_degree, n, c) for j in (0,1)]
 278     if None in degrees:
 279         products = cached_prefix(cache, pairs, n, len(pairs), c)
 280         schedule = ['cached']*len(products)
 281         for pair in pairs[len(products):]:
 282             products.append(seed_product(pair, n, c))
 283             schedule.append('fresh_unstructured')
 284         return finish(products, c, schedule, degrees=degrees, verified=False)
 285     order = sum(degrees)+1
 286     products = cached_prefix(cache, pairs, n, min(order, len(pairs)), c)
 287     schedule = ['cached']*len(products)
 288     weights = [(-1)**(j+1)*comb(order,j) for j in range(1,order+1)]
 289     c['recurrence_coefficients_prepared'] += order
 290     for i in range(len(products), len(pairs)):
 291         if i < order:
 292             products.append(seed_product(pairs[i], n, c))
 293             schedule.append('fresh')
 294         else:
 295             previous = [products[i-j] for j in range(1, order+1)]
 296             coefficients = combine(previous, weights, 2*n+1, c)
 297             products.append(product_digits(coefficients, n, c))
 298             schedule.append('polynomial')
 299     return finish(products, c, schedule, degrees=degrees, verified=True)
