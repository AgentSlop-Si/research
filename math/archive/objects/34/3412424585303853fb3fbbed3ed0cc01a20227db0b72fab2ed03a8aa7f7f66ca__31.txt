0008 class TestLazySpectrum(unittest.TestCase):
0009     def test_point_edits_and_partial_bins(self):
0010         rng=np.random.default_rng(1001)
0011         for n in [17,64,256,1024]:
0012             x=rng.standard_normal(n)+1j*rng.standard_normal(n)
0013             state=LazySpectrum.from_spectrum(np.fft.fft(x))
0014             edits=[]
0015             for _ in range(30):
0016                 j=int(rng.integers(0,n)); d=complex(rng.normal(),rng.normal())
0017                 x[j]+=d; edits.append((j,d)); state.add_edit(j,d)
0018             bins=rng.integers(0,n,size=min(25,n)).tolist()
0019             got=state.bins(bins)
0020             want=np.fft.fft(x)[bins]
0021             self.assertTrue(np.allclose(got,want,rtol=2e-11,atol=2e-10))

0034         rng=np.random.default_rng(1002)
0035         for n in [64,256,1024]:
0036             base=rng.normal(size=n)+1j*rng.normal(size=n)
0037             idx=rng.choice(n,size=min(12,n),replace=False)
0038             vals=rng.normal(size=len(idx))+1j*rng.normal(size=len(idx))
0039             altered=base.copy(); altered[idx]+=vals
0040             for method in ['direct','factorized']:
0041                 s=LazySpectrum.from_spectrum(np.fft.fft(base))
0042                 s.add_edits(zip(idx.tolist(),vals.tolist()))
0043                 got,plan=s.materialize(method=method)
0044                 self.assertTrue(np.allclose(got,np.fft.fft(altered),rtol=2e-11,atol=3e-10))
0045                 self.assertEqual(plan.pending_edits,len(idx))
