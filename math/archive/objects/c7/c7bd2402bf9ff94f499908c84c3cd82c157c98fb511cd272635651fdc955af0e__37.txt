   7 def root(N):
   8     for a in range(2,P):
   9         w=pow(a,(P-1)//N,P)
  10         if pow(w,N,P)==1 and pow(w,N//2,P)!=1:return w
  11     raise AssertionError()
  12 
  13 def qstep(x,w):
  14     N=len(x);out=[0]*N;phase=1
  15     for s,a in enumerate(x):
  16         out[2*s%N]=(out[2*s%N]+a)%P
  17         out[(2*s+1)%N]=(out[(2*s+1)%N]+phase*a)%P
  18         phase=phase*w%P
  19     return out
  20 
  21 def qtstep(x,w):
  22     N=len(x);out=[0]*N;phase=1
  23     for s in range(N):
  24         out[s]=(x[2*s%N]+phase*x[(2*s+1)%N])%P
  25         phase=phase*w%P
  26     return out
  27 
  28 def gamma(k):
  29     ans=0
  30     while k:
  31         ans+=(k//2)*(k&1);k//=2
  32     return ans
  33 
  34 def rev(k,d):
  35     ans=0
  36     for _ in range(d):ans=2*ans+(k&1);k//=2
  37     return ans
  38 
  39 def good(k,L,b):
  40     return k>0 and k.bit_length()==L and '0'*b not in format(k,'b')
  41 
  42 def fold_inputs(x,b):
  43     N=len(x);r=N>>b;us=[]
  44     for c in range(b):
  45         a=b-c;u=[0]*r
  46         for j in range(N>>a):u[j%r]=(u[j%r]+x[j<<a])%P
  47         us.append(u)
  48     return us
  49 
  50 def schur_coefficient(cur,t,b,w,N):
  51     r=len(cur);eta=pow(w,1<<b,P);out=[0]*r
  52     for a,v in enumerate(cur):
  53         k=a% (1<<t)
  54         if good(k,t,b) and k%2:
  55             y=((1<<b)*(a-k)+k)%r
  56             out[y]=(out[y]+pow(w*pow(eta,-1,P)%P,gamma(k),P)*v)%P
  57     return out
  58 
  59 def compressed_source(x,b,w):
  60     N=len(x);n=N.bit_length()-1;d=n-b;r=1<<d
  61     out=[[0]*r for _ in range(n+1)];out[0]=x[::1<<b]
  62     for s in range(1,N):
  63         a=(s&-s).bit_length()-1
  64         if a<b:
  65             L=b-a;y=(s>>a)%r;out[L][y]=(out[L][y]+x[s])%P
  66     cur=x[:]
  67     for t in range(1,d+1):
  68         cur=qstep(cur,w)
  69         for a,v in enumerate(cur):
  70             k=a% (1<<t);c=t-k.bit_length()
  71             if k%2 and c<b and '0'*b not in format(k,'b'):
  72                 out[b+t][a%r]=(out[b+t][a%r]+v)%P
  73     eta=pow(w,1<<b,P);us=fold_inputs(x,b)
  74     for c in range(min(b,d)):
  75         cur=us[c]
  76         for t in range(1,d-c+1):
  77             cur=qstep(cur,eta);v=schur_coefficient(cur,t,b,w,N)
  78             out[b+c+t]=[(a-z)%P for a,z in zip(out[b+c+t],v)]
  79     return out
  80 
  81 def direct_source(x,b,w):
  82     N=len(x);n=N.bit_length()-1;R=list(range(0,N,1<<b));cur=x[:]
  83     for s in R:cur[s]=0
  84     out=[[x[s] for s in R]]
  85     for L in range(1,n+1):
  86         cur=qstep(cur,w);out.append([cur[s] for s in R])
  87         for s in R:cur[s]=0
  88     return out
  89 
  90 def forced_constant(v,N,b,w,H=None):
  91     n=N.bit_length()-1;H=n if H is None else H;d=n-b;r=1<<d;eta=pow(w,1<<b,P)
  92     out=[[0]*N for _ in range(H+1)];cur=v[:]
  93     for L in range(1,min(d,H)+1):
  94         cur=qstep(cur,eta)
  95         for a,u in enumerate(cur):
  96             k=a%(1<<L)
  97             if good(k,L,b):
  98                 y=((1<<b)*(a-k)+k)%N
  99                 out[L][y]=(out[L][y]+pow(w*pow(eta,-1,P)%P,gamma(k),P)*u)%P
 100     if H>d:
 101         assert len(cur)==r
 102         for L in range(d+1,H+1):
 103             for word in range(1<<(L-1),1<<L):
 104                 if good(word,L,b):
 105                     a=word>>(L-d)
 106                     phase=pow(w,gamma(word),P)*pow(pow(eta,gamma(a),P),-1,P)%P
 107                     out[L][word]=phase*cur[a]%P
 108     return out

 110 def direct_forced(v,N,b,w,H=None):
 111     n=N.bit_length()-1;H=n if H is None else H;R=list(range(0,N,1<<b))
 112     cur=[0]*N
 113     for j,s in enumerate(R):cur[s]=v[j]
 114     out=[[0]*N]
 115     for L in range(1,H+1):
 116         cur=qstep(cur,w)
 117         for s in R:cur[s]=0
 118         out.append(cur[:])
 119     return out

 121 def compressed_free(x,b,w):
 122     N=len(x);n=N.bit_length()-1;d=n-b;r=1<<d;eta=pow(w,1<<b,P)
 123     full=x[:]
 124     for _ in range(n):full=qstep(full,w)
 125     us=fold_inputs(x,b);trans=[]
 126     for u in us:
 127         cur=u[:]
 128         for _ in range(d):cur=qstep(cur,eta)
 129         trans.append(cur)
 130     out=[0]*N;correction=[0]*N;mask=[]
 131     for word in range(1,N):
 132         binary=format(word,'0%db'%n)
 133         if '0'*b in binary:continue
 134         c=n-word.bit_length();freq=rev(word,n);a=rev((freq>>c)%r,d)
 135         assert a==word>>(b-c)
 136         phase=pow(w,gamma(word),P)*pow(pow(eta,gamma(a),P),-1,P)%P
 137         correction[word]=phase*trans[c][a]%P
 138         out[word]=(full[word]-correction[word])%P;mask.append(word)
 139     return out,full,correction,mask

 141 def direct_free(x,b,w):
 142     N=len(x);n=N.bit_length()-1;R=list(range(0,N,1<<b));cur=x[:]
 143     for s in R:cur[s]=0
 144     for _ in range(n):
 145         cur=qstep(cur,w)
 146         for s in R:cur[s]=0
 147     return cur

 149 def vapply(x,w):
 150     N=len(x);n=N.bit_length()-1;out=[0]*N
 151     for s,a in enumerate(x):out[rev(s,n)]=pow(pow(w,gamma(s),P),-1,P)*a%P
 152     return out
