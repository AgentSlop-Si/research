0011 def rank(a):
0012     if not a: return 0
0013     a = [[x % P for x in row] for row in a]
0014     rr = 0
0015     for j in range(len(a[0])):
0016         piv = next((i for i in range(rr, len(a)) if a[i][j]), None)
0017         if piv is None: continue
0018         a[rr], a[piv] = a[piv], a[rr]
0019         inv = pow(a[rr][j], -1, P)
0020         a[rr] = [x * inv % P for x in a[rr]]
0021         for i in range(rr + 1, len(a)):
0022             c = a[i][j]
0023             if c: a[i] = [(x - c*y) % P for x,y in zip(a[i], a[rr])]
0024         rr += 1
0025         if rr == len(a): break
0026     return rr

0028 def block(a, rows, cols): return [[a[i][j] for j in cols] for i in rows]
0029 def mm(a,b):
0030     bt = list(zip(*b))
0031     return [[sum(x*y for x,y in zip(row,col)) % P for col in bt] for row in a]
0032 def ident(n): return [[int(i==j) for j in range(n)] for i in range(n)]
0033 def perm(p): return [[int(i==p[j]) for j in range(len(p))] for i in range(len(p))]
0034 def tr(a): return list(map(list,zip(*a)))
0035 
0036 def qs(n,r):
0037     """Sum of r scalar-transition QS1 generators, with transition zeros."""
0038     a = [[0]*n for _ in range(n)]
0039     for i in range(n): a[i][i] = RNG.randrange(P)
0040     for component in range(r):
0041         u,v,g,h = [[RNG.randrange(P) for _ in range(n)] for _ in range(4)]
0042         lower = [RNG.randrange(P) if RNG.randrange(5) else 0 for _ in range(n)]
0043         upper = [RNG.randrange(P) if RNG.randrange(5) else 0 for _ in range(n)]
0044         for j in range(n):
0045             val = v[j]
0046             for i in range(j+1,n):
0047                 a[i][j] = (a[i][j] + u[i]*val) % P
0048                 val = val*lower[i] % P
0049             val = h[j]
0050             for i in range(j-1,-1,-1):
0051                 a[i][j] = (a[i][j] + g[i]*val) % P
0052                 val = val*upper[i] % P
0053     return a

0055 def cut_ranks(a):
0056     n = len(a)
0057     return [(rank(block(a,range(k),range(k,n))),rank(block(a,range(k,n),range(k)))) for k in range(1,n)]
