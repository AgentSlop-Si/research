0012 def eye(n):

0016 def mul(a, b):

0022 def inv(a):
0023     n = len(a)
0024     c = [a[i][:] + eye(n)[i] for i in range(n)]
0025     for k in range(n):
0026         r = next((r for r in range(k, n) if c[r][k]), None)
0027         if r is None:
0028             raise ValueError("singular")
0029         c[k], c[r] = c[r], c[k]
0030         z = pow(c[k][k], -1, P)
0031         c[k] = [(z * x) % P for x in c[k]]
0032         for r in range(n):
0033             if r != k and c[r][k]:
0034                 z = c[r][k]
0035                 c[r] = [(u - z * v) % P for u, v in zip(c[r], c[k])]
0036     return [r[n:] for r in c]

0043 def root(n):
0044     for z in range(2, P):
0045         w = pow(z, (P - 1) // n, P)
0046         if pow(w, n, P) == 1 and all(pow(w, d, P) != 1
0047                 for d in range(1, n) if n % d == 0):
0048             return w
0049     raise AssertionError("no primitive root")

0052 def canonical(n):
0053     w = root(n)
0054     d = [[pow(w, i, P) if i == j else 0 for j in range(n)] for i in range(n)]
0055     z = [[int(i == (j + 1) % n) for j in range(n)] for i in range(n)]
0056     f = [[pow(w, i * j, P) for j in range(n)] for i in range(n)]
0057     assert mul(f, z) == mul(d, f)
0058     assert mul(f, d) == mul(inv(z), f)
0059     return w, d, z, f

0081 def recover_monomial(x, y, u, v):

0083     sx, su = signatures(x, y), signatures(u, v)
0084     assert len(set(sx)) == n and len(set(su)) == n
0085     lookup = {s: i for i, s in enumerate(su)}
0086     if set(sx) != set(su):
0087         return None, "full-fingerprint mismatch"
0088     pi = [lookup[s] for s in sx]
0089     uu = [[u[pi[i]][pi[j]] for j in range(n)] for i in range(n)]
0090     vv = [[v[pi[i]][pi[j]] for j in range(n)] for i in range(n)]
0091     edges = [[] for _ in range(n)]

0093         for i in range(n):
0094             for j in range(n):
0095                 assert bool(a[i][j]) == bool(b[i][j])

0113     assert mul(x, c) == mul(c, u) and mul(y, c) == mul(c, v)
