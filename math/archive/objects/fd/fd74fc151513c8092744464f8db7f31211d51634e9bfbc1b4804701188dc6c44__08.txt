  41 def rank(a,p):
  42     a=[r[:] for r in a]
  43     i=0
  44     for j in range(len(a[0])):
  45         k=next((k for k in range(i,len(a)) if a[k][j]), None)
  46         if k is None:
  47             continue
  48         a[i],a[k]=a[k],a[i]
  49         f=pow(a[i][j],-1,p)
  50         a[i]=[x*f%p for x in a[i]]
  51         for k in range(i+1,len(a)):
  52             f=a[k][j]
  53             if f:
  54                 a[k]=[(x-f*y)%p for x,y in zip(a[k],a[i])]
  55         i+=1
  56         if i==len(a):
  57             break
  58     return i

  61 def inverse(a,p):
  62     n=len(a); a=[r[:] + e for r,e in zip(a,eye(n,p))]
  63     for j in range(n):
  64         k=next(k for k in range(j,n) if a[k][j])
  65         a[j],a[k]=a[k],a[j]
  66         f=pow(a[j][j],-1,p); a[j]=[x*f%p for x in a[j]]
  67         for k in range(n):
  68             if k!=j and a[k][j]:
  69                 f=a[k][j]; a[k]=[(x-f*y)%p for x,y in zip(a[k],a[j])]
  70     return [r[n:] for r in a]

  77 def signed_cycle(n,p,variant):
  78     rng=random.Random(610100+n*13+variant)
  79     order=list(range(n))
  80     if variant:
  81         rng.shuffle(order)
  82     signs=[1]*n
  83     if variant:
  84         signs=[rng.choice((-1,1)) for _ in range(n)]
  85     if math.prod(signs)!=-1:
  86         signs[-1]*=-1
  87     t=zero(n)
  88     for k,v in enumerate(order):
  89         t[order[(k+1)%n]][v]=signs[k]%p
  90     return t,order,signs

  93 def green_scan(order,signs,t,x,p):
  94     n=len(x)
  95     carry=0
  96     for k,v in enumerate(order):
  97         carry=(x[v]+t*signs[(k-1)%n]*carry)%p
  98     last=carry*pow((1+pow(t,n,p))%p,-1,p)%p
  99     out=[0]*n
 100     for k,v in enumerate(order):
 101         last=(x[v]+t*signs[(k-1)%n]*last)%p
 102         out[v]=last
 103     return out

 110 def apply_fft_escape(x,z,p,ledger):
 111     N=len(x)
 112     if N==1:
 113         return x[:]
 114     if N==2:
 115         ledger['arithmetic']+=2
 116         return [(x[0]+x[1])%p,(x[0]-x[1])%p]
 117     n=N//2; half=pow(2,-1,p); t=pow(2,-1,p)
 118     order=list(range(n)); signs=[1]*(n-1)+[-1]
 119     assert pow(t,N,p)!=1
 120     def C(v,negative=False):
 121         g=green_scan(order,signs,t,v,p)
 122         h=green_scan(order,signs,-t%p,v,p)
 123         ledger['green_applications']+=2
 124         ledger['arithmetic']+=9*n+2
 125         return [((b-a) if negative else (a-b))%p for a,b in zip(g,h)]
 126     def Cinv(v,scalar=1):
 127         # Each output uses two signed permuted inputs and two scalar multiplies.
 128         out=[0]*n
 129         c1=scalar*pow(2*t%p,-1,p)%p; c2=-scalar*t*half%p
 130         for k,src in enumerate(order):
 131             nxt=order[(k+1)%n]; prev=order[(k-1)%n]
 132             out[src]=(c1*signs[k]*v[nxt]+c2*signs[(k-1)%n]*v[prev])%p
 133         ledger['signed_permutation_passes']+=2
 134         ledger['arithmetic']+=3*n
 135         return out
 136     u=[(x[j]+x[j+n])*half%p for j in range(n)]
 137     v=[(x[j]-x[j+n])*half%p for j in range(n)]
 138     ledger['arithmetic']+=4*n
 139     v=[pow(z,j,p)*v[j]%p for j in range(n)]
 140     ledger['arithmetic']+=n
 141     u=apply_fft_escape(u,z*z%p,p,ledger)
 142     v=apply_fft_escape(v,z*z%p,p,ledger)
 143     u=[2*a%p for a in u]; ledger['arithmetic']+=n
 144     v=Cinv(v)
 145     b=Cinv(u,-half)
 146     v=[(a+b)%p for a,b in zip(v,b)]; ledger['arithmetic']+=n
 147     a=C(v)
 148     u=[(a+b)%p for a,b in zip(u,a)]; ledger['arithmetic']+=n
 149     v=C(v,negative=True)
 150     out=[0]*N
 151     for j in range(n):
 152         out[2*j]=(u[j]+v[j])%p
 153         out[2*j+1]=(u[j]-v[j])%p
 154     ledger['arithmetic']+=2*n
 155     ledger['output_permutation_words']+=N
 156     return out
