  28 def triangular_solve(field, inverse_diagonal, bands, rhs):
  29     n = len(rhs)
  30     out = [field.zero]*n
  31     for i in range(n-1, -1, -1):
  32         acc = rhs[i]
  33         for k, band in enumerate(bands, 1):
  34             if i+k < n:
  35                 acc = field.sub(acc, field.mul(band[i], out[i+k]))
  36         out[i] = field.mul(inverse_diagonal, acc)
  37     return out

  40 def prepare_band(field, diagonal, bands):
  41     """Prepare (diagonal I + sum band[k][i] shift^(k+1))^-1.
  42 
  43     Width is 1 or 2 and N>=4. Scalar diagonal must be nonzero; singular full
  44     cyclic matrices are rejected by their small closure determinant.
  45     """
  46     if len(bands) not in (1, 2) or len(bands[0]) < 4:
  47         raise ValueError('width 1 or 2 and N>=4 required')
  48     n = len(bands[0])
  49     if any(len(b) != n for b in bands) or diagonal == field.zero:
  50         raise ValueError('equal band lengths and nonzero diagonal required')
  51     c = Counted(field)
  52     inverse_diagonal = c.inv(diagonal)
  53     width = len(bands)
  54     columns = [[field.zero]*n for _ in range(width)]
  55     for k, band in enumerate(bands, 1):
  56         for i in range(n-k, n):
  57             j = i+k-n
  58             columns[j][i] = c.add(columns[j][i], band[i])
  59     responses = [triangular_solve(c, inverse_diagonal, bands, v) for v in columns]
  60     closure = [[c.add(field.one if i == j else field.zero, responses[j][i])
  61                 for j in range(width)] for i in range(width)]
  62     if width == 1:
  63         if closure[0][0] == field.zero:
  64             raise ValueError('singular cyclic closure')
  65         inverse = [[c.inv(closure[0][0])]]
  66     else:
  67         a, b = closure[0]
  68         d, e = closure[1]
  69         determinant = c.sub(c.mul(a, e), c.mul(b, d))
  70         if determinant == field.zero:
  71             raise ValueError('singular cyclic closure')
  72         scale = c.inv(determinant)
  73         inverse = [[c.mul(e, scale), c.neg(c.mul(b, scale))],
  74                    [c.neg(c.mul(d, scale)), c.mul(a, scale)]]
  75     return BandPlan(n, diagonal, inverse_diagonal, [list(b) for b in bands],
  76                     responses, inverse, dict(c.count))

  91 def solve_band(field, plan, rhs):
  92     if len(rhs) != plan.n:
  93         raise ValueError('wrong vector length')
  94     c = Counted(field)
  95     z = triangular_solve(c, plan.inverse_diagonal, plan.bands, rhs)
  96     width = len(plan.bands)
  97     alpha = []
  98     for row in plan.inverse_closure:
  99         acc = c.mul(row[0], z[0])
 100         for j in range(1, width):
 101             acc = c.add(acc, c.mul(row[j], z[j]))
 102         alpha.append(acc)
 103     out = list(z)
 104     for response, coefficient in zip(plan.responses, alpha):
 105         for i in range(plan.n):
 106             out[i] = c.sub(out[i], c.mul(response[i], coefficient))
 107     return out, dict(c.count)

 121 def prepare_rational(field, n, q, shift, grading=None):
 122     if n < 4 or n & (n-1) or shift == field.zero:
 123         raise ValueError('dyadic N>=4 and nonzero shift required')
 124     count = Counter()
 125     if grading is None:
 126         grading = prepare_grading(field, n, q)
 127         count.update(grading['preparation'])
 128     elif grading['n'] != n or grading['q'] != q:
 129         raise ValueError('stale grading plan')
 130     c = Counted(field)
 131     diagonal = c.neg(shift)
 132     source = prepare_band(field, diagonal, [grading['r']])
 133     target = prepare_band(field, diagonal, [grading['d1'], grading['d2']])
 134     count.update(c.count)
 135     count.update(source.preparation)
 136     count.update(target.preparation)
 137     return RationalPlan(n, q, shift, grading, source, target, dict(count))

 140 def generate_related_pair(field, plan, anchor_input, anchor_output):
 141     """Both inverses paid; caller's exact anchor_output=H anchor_input is trusted."""
 142     x, source_count = solve_band(field, plan.source, anchor_input)
 143     y, target_count = solve_band(field, plan.target, anchor_output)
 144     count = Counter(source_count)
 145     count.update(target_count)
 146     return x, y, dict(field_operations=dict(count), source_solve=source_count,
 147                       target_solve=target_count, full_vector_outputs=2*plan.n,

 151 def verified_output(field, plan, anchor_input, anchor_output, query_input):
 152     """Use a supplied shift only after checking its full relation in O(N)."""
 153     if any(len(x) != plan.n for x in (anchor_input, anchor_output, query_input)):
