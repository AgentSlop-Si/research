   9 def field(n):
  10  assert prime_by_trial_division(n) and n%2
  11  for k in range(1,1000):
  12   p=k*n+1
  13   if prime_by_trial_division(p):
  14    for g in range(2,p):
  15     w=pow(g,k,p)
  16     if w!=1 and pow(w,n,p)==1:return p,w
  17  raise AssertionError

  19 def D(n,p,w,a,b):
  20  mat=[[0]*n for _ in range(n)];two=pow(2,-1,n)
  21  for j in range(n):mat[(j+a)%n][j]=pow(w,(a*b*two+b*j)%n,p)
  22  return mat

  24 def qnorm(u,n):return (u[0]**2+u[1]**2)%n
  25 def det(u,v,n):return (u[0]*v[1]-u[1]*v[0])%n
  26 def kap(u,v,n):return ((u[0]+u[1])*v[0]+(u[1]-u[0])*v[1])%n
  27 def directions(n):return [(a,b) for a in range(n) for b in range(n) if a or b]
  28 def line(n,p,w,u,t=None,s=0):
  29  # t is the quadratic coefficient; canonical is -||u||^2/4.
  30  if t is None:t=-qnorm(u,n)*pow(4,-1,n)%n
  31  out=[[0]*n for _ in range(n)];a,b=u;two=pow(2,-1,n)
  32  for m in range(n):
  33   for j in range(n):out[(j+m*a)%n][j]=(out[(j+m*a)%n][j]+pow(w,(t*m*m+s*m+m*m*a*b*two+m*b*j)%n,p))%p
  34  return out

  36 def base_controls(n):
  37  p,w=field(n);f=Field(p);two=pow(2,-1,n);four=pow(4,-1,n);gamma=f.inv(n)*sum(pow(w,j*j%n,p) for j in range(n))%p;assert gamma
  38  F=[[pow(w,i*j%n,p) for j in range(n)] for i in range(n)];symbol={};recon=[[0]*n for _ in range(n)]
  39  for a,b in product(range(n),repeat=2):
  40   dm=D(n,p,w,-a,-b);actual=f.inv(n)*sum(dm[i][j]*F[j][i] for i in range(n) for j in range(n))%p
  41   expected=gamma*pow(w,(-(a*a+b*b)*four)%n,p)%p;assert actual==expected;symbol[a,b]=actual
  42   matrix=D(n,p,w,a,b)
  43   for i in range(n):
  44    for j in range(n):recon[i][j]=(recon[i][j]+actual*matrix[i][j])%p
  45  assert recon==F
  46  classes={'full_rank_anisotropic_lines':0,'rank_one_isotropic_lines':0,'nonvertical_fourier_equivalent_directions':0,'vertical_full_rank_directions':0};factor_checks=0
  47  for u in directions(n):
  48   a,b=u;q=qnorm(u,n);L=line(n,p,w,u)
  49   if not q:assert f.rank(L)==1;classes['rank_one_isotropic_lines']+=1;continue
  50   assert f.rank(L)==n;classes['full_rank_anisotropic_lines']+=1
  51   if not a:
  52    gm=sum(pow(w,(-j*j)%n,p) for j in range(n))%p;assert gm
  53    expected=[[gm*pow(w,j*j%n,p)%p if i==j else 0 for j in range(n)] for i in range(n)];classes['vertical_full_rank_directions']+=1
  54   else:
  55    inva2=pow(a,-2,n);left=-(a-b)**2*four*inva2%n;right=-(a+b)**2*four*inva2%n;slope=q*two*inva2%n;assert slope
  56    expected=[[pow(w,(left*i*i+slope*i*j+right*j*j)%n,p) for j in range(n)] for i in range(n)];classes['nonvertical_fourier_equivalent_directions']+=1
  57   assert expected==L
  58   for scale in (1,2,n-1):
  59    v=(scale*(a-b)%n,scale*(a+b)%n);assert det(u,v,n) and kap(u,v,n)==0 and kap(v,u,n)!=0
  60    Lv=line(n,p,w,v);assert Lv==line(n,p,w,((a-b)%n,(a+b)%n))
  61    actual=f.mul(L,Lv);assert all(gamma*actual[i][j]%p==F[i][j] for i in range(n) for j in range(n));factor_checks+=1
  62  return {'N':n,'field':p,'omega':w,'gamma':gamma,'weyl_coefficients_checked':n*n,'line_classification':classes,'all_compatible_factor_controls':factor_checks},symbol

  64 def pair_controls(n,symbol):
  65  p,w=field(n);f=Field(p);two=pow(2,-1,n);hist={};ind=0;cocycle_checks=0
  66  for u in directions(n):
  67   for v in directions(n):
  68    if not det(u,v,n):continue
  69    ind+=1;sigma=(u[1]*v[0]-u[0]*v[1])%n
  70    # Explicit matrix multiplication validates sign and ordering conventions.
  71    left=f.mul(D(n,p,w,*u),D(n,p,w,*v));right=D(n,p,w,(u[0]+v[0])%n,(u[1]+v[1])%n);phase=pow(w,sigma*two%n,p)
  72    assert all(left[i][j]==phase*right[i][j]%p for i in range(n) for j in range(n));cocycle_checks+=1
  73    T=[[symbol[(m*u[0]+ell*v[0])%n,(m*u[1]+ell*v[1])%n]*pow(w,(-sigma*m*ell*two)%n,p)%p for ell in range(n)] for m in range(n)]
  74    actual=f.rank(T);assert actual==(1 if kap(u,v,n)==0 else n);hist[actual]=hist.get(actual,0)+1
  75    if actual==1:
  76     assert qnorm(u,n) and any(v==(scale*(u[0]-u[1])%n,scale*(u[0]+u[1])%n) for scale in range(1,n))
  77  assert ind==(n*n-1)*(n*n-n)
  78  return {'N':n,'independent_vector_pairs':ind,'rank_histogram':hist,'cocycle_matrix_checks':cocycle_checks,'independent_projective_line_pairs':ind//((n-1)**2),'compatible_projective_line_pairs':hist[1]//((n-1)**2)}

  80 def stronger_degree_controls():
  81  out=[]
  82  for n in (3,5,7,11,13,17,19):
  83   p,w=field(n);f=Field(p);cases=0
  84   for q in (1,2,n-1):
  85    chirp=[pow(w,(-q*m*m*pow(4,-1,n))%n,p) for m in range(n)]
  86    for d in range(n-1):
  87     for e in range(n-1-d):
  88      # Tail zero equations for q_den(C)L=p(C); test absence of a nonzero denominator.
  89      rows=[[chirp[(r-k)%n] for k in range(d+1)] for r in range(e+1,n)]
  90      assert f.rank(rows)==d+1;cases+=1
  91    # Tight e=0 control: inverse cyclic polynomial has all N coefficients nonzero.
  92    circ=[[chirp[(i-j)%n] for j in range(n)] for i in range(n)];inv=f.inverse(circ);coeff=[inv[i][0] for i in range(n)];assert all(coeff)
  93    conv=[sum(coeff[k]*chirp[(r-k)%n] for k in range(n))%p for r in range(n)];assert conv==[1]+[0]*(n-1)
  94   out.append({'N':n,'tail_constraint_full_column_rank_checks':cases,'tight_constant_numerator_degree':n-1})
  95  return out
