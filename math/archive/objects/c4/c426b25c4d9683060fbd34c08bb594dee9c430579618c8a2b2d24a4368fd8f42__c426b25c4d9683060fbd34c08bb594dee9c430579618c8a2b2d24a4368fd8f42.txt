import unittest
from commutator_mod import one_green,commute_shift,rank_mod,modular_commutator_audit
from commutator_audit import analyze_stationary

class CommutatorTests(unittest.TestCase):
    def test_exact_single_green_rank_8(self):
        self.assertLessEqual(rank_mod(commute_shift(one_green(8,24,-7,-7))),2)
    def test_exact_single_green_rank_32(self):
        self.assertLessEqual(rank_mod(commute_shift(one_green(32,72,-31,-30))),2)
    def test_exact_product_rank_16(self):
        x=modular_commutator_audit(16)
        self.assertLessEqual(x['product_rank'],8)
        self.assertEqual(x['Fourier_commutator_rank'],16)
    def test_exact_product_rank_32(self):
        x=modular_commutator_audit(32)
        self.assertLessEqual(x['product_rank'],8)
        self.assertEqual(x['Fourier_commutator_rank'],32)
    def test_exact_product_rank_64(self):
        x=modular_commutator_audit(64)
        self.assertLessEqual(x['product_rank'],8)
        self.assertEqual(x['Fourier_commutator_rank'],64)
    def test_stable_operator_approx_bound(self):
        r=analyze_stationary(32)
        self.assertGreaterEqual(r['actual_spectral_error']+1e-10,r['guaranteed_error_floor'])
    def test_near_resonant_operator_approx_bound(self):
        r=analyze_stationary(32,near_resonant=True)
        self.assertGreaterEqual(r['actual_spectral_error']+1e-10,r['guaranteed_error_floor'])

if __name__=='__main__':unittest.main()
