   8 def roots(n,p):

  14 def reverse(k,n):return int(format(k,'0'+str(n.bit_length()-1)+'b')[::-1],2)
  15 class Schedule:
  16  # Independent flat-heap node state, separate primitive labels, in-place data.
  17  def __init__(self,x,W,p):
  18   self.a=x[:];self.W=W;self.p=p;self.N=len(x);self.state=[0]*(2*self.N);self.saved=[None]*(2*self.N);self.tags=set();self.adds=self.muls=self.current=self.peak=0;self.closed=False
  19  def operation(self,node,j,kind):
  20   tag=(node,j,kind);assert tag not in self.tags;self.tags.add(tag)
  21   if kind=='M':self.muls+=1
  22   else:self.adds+=1
  23  def prepare_branch(self,node,start,size,which):
  24   h=size//2;old=self.state[node]
  25   assert which&old==0
  26   if old==0 and which!=3:
  27    shift=0 if which==1 else h
  28    self.saved[node]=self.a[start+shift:start+shift+h];self.current+=h;self.peak=max(self.peak,self.current)
  29   for j in range(h):
  30    if old==1:u=self.saved[node][j];v=self.a[start+h+j]
  31    elif old==2:u=self.a[start+j];v=self.saved[node][j]
  32    else:u=self.a[start+j];v=self.a[start+h+j]
  33    if which&1:self.operation(node,j,'A');self.a[start+j]=(u+v)%self.p
  34    if which&2:
  35     self.operation(node,j,'D');self.operation(node,j,'M');self.a[start+h+j]=(u-v)*self.W[j*(self.N//size)]%self.p
  36   self.state[node]|=which
  37   if old and self.state[node]==3:self.saved[node]=None;self.current-=h
  38  def preview(self,Q):
  39   if self.closed:raise ValueError('preview after completed graph rejected in corrected lifecycle')
  40   Q=set(Q)
  41   if not Q or any(k<0 or k>=self.N for k in Q):raise ValueError('invalid queries')
  42   assert not any(self.state)
  43   stack=[(1,0,self.N,Q)]
  44   while stack:
  45    node,start,size,q=stack.pop()
  46    if size==1:continue
  47    h=size//2;groups=[{k//2 for k in q if k%2==bit} for bit in [0,1]];which=int(bool(groups[0]))+2*int(bool(groups[1]));self.prepare_branch(node,start,size,which)
  48    for bit in [1,0]:
  49     if groups[bit]:stack.append((2*node+bit,start+bit*h,h,groups[bit]))
  50   assert self.current==self.peak==self.N-len(Q)
  51   return {k:self.a[reverse(k,self.N)] for k in Q}
  52  def full(self):
  53   stack=[(1,0,self.N)]
  54   while stack:
  55    node,start,size=stack.pop()
  56    if size==1:continue
  57    if self.state[node]!=3:self.prepare_branch(node,start,size,3^self.state[node])
  58    h=size//2;stack.extend([(2*node+1,start+h,h),(2*node,start,h)])
  59   self.closed=True;assert self.current==0
  60   assert self.adds==self.N*(self.N.bit_length()-1) and self.muls==self.N//2*(self.N.bit_length()-1)
  61   return [self.a[reverse(k,self.N)] for k in range(self.N)]

  63 def tree_count(N,Q):
  64  adds=muls=backups=eager=0;stack=[(N,set(Q))]
  65  while stack:
  66   size,q=stack.pop()
  67   if size==1:continue
  68   h=size//2;e={k//2 for k in q if not k%2};o={k//2 for k in q if k%2};eager+=3*h
  69   if e:adds+=h;stack.append((h,e))
  70   if o:adds+=h;muls+=h;stack.append((h,o))
  71   if bool(e)!=bool(o):backups+=h
  72  return {'adds':adds,'muls':muls,'backup':backups,'eager_arithmetic':eager}

 110 def discover(R,loader,N,w,p,s=4):
 111  meta={'residual_calls':0,'full_checks':0,'failed_checks':0}
 112  if not all(R):return None,{**meta,'reason':'zero-probe'}
 113  inv=[pow(v,-1,p) for v in R];U=M.compress(inv,s,p)
 114  if not any(U):return None,{**meta,'reason':'zero-resultant'}
 115  b=s*(s+1)*(2*s+1)//6;V=T.trace_from_U(U,b,p);G,_=T.sieve(V,N,p);ds=T.enum_half(G,w,N,p) if len(G)>1 else []
 116  if not ds:return None,{**meta,'reason':'trace-gcd-one'}
 117  powers=[pow(w,j,p) for j in range(N)];r=None
 118  for delta in ds:
 119   g=M.pgcd(M.pencil(inv,powers[delta],0,s,p),M.pencil(inv,powers[delta],1,s,p),p)
 120   if len(g)!=2 or g[0]==0:continue
 121   alpha=-g[0]%p;Y=[v*(1-alpha*pow(w,k*delta,p))%p for k,v in enumerate(inv)];A=M.prony(Y,s,powers,p)
 122   if not A:continue
 123   Ah=M.sparse_eval(A,w,N,p)
 124   if not all(Ah):continue
 125   if r is None:r=loader();meta['residual_calls']+=1
 126   meta['full_checks']+=1;lhs=[sum(a*r[(j-e)%N] for e,a in A.items())%p for j in range(N)];rhs=[(int(j==0)-alpha*int(j==delta))%p for j in range(N)]
 127   if lhs!=rhs:meta['failed_checks']+=1;continue
 128   F=[(1-alpha*pow(w,k*delta,p))*pow(Ah[k],-1,p)%p for k in range(N)]
 129   assert F==M.dft(r,w,p)
 130   return F,{**meta,'reason':'accepted'}
 131  return None,{**meta,'reason':'full-certificate-failed' if meta['failed_checks'] else 'no-candidate'}
