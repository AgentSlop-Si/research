   1 #!/usr/bin/env python3
   2 """Read-only nested ZIP audit, never executing archive members. Standard library."""
   3 from pathlib import Path, PurePosixPath
   4 from zipfile import ZipFile
   5 import hashlib, io, json, re, os
   6 
   7 ROOT=Path(__file__).resolve().parent
   8 def digest(b):return hashlib.sha256(b).hexdigest()
   9 def main():
  10     prior=set(json.loads((ROOT/'PRIOR_HASH_REFERENCE.json').read_text())['hashes'])
  11     ids=json.loads((ROOT/'SOURCE_IDENTITIES.json').read_text())
  12     inventory=[];checks=[];archives=[];new={};budget=0
  13     def visit(data,origin,depth=0):
  14         nonlocal budget
  15         assert depth<32
  16         with ZipFile(io.BytesIO(data)) as z:
  17             names=z.namelist();assert len(names)==len(set(names))
  18             payload={}
  19             for obj in z.infolist():
  20                 path=PurePosixPath(obj.filename)
  21                 assert not path.is_absolute() and '..' not in path.parts
  22                 assert (obj.external_attr>>16)&0o170000 != 0o120000, 'symlink source'
  23                 if obj.is_dir():continue
  24                 budget+=obj.file_size;assert budget<250_000_000
  25                 raw=z.read(obj);payload[obj.filename]=raw
  26                 sha=digest(raw)
  27                 row={'archive':origin,'member':obj.filename,'bytes':len(raw),'sha256':sha,'new_to_prior_source_union':sha not in prior}
  28                 inventory.append(row)
  29                 if sha not in prior:new.setdefault(sha,row)
  30             for name,raw in payload.items():
  31                 if PurePosixPath(name).name.upper().startswith('SHA256SUMS'):
  32                     parent=PurePosixPath(name).parent
  33                     for no,line in enumerate(raw.decode().splitlines(),1):
  34                         if not line.strip() or line.lstrip().startswith('#'):continue
  35                         m=re.match(r'^([0-9a-fA-F]{64})\s+\*?(.+?)\s*$',line)
  36                         assert m,(origin,name,no,line)
  37                         wanted=m.group(2);choices=[str(parent/wanted),wanted]
  38                         target=next((x for x in choices if x in payload),None)
  39                         ok=target is not None and digest(payload[target])==m.group(1).lower()
  40                         checks.append({'archive':origin,'manifest':name,'line':no,'target':target or wanted,'ok':ok})
  41                         assert ok,(origin,name,no,wanted)
  42                 if name.lower().endswith('.zip'):visit(raw,origin+'!/'+name,depth+1)
  43     for identity in ids:
  44         name=identity['file_name'];path=ROOT/'source_archives'/name
  45         if not path.exists():path=ROOT/'downloads'/name
  46         data=path.read_bytes();assert len(data)==identity['size_bytes'];assert digest(data)==identity['sha256']
  47         version=identity['version']
  48         if 'user.library-file-version' in os.listxattr(path):version=int(os.getxattr(path,'user.library-file-version'))
  49         archives.append({'filename':name,'library_file_id':identity['library_file_id'],'file_id':identity['file_id'],'version':version,'bytes':len(data),'sha256':digest(data)})
  50         visit(data,name)
  51     out={'status':'PASS','outer_archives':len(archives),'recursive_member_occurrences':len(inventory),'distinct_member_contents':len({x['sha256'] for x in inventory}),'byte_new_member_contents':len(new),'embedded_checksum_entries':len(checks),'embedded_checksum_failures':sum(not x['ok'] for x in checks),'uploaded_code_executed':False}
  52     for name,obj in [('SOURCE_ARCHIVES.json',archives),('SOURCE_MEMBER_INVENTORY.json',inventory),('NEW_UNIQUE_CONTENT.json',list(new.values())),('SOURCE_MANIFEST_VERIFICATION.json',{'summary':out,'entries':checks})]:
  53         (ROOT/name).write_text(json.dumps(obj,indent=2)+'\n')
  54     print(json.dumps(out,indent=2))
  55 if __name__=='__main__':main()
