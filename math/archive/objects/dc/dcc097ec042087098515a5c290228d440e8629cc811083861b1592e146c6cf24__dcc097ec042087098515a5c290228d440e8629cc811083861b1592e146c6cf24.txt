"""Independent exhaustive v6 median covariance probe. No uploaded imports."""
import itertools, json
from pathlib import Path
import numpy as np

def row_errors(A):
    K,M=A.shape
    phases=np.array([1,1j,-1,-1j])
    out=[]
    for qtail in itertools.product(range(K),repeat=M-1):
      q=np.array((0,)+qtail)
      for itail in itertools.product(range(4),repeat=M-1):
        xi=phases[np.array((0,)+itail)]
        e=np.zeros_like(A,dtype=complex)
        for p in range(K):
          for t in range(M):
            e[p,t]=sum(xi[t]*xi[u].conjugate()*A[(p+q[u]-q[t])%K,u] for u in range(M) if u!=t)
        out.append(e)
    return np.asarray(out)

def probe(A):
    e=row_errors(A); n=len(e); K,M=A.shape
    sumerr=np.zeros_like(A); cov=np.zeros((K,M,M),complex)
    # Exhaust all ordered triples. Median translations remove target A exactly.
    for a in range(n):
      b,c=np.meshgrid(np.arange(n),np.arange(n),indexing='ij')
      trip=np.stack((np.broadcast_to(e[a],(n*n,K,M)),e[b.ravel()],e[c.ravel()]),axis=1)
      med=np.median(trip.real,axis=1)+1j*np.median(trip.imag,axis=1)
      sumerr+=med.sum(axis=0)
      cov+=np.einsum('bpt,bpu->ptu',med,med.conj())
    cov/=n**3
    return {'K':K,'M':M,'states':n,'triples':n**3,'A_real':A.real.tolist(),'A_imag':A.imag.tolist(),'mean_abs_max':float(np.max(np.abs(sumerr/n**3))),'cov_real':cov.real.tolist(),'cov_imag':cov.imag.tolist(),'max_offdiag':float(max(abs(cov[p,t,u]) for p in range(K) for t in range(M) for u in range(M) if t!=u))}

if __name__=='__main__':
  cases=[np.array([[1,2,3,4]],complex),np.array([[1,2,3]],complex),np.array([[1+2j,3-1j,2+1j]],complex),np.array([[1+2j,3-1j,2+1j],[2-1j,1+3j,-2+2j]],complex)]
  out=[probe(A) for A in cases]
  with Path(__file__).with_name('covariance_results.json').open('w') as f:json.dump(out,f,indent=2)
  for o in out: print(o)
