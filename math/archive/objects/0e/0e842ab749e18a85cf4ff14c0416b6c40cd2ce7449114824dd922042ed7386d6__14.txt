0015 def twiddle_plan(n):
0016     return {m: tuple(cmath.exp(-2j*math.pi*k/m) for k in range(m//2))
0017             for m in (1 << j for j in range(1,n.bit_length()))}

0020 def standard_warm(signal, twiddles):
0021     def rec(offset,stride,size):
0022         if size == 1: return (complex(signal[offset]),)
0023         even=rec(offset,stride*2,size//2); odd=rec(offset+stride,stride*2,size//2)
0024         factors=twiddles[size]; half=size//2; out=[0j]*size
0025         for k in range(half):
0026             t=factors[k]*odd[k]; out[k]=even[k]+t; out[k+half]=even[k]-t
0027         return tuple(out)
0028     return rec(0,1,len(signal))

0031 def canonical_no_stats(signal, twiddles=None):
0032     """Original algorithm, removing only counters and summary statistics.
0033     Optional reusable twiddles; dictionaries, DAG, tuple allocation remain.
0034     """
0035     n=len(signal); intern={}; nodes=[]
0036     def build(offset,stride,size):
0037         if size==1:
0038             value=signal[offset]; key=(1,value); spec=(1,-1,-1,value)
0039         else:
0040             left=build(offset,stride*2,size//2); right=build(offset+stride,stride*2,size//2)
0041             key=(size,left,right); spec=(size,left,right,None)
0042         ident=intern.get(key)
0043         if ident is None:
0044             ident=len(nodes); intern[key]=ident; nodes.append(spec)
0045         return ident
0046     root=build(0,1,n); results={}
0047     if twiddles is None: twiddles={}
0048     def evaluate(ident):
0049         if ident in results: return results[ident]
0050         size,left_id,right_id,value=nodes[ident]
0051         if size==1: out=(complex(value),)
0052         else:
0053             left=evaluate(left_id); right=evaluate(right_id); half=size//2
0054             factors=twiddles.get(size)
0055             if factors is None:
0056                 factors=[cmath.exp(-2j*math.pi*k/size) for k in range(half)]
0057                 twiddles[size]=factors
0058             out=[0j]*size
0059             for k in range(half):
0060                 t=factors[k]*right[k]; out[k]=left[k]+t; out[k+half]=left[k]-t
0061             out=tuple(out)
0062         results[ident]=out
0063         return out
0064     return evaluate(root)

0067 def make_table(b):
0068     """Python-object table; mathematical compact complex128 size is reported separately."""
0069     plan=twiddle_plan(b)
0070     return tuple(standard_warm([(key>>j)&1 for j in range(b)],plan) for key in range(1<<b))

0073 def lookup_warm(signal,twiddles,b,table):
0074     """Packing follows natural order of the actual decimated subproblem.
0075     N/b lookups, exactly N bit gathers/shifts/ORs; no dictionary or DAG.
0076     A leaf returns a read-only tuple; upper merges allocate outputs normally.
0077     """
0078     def rec(offset,stride,size):
0079         if size==b:
0080             key=0
0081             for j in range(b): key |= signal[offset+j*stride] << j
0082             return table[key]
0083         even=rec(offset,stride*2,size//2); odd=rec(offset+stride,stride*2,size//2)
0084         factors=twiddles[size]; half=size//2; out=[0j]*size
0085         for k in range(half):
0086             t=factors[k]*odd[k]; out[k]=even[k]+t; out[k+half]=even[k]-t
0087         return tuple(out)
0088     return rec(0,1,len(signal))

0091 def timed_once(fn):
0092     start=time.perf_counter_ns(); value=fn(); elapsed=(time.perf_counter_ns()-start)*1e-9
0093     # deallocation of the returned array is excluded equally for all kernels.
0094     return elapsed,value
