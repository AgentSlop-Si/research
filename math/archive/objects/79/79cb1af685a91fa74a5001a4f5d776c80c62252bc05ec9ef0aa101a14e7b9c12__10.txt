0019 def stat(values):
0020     a=np.asarray(values);m=float(np.median(a))
0021     return {'median_us':m*1e6,'p10_us':float(np.percentile(a,10))*1e6,
0022             'p90_us':float(np.percentile(a,90))*1e6,
0023             'relative_mad':float(np.median(abs(a-m))/m),'samples_us':(a*1e6).tolist()}

0030 def ordinary_setup(lib,n):
0031     lib.fft_plan_create.argtypes=[ct.c_size_t];lib.fft_plan_create.restype=ct.c_void_p
0032     lib.fft_plan_destroy.argtypes=[ct.c_void_p]
0033     times=[]
0034     for _ in range(9):
0035         start=time.perf_counter();p=lib.fft_plan_create(n);elapsed=time.perf_counter()-start
0036         assert p;lib.fft_plan_destroy(p);times.append(elapsed)
0037     return stat(times)

0039 def benchmark_heldout(plan,pool,rounds=13,target_ms=20):
0040     out=np.empty(plan.n,np.complex128);methods=list(METHODS)+['selected'];samples={m:[] for m in methods};repeats={}
0041     for m in methods:
0042         estimate=plan.benchmark(m,pool,out,len(pool)*2)
0043         repeats[m]=max(len(pool),math.ceil(target_ms/1000/estimate/len(pool))*len(pool))
0044     rng=random.Random(0x9182817+plan.n)
0045     for _ in range(rounds):
0046         rng.shuffle(methods)
0047         for m in methods:samples[m].append(plan.benchmark(m,pool,out,repeats[m]))
0048     return {m:dict(stat(s),transforms_per_sample=repeats[m]) for m,s in samples.items()}

0050 def benchmark_public(plan,pool,rounds=7,target_ms=12):
0051     out=np.empty(plan.n,np.complex128)
0052     calls={'plan_execute_out':lambda x:plan.execute(x,out=out),
0053            'plan_execute_allocating':lambda x:plan.execute(x),
0054            'cached_binary_fft_allocating':lambda x:binary_fft(x)}
0055     repeats={};samples={m:[] for m in calls}
0056     for m,f in calls.items():
0057         start=time.perf_counter()
0058         for x in pool:f(x)
0059         est=(time.perf_counter()-start)/len(pool)
0060         repeats[m]=max(len(pool),math.ceil(target_ms/1000/est/len(pool))*len(pool))
0061     rng=random.Random(0x238123+plan.n);order=list(calls)
0062     for _ in range(rounds):
0063         rng.shuffle(order)
0064         for m in order:
0065             start=time.perf_counter()
0066             for r in range(repeats[m]):calls[m](pool[r%len(pool)])
0067             samples[m].append((time.perf_counter()-start)/repeats[m])
0068     return {m:dict(stat(s),transforms_per_sample=repeats[m]) for m,s in samples.items()}
