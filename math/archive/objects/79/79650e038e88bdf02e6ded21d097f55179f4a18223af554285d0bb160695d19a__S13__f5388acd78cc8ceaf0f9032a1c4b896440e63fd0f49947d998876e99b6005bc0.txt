  36     def test_rational_polynomial_identity(self):
  37         for n,p,g in PAIRS:
  38             e=AllLengthGreen(n,p,g)
  39             for k in (1,2,min(7,n)):
  40                 entries={j:self.rng.randrange(1,p) for j in self.rng.sample(range(n),k)}

  43                 self.assertEqual(len(den),k+1)
  44                 b=min(32,1<<(k-1).bit_length())
  45                 inv=inv_series(den,b,p,g)
  46                 check=conv(den,inv,p,g)[:b]
  47                 self.assertEqual(check,[1]+[0]*(b-1))
  48                 out,_=forcing_from_sparse(entries,n,e.omega,p,g)
  49                 check=conv(den,out,p,g)[:n]
  50                 self.assertEqual(check,num+[0]*(n-len(num)))

 180     def test_complex_norm_certificate(self):
 181         try:import numpy as np
 182         except ImportError:self.skipTest('numpy is not installed')
 183         for n in (8,16,32):
 184             w=np.exp(-2j*np.pi/n)
 185             diag=7+w**np.arange(n)
 186             mat=np.diag(diag)+2*np.roll(np.eye(n),1,axis=0)
 187             self.assertLessEqual(np.linalg.cond(mat),2.5+1e-9)
 188             rng=np.random.default_rng(n)
 189             x=rng.normal(size=n)+1j*rng.normal(size=n)
 190             right=7*x+np.roll(x,1)+2*w**(-np.arange(n))*x
 191             true=np.fft.fft(x)/np.sqrt(n)
 192             out=np.linalg.solve(mat,np.fft.fft(right)/np.sqrt(n))
 193             self.assertLess(np.linalg.norm(out-true),1e-10)
 194             indices=np.argsort(np.abs(right))[-max(1,n//4):]
 195             short=np.zeros(n,dtype=complex);short[indices]=right[indices]
 196             approx=np.linalg.solve(mat,np.fft.fft(short)/np.sqrt(n))
 197             err=np.linalg.norm(true-approx)
 198             tail=np.linalg.norm(right-short)
 199             self.assertLessEqual(err,tail/4+1e-10)
 200             self.assertGreaterEqual(err,tail/10-1e-10)
