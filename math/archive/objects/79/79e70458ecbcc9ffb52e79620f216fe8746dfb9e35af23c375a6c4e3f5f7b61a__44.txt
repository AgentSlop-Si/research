43: def brute_tree_optimum(queries,b,base):
44:     # Exhaustively enumerate all trees on a super-root plus four query nodes
45:     # via Prüfer sequences. This independently checks planner optimality.
46:     cold=run(queries,b,base,graph=True)
47:     n=len(b); q=len(queries); nodes=q+1
48:     w={}
49:     for i in range(q):
50:         w[i,q]=fresh_weight(n,cold['context'].m)
51:         for j in range(i+1,q):
52:             s=sum(a!=z for a,z in zip(queries[i],queries[j]))
53:             w[i,j]=2*n*s+2*n-1 if s else 0
54:     best=None; trees=0
55:     for sequence in itertools.product(range(nodes),repeat=nodes-2):
56:         degree=[1]*nodes
57:         for x in sequence: degree[x]+=1
58:         edges=[]
59:         for x in sequence:
60:             leaf=next(i for i in range(nodes) if degree[i]==1)
61:             edges.append(tuple(sorted((leaf,x))))
62:             degree[leaf]-=1; degree[x]-=1
63:         last=[i for i in range(nodes) if degree[i]==1]
64:         edges.append(tuple(last))
65:         cost=sum(w[e] for e in edges)
66:         best=cost if best is None else min(best,cost)
67:         trees+=1
68:     assert best==cold['product_arithmetic']
69:     return trees

72: def coverage_check():
73:     q,d,k=3,4,1
74:     anchors=[(a,)*d for a in range(q)]
75:     inputs=list(itertools.product(range(q),repeat=d))
76:     covered=[x for x in inputs if any(sum(a!=b for a,b in zip(x,c))<=k for c in anchors)]
77:     volume=sum(math.comb(d,j)*(q-1)**j for j in range(k+1))
78:     assert len(covered)==len(anchors)*volume==27
79:     witness=next(x for x in inputs if x not in covered)
80:     return {'alphabet':q,'dimension':d,'radius':k,'anchors':len(anchors),
81:             'ball_volume':volume,'covered':len(covered),'total':len(inputs),
82:             'minimum_anchors_from_volume_bound':math.ceil(q**d/volume),
83:             'uncovered_witness':witness}
