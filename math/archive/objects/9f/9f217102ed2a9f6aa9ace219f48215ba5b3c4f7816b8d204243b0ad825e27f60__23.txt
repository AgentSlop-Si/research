0007 def isprime(n):
0008     if n<2:return False
0009     return all(n%d for d in range(2,math.isqrt(n)+1))
0010 def root_of_order(order,p):
0011     factors=[]; r=order; d=2
0012     while d*d<=r:
0013         if r%d==0:
0014             factors.append(d)
0015             while r%d==0:r//=d
0016         d+=1
0017     if r>1:factors.append(r)
0018     for a in range(2,p):
0019         x=pow(a,(p-1)//order,p)
0020         if pow(x,order,p)==1 and all(pow(x,order//d,p)!=1 for d in factors):return x
0021     raise ValueError('no root')
0022 def field_for(m):
0023     n=m*m; L=1
0024     while L<3*m-2:L*=2
0025     l=math.lcm(n,L); p=l+1
0026     while not isprime(p):p+=l
0027     return p,root_of_order(n,p),L,root_of_order(L,p)
0028 def ntt(a,w,p,inverse=False):
0029     a=list(a); n=len(a); j=0
0030     for i in range(1,n):
0031         bit=n>>1
0032         while j&bit:j^=bit;bit>>=1
0033         j^=bit
0034         if i<j:a[i],a[j]=a[j],a[i]
0035     if inverse:w=pow(w,-1,p)
0036     width=2
0037     while width<=n:
0038         ws=pow(w,n//width,p)
0039         for start in range(0,n,width):
0040             phase=1
0041             for off in range(width//2):
0042                 u=a[start+off];v=a[start+off+width//2]*phase%p
0043                 a[start+off]=(u+v)%p;a[start+off+width//2]=(u-v)%p
0044                 phase=phase*ws%p
0045         width*=2
0046     if inverse:
0047         ni=pow(n,-1,p);a=[x*ni%p for x in a]
0048     return a
0049 def conv(a,b,p,L,w):
0050     assert len(a)+len(b)-1<=L
0051     aa=ntt(a+[0]*(L-len(a)),w,p);bb=ntt(b+[0]*(L-len(b)),w,p)
0052     return ntt([x*y%p for x,y in zip(aa,bb)],w,p,True)[:len(a)+len(b)-1]
0053 def geom_forward(b,q,p,L,w):
0054     m=len(b)
0055     c=[pow(q,r*(r-1)//2,p) for r in range(2*m-1)]
0056     a=[b[j]*pow(c[j],-1,p)%p for j in range(m)]
0057     z=conv(a[::-1],c,p,L,w)
0058     return [z[t+m-1]*pow(c[t],-1,p)%p for t in range(m)]
0059 def inv_geom(c,q,p,L,w):
0060     m=len(c); fact=[1]
0061     for r in range(1,m):fact.append(fact[-1]*(1-pow(q,r,p))%p)
0062     assert all(fact)
0063     fi=[pow(x,-1,p) for x in fact]
0064     tri=[pow(q,r*(r-1)//2,p) for r in range(m)]
0065     left=[c[t]*(-1 if t%2 else 1)*fi[t]%p for t in range(m)]
0066     kernel=[tri[s]*fi[s]%p for s in range(m)]
0067     d=conv(left,kernel,p,L,w)
0068     a=[d[r]*pow(tri[r],-1,p)%p for r in range(m)]
0069     bkernel=[(-1 if s%2 else 1)*tri[s]*fi[s]%p for s in range(m)]
0070     rev=[a[r]*fact[r]%p for r in range(m)][::-1]
0071     e=conv(rev,bkernel,p,L,w)
0072     return [e[m-1-j]*fi[j]%p for j in range(m)]
0073 def dense_geom(b,q,p):
0074     return [sum(b[j]*pow(q,t*j,p) for j in range(len(b)))%p for t in range(len(b))]
0075 def direct_dft(x,omega,p):
0076     return [sum(v*pow(omega,-k*r,p) for r,v in enumerate(x))%p for k in range(len(x))]
0077 def batch_inverse(values,p):
0078     prefix=[1]
0079     for v in values:
0080         assert v
0081         prefix.append(prefix[-1]*v%p)
0082     z=pow(prefix[-1],-1,p);out=[0]*len(values)
0083     for i in range(len(values)-1,-1,-1):
0084         out[i]=z*prefix[i]%p;z=z*values[i]%p
0085     return out
