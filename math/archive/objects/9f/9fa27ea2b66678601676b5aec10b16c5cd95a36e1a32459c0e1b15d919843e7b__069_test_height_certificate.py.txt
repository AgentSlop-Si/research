import unittest
from fractions import Fraction
from height_certificate import ceil_log2, extract


class HeightCertificateTests(unittest.TestCase):
    def test_integer_rounding(self):
        self.assertEqual([ceil_log2(x) for x in range(1, 9)], [0, 1, 2, 2, 3, 3, 3, 3])

    def test_root_and_rational_dag(self):
        spec = {"field_degree_upper_bound": 4, "nodes": [
            {"id":"z", "op":"root", "order":8},
            {"id":"r", "op":"rational", "numerator":3,"denominator":2},
            {"id":"s", "op":"add", "args":["z","r"]},
            {"id":"p", "op":"mul", "args":["s","s"]},
            {"id":"q", "op":"div", "args":["p","r"], "nonzero_proof_reference":"r=3/2"}],
            "shear_coefficient_nodes":["z","s","q"]}
        out = extract(spec)
        self.assertEqual(out["height_budgets"], {"z":0,"r":2,"s":3,"p":6,"q":8})
        self.assertEqual(out["shear_boundary_kappa_upper_bound"]["exponent"], 94)
        self.assertEqual(out["scalar_instruction_kappa_upper_bound"]["exponent"], 202)
        self.assertEqual(out["scalar_instruction_count"], 18)

    def test_division_requires_proof(self):
        with self.assertRaises(ValueError):
            extract({"field_degree_upper_bound":1,"nodes":[
                {"id":"r","op":"rational","numerator":1,"denominator":1},
                {"id":"s","op":"div","args":["r","r"]}],
                "shear_coefficient_nodes":["s"]})

    def test_empty_word(self):
        out = extract({"field_degree_upper_bound":1,"nodes":[],"shear_coefficient_nodes":[]})
        self.assertEqual(out["scalar_instruction_kappa_upper_bound"], {"base":2,"exponent":0})

    def test_six_scalar_instructions_exact(self):
        # Includes zero alpha, whose naive source*=alpha would be singular.
        for alpha in [Fraction(0), Fraction(1), Fraction(-1), Fraction(3,2), Fraction(-7,3)]:
            C=Fraction(8)
            for x0,y0 in [(Fraction(5,7),Fraction(-3,11)),(Fraction(0),Fraction(9))]:
                x,y=x0,y0
                for beta in [C, alpha-C]:
                    self.assertNotEqual(beta, 0)
                    x *= beta
                    y += x
                    x /= beta
                self.assertEqual((x,y),(x0,y0+alpha*x0))

    def test_repeated_squaring_height_growth(self):
        nodes=[{"id":"two","op":"rational","numerator":2,"denominator":1}]
        previous="two"
        for j in range(20):
            name=f"square_{j}"
            nodes.append({"id":name,"op":"mul","args":[previous,previous]})
            previous=name
        out=extract({"field_degree_upper_bound":1,"nodes":nodes,"shear_coefficient_nodes":[previous]})
        self.assertEqual(out["maximum_coefficient_height_budget"],2**20)
        self.assertLess(out["stored_scalar_certificate_exponent_bit_length"],30)


if __name__ == '__main__':
    unittest.main(verbosity=2)
